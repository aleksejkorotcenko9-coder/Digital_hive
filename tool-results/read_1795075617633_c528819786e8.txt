    62→/** «2026-08-29» → «29 августа 2026» (UTC). */
    63→function fmtDateLabel(key: string): string {
    64→  const [y, m, d] = key.split('-').map(Number)
    65→  return `${d} ${MONTHS_GEN[m - 1]} ${y}`
    66→}
    67→
    68→/** «2026-08» → «Август 2026». */
    69→function fmtMonthLabel(key: string): string {
    70→  const [y, m] = key.split('-').map(Number)
    71→  return `${MONTHS_NOM[m - 1]} ${y}`
    72→}

    74→/** ISO → «HH:MM» (UTC). */
    75→function fmtTime(iso: string): string {
    76→  const d = new Date(iso)
    77→  return `${pad2(d.getUTCHours())}:${pad2(d.getUTCMinutes())}`
    78→}
    79→
    80→const dayKeyOf = (d: Date) => d.toISOString().slice(0, 10)
    81→const parseDayKey = (key: string) => new Date(`${key}T00:00:00Z`)
    82→
    83→function addDaysKey(key: string, days: number): string {
    84→  const d = parseDayKey(key)
    85→  d.setUTCDate(d.getUTCDate() + days)
    86→  return dayKeyOf(d)
    87→}
    88→
    89→function addMonthsKey(key: string, months: number): string {
    90→  const [y, m] = key.split('-').map(Number)
    91→  const d = new Date(Date.UTC(y, m - 1 + months, 1))
    92→  return `${d.getUTCFullYear()}-${pad2(d.getUTCMonth() + 1)}`
    93→}

   158→/** Стрелка дельты: зелёный рост, красное падение, серый ноль. */
   159→function Delta({ v, unit }: { v: number; unit: string }) {
   160→  const eps = unit === '°C' ? 0.05 : unit === '%' ? 0.5 : 0.05
   161→  const up = v > eps
   162→  const down = v < -eps
   163→  return (
   164→    <span
   165→      className={cn(
   166→        'inline-flex items-center gap-0.5 font-semibold',
   167→        up && 'text-emerald-600',
   168→        down && 'text-red-600',
   169→        !up && !down && 'text-gray-400',
   170→      )}
   171→    >
   172→      {up ? '↑' : down ? '↓' : '·'} {up || down ? `${v > 0 ? '+' : ''}${unit === '%' ? Math.round(v) : v.toFixed(1)}` : '0'}
   173→    </span>
   174→  )
   175→}

   300→export function HistoryView() {
   301→  const [scope, setScope] = useState<HistoryScope>('day')
   302→  /** Выбранный период: yyyy-MM-dd | yyyy-MM | yyyy; null — «последние записи». */
   303→  const [dateKey, setDateKey] = useState<string | null>(null)
   304→  const [pointFilter, setPointFilter] = useState<'all' | string>('all')
   305→  const [data, setData] = useState<HistoryResponse | null>(null)
   306→  const [calOpen, setCalOpen] = useState(false)
   307→  const [reloadTick, setReloadTick] = useState(0)
   308→
   309→  const [hoverPop, setHoverPop] = useState<PopState | null>(null)
   310→  const [pinnedPop, setPinnedPop] = useState<PopState | null>(null)
   311→  const leaveTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   312→
   313→  const range = data?.range ?? { from: '2026-07-01', to: '2026-08-29' }
   314→
   315→  /* Загрузка: состояние «загрузки/ошибки» производное от ключа периода —
   316→     без setState внутри тела эффекта (правило react-hooks). */
   317→  const defaultYear = range.to.slice(0, 4)
   318→  const currentKey = `${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`
   319→  const [loadedKey, setLoadedKey] = useState<string | null>(null)
   320→  const [errState, setErrState] = useState<{ key: string; message: string } | null>(null)
   321→  const loading = loadedKey !== currentKey
   322→  const error = errState && errState.key === currentKey ? errState.message : null
   323→
   324→  useEffect(() => {
   325→    const ctrl = new AbortController()
   326→    const q = dateKey
   327→      ? scope === 'day'
   328→        ? `date=${dateKey}`
   329→        : scope === 'month'
   330→          ? `month=${dateKey}`
   331→          : `year=${dateKey}`
   332→      : scope === 'year'
   333→        ? `year=${defaultYear}`
   334→        : ''
   335→    fetch(`/api/history?scope=${scope}${q ? `&${q}` : ''}`, { cache: 'no-store', signal: ctrl.signal })
   336→      .then(async (res) => {
   337→        const j = (await res.json().catch(() => ({}))) as HistoryResponse & { error?: string }
   338→        if (!res.ok) throw new Error(j.error ?? `HTTP ${res.status}`)
   339→        return j
   340→      })
   341→      .then((j) => {
   342→        if (ctrl.signal.aborted) return
   343→        setData(j)
   344→        setHoverPop(null)
   345→        setPinnedPop(null)
   346→        setErrState(null)
   347→        const resolvedDate = dateKey ?? j.date
   348→        setLoadedKey(`${scope}|${resolvedDate}|${reloadTick}`)
   349→        if (!dateKey) setDateKey(j.date)
   350→      })
   351→      .catch((e: unknown) => {
   352→        if (e instanceof DOMException && e.name === 'AbortError') return
   353→        setErrState({
   354→          key: `${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`,
   355→          message: e instanceof Error ? e.message : 'Не удалось загрузить журнал',
   356→        })
   357→        setLoadedKey(`${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`)
   358→      })
   359→    return () => ctrl.abort()
   360→  }, [scope, dateKey, reloadTick, defaultYear])
   361→
   362→  /* Всплывающие таблицы */
   363→  const openPop = useCallback(
   364→    (el: HTMLElement, title: string, subtitle: string | undefined, table: React.ReactNode) => {
   365→      if (leaveTimer.current) {
   366→        clearTimeout(leaveTimer.current)
   367→        leaveTimer.current = null
   368→      }
   369→      const r = el.getBoundingClientRect()
   370→      setHoverPop({ rect: { left: r.left, right: r.right, top: r.top }, title, subtitle, table })
   371→    },
   372→    [],
   373→  )
   374→  const closePopSoon = useCallback(() => {
   375→    if (leaveTimer.current) clearTimeout(leaveTimer.current)
   376→    leaveTimer.current = setTimeout(() => setHoverPop(null), 180)
   377→  }, [])
   378→  const keepPop = useCallback(() => {
   379→    if (leaveTimer.current) {
   380→      clearTimeout(leaveTimer.current)
   381→      leaveTimer.current = null
   382→    }
   383→  }, [])
   384→  const pinPop = useCallback(
   385→    (el: HTMLElement, title: string, subtitle: string | undefined, table: React.ReactNode) => {
   386→      const r = el.getBoundingClientRect()
   387→      const st = { rect: { left: r.left, right: r.right, top: r.top }, title, subtitle, table }
   388→      setPinnedPop((prev) =>
   389→        prev && prev.title === title && prev.rect.top === st.rect.top ? null : st,
   390→      )
   391→    },
   392→    [],
   393→  )
   394→  useEffect(() => () => void (leaveTimer.current && clearTimeout(leaveTimer.current)), [])
   395→
   396→  /* Данные журнала с фильтром по улью */
   397→  const pointName = useMemo(() => {
   398→    const m = new Map<string, string>()
   399→    for (const p of data?.points ?? []) m.set(p.id, p.name)
   400→    return m
   401→  }, [data])
   402→
   403→  const filteredLogs = useMemo(() => {
   404→    if (!data?.logs) return []
   405→    return pointFilter === 'all' ? data.logs : data.logs.filter((l) => l.pointId === pointFilter)
   406→  }, [data, pointFilter])
   407→
   408→  /* День: группировка по часам */
   409→  const dayHours = useMemo(() => {
   410→    const map = new Map<number, HistoryLogDto[]>()
   411→    for (const l of filteredLogs) {
   412→      const h = new Date(l.recordedAt).getUTCHours()
   413→      const arr = map.get(h)
   414→      if (arr) arr.push(l)
   415→      else map.set(h, [l])
   416→    }
   417→    return [...map.entries()].sort(([a], [b]) => a - b)
   418→  }, [filteredLogs])
   419→
   420→  /* Сводки периода */
   421→  const daySummary = useMemo(() => {
   422→    if (filteredLogs.length === 0) return null
   423→    const avgT = filteredLogs.reduce((s, l) => s + l.tempC, 0) / filteredLogs.length
   424→    const avgH = filteredLogs.reduce((s, l) => s + l.humidityPct, 0) / filteredLogs.length
   425→    const byPoint = new Map<string, { first: number; last: number }>()
   426→    for (const l of filteredLogs) {
   427→      const acc = byPoint.get(l.pointId) ?? { first: l.weightKg, last: l.weightKg }
   428→      acc.last = l.weightKg
   429→      byPoint.set(l.pointId, acc)
   430→    }
   431→    const gain = [...byPoint.values()].reduce((s, v) => s + (v.last - v.first), 0)
   432→    const alarms = filteredLogs.filter((l) => l.alarm).length
   433→    return { avgT, avgH, gain, alarms, logs: filteredLogs.length }
   434→  }, [filteredLogs])
   435→
   436→  const monthSummary = useMemo(() => {
   437→    const days = (data?.days ?? []).filter((d) => pointFilter === 'all' || d.perPoint.some((p) => p.pointId === pointFilter))
   438→    if (days.length === 0) return null
   439→    const pts = days.flatMap((d) => (pointFilter === 'all' ? d.perPoint : d.perPoint.filter((p) => p.pointId === pointFilter)))
   440→    const avgT = pts.reduce((s, p) => s + p.avgTempC * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)
   441→    const avgH = pts.reduce((s, p) => s + p.avgHumidityPct * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)
   442→    const gain = pts.reduce((s, p) => s + p.deltaWeightKg, 0)
   443→    const alarms = pts.reduce((s, p) => s + p.alarms, 0)
   444→    const logs = pts.reduce((s, p) => s + p.logs, 0)
   445→    return { days: days.length, avgT, avgH, gain, alarms, logs }
   446→  }, [data, pointFilter])
   447→
   448→  const yearSummary = useMemo(() => {
   449→    const months = (data?.months ?? []).filter((m) => pointFilter === 'all' || m.perPoint.some((p) => p.pointId === pointFilter))
   450→    if (months.length === 0) return null
   451→    const pts = months.flatMap((m) => (pointFilter === 'all' ? m.perPoint : m.perPoint.filter((p) => p.pointId === pointFilter)))
   452→    const avgT = pts.reduce((s, p) => s + p.avgTempC * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)
   453→    const avgH = pts.reduce((s, p) => s + p.avgHumidityPct * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)
   454→    const gain = pts.reduce((s, p) => s + p.deltaWeightKg, 0)
   455→    const alarms = pts.reduce((s, p) => s + p.alarms, 0)
   456→    const logs = pts.reduce((s, p) => s + p.logs, 0)
   457→    return { months: months.length, avgT, avgH, gain, alarms, logs }
   458→  }, [data, pointFilter])
   459→
   460→  /* Таблицы для всплывающих окон */
