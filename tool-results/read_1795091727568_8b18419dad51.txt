   150→   150→function CsvCard({ isBeekeeper, hivesApi, reload }: {
   151→   151→  isBeekeeper: boolean
   152→   152→  hivesApi: Array<{ number: number; name: string }>
   153→   153→  reload: () => void
   154→   154→}) {
   155→   155→  const { toast } = useToast()
   156→   156→  const [file, setFile] = useState<File | null>(null)
   157→   157→  const [fallbackHive, setFallbackHive] = useState('')
   158→   158→  const [busy, setBusy] = useState(false)
   159→   159→  const [report, setReport] = useState<string | null>(null)
   160→   160→
   161→   161→  const upload = async () => {
   162→   162→    if (!file || busy) return
   163→   163→    setBusy(true)
   164→   164→    setReport(null)
   165→   165→    try {
   166→   166→      const fd = new FormData()
   167→   167→      fd.append('file', file)
   168→   168→      if (fallbackHive) fd.append('hiveNumber', fallbackHive)
   169→   169→      const res = await fetch('/api/csv', { method: 'POST', body: fd })
   170→   170→      const j = (await res.json()) as {
   171→   171→        ok?: boolean
   172→   172→        inserted?: number
   173→   173→        skipped?: number
   174→   174→        errors?: string[]
   175→   175→        byHive?: Record<string, number>
   176→   176→        error?: string
   177→   177→      }
   178→   178→      if (!res.ok) {
   179→   179→        toast({ title: 'Ошибка загрузки', description: j.error ?? `HTTP ${res.status}`, variant: 'destructive' })
   180→   180→        return
   181→   181→      }
   182→   182→      const byHive = j.byHive ? Object.entries(j.byHive).map(([k, v]) => `№${k} — ${v}`).join(', ') : '—'
   183→   183→      setReport(
   184→   184→        `Загружено измерений: ${j.inserted ?? 0}${j.skipped ? `, пропущено: ${j.skipped}` : ''}. По ульям: ${byHive}` +
   185→   185→        (j.errors && j.errors.length ? `\nПредупреждения:\n• ${j.errors.join('\n• ')}` : ''),
   186→   186→      )
   187→   187→      toast({ title: 'CSV обработан', description: `Добавлено измерений: ${j.inserted ?? 0}` })
   188→   188→      setFile(null)
   189→   189→      reload()
   190→   190→    } catch {
   191→   191→      toast({ title: 'Ошибка сети', description: 'Не удалось отправить файл', variant: 'destructive' })
   192→   192→    } finally {
   193→   193→      setBusy(false)
   194→   194→    }
   195→   195→  }
   196→   196→
   197→   197→  return (
   198→   198→    <Card className="border-red-100 shadow-sm">
   199→   199→      <CardContent className="space-y-3 p-5">
   200→   200→        <p className="flex items-center gap-2 font-bold text-[#2A1A12]">
   201→   201→          <FileUp className="h-4.5 w-4.5 text-red-600" /> Загрузка CSV
   202→   202→        </p>
   203→   203→        <p className="text-xs leading-relaxed text-gray-500">
   204→   204→          Экспорт из программы датчиков: колонки «дата», «улей», «вес», «температура», «влажность»
   205→   205→          (в любом порядке, разделитель «;» или «,», кодировка UTF-8 или windows-1251).
   206→   206→          Если колонки «улей» нет — выберите её ниже.
   207→   207→        </p>
   208→   208→        {isBeekeeper ? (
   209→   209→          <>
   210→   210→            <div className="flex flex-wrap items-center gap-2">
   211→   211→              <Input
   212→   212→                type="file"
   213→   213→                accept=".csv,text/csv"
   214→   214→                onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   215→   215→                className="border-red-200 file:mr-3 file:rounded-lg file:border-0 file:bg-red-100 file:px-3 file:py-1.5 file:text-xs file:font-bold file:text-red-700"
   216→   216→              />
   217→   217→              <a href="/api/csv" download className="shrink-0">
   218→   218→                <Button variant="outline" className="border-red-200 text-red-700 hover:bg-red-50">
   219→   219→                  <Download className="h-4 w-4" /> Шаблон
   220→   220→                </Button>
   221→   221→              </a>
   222→   222→            </div>
   223→   223→            <div className="grid gap-2 sm:grid-cols-[1fr_auto]">
   224→   224→              <HiveSelect hivesApi={hivesApi} value={fallbackHive} onChange={setFallbackHive} />
   225→   225→              <Button onClick={() => void upload()} disabled={!file || busy} className="bg-orange-500 hover:bg-orange-600">
   226→   226→                <FileUp className="h-4 w-4" /> {busy ? 'Загружаем…' : 'Загрузить'}
   227→   227→              </Button>
   228→   228→            </div>
   229→   229→            {report && (
   230→   230→              <pre className="max-h-40 overflow-y-auto whitespace-pre-wrap rounded-xl bg-red-50 p-3 text-xs leading-relaxed text-gray-700">
   231→   231→                {report}
   232→   232→              </pre>
   233→   233→            )}
   234→   234→          </>
   235→   235→        ) : (
   236→   236→          <p className="text-xs text-gray-400">Доступно только пасечнику.</p>
   237→   237→        )}
   238→   238→      </CardContent>
   239→   239→    </Card>
   240→   240→  )
   241→   241→}

   245→   245→function ManualCard({ isBeekeeper, hivesApi, reload }: {
   246→   246→  isBeekeeper: boolean
   247→   247→  hivesApi: Array<{ number: number; name: string }>
   248→   248→  reload: () => void
   249→   249→}) {
   250→   250→  const { toast } = useToast()
   251→   251→  const [hive, setHive] = useState('')
   252→   252→  const [dt, setDt] = useState(() => format(new Date(), "yyyy-MM-dd'T'HH:mm"))
   253→   253→  const [weight, setWeight] = useState('')
   254→   254→  const [temp, setTemp] = useState('')
   255→   255→  const [hum, setHum] = useState('')
   256→   256→  const [busy, setBusy] = useState(false)
   257→   257→
   258→   258→  const submit = async () => {
   259→   259→    if (busy) return
   260→   260→    setBusy(true)
   261→   261→    try {
   262→   262→      const res = await fetch('/api/measurements', {
   263→   263→        method: 'POST',
   264→   264→        headers: { 'Content-Type': 'application/json' },
   265→   265→        body: JSON.stringify({ hiveNumber: Number(hive), ts: dt, weightKg: weight, tempC: temp, humidityPct: hum }),
   266→   266→      })
   267→   267→      const j = (await res.json()) as { ok?: boolean; error?: string }
   268→   268→      if (!res.ok) {
   269→   269→        toast({ title: 'Не сохранено', description: j.error ?? `HTTP ${res.status}`, variant: 'destructive' })
   270→   270→        return
   271→   271→      }
   272→   272→      toast({ title: 'Показание записано', description: `Улей №${hive}, ${dt.replace('T', ' ')}` })
   273→   273→      setWeight('')
   274→   274→      setTemp('')
   275→   275→      setHum('')
   276→   276→      reload()
   277→   277→    } catch {
   278→   278→      toast({ title: 'Ошибка сети', variant: 'destructive' })
   279→   279→    } finally {
   280→   280→      setBusy(false)
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  return (
   285→   285→    <Card className="border-red-100 shadow-sm">
   286→   286→      <CardContent className="space-y-3 p-5">
   287→   287→        <p className="flex items-center gap-2 font-bold text-[#2A1A12]">
   288→   288→          <Waypoints className="h-4.5 w-4.5 text-red-600" /> Внести показание вручную
   289→   289→        </p>
   290→   290→        {isBeekeeper ? (
   291→   291→          <>
   292→   292→            <div className="grid gap-3 sm:grid-cols-2">
   293→   293→              <div className="space-y-1.5">
   294→   294→                <Label className="text-xs text-gray-500">Улей</Label>
   295→   295→                <HiveSelect hivesApi={hivesApi} value={hive} onChange={setHive} />

   300→   300→              </div>

   363→   363→
   364→   364→  return (
