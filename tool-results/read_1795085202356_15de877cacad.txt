   239→            <div className="mt-1.5 h-2 overflow-hidden rounded-full bg-red-50 dark:bg-[#2C211C]">
   240→              <motion.div
   241→                className={cn(
   242→                  'h-full rounded-full bg-gradient-to-r',
   243→                  c.soon ? 'from-amber-400 to-orange-500' : 'from-emerald-400 to-emerald-500',
   244→                )}
   245→                initial={{ width: 0 }}
   246→                animate={{ width: `${c.pct}%` }}
   247→                transition={{ duration: 0.9, ease: 'easeOut', delay: 0.15 }}
   248→              />
   249→            </div>
   250→            <p className="mt-1 text-[10px] text-gray-400">
   251→              {c.note} · {c.pct} % порога
   252→            </p>
   253→          </li>
   254→        ))}
   255→      </ul>
   256→      {soon.length > 0 && (
   257→        <p className="mt-3 rounded-xl border border-amber-200 bg-amber-50 px-3 py-2 text-[11px] font-semibold leading-relaxed text-amber-800">
   258→          Автопредупреждение: до цветения «{soon[0].name}» около {soon[0].daysLeft} дн. — поставьте магазины и
   259→          проверьте вентиляцию заранее.
   260→        </p>
   261→      )}
   262→    </section>
   263→  )
   264→}

   291→      .filter((f) => (done[id] ?? []).includes(f.id))
