    74→/** Максимум температуры за последние 3 дня (пиковая аномалия). */
    75→function maxTempRecent(h: Hive): number {
    76→  const temps = h.series.slice(-3).map((p) => p.tempInC ?? 0)
    77→  return Math.max(...temps)
    78→}

   217→/** Список находок: по выбранному улью + глобальные сценарии, отсортированы по срочности. */
   218→export function findingsFor(hiveId: number): AgentFinding[] {
   219→  const hiveFinding = swarmFinding(getHive(hiveId))
   220→  const list = [pesticideFinding, hiveFinding, weatherFinding, migrationFinding]
   221→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   222→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   223→}

   233→function extractHiveId(q: string): number | null {
   234→  const m = q.match(/(?:№|улье|улья|улью|hive)\s*№?\s*(\d)/)
   235→  if (!m) return null
   236→  const id = Number(m[1])
   237→  return id >= 1 && id <= 4 ? id : null
   238→}

   543→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
   544→export function askAgent(question: string, currentHiveId: number): AgentMessage {
   545→  const q = norm(question)
   546→  const hiveId = extractHiveId(q)
   547→  const h = getHive(hiveId ?? currentHiveId)
   548→
   549→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
   550→
   551→  // Приветствие и сервисные фразы
   552→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
   553→    return greetingAnswer()
   554→  }
   555→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
   556→    return thanksAnswer()
   557→  }
   558→
   559→  // Сценарий пестицидов
   560→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
   561→    return pesticideAnswer()
   562→  }
   563→
   564→  // Кочёвка
   565→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
   566→    return migrationAnswer()
   567→  }
   568→
   569→  // Погода и медосбор
   570→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|когда вывоз|когда летать/)) {
   571→    return weatherAnswer()
   572→  }
   573→
   574→  // Вопрос про конкретный или выбранный улей
   575→  if (has(/влажн|сырост|сыро/)) {
   576→    return humidityAnswer(h)
   577→  }
   578→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
   579→    return swarmAnswer(h)
   580→  }
   581→  if (has(/температ|жар|тепло внутри/)) {
   582→    return climateAnswer(h)
   583→  }
   584→  if (has(/вес|мед\b|меда|привес|набор|тяже/)) {
   585→    return weightAnswer(h)
   586→  }
   587→  if (has(/что происходит|состояни|как (себя|дела)|статус|проверь|проблем|тревог|что с улеем|как улей|расскажи про улей/)) {
   588→    return statusAnswer(h)
   589→  }
   590→
   591→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
   592→  if (hiveId !== null) {
   593→    return statusAnswer(h)
   594→  }
   595→
   596→  return fallbackAnswer()
   597→}
