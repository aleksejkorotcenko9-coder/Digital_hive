   102→/* ------------------------------------------------------------------ */
   103→/* Метрики по рядам (безопасны к пропускам значений)                   */
   104→/* ------------------------------------------------------------------ */
   105→
   106→function lastPoint(h: Hive) {
   107→  return h.series[h.series.length - 1] ?? null
   108→}
   109→
   110→function lastWeight(h: Hive): number | null {
   111→  return lastPoint(h)?.weightKg ?? null
   112→}
   113→
   114→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   115→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   116→function weightDelta(h: Hive, backDays: number): number | null {
   117→  const s = h.series
   118→  if (s.length === 0) return null
   119→  const lastPoint = s[s.length - 1]
   120→  if (lastPoint.weightKg === null) return null
   121→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   122→  let best: number | null = null
   123→  let bestDiff = Infinity
   124→  for (const p of s) {
   125→    if (p.weightKg === null || p === lastPoint) continue
   126→    const diff = Math.abs(new Date(p.date).getTime() - target)
   127→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   128→      bestDiff = diff
   129→      best = p.weightKg
   130→    }
   131→  }
   132→  if (best === null) return null
   133→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   134→}
   135→
   136→const delta2d = (h: Hive) => weightDelta(h, 2)
   137→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   138→
   139→function maxTempRecent(h: Hive, n = 3): number | null {
   140→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   141→  return temps.length ? Math.max(...temps) : null
   142→}
   143→
   144→function lastHumidity(h: Hive): number | null {
   145→  return lastPoint(h)?.humidityPct ?? null
   146→}
   147→
   148→/** Лидер пасеки по 30-дневному привесу. */
   149→function leaderOf(list: Hive[]): Hive | null {
   150→  const withW = list.filter((x) => delta30d(x) !== null)
   151→  if (!withW.length) return null
   152→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   153→}
   154→
   155→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   156→function spanDays(h: Hive): number {
   157→  if (h.series.length < 2) return h.series.length
   158→  const a = new Date(h.series[0].date).getTime()
   159→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   160→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   161→  return Math.round((b - a) / 86400000) + 1
   162→}
   163→
   164→function periodLabel(h: Hive): string {
   165→  const d = spanDays(h)
   166→  if (d >= 25) return 'за месяц'
   167→  if (d === 1) return 'за день'
   168→  if (d === 0) return '—'
   169→  return `за ${d} дн.`
   170→}

   172→/** Порог срабатывания «падения веса» (признак возможного роения). */
   173→const SWARM_DROP = -1.0
   174→
   175→function swarmProbability(drop: number): string {
   176→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   177→  return `Вероятность подготовки к роению — около ${p} %`
   178→}

   184→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   185→  const status = hiveStatus(h, ctx.hives)
   186→  const d2 = delta2d(h)
   187→  const d30 = delta30d(h)
   188→  const t = lastPoint(h)?.tempInC ?? null
   189→  const hum = lastHumidity(h)
   190→
   191→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   192→    return {
   193→      id: `nodata-${h.id}`,
   194→      scenario: 'status',
   195→      hiveId: h.id,
   196→      title: `Улей №${h.id}: пока нет данных`,
   197→      text: `По улью №${h.id} в базе нет измерений. Я не буду гадать: как только появятся показания — загрузите их в разделе «Данные» (CSV, вручную или через API), и я сразу начну следить за весом, температурой и влажностью.`,
   198→      basis: ['База данных платформы', 'Измерения улья №' + h.id],
   199→      urgency: 'this-week',
   200→      urgencyNote: 'Загрузить данные',
   201→      probability: 'Оценка невозможна — данных нет',
   202→      metrics: [{ label: 'Измерений', value: String(h.series.length) }],
   203→      prefill: `Что происходит с ульем №${h.id}?`,
   204→    }
   205→  }
   206→
   207→  if (status.level === 'risk' && d2 !== null) {
   208→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   209→    const now = lastWeight(h) ?? peak
   210→    return {
   211→      id: `swarm-${h.id}`,
   212→      scenario: 'swarm',
   213→      hiveId: h.id,
   214→      title: 'Похоже, семья готовится к роению',
   215→      text: `За последние 2 дня вес улья №${h.id} снизился на ${fmtNum(Math.abs(d2))} кг — с ${fmtNum(peak)} до ${fmtNum(now)} кг, хотя до этого семья стабильно набирала мед.${
   216→        t !== null ? ` Температура внутри поднималась до ${fmtNum(t)} °C (норма около 34,5 °C).` : ''
   217→      }${hum !== null ? ` Влажность держится на отметке ${fmtNum(hum, 0)} % — выше обычной для этого улья.` : ''} Июль — пик сезона роения: часть пчёл с маткой может улететь, и медосбор остановится на 2–3 недели.`,
   218→      basis: [
   219→        `График веса улья №${h.id}`,
   220→        `Вес: ${fmtSigned(d2)} кг за 2 дня`,
   221→        ...(t !== null ? [`Температура: ${fmtNum(t)} °C при норме 34,5 °C`] : []),
   222→        'Сезон: июль — период роения',
   223→      ],
   224→      urgency: 'today',
   225→      urgencyNote: 'Осмотреть улей сегодня до вечера',
   226→      probability: swarmProbability(d2),
   227→      metrics: [
   228→        { label: 'Вес за 2 дня', value: `${fmtSigned(d2)} кг` },
   229→        ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   230→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   231→      ],
   232→      prefill: `Что происходит с ульем №${h.id}?`,
   233→    }
   234→  }
   235→
   236→  if (status.level === 'watch' && h.label !== undefined) {
   237→    const leader = leaderOf(ctx.hives)
   238→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   239→      ? 'отдельно: датчик влажности не отвечает'
   240→      : 'привес хуже, чем у лидера пасеки'
   241→    return {
   242→      id: `slow-${h.id}`,
   243→      scenario: 'status',
   244→      hiveId: h.id,
   245→      title: 'Слабый набор веса',
   246→      text: `Улей №${h.id} прибавил только ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}${
   247→        leader && leader.id !== h.id ? ` — у лидера (улей №${leader.id}) ${fmtSigned(delta30d(leader) ?? 0)} кг` : ''
   248→      }. Температура${t !== null ? ` ${fmtNum(t)} °C` : ''} в норме. Причин может быть две: слабая семья или бедная кормовая база вокруг. ${reason.charAt(0).toUpperCase() + reason.slice(1)}${
   249→        h.humidityOfflineFrom ? ` — молчит с ${h.humidityOfflineFrom}, возможно, сел аккумулятор` : ' — стоит сверить привес с соседними ульями'
   250→      }.`,
   251→      basis: [
   252→        'Сравнение привеса по ульям пасеки',
   253→        `Привес улья №${h.id}: ${fmtSigned(d30 ?? 0)} кг`,
   254→        ...(h.humidityOfflineFrom ? [`Датчик влажности: нет данных с ${h.humidityOfflineFrom}`] : []),
   255→      ],
   256→      urgency: 'this-week',
   257→      urgencyNote: 'Проверить датчик и семью в ближайшие дни',
   258→      probability: 'Диагноз по косвенным признакам — нужна проверка',
   259→      metrics: [
   260→        { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   261→        ...(leader && leader.id !== h.id
   262→          ? [{ label: `У лидера (№${leader.id})`, value: `${fmtSigned(delta30d(leader) ?? 0)} кг` }]
   263→          : []),
   264→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : [{ label: 'Влажность', value: 'нет данных' }]),
   265→      ],
   266→      prefill: `Почему в улье №${h.id} слабый привес?`,
   267→    }
   268→  }
   269→
   270→  return {
   271→    id: `ok-${h.id}`,
   272→    scenario: 'status',
   273→    hiveId: h.id,
   274→    title: 'Семья работает штатно',
   275→    text: `Улей №${h.id} развивается штатно: вес ${fmtNum(lastWeight(h) ?? 0)} кг, привес ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}, за последние 2 дня ${fmtSigned(d2 ?? 0)} кг.${
   276→      t !== null ? ` Температура ${fmtNum(t)} °C — в норме.` : ''
   277→    } Признаков роения нет: вес растёт ровно, без провалов.`,
   278→    basis: [
   279→      `График веса улья №${h.id}`,
   280→      `Привес: ${fmtSigned(d30 ?? 0)} кг`,
   281→      'Температура в норме семьи',
   282→    ],
   283→    urgency: 'ok',
   284→    urgencyNote: 'Продолжать наблюдение',
   285→    metrics: [
   286→      { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   287→      ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   288→      ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   289→    ],
   290→    prefill: `Что происходит с улье №${h.id}?`,
   291→  }
   292→}

   321→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   322→  const w = ctx.weather
   323→  if (!w || w.length === 0) return null
   324→  const good = w.filter((d) => d.flight === 'good')
   325→  const bad = w.filter((d) => d.flight === 'bad')
   326→  const tMax = Math.max(...w.map((d) => d.tMax))
   327→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   328→  return {
   329→    id: 'weather',
   330→    scenario: 'weather',
   331→    hiveId: null,
   332→    title: good.length
   333→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   334→      : 'Ближайшие дни — не для лёта',
   335→    text: good.length
   336→      ? `Ближайшие дни с хорошими условиями: ${good.map((d) => d.full).join(', ')} — до +${tMax} °C, ветер до ${Math.max(...good.map((d) => d.windMs))} м/с, осадки маловероятны.${
   337→          bad.length ? ` Затем, ${bad.map((d) => d.full).join(' и ')}, — дожди (до ${rainProb} %) и ветер до ${Math.max(...bad.map((d) => d.windMs))} м/с: пчёлы останутся в ульях, привес остановится.` : ''
   338→        }`
   339→      : `По прогнозу на неделю погода не летная: дожди и ветер. Пчёлы останутся в ульях, привес остановится — следите за запасами корма по графику веса.`,
   340→    basis: ['Прогноз погоды платформы', 'Календарь цветения медоносов — ГИС'],
   341→    urgency: 'ok',
   342→    urgencyNote: good.length ? 'Использовать окно лёта' : 'Наблюдать за погодой',
   343→    metrics: [
   344→      { label: 'Максимум', value: `+${tMax} °C` },
   345→      ...(bad.length ? [{ label: 'Дождь', value: `до ${rainProb} %` }] : []),
   346→      ...(ctx.sites.length
   347→        ? [{ label: 'Кандидатов кочёвки', value: String(ctx.sites.length) }]
   348→        : []),
   349→    ],
   350→    prefill: 'Когда пчёлам лучше летать за взятком?',
   351→  }
   352→}

   354→function monthGen(date: string): string {
   355→  // «2026-07-20» → «июля» (родительный падеж)
   356→  const months = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря']
   357→  const m = Number(date.slice(5, 7))
   358→  return months[m - 1] ?? ''
   359→}

   361→function migrationFindingOf(ctx: AgentContext): AgentFinding | null {
   362→  const sites = [...ctx.sites].sort((a, b) => b.score - a.score)
   363→  if (!sites.length) return null
   364→  const best = sites[0]
   365→  const worst = sites[sites.length - 1]
   366→  return {
   367→    id: 'migration',
   368→    scenario: 'migration',
   369→    hiveId: null,
   370→    title: `Лучшее место для кочёвки — «${best.name}»`,
   371→    text: `Сравнил ${sites.length} ${
   372→      sites.length === 1 ? 'участок' : sites.length < 5 ? 'участка' : 'участков'
   373→    } вокруг пасеки по 4 факторам: медоносы, погода, расстояние и риски. Лучший — «${best.name}»: ${best.honeyPlant.toLowerCase()} цветёт ${best.bloomWindow}, ${fmtNum(best.distanceKm)} км от пасеки, погода: ${best.weatherNote.toLowerCase()}.${
   374→      best.risks ? ` Учтите: ${best.risks.toLowerCase()}.` : ''
   375→    }${sites.length > 1 && worst.verdict === 'avoid' ? ` «${worst.name.replace(/^Донниковое поле \(/, '').replace(/\)$/, '')}» не советую: ${worst.risks.toLowerCase() || 'низкая оценка участка'}.` : ''}`,
   376→    basis: ['ГИС: медоносы и расстояния', 'Прогноз погоды по участкам', 'Реестр обработок полей'],
   377→    urgency: 'this-week',
   378→    urgencyNote: 'Решить до начала цветения',
   379→    metrics: [
   380→      { label: 'Лучший участок', value: `${fmtNum(best.score)} / 10` },
   381→      { label: 'Расстояние', value: `${fmtNum(best.distanceKm)} км` },
   382→      { label: 'Цветение', value: best.bloomWindow },
   383→    ],
   384→    prefill: 'Куда перевезти пасеку?',
   385→  }

   388→/** Список находок: по выбранному улью + глобальные сценарии, отсортированы по срочности. */
   389→export function findingsFor(hiveId: number, ctx: AgentContext = DEMO_CONTEXT): AgentFinding[] {
   390→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   391→  const list: AgentFinding[] = []
   392→  if (hive) list.push(hiveFinding(hive, ctx))
   393→  const pesticide = pesticideFindingOf(ctx)
   394→  if (pesticide) list.push(pesticide)
   395→  const weatherF = weatherFindingOf(ctx)
   396→  if (weatherF) list.push(weatherF)
   397→  const migrationF = migrationFindingOf(ctx)
   398→  if (migrationF) list.push(migrationF)
   399→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   400→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   401→}

   895→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
   896→export function askAgent(
   897→  question: string,
   898→  currentHiveId: number,
   899→  ctx: AgentContext = DEMO_CONTEXT,
   900→): AgentMessage {
   901→  const q = norm(question)
   902→  const hiveId = extractHiveId(q, ctx)
   903→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
   904→
   905→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
   906→
   907→  // Приветствие и сервисные фразы
   908→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
   909→    return greetingAnswer(ctx)
   910→  }
   911→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
   912→    return thanksAnswer()
   913→  }
   914→
   915→  // Сценарий пестицидов
   916→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
   917→    return pesticideAnswer(ctx)
   918→  }
   919→
   920→  // Кочёвка
   921→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
   922→    return migrationAnswer(ctx)
   923→  }
   924→
   925→  // Погода и медосбор
   926→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|когда вывоз|когда летать/)) {
   927→    return weatherAnswer(ctx)
   928→  }
   929→
   930→  if (!hive) {
   931→    return fallbackAnswer()
   932→  }
   933→
   934→  // Вопрос про конкретный или выбранный улей
   935→  if (has(/влажн|сырост|сыро/)) {
   936→    return humidityAnswer(hive)
   937→  }
   938→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
   939→    return swarmAnswer(hive, ctx)
   940→  }
   941→  if (has(/температ|жар|тепло внутри/)) {
   942→    return climateAnswer(hive)
   943→  }
   944→  if (has(/вес|мед\b|меда|привес|набор|тяже/)) {
   945→    return weightAnswer(hive, ctx)
   946→  }
   947→  if (has(/что происходит|состояни|как (себя|дела)|статус|проверь|проблем|тревог|что с улеем|как улей|расскажи про улей/)) {
   948→    return statusAnswer(hive, ctx)
   949→  }
   950→
   951→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
   952→  if (hiveId !== null) {
   953→    return statusAnswer(hive, ctx)
   954→  }
   955→
   956→  return fallbackAnswer()
   957→}
