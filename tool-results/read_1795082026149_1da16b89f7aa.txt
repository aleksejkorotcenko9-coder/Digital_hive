    96→export function DashboardView() {
    97→  const setView = useAppState((s) => s.setView)
    98→  const hiveId = useAppState((s) => s.hiveId)
    99→  const setHive = useAppState((s) => s.setHive)
   100→  const { status, data, reload } = useRealData()
   101→
   102→  // Синхронизация с картой: ульи дашборда = точки на карте, показания — с
   103→  // датчиков. Опрос каждые 5 секунд — числа «живут» вместе с картой.
   104→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
   105→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
   106→  useEffect(() => {
   107→    let alive = true
   108→    const load = async () => {
   109→      try {
   110→        const res = await fetch('/api/map-points', { cache: 'no-store' })
   111→        if (!res.ok) return
   112→        const j = (await res.json()) as { points: MapPointDto[] }
   113→        if (alive) setPoints(j.points)
   114→      } catch {
   115→        // датчики подтянутся на следующем опросе
   116→      }
   117→    }
   118→    void load()
   119→    const timer = setInterval(() => void load(), 5000)
   120→    return () => {
   121→      alive = false
   122→      clearInterval(timer)
   123→    }
   124→  }, [])
   125→
   126→  // Погода по фактическому центру пасеки (среднее координат точек).
   127→  // Эффект зависит от ЧИСЕЛ координат: пересчёт центра при каждом опросе
   128→  // точек не вызывает перезапрос погоды, пока координаты не изменились.
   129→  const center = useMemo<[number, number]>(
   130→    () =>
   131→      (points && points.length > 0 ? pointsCenter(points) : null) ?? DEFAULT_PLACE.center,
   132→    [points],
   133→  )
   134→  const [wLat, wLon] = center
   135→  useEffect(() => {
   136→    let alive = true
   137→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
   138→    const load = async () => {
   139→      try {
   140→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   141→        if (!res.ok) return
   142→        const j = (await res.json()) as WeatherResponse
   143→        if (alive) setWeather(j)
   144→      } catch {
   145→        // агент честно скажет, если погоды нет
   146→      }
   147→    }
   148→    void load()
   149→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   150→    return () => {
   151→      alive = false
   152→      clearInterval(timer)
   153→    }
   154→  }, [wLat, wLon])
   155→  const agentWeather = useMemo(() => (weather ? weatherResponseToDays(weather) : null), [weather])
   156→
   157→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   158→
   159→  // выбранный улей всегда существует в синхронизированном наборе
   160→  useEffect(() => {
   161→    if (syncedHives && syncedHives.length > 0 && !syncedHives.some((h) => h.id === hiveId)) {
   162→      setHive(syncedHives[0].id)
   163→    }
   164→  }, [syncedHives, hiveId, setHive])
   165→
   166→  if (status === 'idle' || status === 'loading') {
   167→    return <DashboardSkeleton />
   168→  }
   169→  if (status === 'error') {
   170→    return (
   171→      <Card className="mx-auto max-w-lg border-red-100">
   172→        <CardContent className="p-6 text-center">
   173→          <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   174→          <h2 className="mt-2 font-bold text-gray-900">Не удалось загрузить данные</h2>
   175→          <p className="mt-1 text-sm text-gray-500">
   176→            Проверьте, что база данных доступна, и попробуйте ещё раз.
   177→          </p>
   178→          <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   179→            Повторить загрузку
   180→          </Button>
   181→        </CardContent>
   182→      </Card>
   183→    )
   184→  }
   185→  // Схема и диаграммы — ТО ЖЕ точки, что на карте пасеки: один набор ульев
   186→  // для карты, вкладки «Датчики» и дашборда. Из базы — журнал и площадки кочёвки.
   187→  if (!points || !data || syncedHives === null) return <DashboardSkeleton />
   188→  if (syncedHives.length === 0) {
   189→    return <EmptyMapState onOpenMap={() => setView('map')} />
   190→  }
   191→  return (
   192→    <DashboardContent
   193→      hives={syncedHives}
   194→      sites={data.sites.map(siteDtoToSite)}
   195→      weather={agentWeather?.days ?? null}
   196→      events={data.events}
   197→      center={center}
   198→    />
   199→  )
   200→}

   206→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/50" />
   207→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/40" />

   249→function DashboardContent({ hives, sites, weather, events, center }: ContentProps) {
   250→  const hiveId = useAppState((s) => s.hiveId)
   251→  const setHive = useAppState((s) => s.setHive)
   252→  const setView = useAppState((s) => s.setView)
   253→  const ask = useAppState((s) => s.ask)
   254→  const reminders = useAppState((s) => s.reminders)
   255→  const [openFinding, setOpenFinding] = useState<string | null>(null)
   256→
   257→  const hive = hives.find((x) => x.id === hiveId) ?? hives[0]
   258→  const status = hiveStatus(hive, hives)
   259→  const ctx = makeContext(hives, { weather, sites, apiaryName: 'Моя пасека' })
   260→  const findings = findingsFor(hive.id, ctx)
   261→
   262→  const riskyHive = hives.find((x) => hiveStatus(x, hives).level === 'risk')
   263→  const watchHive = hives.find((x) => hiveStatus(x, hives).level === 'watch' && x.id !== riskyHive?.id)
   264→
   265→  const lastPoint = hive.series[hive.series.length - 1]
   266→  const d2 = weightDeltaSafe(hive, 2)
   267→  const lastWeight = lastPoint?.weightKg ?? null
   268→  const lastTemp = lastPoint?.tempInC ?? null
   269→  const lastHum = lastPoint?.humidityPct ?? null
   270→  const period = periodCaption(hive)
   271→
   272→  // показатели пасеки целиком
   273→  const withTelemetry = hives.filter((h) => h.series.length > 0 && h.series[h.series.length - 1].weightKg !== null)
   274→  const gauge = apiaryGauge(hives.map((h) => hiveStatus(h, hives).level), hives.length)
   275→  const leader = bestHiveOf(hives)
   276→
   277→  const upcomingReminders = reminders
   278→    .filter((r) => !r.done)
   279→    .sort((a, b) => a.remindAt.localeCompare(b.remindAt))
   280→    .slice(0, 4)
   281→
   282→  const todayLabel = format(new Date(), 'EEEE, d MMMM yyyy', { locale: ru })
   283→
   284→  return (
   285→    <div className="space-y-5">
   286→      {/* Верхний баннер срочности */}
   287→      {riskyHive ? (
   288→        <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-[#D93E00] to-[#FF5A1F] p-4 text-white shadow-lg shadow-red-100 md:p-5">
   289→          <Split className="absolute -right-3 -top-3 h-28 w-28 rotate-12 text-white/15" />
   290→          <div className="relative flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   291→            <div className="flex items-start gap-3">
   292→              <AlertTriangle className="mt-0.5 h-6 w-6 shrink-0" />
   293→              <div>
   294→                <p className="text-sm font-bold uppercase tracking-wide text-white/90">
   295→                  Сигнал данных · возможное роение
   296→                </p>
   297→                <p className="mt-1 text-sm leading-snug md:text-base">
   298→                  Улей №{riskyHive.id}: вес снизился на {fmtSigned(weightDeltaSafe(riskyHive, 2) ?? 0)} кг за 2 дня при
   299→                  тёплой погоде — похоже на подготовку семьи к делению.
   300→                </p>
   301→              </div>
   302→            </div>
   303→            <Button
   304→              className="w-fit shrink-0 bg-white font-semibold text-orange-700 hover:bg-orange-50"
   305→              onClick={() => ask(`Что происходит с ульем №${riskyHive.id}?`)}
   306→            >
   307→              Что делать? <ArrowRight className="h-4 w-4" />
   308→            </Button>
   309→          </div>
   310→        </div>
   311→      ) : watchHive ? (
   312→        <div className="flex items-center gap-3 rounded-2xl border border-amber-200 bg-amber-50 px-4 py-3 text-amber-800">
   313→          <AlertTriangle className="h-5 w-5 shrink-0 text-amber-500" />
   314→          <p className="text-sm">
   315→            Улей №{watchHive.id}: {hiveStatus(watchHive, hives).label.toLowerCase()} — проверьте семью в ближайшие дни.
   316→          </p>
   317→        </div>
   318→      ) : (
   319→        <div className="flex items-center gap-3 rounded-2xl border border-emerald-200 bg-emerald-50 px-4 py-3 text-emerald-800">
   320→          <Sun className="h-5 w-5 shrink-0 text-emerald-500" />
   321→          <p className="text-sm">
   322→            По данным датчиков тревожных сигналов нет — все семьи с телеметрией работают штатно.
   323→          </p>
   324→        </div>
   325→      )}
   326→
   327→      {/* Заголовок пасеки */}
   328→      <div className="flex flex-wrap items-end justify-between gap-3">
   329→        <div>
   330→          <h1 className="text-2xl font-extrabold tracking-tight text-[#2A1A12] md:text-3xl">
   331→            Моя пасека · данные с датчиков
   332→          </h1>
   333→          <p className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-1 text-sm text-gray-500">
   334→            <span className="flex items-center gap-1">
   335→              <MapPin className="h-3.5 w-3.5 text-red-500" />
   336→              {`${hives.length} ${hiveWord(hives.length)} на карте · данные с датчиков`}
   337→            </span>
   338→            <span className="flex items-center gap-1">
   339→              <CalendarDays className="h-3.5 w-3.5 text-red-500" /> {todayLabel}
   340→            </span>
   341→          </p>
   342→        </div>
   343→      </div>
   344→
   345→      {/* Три колонки, как на референсе */}
   346→      <div className="grid grid-cols-1 gap-4 xl:grid-cols-[320px_minmax(0,1fr)_340px]">
   347→        {/* Колонка 1: погода пасеки (по координатам её точек на карте) */}
   348→        <div className="order-2 xl:order-1">
   349→          <WeatherPanel coords={center} place="Моя пасека" />
   350→        </div>
   351→
   352→        {/* Колонка 2: состояние пасеки */}

   355→            <div className="flex flex-wrap items-center justify-between gap-3">
   356→              <h2 className="flex items-center gap-1.5 text-sm font-bold text-[#2A1A12]">
   357→                <TrendingUp className="h-4 w-4 text-red-500" /> Состояние пасеки
   358→              </h2>
   359→              <span className="text-[11px] text-gray-400">
   360→                {withTelemetry.length} из {hives.length} семей с телеметрией
   361→              </span>
   362→            </div>
   363→
   364→            <StatusGauge info={gauge} className="mx-auto mt-2" />

   370→                label="С датчиками"

   372→              />

   377→                hint={`улей №${hive.id}`}
   378→              />
   379→              <Tile
   380→                icon={<Droplets className="h-3.5 w-3.5" />}
   381→                label="Влажность"
   382→                value={lastHum !== null ? `${fmtNum(lastHum, 0)} %` : '—'}

   634→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   635→function weightTotal(h: Hive): number {
   636→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   637→  const first = h.series[0]?.weightKg ?? null
   638→  if (w === null || first === null) return 0
   639→  return Math.round((w - first) * 10) / 10
   640→}
