   189→   189→              Добавьте улья кнопкой «+ Новый улей» и загрузите телеметрию — файлом CSV, через форму

   213→   213→
   214→   214→  return (
   215→   215→    <DashboardContent
   216→   216→      mode="demo"
   217→   217→      hives={syncedHives}

   687→   687→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   688→   688→function weightTotal(h: Hive): number {
   689→   689→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   690→   690→  const first = h.series[0]?.weightKg ?? null
   691→   691→  if (w === null || first === null) return 0
   692→   692→  return Math.round((w - first) * 10) / 10
   693→   693→}
