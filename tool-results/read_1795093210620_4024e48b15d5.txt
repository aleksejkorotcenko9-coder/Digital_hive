     4→ * Экран «ИИ-агент»: чат пчеловода с цифровым помощником.

     9→ * Голосовой ввод работает как рация (push-to-talk): кнопку микрофона нужно

    11→ * завершается, речь распознаётся (Web Speech или сервер /api/asr) и текст
    12→ * подставляется в поле ввода.
    13→ * Улии и показания синхронизированы с картой: агент говорит по тем же точкам
    14→ * и датчикам, что видит пчеловод на карте и дашборде.
    15→ * История переписки сохраняется в SQLite (общая лента пасеки).

    71→        const j = (await res.json()) as { points: MapPointDto[] }

    73→      } catch {
    74→        // датчики подтянутся на следующем опросе

    76→    }
    77→    void load()
    78→    const timer = setInterval(() => void load(), 5000)
    79→    return () => {
    80→      alive = false
    81→      clearInterval(timer)
    82→    }
    83→  }, [])
    84→
    85→  // Погода по фактическому центру пасеки (среднее координат точек), те же
    86→  // координаты, что на дашборде, чтобы агент называл те же числа, что видит
    87→  // пчеловод. Эффект зависит от ЧИСЕЛ координат: опрос точек каждые 5 с
    88→  // не вызывает перезапрос погоды, пока координаты не изменились.
    89→  const center = useMemo<[number, number]>(
    90→    () =>
    91→      (apiaryPoints && apiaryPoints.length > 0 ? pointsCenter(apiaryPoints) : null) ??
    92→      DEFAULT_PLACE.center,
    93→    [apiaryPoints],
    94→  )
    95→  const [wLat, wLon] = center
    96→  useEffect(() => {
    97→    let cancelled = false
    98→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
    99→    const load = async () => {
   100→      try {
   101→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   102→        if (!res.ok) throw new Error('weather')
   103→        const j = (await res.json()) as WeatherResponse
   104→        if (!cancelled) setWResp(j)
   105→      } catch {
   106→        // погода не критична: агент честно скажет, если данных нет
   107→      }
   108→    }
   109→    void load()
   110→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   111→    return () => {
   112→      cancelled = true
   113→      clearInterval(timer)
   114→    }
   115→  }, [wLat, wLon])
   116→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   117→
   118→  // Ульи и показания — те же точки, что на карте (синхронизация с /api/map-points)
   119→  const pointsCtx = useMemo<AgentContext | null>(() => {
   120→    if (!apiaryPoints) return null
   121→    const points: AgentPoint[] = apiaryPoints.map((p) => ({
   122→      id: p.id,
   123→      name: p.name,
   124→      lat: p.lat,
   125→      lon: p.lon,
   126→      hiveCount: p.hiveCount,
   127→      hasSensor: p.hasSensor,
   128→      sensorOn: p.sensorOn,
   129→      sensor: p.sensor,
   130→    }))
   131→    return makeContext(pointsToHives(apiaryPoints), {
   132→      weather: agentWeather?.days ?? null,
   133→      weatherNow: agentWeather?.now ?? null,
   134→      pesticide: null, // реестр обработок не подключён — агент честно об этом скажет
   135→      sites: data ? data.sites.map(siteDtoToSite) : [],
   136→      points,
   137→      apiaryName: 'Моя пасека',
   138→    })
   139→  }, [apiaryPoints, agentWeather, data])
   140→
   141→  // контекст мемоизирован, чтобы не пересоздаваться на каждом рендере
   142→  // (иначе эффекты с ctx в зависимостях зацикливаются)
   143→  const ctx: AgentContext | null = pointsCtx
   144→
   145→  const [items, setItems] = useState<ChatItem[]>([])
   146→  const [input, setInput] = useState('')
   147→  const [thinking, setThinking] = useState(false)
   148→  const thinkingRef = useRef(false)
   149→  const scrollRef = useRef<HTMLDivElement>(null)
   150→
   151→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   152→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   153→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   154→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   155→  const srClass = useMemo(() => getSpeechRecognition(), [])
   156→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   157→  const [voice, setVoice] = useState<VoiceState>('idle')
   158→  const speechRef = useRef<SpeechRecognitionLike | null>(null)
   159→  const recorderRef = useRef<MediaRecorder | null>(null)
   160→  const streamRef = useRef<MediaStream | null>(null)
   161→  const chunksRef = useRef<Blob[]>([])
   162→  const autoStopRef = useRef<number | null>(null)
   163→  const baseInputRef = useRef('')
   164→  // push-to-talk: кнопка микрофона зажата и момент начала удержания
   165→  // (по длительности отличаем намеренное удержание от случайного тапа)
   166→  const holdActiveRef = useRef(false)
   167→  const holdStartedAtRef = useRef(0)
   168→
   169→  const clearAutoStop = () => {
   170→    if (autoStopRef.current !== null) {
   171→      window.clearTimeout(autoStopRef.current)
   172→      autoStopRef.current = null
   173→    }
   174→  }
   175→
   176→  const stopVoice = useCallback(() => {
   177→    clearAutoStop()
   178→    if (speechRef.current) {
   179→      try {
   180→        speechRef.current.stop()
   181→      } catch {
   182→        // уже остановлен
   183→      }
   184→      return
   185→    }
   186→    const rec = recorderRef.current
   187→    if (rec && rec.state !== 'inactive') rec.stop()
   188→  }, [])
   189→
   190→  /** Отпускание кнопки (push-to-talk): завершаем запись. */
   191→  const releaseHold = useCallback(() => {
   192→    if (!holdActiveRef.current) return
   193→    holdActiveRef.current = false
   194→    const wasRecording = voice === 'recording'
   195→    const heldMs = Date.now() - holdStartedAtRef.current
   196→    stopVoice()
   197→    // мгновенный тап вместо удержания — вежливо подсказываем, как пользоваться
   198→    if (wasRecording && heldMs < 600) {
   199→      toast({
   200→        title: 'Короткое нажатие',
   201→        description: 'Зажмите кнопку микрофона и говорите, отпустите — вопрос появится в поле ввода.',
   202→      })
   203→    }
   204→  }, [stopVoice, voice, toast])
   205→
   206→  /** Запасной путь: собранный blob → WAV → /api/asr → текст. */
   207→  const finishVoice = useCallback(async () => {
   208→    // поток больше не нужен
   209→    streamRef.current?.getTracks().forEach((t) => t.stop())
   210→    streamRef.current = null
   211→    recorderRef.current = null
   212→    const blob = new Blob(chunksRef.current, { type: chunksRef.current[0]?.type || 'audio/webm' })
   213→    chunksRef.current = []
   214→    if (blob.size < 2048) {
   215→      setVoice('idle')
   216→      toast({ title: 'Запись пустая', description: 'Зажмите кнопку микрофона и скажите вопрос голосом.' })
   217→      return
   218→    }
   219→    setVoice('transcribing')
   220→    try {
   221→      const wav = await blobToWavBase64(blob)
   222→      const res = await fetch('/api/asr', {
   223→        method: 'POST',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({ audio: wav }),
   226→      })
   227→      const j = (await res.json().catch(() => ({}))) as { text?: string; error?: string }
   228→      if (!res.ok || !j.text?.trim()) {
   229→        toast({ title: 'Речь не распознана', description: j.error ?? 'Попробуйте сказать чуть чётче.', variant: 'destructive' })
   230→        return
   231→      }
   232→      // текст подставляем в поле ввода — пчеловод может поправить и отправить
   233→      const base = baseInputRef.current
   234→      const text = j.text.trim()
   235→      setInput(base ? `${base} ${text}` : text)
   236→      toast({ title: 'Голос распознан', description: text })
   237→    } catch {
   238→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   239→    } finally {
   240→      setVoice('idle')
   241→    }
   242→  }, [toast])
   243→
   244→  const startVoice = useCallback(async () => {
   245→    // запись уже идёт — второй старт (второй палец/указатель) не нужен
   246→    if (speechRef.current || recorderRef.current) return
   247→    clearAutoStop()
   248→    baseInputRef.current = input.trim()
   249→
   250→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   251→    if (srClass) {
   252→      const rec = new srClass()
   253→      rec.lang = VOICE_LANG
   254→      rec.continuous = true
   255→      rec.interimResults = true
   256→      rec.maxAlternatives = 1
   257→      let finalText = ''
   258→      rec.onresult = (e) => {
   259→        let interim = ''
   260→        for (let i = e.resultIndex; i < e.results.length; i++) {
   261→          const r = e.results[i]
   262→          const t = r[0]?.transcript ?? ''
   263→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   264→          else interim += t
   265→        }
   266→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   267→        if (text) {
   268→          const base = baseInputRef.current
   269→          setInput(base ? `${base} ${text}` : text)
   270→        }
   271→      }
   272→      rec.onerror = (ev) => {
   273→        const code = ev.error ?? ''
   274→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   275→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   276→        } else if (code === 'no-speech') {
   277→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   278→        } else if (code !== 'aborted') {
   279→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   280→        }
   281→      }
   282→      rec.onend = () => {
   283→        speechRef.current = null
   284→        clearAutoStop()
   285→        setVoice('idle')
   286→      }
   287→      speechRef.current = rec
   288→      try {
   289→        rec.start()
   290→      } catch {
   291→        toast({ title: 'Не получилось начать', description: 'Зажмите кнопку микрофона ещё раз.', variant: 'destructive' })
   292→        return
   293→      }
   294→      setVoice('recording')
   295→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   296→      // кнопку отпустили раньше, чем распознавание успело стартовать
   297→      if (!holdActiveRef.current) stopVoice()
   298→      return
   299→    }
   300→
   301→    // 2) запасной путь: запись с микрофона → серверное распознавание
   302→    try {
   303→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   304→      streamRef.current = stream
   305→      chunksRef.current = []
   306→      const rec = new MediaRecorder(stream)
   307→      rec.ondataavailable = (e) => {
   308→        if (e.data.size > 0) chunksRef.current.push(e.data)
   309→      }
   310→      rec.onstop = () => void finishVoice()
   311→      rec.start()
   312→      recorderRef.current = rec
   313→      setVoice('recording')
   314→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   315→      // кнопку отпустили, пока запрашивался доступ к микрофону
   316→      if (!holdActiveRef.current) stopVoice()
   317→    } catch {
   318→      toast({
   319→        title: 'Микрофон недоступен',
   320→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   321→        variant: 'destructive',
   322→      })
   323→    }
   324→  }, [srClass, input, finishVoice, stopVoice, toast])
   325→
   326→  // при размонтировании прерываем распознавание и закрываем поток
   327→  useEffect(
   328→    () => () => {
   329→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   330→      try {
   331→        speechRef.current?.abort()
   332→      } catch {
   333→        // нет активного распознавания
   334→      }
   335→      streamRef.current?.getTracks().forEach((t) => t.stop())
   336→    },
   337→    [],
   338→  )
   339→
   340→  /* История из базы — общая лента пасеки, без привязки к улью */
   341→  useEffect(() => {
   342→    if (!user || realStatus !== 'ready' || !ctx) return
   343→    let cancelled = false
   344→    void (async () => {
   345→      try {
   346→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   347→        if (!res.ok) throw new Error('history')
   348→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   349→        if (cancelled) return
   350→        const restored: ChatItem[] = j.items.map((m) => ({
   351→          role: m.role,
   352→          text: m.text,
   353→          ...(m.role === 'agent' && m.agentJson
   354→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   355→            : {}),
   356→        }))
   357→        if (restored.length === 0) {
   358→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   359→        } else {
   360→          setItems(restored.slice(-30))
   361→        }
   362→      } catch {
   363→        if (!cancelled) {
   364→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   365→        }
   366→      }
   367→    })()
   368→    return () => {
   369→      cancelled = true
   370→    }
   371→  }, [user, realStatus, ctx])
   372→
   373→  const sendUser = useCallback(
   374→    (rawText: string) => {
   375→      const trimmed = rawText.trim()
   376→      if (!trimmed || thinkingRef.current) return
   377→      // отправка текста прерывает голосовой ввод (кнопка могла быть ещё зажата)
   378→      if (holdActiveRef.current) {
   379→        holdActiveRef.current = false
   380→        stopVoice()
   381→      }
   382→      const context = ctx
   383→      if (!context) return
   384→      thinkingRef.current = true
   385→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   386→      setInput('')
   387→      setThinking(true)
   388→      const started = Date.now()
   389→
   390→      const run = async () => {
   391→        let answer: AgentMessage
   392→        try {
   393→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   394→          const res = await fetch('/api/agent', {
   395→            method: 'POST',
   396→            headers: { 'Content-Type': 'application/json' },
   397→            body: JSON.stringify({ question: trimmed }),
   398→          })
   399→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   400→          const j = (await res.json()) as { answer: AgentMessage }
   401→          answer = j.answer
   402→        } catch {
   403→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   404→        }
   405→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   406→        const elapsed = Date.now() - started
   407→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   408→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   409→        thinkingRef.current = false
   410→        setThinking(false)
   411→        // сохраняем переписку в базу — общая лента пасеки
   412→        if (useAppState.getState().user) {
   413→          void fetch('/api/chat', {
   414→            method: 'POST',
   415→            headers: { 'Content-Type': 'application/json' },
   416→            body: JSON.stringify({
   417→              userText: trimmed,
   418→              agentText: answer.text,
   419→              agent: answer,
   420→            }),
   421→          }).catch(() => undefined)
   422→        }
   423→      }
   424→      void run()
   425→    },
   426→    [ctx, stopVoice],
   427→  )
   428→
   429→  // consume prefill, заданного до монтирования + подписка на новые вопросы из дашборда
   430→  useEffect(() => {
   431→    useAppState.setState({ chatPrefill: null })
   432→    const unsub = useAppState.subscribe((state, prev) => {
   433→      if (state.chatPrefill && state.chatPrefill !== prev.chatPrefill) {
   434→        useAppState.setState({ chatPrefill: null })
   435→        sendUser(state.chatPrefill)
   436→      }
   437→    })
   438→    return unsub
   439→  }, [sendUser])
   440→
   441→  useEffect(() => {
   442→    const el = scrollRef.current
   443→    if (el) el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' })
   444→  }, [items, thinking])
   445→
   446→  /* Контекст ещё не готов — экран загрузки; пустая карта — честная подсказка */
   447→  if (!ctx || ctx.hives.length === 0) {
   448→    const noPoints = apiaryPoints !== null && apiaryPoints.length === 0
   449→    return (
   450→      <Card className="border-red-100">
   451→        <CardContent className="p-10 text-center">
   452→          <HexChip className="mx-auto h-12 w-12 animate-pulse bg-red-100">
   453→            <Bot className="h-6 w-6 text-red-600" />
   454→          </HexChip>
