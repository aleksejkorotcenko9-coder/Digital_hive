    33→    33→ts
    34→    34→/**
    35→    35→ * Подтверждение email: коды и токены.
    36→    36→ *
    37→    37→ * КОД (основной способ): 6 цифр, crypto.randomInt. В БД хранится ТОЛЬКО
    38→    38→ * SHA-256 хеш кода (утечка базы не раскрывает код), сырой код уходит
    39→    39→ * пользователю в письме. Срок жизни — 30 минут, код одноразовый;
    40→    40→ * не более 5 неверных попыток ввода.
    41→    41→ *
    42→    42→ * ТОКЕН-ССЫЛКА (legacy, остался для /api/auth/verify): randomBytes(32) → hex,
    43→    43→ * в БД SHA-256 хеш, срок 24 часа, одноразовый.
    44→    44→ *
    45→    45→ * Плюс rate limit повторной отправки: не более 3 запросов на один email в час.
    46→    46→ * Хранение в памяти процесса (Map) — достаточно для этого приложения.
    47→    47→ */
    48→    48→import { createHash, randomBytes, randomInt } from 'node:crypto'
    49→    49→import { db } from './db'
    50→    50→
    51→    51→export const VERIFICATION_TTL_MS = 24 * 3600 * 1000 // 24 часа (для токена-ссылки)
    52→    52→export const CODE_TTL_MS = 30 * 60 * 1000 // 30 минут (для кода из письма)
    53→    53→const MAX_CODE_ATTEMPTS = 5
    54→    54→
    55→    55→function hashToken(raw: string): string {
    56→    56→  return createHash('sha256').update(raw).digest('hex')
    57→    57→}
    58→    58→
    59→    59→/**
    60→    60→ * Создаёт новый токен подтверждения для пользователя. Предыдущие токены
    61→    61→ * пользователя удаляются — действительна всегда только последняя ссылка.
    62→    62→ * Возвращает СЫРОЙ токен (только для ссылки в письме).
    63→    63→ */
    64→    64→export async function issueVerificationToken(userId: string): Promise<string> {
    65→    65→  const raw = randomBytes(32).toString('hex')
    66→    66→  await db.verificationToken.deleteMany({ where: { userId } })
    67→    67→  await db.verificationToken.create({
    68→    68→    data: {
    69→    69→      userId,
    70→    70→      tokenHash: hashToken(raw),
    71→    71→      expiresAt: new Date(Date.now() + VERIFICATION_TTL_MS),
    72→    72→    },
    73→    73→  })
    74→    74→  return raw
    75→    75→}

    77→    77→export type ConsumeResult = { ok: true; userId: string } | { ok: false }
    78→    78→
    79→    79→/* ------------------------------------------------------------------ */
    80→    80→/* КОД ПОДТВЕРЖДЕНИЯ (основной способ)                                  */
    81→    81→/* ------------------------------------------------------------------ */
    82→    82→
    83→    83→/** userId → счётчик неверных попыток ввода кода. */
    84→    84→const codeAttempts = new Map<string, { count: number; firstAt: number }>()
    85→    85→
    86→    86→function resetAttempts(userId: string): void {
    87→    87→  codeAttempts.delete(userId)
    88→    88→}
    89→    89→
    90→    90→function failAttempt(userId: string): number {
    91→    91→  const now = Date.now()
    92→    92→  const rec = codeAttempts.get(userId)
    93→    93→  if (!rec || now - rec.firstAt > CODE_TTL_MS) {
    94→    94→    codeAttempts.set(userId, { count: 1, firstAt: now })
    95→    95→    return 1
    96→    96→  }
    97→    97→  rec.count += 1
    98→    98→  return rec.count
    99→    99→}
   100→   100→
   101→   101→function attemptsLeft(userId: string): number {
   102→   102→  const rec = codeAttempts.get(userId)
   103→   103→  if (!rec || Date.now() - rec.firstAt > CODE_TTL_MS) return MAX_CODE_ATTEMPTS
   104→   104→  return Math.max(0, MAX_CODE_ATTEMPTS - rec.count)
   105→   105→}
   106→   106→
   107→   107→/**
   108→   108→ * Создаёт новый 6-значный код подтверждения для пользователя. Предыдущие
   109→   109→ * записи пользователя удаляются — действителен всегда только последний код.
   110→   110→ * Возвращает СЫРОЙ код (только для письма).
   111→   111→ */
   112→   112→export async function issueVerificationCode(userId: string): Promise<string> {
   113→   113→  const code = String(randomInt(100000, 1000000))
   114→   114→  await db.verificationToken.deleteMany({ where: { userId } })
   115→   115→  await db.verificationToken.create({
   116→   116→    data: {
   117→   117→      userId,
   118→   118→      tokenHash: hashToken(code),
   119→   119→      expiresAt: new Date(Date.now() + CODE_TTL_MS),
   120→   120→    },
   121→   121→  })
   122→   122→  resetAttempts(userId)
   123→   123→  return code
   124→   124→}
   125→   125→
   126→   126→export type CodeConsumeResult = 'ok' | 'invalid' | 'expired' | 'locked'
   127→   127→
   128→   128→/**
   129→   129→ * Проверяет код из письма по email. Валидный и не истёкший — ставит
   130→   130→ * User.emailVerified и удаляет запись (одноразовость).
   131→   131→ * 'invalid' — неверный код (или аккаунта нет); 'expired' — срок вышел;
   132→   132→ * 'locked' — слишком много неверных попыток.
   133→   133→ */
   134→   134→export async function consumeVerificationCode(email: string, code: string): Promise<CodeConsumeResult> {
   135→   135→  const user = await db.user.findUnique({ where: { email } })
   136→   136→  if (!user) return 'invalid'
   137→   137→
   138→   138→  const record = await db.verificationToken.findFirst({ where: { userId: user.id } })
   139→   139→  if (!record) return 'expired'
   140→   140→
   141→   141→  if (record.expiresAt.getTime() < Date.now()) {
   142→   142→    await db.verificationToken.delete({ where: { id: record.id } }).catch(() => undefined)
   143→   143→    return 'expired'
   144→   144→  }
   145→   145→
   146→   146→  if (attemptsLeft(user.id) <= 0) return 'locked'
   147→   147→
   148→   148→  if (hashToken(code) !== record.tokenHash) {
   149→   149→    const fails = failAttempt(user.id)
   150→   150→    return fails >= MAX_CODE_ATTEMPTS ? 'locked' : 'invalid'
   151→   151→  }
   152→   152→
   153→   153→  await db.$transaction([
   154→   154→    db.user.update({ where: { id: user.id }, data: { emailVerified: new Date() } }),
   155→   155→    db.verificationToken.delete({ where: { id: record.id } }),
   156→   156→  ])
   157→   157→  resetAttempts(user.id)
   158→   158→  return 'ok'
   159→   159→}
   160→   160→
   161→   161→/**
   162→   162→ * Проверяет сырой токен из ссылки. Если он валиден и не истёк — ставит
   163→   163→ * User.emailVerified = текущая дата и удаляет запись токена (одноразовость).
   164→   164→ * Невалидный / истёкший / отсутствующий → { ok: false }.
   165→   165→ */
   166→   166→export async function consumeVerificationToken(raw: string | null): Promise<ConsumeResult> {
   167→   167→  if (!raw) return { ok: false }
   168→   168→  const record = await db.verificationToken.findUnique({ where: { tokenHash: hashToken(raw) } })
   169→   169→  if (!record) return { ok: false }
   170→   170→
   171→   171→  if (record.expiresAt.getTime() < Date.now()) {
   172→   172→    // Истёкший токен сразу чистим, чтобы не копился мусор.
   173→   173→    await db.verificationToken.delete({ where: { id: record.id } }).catch(() => undefined)
   174→   174→    return { ok: false }
   175→   175→  }
   176→   176→
   177→   177→  await db.$transaction([
   178→   178→    db.user.update({ where: { id: record.userId }, data: { emailVerified: new Date() } }),
   179→   179→    db.verificationToken.delete({ where: { id: record.id } }),
   180→   180→  ])
   181→   181→  return { ok: true, userId: record.userId }
   182→   182→}
   183→   183→
   184→   184→/* ------------------------------------------------------------------ */
   185→   185→/* Rate limit повторной отправки: 3 запроса на email в час              */
   186→   186→/* ------------------------------------------------------------------ */
   187→   187→
   188→   188→const RESEND_WINDOW_MS = 3600 * 1000 // 1 час
   189→   189→const RESEND_MAX_PER_WINDOW = 3
   190→   190→
   191→   191→/** email (в нижнем регистре) → времена запросов внутри текущего часа. */
   192→   192→const resendHits = new Map<string, number[]>()
   193→   193→
   194→   194→export function checkResendRateLimit(email: string): { allowed: boolean; retryAfterSec: number } {
   195→   195→  const now = Date.now()
   196→   196→  const list = (resendHits.get(email) ?? []).filter((t) => now - t < RESEND_WINDOW_MS)
   197→   197→
   198→   198→  if (list.length >= RESEND_MAX_PER_WINDOW) {
   199→   199→    const retryAfterSec = Math.max(1, Math.ceil((RESEND_WINDOW_MS - (now - list[0])) / 1000))
   200→   200→    resendHits.set(email, list)
   201→   201→    return { allowed: false, retryAfterSec }
   202→   202→  }
   203→   203→
   204→   204→  list.push(now)
   205→   205→  resendHits.set(email, list)
   206→   206→  return { allowed: true, retryAfterSec: 0 }
   207→   207→}

   295→   295→/**
   296→   296→ * Отправляет письмо с кодом подтверждения. Бросает исключение только при
   297→   297→ * настроенном, но неудавшемся SMTP-вызове — вызывающий код решает, как реагировать.
   298→   298→ */
   299→   299→export async function sendVerificationEmail(to: string, name: string, code: string): Promise<SendResult> {
   300→   300→  // SMTP не настроен — dev-fallback: печатаем код в консоль сервера.
   301→   301→  if (!isSmtpConfigured()) {
   302→   302→    console.log(`[email] SMTP не настроен, письмо НЕ отправлено. Код подтверждения для ${to}: ${code}`)
   303→   303→    return 'dev-logged'
   304→   304→  }
   305→   305→
   306→   306→  const port = Number(process.env.EMAIL_SERVER_PORT ?? 587)
   307→   307→  const transporter = nodemailer.createTransport({
   308→   308→    host: process.env.EMAIL_SERVER_HOST,
   309→   309→    port,
   310→   310→    secure: port === 465, // 465 — implicit TLS, 587 — STARTTLS
   311→   311→    auth: process.env.EMAIL_SERVER_USER
   312→   312→      ? {
   313→   313→          user: process.env.EMAIL_SERVER_USER,
   314→   314→          pass: process.env.EMAIL_SERVER_PASSWORD,
   315→   315→        }
   316→   316→      : undefined,
   317→   317→    // CHANGED: Mail.ru обрывает TLS-рукопожатие без явного минимального TLS 1.2
   318→   318→    tls: { minVersion: 'TLSv1.2' },
   319→   319→  })
   320→   320→
   321→   321→  const info = await transporter.sendMail({
   322→   322→    from: process.env.EMAIL_FROM,
   323→   323→    to,
   324→   324→    subject: `Код подтверждения — ${APP_NAME}`,
   325→   325→    html: buildVerificationHtml(name, code),
   326→   326→    text: buildVerificationText(name, code),
   327→   327→    // CHANGED: просим Gmail прислать отчёт при отказе/задержке — придёт в ящик отправителя
   328→   328→    dsn: { notify: ['FAILURE', 'DELAY'], ret: 'HDRS' },
   329→   329→  })
   330→   330→  console.log(`[email] письмо с кодом отправлено на ${to} (${info.messageId}). Код: ${code}`)
   331→   331→  return 'sent'
   332→   332→}

   360→   360→/**
   361→   361→ * Регистрация нового пользователя: имя + email + пароль.
   362→   362→ * CHANGED: сессия/cookie НЕ выдаются — доступ откроется только после
   363→   363→ * подтверждения email кодом из письма.

   365→   365→export async function POST(req: Request) {
   366→   366→  try {
   367→   367→    await ensureSeeded()
   368→   368→    const body = (await req.json()) as { name?: string; email?: string; password?: string; role?: string }
   369→   369→    const name = (body.name ?? '').trim()
   370→   370→    const email = (body.email ?? '').trim().toLowerCase()
   371→   371→    const password = body.password ?? ''
   372→   372→    const role = body.role === 'BEEKEEPER' ? 'BEEKEEPER' : 'HELPER'
   373→   373→
   374→   374→    if (name.length < 2 || name.length > 40) {
   375→   375→      return NextResponse.json({ error: 'Имя: от 2 до 40 символов' }, { status: 400 })
   376→   376→    }
   377→   377→    if (!EMAIL_RE.test(email)) {
   378→   378→      return NextResponse.json({ error: 'Введите корректный email' }, { status: 400 })
   379→   379→    }
   380→   380→    if (password.length < 4) {
   381→   381→      return NextResponse.json({ error: 'Пароль — минимум 4 символа' }, { status: 400 })
   382→   382→    }
   383→   383→
