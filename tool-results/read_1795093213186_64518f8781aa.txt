   151→   151→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */

   162→   162→  const autoStopRef = useRef<number | null>(null)

   234→   234→      const text = j.text.trim()
   235→   235→      setInput(base ? `${base} ${text}` : text)

   237→   237→    } catch {
   238→   238→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   239→   239→    } finally {
   240→   240→      setVoice('idle')
   241→   241→    }
   242→   242→  }, [toast])
   243→   243→
   244→   244→  const startVoice = useCallback(async () => {

   249→   249→
   250→   250→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе

   252→   252→      const rec = new srClass()

   264→   264→          else interim += t

   268→   268→          const base = baseInputRef.current
   269→   269→          setInput(base ? `${base} ${text}` : text)
   270→   270→        }
   271→   271→      }

   280→   280→        }
   281→   281→      }

   284→   284→        clearAutoStop()

   286→   286→      }

   293→   293→      }

   297→   297→      if (!holdActiveRef.current) stopVoice()
   298→   298→      return
   299→   299→    }
   300→   300→

   310→   310→      rec.onstop = () => void finishVoice()

   313→   313→      setVoice('recording')
   314→   314→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)

   316→   316→      if (!holdActiveRef.current) stopVoice()
   317→   317→    } catch {
   318→   318→      toast({
   319→   319→        title: 'Микрофон недоступен',
   320→   320→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   321→   321→        variant: 'destructive',
   322→   322→      })
   323→   323→    }
   324→   324→  }, [srClass, input, finishVoice, stopVoice, toast])
   325→   325→
   326→   326→  // при размонтировании прерываем распознавание и закрываем поток
   327→   327→  useEffect(
   328→   328→    () => () => {
   329→   329→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   330→   330→      try {
   331→   331→        speechRef.current?.abort()
   332→   332→      } catch {
   333→   333→        // нет активного распознавания

   337→   337→    [],
   338→   338→  )
   339→   339→
   340→   340→  /* История из базы — общая лента пасеки, без привязки к улью */
   341→   341→  useEffect(() => {
   342→   342→    if (!user || realStatus !== 'ready' || !ctx) return
   343→   343→    let cancelled = false
   344→   344→    void (async () => {
   345→   345→      try {
   346→   346→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   347→   347→        if (!res.ok) throw new Error('history')
   348→   348→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   349→   349→        if (cancelled) return
   350→   350→        const restored: ChatItem[] = j.items.map((m) => ({
   351→   351→          role: m.role,
   352→   352→          text: m.text,
   353→   353→          ...(m.role === 'agent' && m.agentJson
   354→   354→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   355→   355→            : {}),
   356→   356→        }))
   357→   357→        if (restored.length === 0) {
   358→   358→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   359→   359→        } else {
   360→   360→          setItems(restored.slice(-30))
   361→   361→        }
   362→   362→      } catch {
   363→   363→        if (!cancelled) {
   364→   364→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   365→   365→        }
   366→   366→      }
   367→   367→    })()
   368→   368→    return () => {
   369→   369→      cancelled = true
   370→   370→    }
   371→   371→  }, [user, realStatus, ctx])
   372→   372→
   373→   373→  const sendUser = useCallback(
   374→   374→    (rawText: string) => {
   375→   375→      const trimmed = rawText.trim()
   376→   376→      if (!trimmed || thinkingRef.current) return
   377→   377→      // отправка текста прерывает голосовой ввод (кнопка могла быть ещё зажата)
   378→   378→      if (holdActiveRef.current) {
   379→   379→        holdActiveRef.current = false
   380→   380→        stopVoice()
   381→   381→      }
   382→   382→      const context = ctx
   383→   383→      if (!context) return
