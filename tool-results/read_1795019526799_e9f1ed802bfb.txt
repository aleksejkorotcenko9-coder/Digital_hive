   603→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   604→function weightTotal(h: Hive): number {
   605→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   606→  const first = h.series[0]?.weightKg ?? null
   607→  if (w === null || first === null) return 0
   608→  return Math.round((w - first) * 10) / 10
   609→}
