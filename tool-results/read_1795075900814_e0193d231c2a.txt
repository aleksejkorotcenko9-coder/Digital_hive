     4→ * Экран «ИИ-агент»: чат пчеловода с цифровым помощником.

     6→ * на карте (температура, вес, влажность, шум) и реальная погода из интернета.
     7→ * Темы строго ограничены: погода и ульи — отвечает, всё остальное — вежливо
     8→ * отказывается. Модель, на которой работает агент, пользователю не показывается.
     9→ * Голосовой ввод работает как рация (push-to-talk): кнопку микрофона нужно
    10→ * зажать и говорить — пока она удерживается, идёт запись; при отпускании запись
    11→ * завершается, речь распознаётся (Web Speech или сервер /api/asr) и текст
    12→ * подставляется в поле ввода.
    13→ * Улии и показания синхронизированы с картой: агент говорит по тем же точкам
    14→ * и датчикам, что видит пчеловод на карте и дашборде.
    15→ * История переписки сохраняется в SQLite (общая лента пасеки).
    16→ */

    69→        if (!res.ok) return
    70→        const j = (await res.json()) as { points: MapPointDto[] }

    72→      } catch {
    73→        // датчики подтянутся на следующем опросе
    74→      }
    75→    }
    76→    void load()
    77→    const timer = setInterval(() => void load(), 5000)
    78→    return () => {
    79→      alive = false
    80→      clearInterval(timer)
    81→    }
    82→  }, [])
    83→
    84→  // Погода по фактическому центру пасеки (среднее координат точек), те же
    85→  // координаты, что на дашборде, чтобы агент называл те же числа, что видит
    86→  // пчеловод. Эффект зависит от ЧИСЕЛ координат: опрос точек каждые 5 с
    87→  // не вызывает перезапрос погоды, пока координаты не изменились.
    88→  const center = useMemo<[number, number]>(
    89→    () =>
    90→      (apiaryPoints && apiaryPoints.length > 0 ? pointsCenter(apiaryPoints) : null) ??
    91→      DEFAULT_PLACE.center,
    92→    [apiaryPoints],
    93→  )
    94→  const [wLat, wLon] = center
    95→  useEffect(() => {
    96→    let cancelled = false
    97→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
    98→    const load = async () => {
    99→      try {
   100→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   101→        if (!res.ok) throw new Error('weather')
   102→        const j = (await res.json()) as WeatherResponse
   103→        if (!cancelled) setWResp(j)
   104→      } catch {
   105→        // погода не критична: агент честно скажет, если данных нет
   106→      }
   107→    }
   108→    void load()
   109→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   110→    return () => {
   111→      cancelled = true
   112→      clearInterval(timer)
   113→    }
   114→  }, [wLat, wLon])
   115→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   116→
   117→  // Ульи и показания — те же точки, что на карте (синхронизация с /api/map-points)
   118→  const pointsCtx = useMemo<AgentContext | null>(() => {
   119→    if (!apiaryPoints) return null
   120→    const points: AgentPoint[] = apiaryPoints.map((p) => ({
   121→      id: p.id,
   122→      name: p.name,
   123→      lat: p.lat,
   124→      lon: p.lon,
   125→      hiveCount: p.hiveCount,
   126→      hasSensor: p.hasSensor,
   127→      sensorOn: p.sensorOn,
   128→      sensor: p.sensor,
   129→    }))
   130→    return makeContext(pointsToHives(apiaryPoints), {
   131→      weather: agentWeather?.days ?? null,
   132→      weatherNow: agentWeather?.now ?? null,
   133→      pesticide: null, // реестр обработок не подключён — агент честно об этом скажет
   134→      sites: data ? data.sites.map(siteDtoToSite) : [],
   135→      points,
   136→      apiaryName: 'Моя пасека',
   137→    })
   138→  }, [apiaryPoints, agentWeather, data])
   139→
   140→  // контекст мемоизирован, чтобы не пересоздаваться на каждом рендере
   141→  // (иначе эффекты с ctx в зависимостях зацикливаются)
   142→  const ctx: AgentContext | null = pointsCtx
   143→
   144→  const [items, setItems] = useState<ChatItem[]>([])
   145→  const [input, setInput] = useState('')
   146→  const [thinking, setThinking] = useState(false)
   147→  const thinkingRef = useRef(false)
   148→  const scrollRef = useRef<HTMLDivElement>(null)
   149→
   150→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   151→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   152→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   153→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   154→  const srClass = useMemo(() => getSpeechRecognition(), [])
   155→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   156→  const [voice, setVoice] = useState<VoiceState>('idle')
   157→  const speechRef = useRef<SpeechRecognitionLike | null>(null)
   158→  const recorderRef = useRef<MediaRecorder | null>(null)
   159→  const streamRef = useRef<MediaStream | null>(null)
   160→  const chunksRef = useRef<Blob[]>([])
   161→  const autoStopRef = useRef<number | null>(null)
   162→  const baseInputRef = useRef('')
   163→  // push-to-talk: кнопка микрофона зажата и момент начала удержания
   164→  // (по длительности отличаем намеренное удержание от случайного тапа)
   165→  const holdActiveRef = useRef(false)
   166→  const holdStartedAtRef = useRef(0)
   167→
   168→  const clearAutoStop = () => {
   169→    if (autoStopRef.current !== null) {
   170→      window.clearTimeout(autoStopRef.current)
   171→      autoStopRef.current = null
   172→    }
   173→  }
   174→
   175→  const stopVoice = useCallback(() => {
   176→    clearAutoStop()
   177→    if (speechRef.current) {
   178→      try {
   179→        speechRef.current.stop()
   180→      } catch {
   181→        // уже остановлен
   182→      }
   183→      return
   184→    }
   185→    const rec = recorderRef.current
   186→    if (rec && rec.state !== 'inactive') rec.stop()
   187→  }, [])
   188→
   189→  /** Отпускание кнопки (push-to-talk): завершаем запись. */
   190→  const releaseHold = useCallback(() => {
   191→    if (!holdActiveRef.current) return
   192→    holdActiveRef.current = false
   193→    const wasRecording = voice === 'recording'
   194→    const heldMs = Date.now() - holdStartedAtRef.current
   195→    stopVoice()
   196→    // мгновенный тап вместо удержания — вежливо подсказываем, как пользоваться
   197→    if (wasRecording && heldMs < 600) {
   198→      toast({
   199→        title: 'Короткое нажатие',
   200→        description: 'Зажмите кнопку микрофона и говорите, отпустите — вопрос появится в поле ввода.',
   201→      })
   202→    }
   203→  }, [stopVoice, voice, toast])
   204→
   205→  /** Запасной путь: собранный blob → WAV → /api/asr → текст. */
   206→  const finishVoice = useCallback(async () => {
   207→    // поток больше не нужен
   208→    streamRef.current?.getTracks().forEach((t) => t.stop())
   209→    streamRef.current = null
   210→    recorderRef.current = null
   211→    const blob = new Blob(chunksRef.current, { type: chunksRef.current[0]?.type || 'audio/webm' })
   212→    chunksRef.current = []
   213→    if (blob.size < 2048) {
   214→      setVoice('idle')
   215→      toast({ title: 'Запись пустая', description: 'Зажмите кнопку микрофона и скажите вопрос голосом.' })
   216→      return
   217→    }
   218→    setVoice('transcribing')
   219→    try {
   220→      const wav = await blobToWavBase64(blob)
   221→      const res = await fetch('/api/asr', {
   222→        method: 'POST',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        body: JSON.stringify({ audio: wav }),
   225→      })
   226→      const j = (await res.json().catch(() => ({}))) as { text?: string; error?: string }
   227→      if (!res.ok || !j.text?.trim()) {
   228→        toast({ title: 'Речь не распознана', description: j.error ?? 'Попробуйте сказать чуть чётче.', variant: 'destructive' })
   229→        return
   230→      }
   231→      // текст подставляем в поле ввода — пчеловод может поправить и отправить
   232→      const base = baseInputRef.current
   233→      const text = j.text.trim()
   234→      setInput(base ? `${base} ${text}` : text)
   235→      toast({ title: 'Голос распознан', description: text })
   236→    } catch {
   237→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   238→    } finally {
   239→      setVoice('idle')
   240→    }
   241→  }, [toast])
   242→
   243→  const startVoice = useCallback(async () => {
   244→    // запись уже идёт — второй старт (второй палец/указатель) не нужен
   245→    if (speechRef.current || recorderRef.current) return
   246→    clearAutoStop()
   247→    baseInputRef.current = input.trim()
   248→
   249→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   250→    if (srClass) {
   251→      const rec = new srClass()
   252→      rec.lang = VOICE_LANG
   253→      rec.continuous = true
   254→      rec.interimResults = true
   255→      rec.maxAlternatives = 1
   256→      let finalText = ''
   257→      rec.onresult = (e) => {
   258→        let interim = ''
   259→        for (let i = e.resultIndex; i < e.results.length; i++) {
   260→          const r = e.results[i]
   261→          const t = r[0]?.transcript ?? ''
   262→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   263→          else interim += t
   264→        }
   265→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   266→        if (text) {
   267→          const base = baseInputRef.current
   268→          setInput(base ? `${base} ${text}` : text)
   269→        }
   270→      }
   271→      rec.onerror = (ev) => {
   272→        const code = ev.error ?? ''
   273→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   274→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   275→        } else if (code === 'no-speech') {
   276→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   277→        } else if (code !== 'aborted') {
   278→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   279→        }
   280→      }
   281→      rec.onend = () => {
   282→        speechRef.current = null
   283→        clearAutoStop()
   284→        setVoice('idle')
   285→      }
   286→      speechRef.current = rec
   287→      try {
   288→        rec.start()
   289→      } catch {
   290→        toast({ title: 'Не получилось начать', description: 'Зажмите кнопку микрофона ещё раз.', variant: 'destructive' })
   291→        return
   292→      }
   293→      setVoice('recording')
   294→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   295→      // кнопку отпустили раньше, чем распознавание успело стартовать
   296→      if (!holdActiveRef.current) stopVoice()
   297→      return
   298→    }
   299→
   300→    // 2) запасной путь: запись с микрофона → серверное распознавание
   301→    try {
   302→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   303→      streamRef.current = stream
   304→      chunksRef.current = []
   305→      const rec = new MediaRecorder(stream)
   306→      rec.ondataavailable = (e) => {
   307→        if (e.data.size > 0) chunksRef.current.push(e.data)
   308→      }
   309→      rec.onstop = () => void finishVoice()
   310→      rec.start()
   311→      recorderRef.current = rec
   312→      setVoice('recording')
   313→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   314→      // кнопку отпустили, пока запрашивался доступ к микрофону
   315→      if (!holdActiveRef.current) stopVoice()
   316→    } catch {
   317→      toast({
   318→        title: 'Микрофон недоступен',
   319→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   320→        variant: 'destructive',
   321→      })
   322→    }
   323→  }, [srClass, input, finishVoice, stopVoice, toast])
   324→
   325→  // при размонтировании прерываем распознавание и закрываем поток
   326→  useEffect(
   327→    () => () => {
   328→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   329→      try {
   330→        speechRef.current?.abort()
   331→      } catch {
   332→        // нет активного распознавания
   333→      }
   334→      streamRef.current?.getTracks().forEach((t) => t.stop())
   335→    },
   336→    [],
   337→  )
   338→
   339→  /* История из базы — общая лента пасеки, без привязки к улью */
   340→  useEffect(() => {
   341→    if (!user || realStatus !== 'ready' || !ctx) return
   342→    let cancelled = false
   343→    void (async () => {
   344→      try {
   345→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   346→        if (!res.ok) throw new Error('history')
   347→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   348→        if (cancelled) return
   349→        const restored: ChatItem[] = j.items.map((m) => ({
   350→          role: m.role,
   351→          text: m.text,
   352→          ...(m.role === 'agent' && m.agentJson
   353→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   354→            : {}),
   355→        }))
   356→        if (restored.length === 0) {
   357→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   358→        } else {
   359→          setItems(restored.slice(-30))
   360→        }
   361→      } catch {
   362→        if (!cancelled) {
   363→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   364→        }
   365→      }
   366→    })()
   367→    return () => {
   368→      cancelled = true
   369→    }
   370→  }, [user, realStatus, ctx])
   371→
   372→  const sendUser = useCallback(
   373→    (rawText: string) => {
   374→      const trimmed = rawText.trim()
   375→      if (!trimmed || thinkingRef.current) return
   376→      // отправка текста прерывает голосовой ввод (кнопка могла быть ещё зажата)
   377→      if (holdActiveRef.current) {
   378→        holdActiveRef.current = false
   379→        stopVoice()
   380→      }
   381→      const context = ctx
   382→      if (!context) return
   383→      thinkingRef.current = true
   384→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   385→      setInput('')
   386→      setThinking(true)
   387→      const started = Date.now()
   388→
   389→      const run = async () => {
   390→        let answer: AgentMessage
   391→        try {
   392→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   393→          const res = await fetch('/api/agent', {
   394→            method: 'POST',
   395→            headers: { 'Content-Type': 'application/json' },
   396→            body: JSON.stringify({ question: trimmed }),
   397→          })
   398→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   399→          const j = (await res.json()) as { answer: AgentMessage }
   400→          answer = j.answer
   401→        } catch {
   402→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   403→        }
   404→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   405→        const elapsed = Date.now() - started
   406→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   407→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   408→        thinkingRef.current = false
   409→        setThinking(false)
   410→        // сохраняем переписку в базу — общая лента пасеки
   411→        if (useAppState.getState().user) {
   412→          void fetch('/api/chat', {
   413→            method: 'POST',
   414→            headers: { 'Content-Type': 'application/json' },
   415→            body: JSON.stringify({
   416→              userText: trimmed,
   417→              agentText: answer.text,
   418→              agent: answer,
   419→            }),
   420→          }).catch(() => undefined)
   421→        }
   422→      }
   423→      void run()
   424→    },
   425→    [ctx, stopVoice],
   426→  )
   427→
   428→  // consume prefill, заданного до монтирования + подписка на новые вопросы из дашборда
   429→  useEffect(() => {
   430→    useAppState.setState({ chatPrefill: null })
   431→    const unsub = useAppState.subscribe((state, prev) => {
   432→      if (state.chatPrefill && state.chatPrefill !== prev.chatPrefill) {
   433→        useAppState.setState({ chatPrefill: null })
   434→        sendUser(state.chatPrefill)
   435→      }
   436→    })
   437→    return unsub
   438→  }, [sendUser])
   439→
   440→  useEffect(() => {
   441→    const el = scrollRef.current
   442→    if (el) el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' })
   443→  }, [items, thinking])
   444→
   445→  /* Контекст ещё не готов — экран загрузки; пустая карта — честная подсказка */
   446→  if (!ctx || ctx.hives.length === 0) {
   447→    const noPoints = apiaryPoints !== null && apiaryPoints.length === 0
   448→    return (
   449→      <Card className="border-red-100">
   450→        <CardContent className="p-10 text-center">
   451→          <HexChip className="mx-auto h-12 w-12 animate-pulse bg-red-100">
   452→            <Bot className="h-6 w-6 text-red-600" />
   453→          </HexChip>
