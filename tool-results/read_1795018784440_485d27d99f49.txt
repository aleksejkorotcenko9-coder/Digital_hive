     3→/**
     4→ * «Карта пасеки» (Яндекс Карты, JS API 2.1 — без ключа, подключение через src/lib/ymaps.ts).
     5→ * — Клик по карте выбирает точку: карточка погоды в этом месте (Open-Meteo
     6→ *   через /api/weather?lat&lon: сейчас + 3 дня, влажность, ветер, давление).
     7→ * — Кнопка «Добавить улей»: вводим код датчика с его корпуса — датчик находится
     8→ *   в каталоге (lib/sensor-catalog.ts), его GPS-позиция сразу показывается на
     9→ *   карте (синяя метка-превью), после подтверждения улей привязывается в этой
    10→ *   точке. На пасечника — до трёх ульев; есть датчики «пасеки в поле» подальше
    11→ *   от города. Точки хранятся в БД.
    12→ * — Список точек справа: переключение датчика, удаление, перелёт к точке.
    13→ * Анимации: появление карточки погоды, маркеров, превью GPS и пунктов списка.
    14→ */

    64→export function MapPointsView() {
    65→  const user = useAppState((s) => s.user)
    66→  const { toast } = useToast()
    67→  const isBeekeeper = user?.role === 'BEEKEEPER'
    68→
    69→  const containerRef = useRef<HTMLDivElement | null>(null)
    70→  const mapRef = useRef<YmapsMap | null>(null)
    71→  const ymRef = useRef<YmapsApi | null>(null)
    72→  const pointsColRef = useRef<YmapsGeoCollection | null>(null)
    73→  const selColRef = useRef<YmapsGeoCollection | null>(null)
    74→  const prevColRef = useRef<YmapsGeoCollection | null>(null)
    75→  const markerRefs = useRef<Map<string, YmapsGeoObject>>(new Map())
    76→  const layoutRef = useRef<unknown>(null)
    77→  const clickRef = useRef<(lat: number, lon: number) => void>(() => undefined)
    78→  const [mapReady, setMapReady] = useState(false)
    79→  const [mapError, setMapError] = useState<string | null>(null)
    80→
    81→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
    82→  const [selected, setSelected] = useState<{ lat: number; lon: number } | null>(null)
    83→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
    84→  const [weatherLoading, setWeatherLoading] = useState(false)
    85→
    86→  // диалог привязки улья по коду датчика
    87→  const [dialogOpen, setDialogOpen] = useState(false)
    88→  const [code, setCode] = useState('')
    89→  const [name, setName] = useState('')
    90→  const [busy, setBusy] = useState(false)
    91→
    92→  // Как только введён код из каталога — сразу знаем GPS-позицию датчика
    93→  const found = useMemo(() => findSensor(code), [code])
    94→  const showNotFound = normalizeCode(code).length >= 4 && !found
    95→
    96→  /* ---------- карта (Яндекс Карты) ---------- */
    97→  useEffect(() => {
    98→    let disposed = false
    99→    void (async () => {
   100→      let ym: YmapsApi
   101→      try {
   102→        ym = await loadYmaps()
   103→      } catch (err) {
   104→        if (!disposed) {
   105→          setMapError(err instanceof Error ? err.message : 'Не удалось загрузить карту — проверьте соединение')
   106→        }
   107→        return
   108→      }
   109→      if (disposed || !containerRef.current) return
   110→
   111→      // controls: переключатель «Схема / Спутник / Гибрид» + масштаб и геолокация;
   112→      // suppressMapOpenBlock — без блока «Открыть в Яндекс Картах»
   113→      const map = new ym.Map(containerRef.current, {
   114→        center: UFA,
   115→        zoom: 10,
   116→        controls: ['zoomControl', 'typeSelector', 'fullscreenControl', 'geolocationControl', 'rulerControl'],
   117→        suppressMapOpenBlock: true,
   118→      })
   119→      map.behaviors.enable('scrollZoom')
   120→
   121→      // События меток всплывают до карты: выбираем точку только по клику «в пустое место»
   122→      map.events.add('click', (e) => {
   123→        if (e.get('target') !== map) return
   124→        const coords = e.get('coords') as YmapsCoords | undefined
   125→        if (!coords) return
   126→        clickRef.current(coords[0], coords[1])
   127→      })
   128→
   129→      const pointsCol = new ym.GeoObjectCollection()
   130→      const selCol = new ym.GeoObjectCollection()
   131→      const prevCol = new ym.GeoObjectCollection()
   132→      map.geoObjects.add(pointsCol).add(selCol).add(prevCol)
   133→
   134→      // HTML-макет метки улья (круг 🐝 + подпись с числом ульев)
   135→      layoutRef.current = ym.templateLayoutFactory.createClass(HIVE_ICON_TPL)
   136→
   137→      ymRef.current = ym
   138→      mapRef.current = map
   139→      pointsColRef.current = pointsCol
   140→      selColRef.current = selCol
   141→      prevColRef.current = prevCol
   142→      setMapReady(true)
   143→    })()
   144→    return () => {
   145→      disposed = true
   146→      mapRef.current?.destroy()
   147→      mapRef.current = null
   148→      pointsColRef.current = null
   149→      selColRef.current = null
   150→      prevColRef.current = null
   151→      markerRefs.current.clear()
   152→      setMapReady(false)
   153→    }
   154→  }, [])
   155→
   156→  /* ---------- данные точек ---------- */
   157→  const load = useCallback(async () => {
   158→    try {
   159→      const res = await fetch('/api/map-points', { cache: 'no-store' })
   160→      if (!res.ok) return
   161→      const j = (await res.json()) as { points: MapPointDto[] }
   162→      setPoints(j.points)
   163→    } catch {
   164→      // карта и погода продолжают работать, точки подтянутся при следующем обновлении
   165→    }
   166→  }, [])
   167→
   168→  useEffect(() => {
   169→    void load()
   170→  }, [load])
   171→
   172→  /* ---------- курсор и клики ---------- */
   173→  const onMapClick = useCallback((lat: number, lon: number) => {
   174→    // Клик по карте — только выбор точки для погоды; улей привязывается
   175→    // по коду датчика в диалоге, вручную координаты не задаются
   176→    setSelected({ lat, lon })
   177→  }, [])
   178→
   179→  useEffect(() => {
   180→    clickRef.current = onMapClick
   181→  }, [onMapClick])
   182→
   183→  useEffect(() => {
   184→    if (!mapReady) return
   185→    // Стандартный курсор «ладошка»: точки больше не ставятся кликом —
   186→    // улей привязывается по коду датчика в диалоге
   187→    mapRef.current?.cursors.push('grab')
   188→  }, [mapReady])
   189→
   190→  /* ---------- маркеры точек ---------- */
   191→  useEffect(() => {
   192→    const ym = ymRef.current
   193→    const col = pointsColRef.current
   194→    const layout = layoutRef.current
   195→    if (!mapReady || !ym || !col || layout === null || !points) return
   196→    col.removeAll()
   197→    markerRefs.current.clear()
   198→    for (const p of points) {
   199→      const active = p.sensor.reason === 'active'
   200→      const color = active ? '#22c55e' : p.hasSensor ? '#f59e0b' : '#94a3b8'
   201→      const readings = active
   202→        ? `<br>🌡 ${p.sensor.tempC} °C · 💧 ${p.sensor.humidityPct} %<br>⚖ ${p.sensor.weightKg} кг · 🔊 ${p.sensor.noiseDb} дБ`
   203→        : `<br><i>Данных об улье нет — датчик ${SENSOR_LABEL[p.sensor.reason]}.</i>`
   204→      const pm = new ym.Placemark(
   205→        [p.lat, p.lon],
   206→        {
   207→          color,
   208→          label: `${p.hiveCount} ул.`,
   209→          balloonContentHeader: p.name,
   210→          balloonContentBody: `Датчик: <b>${p.sensorCode ?? '—'}</b><br>GPS: ${p.lat.toFixed(4)}, ${p.lon.toFixed(4)}<br>Состояние: ${SENSOR_LABEL[p.sensor.reason]}${readings}`,
   211→        },
   212→        {
   213→          iconLayout: layout,
   214→          iconShape: HIVE_ICON_SHAPE,
   215→          iconOffset: [-17, -17],
   216→          hideIconOnBalloonOpen: false,
   217→        },
   218→      )
   219→      col.add(pm)
   220→      markerRefs.current.set(p.id, pm)
   221→    }
   222→  }, [points, mapReady])
   223→
   224→  /* ---------- маркер выбранной точки ---------- */
   225→  useEffect(() => {
   226→    const ym = ymRef.current
   227→    const col = selColRef.current
   228→    if (!mapReady || !ym || !col) return
   229→    col.removeAll()
   230→    if (!selected) return
   231→    col.add(
   232→      new ym.Placemark(
   233→        [selected.lat, selected.lon],
   234→        { balloonContentBody: 'Выбранная точка — погода справа' },
   235→        { preset: 'islands#redDotIcon', iconColor: '#ef4444', hideIconOnBalloonOpen: false },
   236→      ),
   237→    )
   238→  }, [selected, mapReady])
   239→
   240→  /* ---------- превью GPS-позиции датчика при вводе кода ---------- */
   241→  useEffect(() => {
   242→    const ym = ymRef.current
   243→    const col = prevColRef.current
   244→    const map = mapRef.current
   245→    if (!mapReady || !ym || !col) return
   246→    col.removeAll()
   247→    if (!dialogOpen || !found) return
   248→    // Синяя метка «здесь будет улей»: позиция, которую передаёт датчик
   249→    col.add(
   250→      new ym.Placemark(
   251→        [found.lat, found.lon],
   252→        {
   253→          balloonContentHeader: 'Позиция по GPS датчика',
   254→          balloonContentBody: `Датчик: <b>${found.code}</b> · ${found.model}<br>GPS: ${found.lat.toFixed(4)}, ${found.lon.toFixed(4)}<br>${found.note}`,
   255→        },
   256→        { preset: 'islands#blueCircleDotIcon', iconColor: '#2563eb', hideIconOnBalloonOpen: false },
   257→      ),
   258→    )
   259→    if (map) {
   260→      if (map.getZoom() < 12) map.setZoom(12)
   261→      map.panTo([found.lat, found.lon], { duration: 700, flying: true })
   262→    }
   263→  }, [dialogOpen, found, mapReady])
   264→
   265→  /* ---------- погода выбранной точки ---------- */
   266→  useEffect(() => {
   267→    if (!selected) {
   268→      setWeather(null)
   269→      return
   270→    }
   271→    const ctl = new AbortController()
   272→    setWeatherLoading(true)
   273→    setWeather(null)
   274→    fetch(`/api/weather?lat=${selected.lat}&lon=${selected.lon}`, {
   275→      signal: ctl.signal,
   276→      cache: 'no-store',
   277→    })
   278→      .then((r) => r.json() as Promise<WeatherResponse>)
   279→      .then((j) => setWeather(j))
   280→      .catch(() => undefined)
   281→      .finally(() => {
   282→        if (!ctl.signal.aborted) setWeatherLoading(false)
   283→      })
   284→    return () => ctl.abort()
   285→  }, [selected])
   286→
   287→  /** Плавный перелёт к точке: зум до 12, полёт Яндекса, затем открываем балун метки. */
   288→  const flyMapTo = (lat: number, lon: number, balloonId?: string) => {
   289→    const map = mapRef.current
   290→    if (!map) return
   291→    if (map.getZoom() < 12) map.setZoom(12)
   292→    map.panTo([lat, lon], { duration: 700, flying: true })
   293→    if (balloonId) {
   294→      setTimeout(() => markerRefs.current.get(balloonId)?.balloon.open(), 750)
   295→    }
   296→  }
   297→
   298→  /* ---------- привязать улей по коду датчика ---------- */
   299→  const place = async () => {
   300→    if (!found) return
   301→    setBusy(true)
   302→    try {
   303→      const res = await fetch('/api/map-points', {
   304→        method: 'POST',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify({ sensorCode: found.code, name: name.trim() }),
   307→      })
   308→      const j = (await res.json().catch(() => ({}))) as { point?: MapPointDto; error?: string }
   309→      if (!res.ok || !j.point) {
   310→        toast({ title: 'Не получилось', description: j.error ?? 'Ошибка сервера', variant: 'destructive' })
   311→        return
   312→      }
   313→      setPoints((prev) => [...(prev ?? []), j.point as MapPointDto])
   314→      toast({
   315→        title: 'Улей добавлен',
   316→        description: `Позиция определена по датчику ${j.point.sensorCode}: ${j.point.lat.toFixed(4)}, ${j.point.lon.toFixed(4)}`,
   317→      })
   318→      setDialogOpen(false)
   319→      setCode('')
   320→      setName('')
   321→      flyMapTo(j.point.lat, j.point.lon, j.point.id)
   322→    } finally {
   323→      setBusy(false)
   324→    }
   325→  }
   326→
   327→  const patchSensor = async (
   328→    p: MapPointDto,
   329→    body: { sensorOn?: boolean; hasSensor?: boolean },
   330→    okMsg: string,
   331→  ) => {
   332→    const res = await fetch(`/api/map-points/${p.id}`, {
   333→      method: 'PATCH',
   334→      headers: { 'Content-Type': 'application/json' },
   335→      body: JSON.stringify(body),
   336→    })
   337→    if (!res.ok) {
   338→      toast({ title: 'Не получилось', variant: 'destructive' })
   339→      return
   340→    }
   341→    toast({ title: okMsg })
   342→    // оптимистично правим карточку, серверный ответ уточнит детали
   343→    setPoints((prev) =>
   344→      prev
   345→        ? prev.map((x) => {
   346→            if (x.id !== p.id) return x
   347→            const hasSensor = body.hasSensor ?? x.hasSensor
   348→            const sensorOn = body.sensorOn ?? (body.hasSensor !== undefined ? body.hasSensor : x.sensorOn)
   349→            return { ...x, hasSensor, sensorOn }
   350→          })
   351→        : prev,
   352→    )
   353→    void load()
   354→  }
   355→
   356→  const removePoint = async (p: MapPointDto) => {
   357→    const res = await fetch(`/api/map-points/${p.id}`, { method: 'DELETE' })
   358→    if (!res.ok) {
   359→      toast({ title: 'Не удалось удалить точку', variant: 'destructive' })
   360→      return
   361→    }
   362→    setPoints((prev) => prev?.filter((x) => x.id !== p.id) ?? null)
   363→    toast({ title: 'Точка удалена с карты' })
   364→  }
   365→
   366→  const flyTo = (p: MapPointDto) => flyMapTo(p.lat, p.lon, p.id)
   367→
   368→  const list = points ?? []
   369→  const summary = useMemo(() => {
   370→    const hives = list.reduce((s, p) => s + p.hiveCount, 0)
   371→    const sensors = list.filter((p) => p.hasSensor && p.sensorOn).length
   372→    return { hives, sensors }
   373→  }, [list])
   374→
   375→  return (
   376→    <div className="mx-auto max-w-6xl space-y-5">
   377→      {/* Шапка */}
   378→      <div className="flex flex-wrap items-center justify-between gap-2">
   379→        <div>
   380→          <h1 className="text-2xl font-extrabold tracking-tight text-[#2A1A12]">Карта пасеки</h1>
   381→          <p className="mt-1 max-w-2xl text-sm text-gray-500">
   382→            Кликните по карте — покажу погоду в этой точке. Улей привязывается по коду датчика:
   383→            позицию задаёт GPS самого датчика. Можно привязать до трёх ульев — как настоящая пасека.
   384→          </p>
   385→        </div>
   386→        {isBeekeeper && list.length < HIVE_LIMIT && (
   387→          <Button
   388→            onClick={() => setDialogOpen(true)}
   389→            className="gap-1.5 bg-gradient-to-r from-red-600 to-orange-500 font-bold active:scale-95"
   390→          >
   391→            <MapPinPlus className="h-4 w-4" />
   392→            Добавить улей
   393→          </Button>
   394→        )}
   395→      </div>
   396→
   397→      {/* Карта + правая колонка */}
   398→      <div className="grid gap-4 xl:grid-cols-[1fr_340px]">
   399→        <motion.div
   400→          initial={{ opacity: 0, scale: 0.98 }}
   401→          animate={{ opacity: 1, scale: 1 }}
   402→          transition={ease}
   403→          className="relative overflow-hidden rounded-2xl border border-red-100 shadow-sm"
   404→        >
   405→          <div ref={containerRef} className="h-[440px] w-full md:h-[520px]" />
   406→          {!mapReady && !mapError && (
   407→            <div className="absolute inset-0 flex items-center justify-center gap-2 bg-red-50/60 text-xs font-semibold text-gray-500">
   408→              <RefreshCcw className="h-4 w-4 animate-spin" /> Загружаем карту…
   409→            </div>
   410→          )}
   411→          {mapError && (
   412→            <div className="absolute inset-0 flex items-center justify-center bg-red-50/90 px-6 text-center text-xs font-semibold text-red-700">
   413→              {mapError}
   414→            </div>
   415→          )}
   416→        </motion.div>
   417→
   418→        <div className="space-y-4">
   419→          {/* Погода выбранной точки */}
   420→          <Card className="border-red-100 shadow-sm">
   421→            <CardContent className="p-4">
   422→              <h2 className="flex items-center gap-2 text-sm font-bold text-[#2A1A12]">
   423→                <MapPin className="h-4 w-4 text-red-600" />
   424→                Погода в точке
   425→                {selected && (
   426→                  <span className="ml-auto text-[10px] font-medium tabular-nums text-gray-400">
   427→                    {selected.lat.toFixed(3)}, {selected.lon.toFixed(3)}
   428→                  </span>
   429→                )}
   430→              </h2>
   431→
   432→              {!selected && (
   433→                <p className="mt-3 text-xs text-gray-500">
   434→                  Выберите точку на карте — покажу все условия: температуру, ветер, влажность,
   435→                  давление и прогноз на 3 дня.
   436→                </p>
   437→              )}
   438→
   439→              <AnimatePresence mode="wait">
   440→                {selected && (
   441→                  <motion.div
   442→                    key={`${selected.lat.toFixed(4)},${selected.lon.toFixed(4)}`}
   443→                    initial={{ opacity: 0, y: 12 }}
   444→                    animate={{ opacity: 1, y: 0 }}
   445→                    exit={{ opacity: 0, y: -8 }}
   446→                    transition={ease}
   447→                    className="mt-3"
   448→                  >
   449→                    {weatherLoading || !weather ? (
   450→                      <div className="flex items-center justify-center gap-2 py-8 text-xs text-gray-400">
   451→                        <RefreshCcw className="h-4 w-4 animate-spin" /> Запрашиваем Open-Meteo…
   452→                      </div>
   453→                    ) : (
   454→                      <>
   455→                        <div className="flex items-center gap-3">
