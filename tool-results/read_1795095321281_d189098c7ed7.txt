   133→/**
   134→ * Статус заклещенности: Норма <2 % / Риск 2–3 % / Критично >3 %.
   135→ * label — русский фолбэк из данных; key — ссылка на базовый словарь i18n
   136→ * (dash.varroa.status.* уже переведены на все языки в i18n.ts).
   137→ */
   138→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; key: string; label: string; cls: string } {
   139→  if (pct > 3) return { level: 'crit', key: 'dash.varroa.status.crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   140→  if (pct >= 2) return { level: 'risk', key: 'dash.varroa.status.risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   141→  return { level: 'ok', key: 'dash.varroa.status.ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   142→}

   144→/** Состояние воздуха по индексу ЛОС (0…100). key → demo.voc.* в словаре DEMO_DICTS. */
   145→export function vocState(v: number): { key: string; label: string; cls: string } {
   146→  if (v > 60) return { key: 'demo.voc.hot', label: 'Духота — семье не хватает объема', cls: 'text-red-600' }
   147→  if (v > 30) return { key: 'demo.voc.mid', label: 'Умеренная загрузка воздухом', cls: 'text-amber-600' }
   148→  return { key: 'demo.voc.ok', label: 'Воздух чистый', cls: 'text-emerald-600' }
   149→}

   151→/** Состояние трафика на летке относительно нормы пасеки (45 пч/мин). key → demo.traffic.* в словаре DEMO_DICTS. */
   152→export function trafficState(v: number): { key: string; label: string; cls: string } {
   153→  if (v >= 55) return { key: 'demo.traffic.high', label: 'Интенсивный лёт', cls: 'text-emerald-600' }
   154→  if (v >= 30) return { key: 'demo.traffic.active', label: 'Активный лёт', cls: 'text-emerald-600' }
   155→  if (v >= 18) return { key: 'demo.traffic.weak', label: 'Лёт ослаблен', cls: 'text-amber-600' }
   156→  return { key: 'demo.traffic.low', label: 'Лёт резко снижен', cls: 'text-red-600' }
   157→}

   159→/** Индекс 0…10 → подпись и тон шкалы. */
   160→export function demoGaugeInfo(value: number): { label: string; tone: 'ok' | 'normal' | 'watch' | 'alert' } {
   161→  if (value < 4) return { label: 'Тревожное состояние', tone: 'alert' }
   162→  if (value < 6) return { label: 'Требует внимания', tone: 'watch' }
   163→  if (value < 8.5) return { label: 'Нормальное состояние', tone: 'normal' }
   164→  return { label: 'Отличное состояние', tone: 'ok' }
   165→}

   629→export function demoHiveById(id: number): DemoHive {
   630→  if (id === 0) return demoAggregateHive()
   631→  return DEMO_HIVES.find((h) => h.id === id) ?? DEMO_HIVES[0]
   632→}

   634→/* ------------------------------------------------------------------ */
   635→/* Сводный вид «Вся пасека» (id = 0): средние показатели по 4 ульям     */
   636→/* ------------------------------------------------------------------ */
   637→
   638→const avg = (xs: number[]) => xs.reduce((s, x) => s + x, 0) / xs.length
   639→
   640→/**
   641→ * Сводный улей: показатели пасеки целиком — средние климат и акустика,
   642→ * суммарные вес и привес, средняя заклещенность. Предписания — самые важные
   643→ * из каждого улья, с меткой улья-источника.
   644→ */
   645→export function demoAggregateHive(): DemoHive {
   646→  const hs = DEMO_HIVES
   647→  const base: DemoTelemetry = {
   648→    tempC: Math.round(avg(hs.map((h) => h.base.tempC)) * 10) / 10,
   649→    humidityPct: Math.round(avg(hs.map((h) => h.base.humidityPct))),
   650→    weightKg: Math.round(hs.reduce((s, h) => s + h.base.weightKg, 0) * 10) / 10,

   694→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   695→function seeded(seed: number): () => number {
   696→  let a = seed >>> 0
   697→  return () => {
   698→    a |= 0
   699→    a = (a + 0x6d2b79f5) | 0
   700→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   701→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   702→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   703→  }
   704→}

   706→/** Температура и влажность за 14 дней: колебания вокруг базовых значений. */
   707→export function demoClimateSeries(h: DemoHive, days = 14): DayPoint[] {
   708→  const rnd = seeded(1000 + h.id * 97)
   709→  const out: DayPoint[] = []
   710→  const nowMs = Date.now()
   711→  for (let k = days - 1; k >= 0; k--) {
   712→    const t = new Date(nowMs - k * 86400000)
   713→    const recent = k <= 1
   714→    const tempC = recent
   715→      ? h.base.tempC + (rnd() - 0.5) * 0.2
   716→      : h.base.tempC - 0.4 + rnd() * 0.8
   717→    const humidityPct = recent
   718→      ? h.base.humidityPct + (rnd() - 0.5) * 2
   719→      : h.base.humidityPct - 4 + rnd() * 8
   720→    out.push({
   721→      date: format(t, 'yyyy-MM-dd'),
   722→      label: format(t, 'd.MM'),
   723→      full: format(t, 'd MMMM', { locale: ru }),
   724→      weightKg: null,
   725→      tempInC: Math.round(tempC * 10) / 10,
   726→      humidityPct: Math.round(humidityPct),
   727→    })
   728→  }
   729→  // последние две точки точно соответствуют «сейчас»
   730→  out[out.length - 1].tempInC = h.base.tempC
   731→  out[out.length - 1].humidityPct = h.base.humidityPct
   732→  return out
   733→}

   755→export interface VarroaPoint {
   756→  label: string
   757→  /** Историческое значение заклещенности, %. */
   758→  pct: number | null
   759→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   760→  forecast: number | null
   761→  /** Отметка проведенной обработки. */
   762→  treated: boolean
   763→}
   764→
   765→/**
   766→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   767→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   768→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   769→ *
   770→ * opts.overridePct — «текущее» значение после проведения обработки или нового
   771→ * замера (кривая сходится к нему); opts.treatedToday — сегодня была обработка:
   772→ * на последней точке рисуется дип, а прогноз считается от нового значения.
   773→ */
   774→export function demoVarroaSeries(
   775→  h: DemoHive,
   776→  days = 30,
   777→  opts?: { overridePct?: number; treatedToday?: boolean },
   778→): VarroaPoint[] {
   779→  const rnd = seeded(3000 + h.id * 57)
   780→  const out: VarroaPoint[] = []
   781→  const nowMs = Date.now()
   782→
   783→  const curPct = opts?.overridePct ?? h.base.varroaPct
   784→  const treatedToday = opts?.treatedToday ?? false
   785→  // прогноз: после обработки семья восстанавливается (+0,1 за 14 дн),
   786→  // без обработки критический улей продолжает расти (+1,6)
   787→  const futurePct =
   788→    opts?.overridePct !== undefined || treatedToday
   789→      ? Math.round((curPct + (treatedToday ? 0.1 : 0.2)) * 10) / 10
   790→      : h.base.varroaForecast14d
   791→
   792→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   793→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   794→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   795→
   796→  for (let k = days - 1; k >= 0; k--) {
   797→    const t = new Date(nowMs - k * 86400000)
   798→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   799→    let pct = startPct + (curPct - startPct) * progress
   800→    // дип после обработки: −1,2 % в течение 3 дней
   801→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   802→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   803→    }
   804→    pct += (rnd() - 0.5) * 0.12
   805→    out.push({
   806→      label: format(t, 'd.MM'),
   807→      pct: Math.max(0.2, Math.round(pct * 10) / 10),
   808→      forecast: null,
   809→      treated: treatDay > 0 && k === treatDay,
   810→    })
   811→  }
   812→  out[out.length - 1].pct = curPct
   813→
   814→  // свежая обработка: дип на последней точке (кривая резко падает)
   815→  if (treatedToday && out.length >= 2) {
   816→    out[out.length - 2].treated = true
   817→  }
   818→
   819→  // прогноз: через 14 дней (пунктир от текущей точки)
   820→  const future = new Date(nowMs + 14 * 86400000)
   821→  out.push({
   822→    label: format(future, 'd.MM'),
   823→    pct: null,
   824→    forecast: futurePct,
   825→    treated: false,
   826→  })
   827→  // сегмент начинается с последней исторической точки
   828→  out[out.length - 2].forecast = curPct
   829→  return out
   830→}
   831→
   832→/**
   833→ * Привесы за 7 дней: столбцы суточного привеса + ночная температура
   834→ * (ночные похолодания объясняют провалы приноса нектара).
   835→ */
   836→export function demoGainSeries(h: DemoHive, days = 7): Array<{ label: string; gainKg: number; nightTempC: number }> {
   837→  const rnd = seeded(4000 + h.id * 83)
   838→  const out: Array<{ label: string; gainKg: number; nightTempC: number }> = []
   839→  const nowMs = Date.now()
   840→  for (let k = days - 1; k >= 0; k--) {
   841→    const t = new Date(nowMs - k * 86400000)
   842→    // привес качается вокруг базового: погода и сила семьи
   843→    const gainKg = k === 0 ? h.base.dailyGainKg : Math.max(-0.1, h.base.dailyGainKg + (rnd() - 0.45) * 0.5)
   844→    // ночная температура: волна + случайные похолодания (2 прохладные ночи)
   845→    const wave = Math.sin(((days - k) / days) * Math.PI * 1.4) * 3
   846→    const dip = k === days - 2 || k === 1 ? -3.5 : 0
   847→    const nightTempC = 14 + wave + dip + (rnd() - 0.5) * 2
   848→    out.push({
   849→      label: format(t, 'd.MM'),
   850→      gainKg: Math.round(gainKg * 100) / 100,
   851→      nightTempC: Math.round(nightTempC * 10) / 10,
   852→    })
   853→  }
   854→  return out
   855→}
