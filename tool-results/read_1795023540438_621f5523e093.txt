    98→export function DashboardView() {
    99→  const mode = useAppState((s) => s.mode)
   100→  const setView = useAppState((s) => s.setView)
   101→  const hiveId = useAppState((s) => s.hiveId)
   102→  const setHive = useAppState((s) => s.setHive)
   103→  const { status, data, reload } = useRealData()
   104→
   105→  // Синхронизация с картой: ульи дашборда = точки на карте, показания — с датчиков.
   106→  // Опрос каждые 5 секунд — как на вкладке «Датчики», числа «живут» вместе с картой.
   107→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
   108→  const [wResp, setWResp] = useState<WeatherResponse | null>(null)
   109→  useEffect(() => {
   110→    if (mode !== 'demo') return
   111→    let alive = true
   112→    const load = async () => {
   113→      try {
   114→        const res = await fetch('/api/map-points', { cache: 'no-store' })
   115→        if (!res.ok) return
   116→        const j = (await res.json()) as { points: MapPointDto[] }
   117→        if (alive) setPoints(j.points)
   118→      } catch {
   119→        // датчики подтянутся на следующем опросе
   120→      }
   121→    }
   122→    void load()
   123→    const timer = setInterval(() => void load(), 5000)
   124→    return () => {
   125→      alive = false
   126→      clearInterval(timer)
   127→    }
   128→  }, [mode])
   129→
   130→  // Погода Open-Meteo — та же, что показывает панель погоды на дашборде
   131→  useEffect(() => {
   132→    if (mode !== 'demo') return
   133→    let alive = true
   134→    const load = async () => {
   135→      try {
   136→        const res = await fetch('/api/weather', { cache: 'no-store' })
   137→        if (!res.ok) return
   138→        const j = (await res.json()) as WeatherResponse
   139→        if (alive) setWResp(j)
   140→      } catch {
   141→        // агент честно скажет, если погоды нет
   142→      }
   143→    }
   144→    void load()
   145→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   146→    return () => {
   147→      alive = false
   148→      clearInterval(timer)
   149→    }
   150→  }, [mode])
   151→
   152→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   153→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   154→
   155→  // выбранный улей всегда существует в синхронизированном наборе
   156→  useEffect(() => {
   157→    if (mode === 'demo' && syncedHives && syncedHives.length > 0 && !syncedHives.some((h) => h.id === hiveId)) {
   158→      setHive(syncedHives[0].id)
   159→    }
   160→  }, [mode, syncedHives, hiveId, setHive])
   161→
   162→  if (mode === 'real') {
   163→    if (status === 'idle' || status === 'loading') {
   164→      return <DashboardSkeleton />
   165→    }
   166→    if (status === 'error') {
   167→      return (
   168→        <Card className="mx-auto max-w-lg border-red-100">
   169→          <CardContent className="p-6 text-center">
   170→            <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   171→            <h2 className="mt-2 font-bold text-gray-900">Не удалось загрузить данные</h2>
   172→            <p className="mt-1 text-sm text-gray-500">
   173→              Проверьте, что база данных доступна, и попробуйте ещё раз.
   174→            </p>
   175→            <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   176→              Повторить загрузку
   177→            </Button>
   178→          </CardContent>
   179→        </Card>
   180→      )
   181→    }
   182→    if (!data || data.hives.length === 0) {
   183→      return (
   184→        <Card className="mx-auto max-w-lg border-dashed border-red-300 bg-red-50/40">
   185→          <CardContent className="p-8 text-center">
   186→            <ClipboardList className="mx-auto h-8 w-8 text-red-300" />
   187→            <h2 className="mt-3 text-lg font-extrabold text-[#2A1A12]">В базе пока нет ульев</h2>
   188→            <p className="mx-auto mt-1.5 max-w-sm text-sm leading-relaxed text-gray-500">
   189→              Добавьте улья кнопкой «+ Новый улей» и загрузите телеметрию — файлом CSV, через форму
   190→              ручного ввода или автоматически по API-ключу. Агент сразу начнёт следить за семьями.
   191→            </p>
   192→            <Button onClick={() => setView('data')} className="mt-5 bg-gradient-to-r from-red-600 to-orange-500 hover:from-red-700 hover:to-orange-600">
   193→              Перейти в раздел «Данные»
   194→            </Button>
   195→          </CardContent>
   196→        </Card>
   197→      )
   198→    }
   199→    return (
   200→      <DashboardContent
   201→        mode="real"
   202→        hives={data.hives.map(apiHiveToHive)}
   203→        sites={data.sites.map(siteDtoToSite)}
   204→        weather={null}
   205→        pesticide={null}
   206→        events={data.events}
   207→      />
   208→    )
   209→  }
   210→
   211→  if (!syncedHives) return <DashboardSkeleton />
   212→  if (syncedHives.length === 0) return <EmptyMapState onOpenMap={() => setView('map')} />
   213→
   214→  return (
   215→    <DashboardContent
   216→      mode="demo"
   217→      hives={syncedHives}
   218→      sites={migrationSites}
   219→      weather={agentWeather?.days ?? null}
   220→      pesticide={pesticideAlert}
   221→      events={[]}
   222→    />
   223→  )
   224→}

   230→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/50" />
   231→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/40" />

   294→
   295→  // показатели пасеки целиком

   304→

   317→                <p className="text-sm font-bold uppercase tracking-wide text-white/90">
   318→                  Срочное предупреждение · пестициды
   319→                </p>
   320→                <p className="mt-1 text-sm leading-snug md:text-base">
   321→                  {pesticide.dateLabel} поле {pesticide.crop} {pesticide.field} обработают химикатом —{' '}

   687→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   688→function weightTotal(h: Hive): number {
   689→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   690→  const first = h.series[0]?.weightKg ?? null
   691→  if (w === null || first === null) return 0
   692→  return Math.round((w - first) * 10) / 10
   693→}
