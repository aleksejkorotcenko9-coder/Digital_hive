   120→function lastPoint(h: Hive) {
   121→  return h.series[h.series.length - 1] ?? null
   122→}
   123→
   124→function lastWeight(h: Hive): number | null {
   125→  return lastPoint(h)?.weightKg ?? null
   126→}
   127→
   128→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   129→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   130→function weightDelta(h: Hive, backDays: number): number | null {
   131→  const s = h.series
   132→  if (s.length === 0) return null
   133→  const lastPoint = s[s.length - 1]
   134→  if (lastPoint.weightKg === null) return null
   135→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   136→  let best: number | null = null
   137→  let bestDiff = Infinity
   138→  for (const p of s) {
   139→    if (p.weightKg === null || p === lastPoint) continue
   140→    const diff = Math.abs(new Date(p.date).getTime() - target)
   141→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   142→      bestDiff = diff
   143→      best = p.weightKg
   144→    }
   145→  }
   146→  if (best === null) return null
   147→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   148→}
   149→
   150→const delta2d = (h: Hive) => weightDelta(h, 2)
   151→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   152→
   153→function maxTempRecent(h: Hive, n = 3): number | null {
   154→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   155→  return temps.length ? Math.max(...temps) : null
   156→}
   157→
   158→function lastHumidity(h: Hive): number | null {
   159→  return lastPoint(h)?.humidityPct ?? null
   160→}
   161→
   162→/** Лидер пасеки по 30-дневному привесу. */
   163→function leaderOf(list: Hive[]): Hive | null {
   164→  const withW = list.filter((x) => delta30d(x) !== null)
   165→  if (!withW.length) return null
   166→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   167→}
   168→
   169→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   170→function spanDays(h: Hive): number {
   171→  if (h.series.length < 2) return h.series.length
   172→  const a = new Date(h.series[0].date).getTime()
   173→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   174→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   175→  return Math.round((b - a) / 86400000) + 1
   176→}
   177→
   178→function periodLabel(h: Hive, lang: Lang): string {
   179→  const d = spanDays(h)
   180→  if (d >= 25) return tr(lang, 'agent.period.month')
   181→  if (d === 1) return tr(lang, 'agent.period.day')
   182→  if (d === 0) return tr(lang, 'agent.period.dash')
   183→  return tr(lang, 'agent.period.days', { n: d })
   184→}

   186→/** Первая буква — заглавная (метрики вида «За месяц» и причины находок). */
   187→function cap(s: string): string {
   188→  return s.charAt(0).toUpperCase() + s.slice(1)
   189→}

   195→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   196→  return tr(lang, 'agent.swarm.prob', { p })

   212→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   213→  const l = ctx.lang ?? 'ru'
   214→  const status = hiveStatus(h, ctx.hives)
   215→  const d2 = delta2d(h)
   216→  const d30 = delta30d(h)
   217→  const t = lastPoint(h)?.tempInC ?? null
   218→  const hum = lastHumidity(h)
   219→
   220→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   221→    return {
   222→      id: `nodata-${h.id}`,
   223→      scenario: 'status',
   224→      hiveId: h.id,
   225→      title: tr(l, 'agent.find.nodata.title', { n: h.id }),
   226→      text: tr(l, 'agent.find.nodata.text', { n: h.id }),
   227→      basis: [tr(l, 'agent.basis.db'), tr(l, 'agent.basis.measurements', { n: h.id })],
   228→      urgency: 'this-week',
   229→      urgencyNote: tr(l, 'agent.note.loadData'),
   230→      probability: tr(l, 'agent.prob.nodata'),
   231→      metrics: [{ label: tr(l, 'agent.metric.measurements'), value: String(h.series.length) }],
   232→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   233→    }
   234→  }
   235→
   236→  if (status.level === 'risk' && d2 !== null) {
   237→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   238→    const now = lastWeight(h) ?? peak
   239→    return {
   240→      id: `swarm-${h.id}`,
   241→      scenario: 'swarm',
   242→      hiveId: h.id,
   243→      title: tr(l, 'agent.find.swarm.title'),
   244→      text: tr(l, 'agent.find.swarm.text', {
   245→        n: h.id,
   246→        w: fmtNum(Math.abs(d2)),
   247→        peak: fmtNum(peak),
   248→        now: fmtNum(now),
   249→        temp: t !== null ? tr(l, 'agent.find.swarm.temp', { t: fmtNum(t) }) : '',
   250→        hum: hum !== null ? tr(l, 'agent.find.swarm.hum', { h: fmtNum(hum, 0) }) : '',
   251→      }),
   252→      basis: [
   253→        tr(l, 'agent.basis.weightChart', { n: h.id }),
   254→        tr(l, 'agent.basis.weight2d', { w: fmtSigned(d2) }),
   255→        ...(t !== null ? [tr(l, 'agent.basis.tempNormCheck', { t: fmtNum(t) })] : []),
   256→        tr(l, 'agent.basis.seasonJuly'),
   257→      ],
   258→      urgency: 'today',
   259→      urgencyNote: tr(l, 'agent.note.inspectTodayEvening'),
   260→      probability: swarmProbability(d2, l),
   261→      metrics: [
   262→        { label: tr(l, 'agent.metric.weight2d'), value: tr(l, 'agent.value.kg', { w: fmtSigned(d2) }) },
   263→        ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   264→        ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),
   265→      ],
   266→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   267→    }
   268→  }
   269→
   270→  if (status.level === 'watch' && h.label !== undefined) {
   271→    const leader = leaderOf(ctx.hives)
   272→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   273→      ? tr(l, 'agent.find.slow.reasonSensor')
   274→      : tr(l, 'agent.find.slow.reasonLeader')
   275→    return {
   276→      id: `slow-${h.id}`,
   277→      scenario: 'status',
   278→      hiveId: h.id,
   279→      title: tr(l, 'agent.find.slow.title'),
   280→      text: tr(l, 'agent.find.slow.text', {
   281→        n: h.id,
   282→        w: fmtSigned(d30 ?? 0),
   283→        period: periodLabel(h, l),
   284→        leader:
   285→          leader && leader.id !== h.id
   286→            ? tr(l, 'agent.find.slow.leader', { n: leader.id, w: fmtSigned(delta30d(leader) ?? 0) })
   287→            : '',
   288→        temp: t !== null ? tr(l, 'agent.inline.temp', { t: fmtNum(t) }) : '',
   289→        reason: cap(reason),
   290→        extra: h.humidityOfflineFrom
   291→          ? tr(l, 'agent.find.slow.offline', { date: h.humidityOfflineFrom })
   292→          : tr(l, 'agent.find.slow.compare'),
   293→      }),
   294→      basis: [
   295→        tr(l, 'agent.basis.gainCompare'),
   296→        tr(l, 'agent.basis.slowGain', { n: h.id, w: fmtSigned(d30 ?? 0) }),
   297→        ...(h.humidityOfflineFrom ? [tr(l, 'agent.basis.humOffline', { date: h.humidityOfflineFrom })] : []),
   298→      ],
   299→      urgency: 'this-week',
   300→      urgencyNote: tr(l, 'agent.note.checkSensorDays'),
   301→      probability: tr(l, 'agent.prob.indirect'),
   302→      metrics: [
   303→        { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   304→        ...(leader && leader.id !== h.id
   305→          ? [{ label: tr(l, 'agent.metric.leaderN', { n: leader.id }), value: tr(l, 'agent.value.kg', { w: fmtSigned(delta30d(leader) ?? 0) }) }]
   306→          : []),
   307→        ...(hum !== null
   308→          ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }]
   309→          : [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.metric.nodata') }]),
   310→      ],
   311→      prefill: tr(l, 'agent.prefill.slowGain', { n: h.id }),
   312→    }
   313→  }
   314→
   315→  return {
   316→    id: `ok-${h.id}`,
   317→    scenario: 'status',
   318→    hiveId: h.id,
   319→    title: tr(l, 'agent.find.ok.title'),
   320→    text: tr(l, 'agent.find.ok.text', {
   321→      n: h.id,
   322→      w: fmtNum(lastWeight(h) ?? 0),
   323→      g: fmtSigned(d30 ?? 0),
   324→      period: periodLabel(h, l),
   325→      d2: fmtSigned(d2 ?? 0),
   326→      temp: t !== null ? tr(l, 'agent.find.ok.temp', { t: fmtNum(t) }) : '',
   327→    }),
   328→    basis: [
   329→      tr(l, 'agent.basis.weightChart', { n: h.id }),
   330→      tr(l, 'agent.basis.gain', { w: fmtSigned(d30 ?? 0) }),
   331→      tr(l, 'agent.basis.tempOkFamily'),
   332→    ],
   333→    urgency: 'ok',
   334→    urgencyNote: tr(l, 'agent.note.observe'),
   335→    metrics: [
   336→      { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   337→      ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   338→      ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),
   339→    ],
   340→    prefill: tr(l, 'agent.prefill.whatOk', { n: h.id }),
   341→  }
   342→}

   378→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   379→  const w = ctx.weather
   380→  if (!w || w.length === 0) return null
   381→  const good = w.filter((d) => d.flight === 'good')
   382→  const bad = w.filter((d) => d.flight === 'bad')
   383→  const tMax = Math.max(...w.map((d) => d.tMax))
   384→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   385→  const l = ctx.lang ?? 'ru'
   386→  return {
   387→    id: 'weather',
   388→    scenario: 'weather',
   389→    hiveId: null,
   390→    title: good.length
   391→      ? tr(l, 'agent.find.weather.title', { days: good.map((d) => dayMonth(d.date, l)).join(', ').slice(0, 80) })
   392→      : tr(l, 'agent.find.weather.titleBad'),
   393→    text: good.length
   394→      ? tr(l, 'agent.find.weather.textGood', {
   395→          list: good.map((d) => d.full).join(', '),
   396→          tmax: fmtSigned(tMax, 0),
   397→          wind: Math.max(...good.map((d) => d.windMs)),
   398→          bad: bad.length
   399→            ? tr(l, 'agent.find.weather.bad', {
   400→                days: bad.map((d) => d.full).join(tr(l, 'agent.misc.and')),
   401→                p: rainProb,
   402→                wind: Math.max(...bad.map((d) => d.windMs)),
   403→              })
   404→            : '',
   405→        })
   406→      : tr(l, 'agent.find.weather.textBad'),
   407→    basis: [tr(l, 'agent.basis.weather'), tr(l, 'agent.basis.bloomCalendar')],
   408→    urgency: 'ok',
   409→    urgencyNote: good.length ? tr(l, 'agent.note.useFlightWindow') : tr(l, 'agent.note.watchWeather'),
   410→    metrics: [
   411→      { label: tr(l, 'agent.metric.max'), value: tr(l, 'agent.value.tempC', { t: fmtSigned(tMax, 0) }) },
   412→      ...(bad.length ? [{ label: tr(l, 'agent.metric.rain'), value: tr(l, 'agent.value.uptoPct', { p: rainProb }) }] : []),
   413→      ...(ctx.sites.length
   414→        ? [{ label: tr(l, 'agent.metric.migrationCandidates'), value: String(ctx.sites.length) }]
   415→        : []),
   416→    ],
   417→    prefill: tr(l, 'agent.prefill.weather'),
   418→  }
   419→}

   421→/** «2026-07-20» → «20 июля» — форма «число + месяц» для заголовка находки. */
   422→function dayMonth(date: string, lang: Lang): string {
   423→  const m = Number(date.slice(5, 7))
   424→  return tr(lang, 'agent.date.dayMonth', {
   425→    d: Number(date.slice(8)),
   426→    m: m >= 1 && m <= 12 ? tr(lang, `agent.month.${m}`) : '',
   427→  })
   428→}

   430→function migrationFindingOf(ctx: AgentContext): AgentFinding | null {
   431→  const sites = [...ctx.sites].sort((a, b) => b.score - a.score)
   432→  if (!sites.length) return null
   433→  const best = sites[0]
   434→  const worst = sites[sites.length - 1]
   435→  const l = ctx.lang ?? 'ru'
   436→  const counted = `${sites.length} ${tr(
   437→    l,
   438→    sites.length === 1 ? 'agent.unit.site.one' : sites.length < 5 ? 'agent.unit.site.few' : 'agent.unit.site.many',
   439→  )}`

   473→/** Список находок: по выбранному улью + глобальные сценарии, отсортированы по срочности. */
   474→export function findingsFor(hiveId: number, ctx: AgentContext): AgentFinding[] {
   475→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   476→  const list: AgentFinding[] = []
   477→  if (hive) list.push(hiveFinding(hive, ctx))
   478→  const pesticide = pesticideFindingOf(ctx)
   479→  if (pesticide) list.push(pesticide)
   480→  const weatherF = weatherFindingOf(ctx)
   481→  if (weatherF) list.push(weatherF)
   482→  const migrationF = migrationFindingOf(ctx)
   483→  if (migrationF) list.push(migrationF)
   484→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   485→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   486→}

  1613→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
  1614→export function askAgent(
  1615→  question: string,
  1616→  currentHiveId: number,
  1617→  ctx: AgentContext,
  1618→): AgentMessage {
  1619→  const q = norm(question)
  1620→  const hiveId = extractHiveId(q, ctx)
  1621→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
  1622→
  1623→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
  1624→
  1625→  // Приветствие и сервисные фразы
  1626→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь|справк/)) {
  1627→    return greetingAnswer(ctx)
  1628→  }
  1629→  // Вопрос о самом агенте: модель и устройство не раскрываем
  1630→  if (has(/кто ты|кто такой|как тебя зовут|ты бот|чат ?бот|какая (ты )?модель|какой (ты )?модель|на (чем|чём) ты (работаешь|сделан|написан)|что ты за|нейросет|искусственн|\bllm\b|\bgpt\b/)) {
  1631→    return identityAnswer(ctx)
  1632→  }
  1633→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
  1634→    return thanksAnswer(ctx)
  1635→  }
  1636→
  1637→  // Здоровье пчёл и уход: болезни, лечение, профилактика, подкормки, сезонный уход.
  1638→  // Ветка стоит ДО пестицидов: «обработать от клеща» — про лечение пчёл, а не про поле.
  1639→  if (
  1640→    has(
  1641→      /боле[етн]|болезн|заболел|недуг|варроа|клещ|нозем|аскосфер|известков|мумии|гнил|восковая моль|(^|[^а-я])моль(?![а-я])|лечен|лечит|вылеч|бипин|апивар|акара|щавелев|оксал|заклещ|полоск|подкорм|сироп|канди|сахарн|кормушк|ветеринар|ветврач|ветстанц|ветслужб|ветлабор|дезинфекц|осирот|нет матк|безматоч|матка пропал|матка потерян|матка исчез|трутовк|понос|испачкан|зим(ов|н)|к зиме|в зиму|омшаник|пчел.{0,14}(умира|погиб|гибнут)/,
  1642→    )
  1643→  ) {
  1644→    return beeCareAnswer(question, ctx)
  1645→  }
  1646→
  1647→  // Сценарий пестицидов
  1648→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
  1649→    return pesticideAnswer(ctx)
  1650→  }
  1651→
  1652→  // Кочёвка
  1653→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
  1654→    return migrationAnswer(ctx)
  1655→  }
  1656→
  1657→  // Погода и медосбор (включая температуру ВОЗДЕУХА — не путать с температурой улья)
  1658→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|снег|мороз|когда вывоз|когда летать|температура (воздуха|за окном|на улице|снаружи)/)) {
  1659→    return weatherAnswer(ctx)
  1660→  }
  1661→
  1662→  if (!hive) {
  1663→    return fallbackAnswer(ctx)
  1664→  }
  1665→
  1666→  // Темы по ульям: если номер улья не назван — отвечаем сразу по ВСЕМ ульям
  1667→  if (has(/влажн|сырост|сыро/)) {
  1668→    return hiveId ? humidityAnswer(hive, ctx) : humidityAllAnswer(ctx)
  1669→  }
