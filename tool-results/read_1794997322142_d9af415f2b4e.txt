   610→   610→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   611→   611→function weightTotal(h: Hive): number {
   612→   612→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   613→   613→  const first = h.series[0]?.weightKg ?? null
   614→   614→  if (w === null || first === null) return 0
   615→   615→  return Math.round((w - first) * 10) / 10
   616→   616→}
