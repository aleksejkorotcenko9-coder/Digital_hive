   100→/* Метрики по рядам (безопасны к пропускам значений)                   */
   101→/* ------------------------------------------------------------------ */
   102→
   103→function lastPoint(h: Hive) {
   104→  return h.series[h.series.length - 1] ?? null
   105→}
   106→
   107→function lastWeight(h: Hive): number | null {
   108→  return lastPoint(h)?.weightKg ?? null
   109→}
   110→
   111→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   112→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   113→function weightDelta(h: Hive, backDays: number): number | null {
   114→  const s = h.series
   115→  if (s.length === 0) return null
   116→  const lastPoint = s[s.length - 1]
   117→  if (lastPoint.weightKg === null) return null
   118→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   119→  let best: number | null = null
   120→  let bestDiff = Infinity
   121→  for (const p of s) {
   122→    if (p.weightKg === null || p === lastPoint) continue
   123→    const diff = Math.abs(new Date(p.date).getTime() - target)
   124→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   125→      bestDiff = diff
   126→      best = p.weightKg
   127→    }
   128→  }
   129→  if (best === null) return null
   130→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   131→}
   132→
   133→const delta2d = (h: Hive) => weightDelta(h, 2)
   134→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   135→
   136→function maxTempRecent(h: Hive, n = 3): number | null {
   137→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   138→  return temps.length ? Math.max(...temps) : null
   139→}
   140→
   141→function lastHumidity(h: Hive): number | null {
   142→  return lastPoint(h)?.humidityPct ?? null
   143→}
   144→
   145→/** Лидер пасеки по 30-дневному привесу. */
   146→function leaderOf(list: Hive[]): Hive | null {
   147→  const withW = list.filter((x) => delta30d(x) !== null)
   148→  if (!withW.length) return null
   149→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   150→}
   151→
   152→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   153→function spanDays(h: Hive): number {
   154→  if (h.series.length < 2) return h.series.length
   155→  const a = new Date(h.series[0].date).getTime()
   156→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   157→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   158→  return Math.round((b - a) / 86400000) + 1
   159→}
   160→
   161→function periodLabel(h: Hive): string {
   162→  const d = spanDays(h)
   163→  if (d >= 25) return 'за месяц'
   164→  if (d === 1) return 'за день'
   165→  if (d === 0) return '—'
   166→  return `за ${d} дн.`
   167→}
   168→
   169→/** Порог срабатывания «падения веса» (признак возможного роения). */
   170→const SWARM_DROP = -1.0
   171→
   172→function swarmProbability(drop: number): string {
   173→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   174→  return `Вероятность подготовки к роению — около ${p} %`
   175→}

   181→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   182→  const status = hiveStatus(h, ctx.hives)
   183→  const d2 = delta2d(h)
   184→  const d30 = delta30d(h)
   185→  const t = lastPoint(h)?.tempInC ?? null
   186→  const hum = lastHumidity(h)
   187→
   188→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   189→    return {
   190→      id: `nodata-${h.id}`,
   191→      scenario: 'status',
   192→      hiveId: h.id,
   193→      title: `Улей №${h.id}: пока нет данных`,
   194→      text: `По улью №${h.id} в базе нет измерений. Я не буду гадать: как только появятся показания — загрузите их в разделе «Данные» (CSV, вручную или через API), и я сразу начну следить за весом, температурой и влажностью.`,
   195→      basis: ['База данных платформы', 'Измерения улья №' + h.id],
   196→      urgency: 'this-week',
   197→      urgencyNote: 'Загрузить данные',
   198→      probability: 'Оценка невозможна — данных нет',
   199→      metrics: [{ label: 'Измерений', value: String(h.series.length) }],
   200→      prefill: `Что происходит с ульем №${h.id}?`,
   201→    }
   202→  }
   203→
   204→  if (status.level === 'risk' && d2 !== null) {
   205→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   206→    const now = lastWeight(h) ?? peak
   207→    return {
   208→      id: `swarm-${h.id}`,
   209→      scenario: 'swarm',
   210→      hiveId: h.id,
   211→      title: 'Похоже, семья готовится к роению',
   212→      text: `За последние 2 дня вес улья №${h.id} снизился на ${fmtNum(Math.abs(d2))} кг — с ${fmtNum(peak)} до ${fmtNum(now)} кг, хотя до этого семья стабильно набирала мед.${
   213→        t !== null ? ` Температура внутри поднималась до ${fmtNum(t)} °C (норма около 34,5 °C).` : ''
   214→      }${hum !== null ? ` Влажность держится на отметке ${fmtNum(hum, 0)} % — выше обычной для этого улья.` : ''} Июль — пик сезона роения: часть пчёл с маткой может улететь, и медосбор остановится на 2–3 недели.`,
   215→      basis: [
   216→        `График веса улья №${h.id}`,
   217→        `Вес: ${fmtSigned(d2)} кг за 2 дня`,
   218→        ...(t !== null ? [`Температура: ${fmtNum(t)} °C при норме 34,5 °C`] : []),
   219→        'Сезон: июль — период роения',
   220→      ],
   221→      urgency: 'today',
   222→      urgencyNote: 'Осмотреть улей сегодня до вечера',
   223→      probability: swarmProbability(d2),
   224→      metrics: [
   225→        { label: 'Вес за 2 дня', value: `${fmtSigned(d2)} кг` },
   226→        ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   227→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   228→      ],
   229→      prefill: `Что происходит с ульем №${h.id}?`,
   230→    }
   231→  }
   232→
   233→  if (status.level === 'watch' && h.label !== undefined) {
   234→    const leader = leaderOf(ctx.hives)
   235→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   236→      ? 'отдельно: датчик влажности не отвечает'
   237→      : 'привес хуже, чем у лидера пасеки'
   238→    return {
   239→      id: `slow-${h.id}`,
   240→      scenario: 'status',
   241→      hiveId: h.id,
   242→      title: 'Слабый набор веса',
   243→      text: `Улей №${h.id} прибавил только ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}${
   244→        leader && leader.id !== h.id ? ` — у лидера (улей №${leader.id}) ${fmtSigned(delta30d(leader) ?? 0)} кг` : ''
   245→      }. Температура${t !== null ? ` ${fmtNum(t)} °C` : ''} в норме. Причин может быть две: слабая семья или бедная кормовая база вокруг. ${reason.charAt(0).toUpperCase() + reason.slice(1)}${
   246→        h.humidityOfflineFrom ? ` — молчит с ${h.humidityOfflineFrom}, возможно, сел аккумулятор` : ' — стоит сверить привес с соседними ульями'
   247→      }.`,
   248→      basis: [
   249→        'Сравнение привеса по ульям пасеки',
   250→        `Привес улья №${h.id}: ${fmtSigned(d30 ?? 0)} кг`,
   251→        ...(h.humidityOfflineFrom ? [`Датчик влажности: нет данных с ${h.humidityOfflineFrom}`] : []),
   252→      ],
   253→      urgency: 'this-week',
   254→      urgencyNote: 'Проверить датчик и семью в ближайшие дни',
   255→      probability: 'Диагноз по косвенным признакам — нужна проверка',
   256→      metrics: [
   257→        { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   258→        ...(leader && leader.id !== h.id
   259→          ? [{ label: `У лидера (№${leader.id})`, value: `${fmtSigned(delta30d(leader) ?? 0)} кг` }]
   260→          : []),
   261→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : [{ label: 'Влажность', value: 'нет данных' }]),
   262→      ],
   263→      prefill: `Почему в улье №${h.id} слабый привес?`,
   264→    }
   265→  }
   266→
   267→  return {
   268→    id: `ok-${h.id}`,
   269→    scenario: 'status',
   270→    hiveId: h.id,
   271→    title: 'Семья работает штатно',
   272→    text: `Улей №${h.id} развивается штатно: вес ${fmtNum(lastWeight(h) ?? 0)} кг, привес ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}, за последние 2 дня ${fmtSigned(d2 ?? 0)} кг.${
   273→      t !== null ? ` Температура ${fmtNum(t)} °C — в норме.` : ''
   274→    } Признаков роения нет: вес растёт ровно, без провалов.`,
   275→    basis: [
   276→      `График веса улья №${h.id}`,
   277→      `Привес: ${fmtSigned(d30 ?? 0)} кг`,
   278→      'Температура в норме семьи',
   279→    ],
   280→    urgency: 'ok',
   281→    urgencyNote: 'Продолжать наблюдение',
   282→    metrics: [
   283→      { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   284→      ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   285→      ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   286→    ],
   287→    prefill: `Что происходит с улье №${h.id}?`,
   288→  }
   289→}

   318→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   319→  const w = ctx.weather
   320→  if (!w || w.length === 0) return null
   321→  const good = w.filter((d) => d.flight === 'good')
   322→  const bad = w.filter((d) => d.flight === 'bad')
   323→  const tMax = Math.max(...w.map((d) => d.tMax))
   324→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   325→  return {
   326→    id: 'weather',
   327→    scenario: 'weather',
   328→    hiveId: null,
   329→    title: good.length
   330→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   331→      : 'Ближайшие дни — не для лёта',
   332→    text: good.length
   333→      ? `Ближайшие дни с хорошими условиями: ${good.map((d) => d.full).join(', ')} — до +${tMax} °C, ветер до ${Math.max(...good.map((d) => d.windMs))} м/с, осадки маловероятны.${
   334→          bad.length ? ` Затем, ${bad.map((d) => d.full).join(' и ')}, — дожди (до ${rainProb} %) и ветер до ${Math.max(...bad.map((d) => d.windMs))} м/с: пчёлы останутся в ульях, привес остановится.` : ''
   335→        }`
   336→      : `По прогнозу на неделю погода не летная: дожди и ветер. Пчёлы останутся в ульях, привес остановится — следите за запасами корма по графику веса.`,
   337→    basis: ['Прогноз погоды платформы', 'Календарь цветения медоносов — ГИС'],
   338→    urgency: 'ok',
   339→    urgencyNote: good.length ? 'Использовать окно лёта' : 'Наблюдать за погодой',
   340→    metrics: [
   341→      { label: 'Максимум', value: `+${tMax} °C` },
   342→      ...(bad.length ? [{ label: 'Дождь', value: `до ${rainProb} %` }] : []),
   343→      ...(ctx.sites.length
   344→        ? [{ label: 'Кандидатов кочёвки', value: String(ctx.sites.length) }]
   345→        : []),
   346→    ],
   347→    prefill: 'Когда пчёлам лучше летать за взятком?',
   348→  }
   349→}

   351→function monthGen(date: string): string {
   352→  // «2026-07-20» → «июля» (родительный падеж)
   353→  const months = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря']
   354→  const m = Number(date.slice(5, 7))
   355→  return months[m - 1] ?? ''
   356→}

   358→function migrationFindingOf(ctx: AgentContext): AgentFinding | null {
   359→  const sites = [...ctx.sites].sort((a, b) => b.score - a.score)
   360→  if (!sites.length) return null
   361→  const best = sites[0]
   362→  const worst = sites[sites.length - 1]
   363→  return {
   364→    id: 'migration',
   365→    scenario: 'migration',
   366→    hiveId: null,
   367→    title: `Лучшее место для кочёвки — «${best.name}»`,
   368→    text: `Сравнил ${sites.length} ${
   369→      sites.length === 1 ? 'участок' : sites.length < 5 ? 'участка' : 'участков'
   370→    } вокруг пасеки по 4 факторам: медоносы, погода, расстояние и риски. Лучший — «${best.name}»: ${best.honeyPlant.toLowerCase()} цветёт ${best.bloomWindow}, ${fmtNum(best.distanceKm)} км от пасеки, погода: ${best.weatherNote.toLowerCase()}.${
   371→      best.risks ? ` Учтите: ${best.risks.toLowerCase()}.` : ''
   372→    }${sites.length > 1 && worst.verdict === 'avoid' ? ` «${worst.name.replace(/^Донниковое поле \(/, '').replace(/\)$/, '')}» не советую: ${worst.risks.toLowerCase() || 'низкая оценка участка'}.` : ''}`,
   373→    basis: ['ГИС: медоносы и расстояния', 'Прогноз погоды по участкам', 'Реестр обработок полей'],
   374→    urgency: 'this-week',
   375→    urgencyNote: 'Решить до начала цветения',
   376→    metrics: [
   377→      { label: 'Лучший участок', value: `${fmtNum(best.score)} / 10` },
   378→      { label: 'Расстояние', value: `${fmtNum(best.distanceKm)} км` },
   379→      { label: 'Цветение', value: best.bloomWindow },
   380→    ],
   381→    prefill: 'Куда перевезти пасеку?',
   382→  }
   383→}

   385→/** Список находок: по выбранному улью + глобальные сценарии, отсортированы по срочности. */
   386→export function findingsFor(hiveId: number, ctx: AgentContext = DEMO_CONTEXT): AgentFinding[] {
   387→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   388→  const list: AgentFinding[] = []
   389→  if (hive) list.push(hiveFinding(hive, ctx))
   390→  const pesticide = pesticideFindingOf(ctx)
   391→  if (pesticide) list.push(pesticide)
   392→  const weatherF = weatherFindingOf(ctx)
   393→  if (weatherF) list.push(weatherF)
   394→  const migrationF = migrationFindingOf(ctx)
   395→  if (migrationF) list.push(migrationF)
   396→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   397→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   398→}

   408→function extractHiveId(q: string, ctx: AgentContext): number | null {
   409→  const m = q.match(/(?:№|улье|улья|улью|улье|hive)\s*№?\s*(\d+)/)
   410→  if (!m) return null
   411→  const id = Number(m[1])
   412→  return ctx.hives.some((h) => h.id === id) ? id : null
   413→}

   575→function weightAnswer(h: Hive, ctx: AgentContext): AgentMessage {
   576→  const d2 = delta2d(h)
   577→  const d30 = delta30d(h)
   578→  const w = lastWeight(h)

   580→
   581→  if (h.series.length === 0 || w === null) {
   582→    return {
   583→      scenario: 'status',
   584→      hiveId: h.id,
   585→      text: `По весу улья №${h.id} данных пока нет — не буду называть цифр. Внесите показание вручную в разделе «Данные» или загрузите CSV: после этого я покажу вес, привес за 2 дня и за месяц и сравню с остальными ульями.`,
   586→      basis: ['База данных платформы'],
   587→      urgency: 'this-week',
   588→      urgencyNote: 'Загрузить данные',
   589→      probability: 'Данных недостаточно',
   590→      metrics: [{ label: 'Вес', value: 'нет данных' }],
   591→    }
   592→  }
   593→
   594→  const others = ctx.hives
   595→    .map((x) => ({ x, d: delta30d(x) }))
   596→    .filter((o): o is { x: Hive; d: number } => o.d !== null)
   597→    .map((o) => `№${o.x.id} — ${fmtSigned(o.d)} кг`)
   598→    .join(', ')
   599→
   600→  return {
   601→    scenario: 'status',
   602→    hiveId: h.id,
   603→    text: `Вес улья №${h.id} сейчас ${fmtNum(w)} кг. За последние 2 дня ${fmtSigned(d2 ?? 0)} кг, ${periodLabel(h)} ${fmtSigned(d30 ?? 0)} кг.${risky ? ` Настораживает само падение веса при тёплой погоде.` : ''}\n\nДля сравнения по пасеке: ${others || 'весов с данными больше нет'}.    605→      `График веса улья №${h.id}`,

   892→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
   893→export function askAgent(
   894→  question: string,
   895→  currentHiveId: number,
   896→  ctx: AgentContext = DEMO_CONTEXT,
   897→): AgentMessage {
   898→  const q = norm(question)
   899→  const hiveId = extractHiveId(q, ctx)
   900→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
   901→
   902→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
   903→
   904→  // Приветствие и сервисные фразы
   905→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
   906→    return greetingAnswer(ctx)
   907→  }
   908→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
   909→    return thanksAnswer()
   910→  }
   911→
   912→  // Сценарий пестицидов
   913→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
   914→    return pesticideAnswer(ctx)
   915→  }
   916→
   917→  // Кочёвка
   918→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
   919→    return migrationAnswer(ctx)
   920→  }
   921→
   922→  // Погода и медосбор
   923→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|когда вывоз|когда летать/)) {
   924→    return weatherAnswer(ctx)
   925→  }
   926→
   927→  if (!hive) {
   928→    return fallbackAnswer()
   929→  }
   930→
   931→  // Вопрос про конкретный или выбранный улей
   932→  if (has(/влажн|сырост|сыро/)) {
   933→    return humidityAnswer(hive)
   934→  }
   935→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
   936→    return swarmAnswer(hive, ctx)
   937→  }
   938→  if (has(/температ|жар|тепло внутри/)) {
   939→    return climateAnswer(hive)
   940→  }
   941→  if (has(/вес|мед\b|меда|привес|набор|тяже/)) {
   942→    return weightAnswer(hive, ctx)
   943→  }
   944→  if (has(/что происходит|состояни|как (себя|дела)|статус|проверь|проблем|тревог|что с улеем|как улей|расскажи про улей/)) {
   945→    return statusAnswer(hive, ctx)
   946→  }
   947→
   948→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
   949→  if (hiveId !== null) {
   950→    return statusAnswer(hive, ctx)
   951→  }
   952→
   953→  return fallbackAnswer()
   954→}
