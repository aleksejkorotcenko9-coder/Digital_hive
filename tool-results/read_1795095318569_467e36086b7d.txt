   120→   120→function lastPoint(h: Hive) {
   121→   121→  return h.series[h.series.length - 1] ?? null
   122→   122→}
   123→   123→
   124→   124→function lastWeight(h: Hive): number | null {
   125→   125→  return lastPoint(h)?.weightKg ?? null

   127→   127→
   128→   128→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   129→   129→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   130→   130→function weightDelta(h: Hive, backDays: number): number | null {
   131→   131→  const s = h.series
   132→   132→  if (s.length === 0) return null
   133→   133→  const lastPoint = s[s.length - 1]
   134→   134→  if (lastPoint.weightKg === null) return null
   135→   135→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   136→   136→  let best: number | null = null
   137→   137→  let bestDiff = Infinity
   138→   138→  for (const p of s) {
   139→   139→    if (p.weightKg === null || p === lastPoint) continue
   140→   140→    const diff = Math.abs(new Date(p.date).getTime() - target)
   141→   141→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   142→   142→      bestDiff = diff
   143→   143→      best = p.weightKg
   144→   144→    }
   145→   145→  }
   146→   146→  if (best === null) return null
   147→   147→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   148→   148→}
   149→   149→
   150→   150→const delta2d = (h: Hive) => weightDelta(h, 2)
   151→   151→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   152→   152→
   153→   153→function maxTempRecent(h: Hive, n = 3): number | null {
   154→   154→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   155→   155→  return temps.length ? Math.max(...temps) : null
   156→   156→}
   157→   157→
   158→   158→function lastHumidity(h: Hive): number | null {
   159→   159→  return lastPoint(h)?.humidityPct ?? null
   160→   160→}
   161→   161→
   162→   162→/** Лидер пасеки по 30-дневному привесу. */
   163→   163→function leaderOf(list: Hive[]): Hive | null {
   164→   164→  const withW = list.filter((x) => delta30d(x) !== null)
   165→   165→  if (!withW.length) return null
   166→   166→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   167→   167→}
   168→   168→
   169→   169→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   170→   170→function spanDays(h: Hive): number {
   171→   171→  if (h.series.length < 2) return h.series.length
   172→   172→  const a = new Date(h.series[0].date).getTime()
   173→   173→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   174→   174→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   175→   175→  return Math.round((b - a) / 86400000) + 1
   176→   176→}
   177→   177→
   178→   178→function periodLabel(h: Hive, lang: Lang): string {
   179→   179→  const d = spanDays(h)
   180→   180→  if (d >= 25) return tr(lang, 'agent.period.month')
   181→   181→  if (d === 1) return tr(lang, 'agent.period.day')
   182→   182→  if (d === 0) return tr(lang, 'agent.period.dash')
   183→   183→  return tr(lang, 'agent.period.days', { n: d })
   184→   184→}

   186→   186→/** Первая буква — заглавная (метрики вида «За месяц» и причины находок). */
   187→   187→function cap(s: string): string {
   188→   188→  return s.charAt(0).toUpperCase() + s.slice(1)

   194→   194→function swarmProbability(drop: number, lang: Lang): string {
   195→   195→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   196→   196→  return tr(lang, 'agent.swarm.prob', { p })

   212→   212→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   213→   213→  const l = ctx.lang ?? 'ru'
   214→   214→  const status = hiveStatus(h, ctx.hives)
   215→   215→  const d2 = delta2d(h)
   216→   216→  const d30 = delta30d(h)
   217→   217→  const t = lastPoint(h)?.tempInC ?? null
   218→   218→  const hum = lastHumidity(h)
   219→   219→
   220→   220→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   221→   221→    return {
   222→   222→      id: `nodata-${h.id}`,
   223→   223→      scenario: 'status',
   224→   224→      hiveId: h.id,
   225→   225→      title: tr(l, 'agent.find.nodata.title', { n: h.id }),
   226→   226→      text: tr(l, 'agent.find.nodata.text', { n: h.id }),
   227→   227→      basis: [tr(l, 'agent.basis.db'), tr(l, 'agent.basis.measurements', { n: h.id })],
   228→   228→      urgency: 'this-week',
   229→   229→      urgencyNote: tr(l, 'agent.note.loadData'),
   230→   230→      probability: tr(l, 'agent.prob.nodata'),
   231→   231→      metrics: [{ label: tr(l, 'agent.metric.measurements'), value: String(h.series.length) }],
   232→   232→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   233→   233→    }
   234→   234→  }
   235→   235→
   236→   236→  if (status.level === 'risk' && d2 !== null) {
   237→   237→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   238→   238→    const now = lastWeight(h) ?? peak
   239→   239→    return {
   240→   240→      id: `swarm-${h.id}`,
   241→   241→      scenario: 'swarm',
   242→   242→      hiveId: h.id,
   243→   243→      title: tr(l, 'agent.find.swarm.title'),
   244→   244→      text: tr(l, 'agent.find.swarm.text', {
   245→   245→        n: h.id,
   246→   246→        w: fmtNum(Math.abs(d2)),
   247→   247→        peak: fmtNum(peak),
   248→   248→        now: fmtNum(now),
   249→   249→        temp: t !== null ? tr(l, 'agent.find.swarm.temp', { t: fmtNum(t) }) : '',
   250→   250→        hum: hum !== null ? tr(l, 'agent.find.swarm.hum', { h: fmtNum(hum, 0) }) : '',
   251→   251→      }),
   252→   252→      basis: [
   253→   253→        tr(l, 'agent.basis.weightChart', { n: h.id }),
   254→   254→        tr(l, 'agent.basis.weight2d', { w: fmtSigned(d2) }),
   255→   255→        ...(t !== null ? [tr(l, 'agent.basis.tempNormCheck', { t: fmtNum(t) })] : []),
   256→   256→        tr(l, 'agent.basis.seasonJuly'),
   257→   257→      ],
   258→   258→      urgency: 'today',
   259→   259→      urgencyNote: tr(l, 'agent.note.inspectTodayEvening'),
   260→   260→      probability: swarmProbability(d2, l),
   261→   261→      metrics: [
   262→   262→        { label: tr(l, 'agent.metric.weight2d'), value: tr(l, 'agent.value.kg', { w: fmtSigned(d2) }) },
   263→   263→        ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   264→   264→        ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),
   265→   265→      ],
   266→   266→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   267→   267→    }
   268→   268→  }
   269→   269→
   270→   270→  if (status.level === 'watch' && h.label !== undefined) {
   271→   271→    const leader = leaderOf(ctx.hives)
   272→   272→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   273→   273→      ? tr(l, 'agent.find.slow.reasonSensor')
   274→   274→      : tr(l, 'agent.find.slow.reasonLeader')
   275→   275→    return {
   276→   276→      id: `slow-${h.id}`,
   277→   277→      scenario: 'status',
   278→   278→      hiveId: h.id,
   279→   279→      title: tr(l, 'agent.find.slow.title'),
   280→   280→      text: tr(l, 'agent.find.slow.text', {
   281→   281→        n: h.id,
   282→   282→        w: fmtSigned(d30 ?? 0),
   283→   283→        period: periodLabel(h, l),
   284→   284→        leader:
   285→   285→          leader && leader.id !== h.id
   286→   286→            ? tr(l, 'agent.find.slow.leader', { n: leader.id, w: fmtSigned(delta30d(leader) ?? 0) })
   287→   287→            : '',
   288→   288→        temp: t !== null ? tr(l, 'agent.inline.temp', { t: fmtNum(t) }) : '',
   289→   289→        reason: cap(reason),
   290→   290→        extra: h.humidityOfflineFrom
   291→   291→          ? tr(l, 'agent.find.slow.offline', { date: h.humidityOfflineFrom })
   292→   292→          : tr(l, 'agent.find.slow.compare'),
   293→   293→      }),
   294→   294→      basis: [
   295→   295→        tr(l, 'agent.basis.gainCompare'),
   296→   296→        tr(l, 'agent.basis.slowGain', { n: h.id, w: fmtSigned(d30 ?? 0) }),
   297→   297→        ...(h.humidityOfflineFrom ? [tr(l, 'agent.basis.humOffline', { date: h.humidityOfflineFrom })] : []),
   298→   298→      ],
   299→   299→      urgency: 'this-week',
   300→   300→      urgencyNote: tr(l, 'agent.note.checkSensorDays'),
   301→   301→      probability: tr(l, 'agent.prob.indirect'),
   302→   302→      metrics: [
   303→   303→        { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   304→   304→        ...(leader && leader.id !== h.id
   305→   305→          ? [{ label: tr(l, 'agent.metric.leaderN', { n: leader.id }), value: tr(l, 'agent.value.kg', { w: fmtSigned(delta30d(leader) ?? 0) }) }]
   306→   306→          : []),
   307→   307→        ...(hum !== null
   308→   308→          ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }]
   309→   309→          : [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.metric.nodata') }]),
   310→   310→      ],
   311→   311→      prefill: tr(l, 'agent.prefill.slowGain', { n: h.id }),
   312→   312→    }
   313→   313→  }
   314→   314→
   315→   315→  return {
   316→   316→    id: `ok-${h.id}`,
   317→   317→    scenario: 'status',
   318→   318→    hiveId: h.id,
   319→   319→    title: tr(l, 'agent.find.ok.title'),
   320→   320→    text: tr(l, 'agent.find.ok.text', {
   321→   321→      n: h.id,
   322→   322→      w: fmtNum(lastWeight(h) ?? 0),
   323→   323→      g: fmtSigned(d30 ?? 0),
   324→   324→      period: periodLabel(h, l),
   325→   325→      d2: fmtSigned(d2 ?? 0),
   326→   326→      temp: t !== null ? tr(l, 'agent.find.ok.temp', { t: fmtNum(t) }) : '',
   327→   327→    }),
   328→   328→    basis: [
   329→   329→      tr(l, 'agent.basis.weightChart', { n: h.id }),
   330→   330→      tr(l, 'agent.basis.gain', { w: fmtSigned(d30 ?? 0) }),
   331→   331→      tr(l, 'agent.basis.tempOkFamily'),
   332→   332→    ],
   333→   333→    urgency: 'ok',
   334→   334→    urgencyNote: tr(l, 'agent.note.observe'),
   335→   335→    metrics: [
   336→   336→      { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   337→   337→      ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   338→   338→      ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),
   339→   339→    ],
   340→   340→    prefill: tr(l, 'agent.prefill.whatOk', { n: h.id }),
   341→   341→  }
   342→   342→}

   378→   378→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   379→   379→  const w = ctx.weather
   380→   380→  if (!w || w.length === 0) return null
   381→   381→  const good = w.filter((d) => d.flight === 'good')

   474→   474→export function findingsFor(hiveId: number, ctx: AgentContext): AgentFinding[] {
   475→   475→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   476→   476→  const list: AgentFinding[] = []
   477→   477→  if (hive) list.push(hiveFinding(hive, ctx))
   478→   478→  const pesticide = pesticideFindingOf(ctx)
   479→   479→  if (pesticide) list.push(pesticide)
   480→   480→  const weatherF = weatherFindingOf(ctx)
   481→   481→  if (weatherF) list.push(weatherF)
   482→   482→  const migrationF = migrationFindingOf(ctx)
   483→   483→  if (migrationF) list.push(migrationF)
   484→   484→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   485→   485→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   486→   486→}
