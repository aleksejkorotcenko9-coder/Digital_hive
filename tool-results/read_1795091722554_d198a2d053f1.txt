    71→/** «2026-08-29» → «29 августа 2026» (UTC). */
    72→function fmtDateLabel(key: string): string {
    73→  const [y, m, d] = key.split('-').map(Number)
    74→  return `${d} ${MONTHS_GEN[m - 1]} ${y}`
    75→}

    77→/** «2026-08» → «Август 2026». */
    78→function fmtMonthLabel(key: string): string {
    79→  const [y, m] = key.split('-').map(Number)
    80→  return `${MONTHS_NOM[m - 1]} ${y}`
    81→}

    83→/** ISO → «HH:MM» (UTC). */
    84→function fmtTime(iso: string): string {
    85→  const d = new Date(iso)
    86→  return `${pad2(d.getUTCHours())}:${pad2(d.getUTCMinutes())}`
    87→}
    88→
    89→const dayKeyOf = (d: Date) => d.toISOString().slice(0, 10)
    90→const parseDayKey = (key: string) => new Date(`${key}T00:00:00Z`)
    91→
    92→function addDaysKey(key: string, days: number): string {
    93→  const d = parseDayKey(key)
    94→  d.setUTCDate(d.getUTCDate() + days)
    95→  return dayKeyOf(d)
    96→}

    98→function addMonthsKey(key: string, months: number): string {
    99→  const [y, m] = key.split('-').map(Number)
   100→  const d = new Date(Date.UTC(y, m - 1 + months, 1))
   101→  return `${d.getUTCFullYear()}-${pad2(d.getUTCMonth() + 1)}`
   102→}

   168→/** Стрелка дельты: зелёный рост, красное падение, серый ноль. */
   169→function Delta({ v, unit }: { v: number; unit: string }) {
   170→  const eps = unit === '°C' ? 0.05 : unit === '%' ? 0.5 : 0.05
   171→  const up = v > eps
   172→  const down = v < -eps
   173→  return (
   174→    <span
   175→      className={cn(
   176→        'inline-flex items-center gap-0.5 font-semibold',
   177→        up && 'text-emerald-600',
   178→        down && 'text-red-600',
   179→        !up && !down && 'text-gray-400',
   180→      )}
   181→    >
   182→      {up ? '↑' : down ? '↓' : '·'} {up || down ? `${v > 0 ? '+' : ''}${unit === '%' ? Math.round(v) : v.toFixed(1)}` : '0'}
   183→    </span>
   184→  )
   185→}

   383→  const [data, setData] = useState<HistoryResponse | null>(null)
   384→  const [calOpen, setCalOpen] = useState(false)
   385→  const [reloadTick, setReloadTick] = useState(0)
   386→
   387→  const [hoverPop, setHoverPop] = useState<PopState | null>(null)
   388→  const [pinnedPop, setPinnedPop] = useState<PopState | null>(null)
   389→  const leaveTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   390→
   391→  const range = data?.range ?? { from: '2026-07-01', to: '2026-08-29' }
   392→
   393→  /* Загрузка: состояние «загрузки/ошибки» производное от ключа периода —
   394→     без setState внутри тела эффекта (правило react-hooks). */
   395→  const defaultYear = range.to.slice(0, 4)
   396→  const currentKey = `${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`
   397→  const [loadedKey, setLoadedKey] = useState<string | null>(null)
   398→  const [errState, setErrState] = useState<{ key: string; message: string } | null>(null)
   399→  const loading = loadedKey !== currentKey
   400→  const error = errState && errState.key === currentKey ? errState.message : null
   401→
   402→  useEffect(() => {
   403→    const ctrl = new AbortController()
   404→    const q = dateKey
   405→      ? scope === 'day'
   406→        ? `date=${dateKey}`
   407→        : scope === 'month'
   408→          ? `month=${dateKey}`
   409→          : `year=${dateKey}`
   410→      : scope === 'year'
   411→        ? `year=${defaultYear}`
   412→        : ''
   413→    fetch(`/api/history?scope=${scope}${q ? `&${q}` : ''}`, { cache: 'no-store', signal: ctrl.signal })
   414→      .then(async (res) => {
   415→        const j = (await res.json().catch(() => ({}))) as HistoryResponse & { error?: string }
   416→        if (!res.ok) throw new Error(j.error ?? `HTTP ${res.status}`)
   417→        return j
   418→      })
   419→      .then((j) => {
   420→        if (ctrl.signal.aborted) return
   421→        setData(j)
   422→        setHoverPop(null)
   423→        setPinnedPop(null)
   424→        setErrState(null)
   425→        const resolvedDate = dateKey ?? j.date
   426→        setLoadedKey(`${scope}|${resolvedDate}|${reloadTick}`)
   427→        if (!dateKey) setDateKey(j.date)
   428→      })
   429→      .catch((e: unknown) => {
   430→        if (e instanceof DOMException && e.name === 'AbortError') return
   431→        setErrState({
   432→          key: `${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`,
   433→          message: e instanceof Error ? e.message : 'Не удалось загрузить журнал',
   434→        })
   435→        setLoadedKey(`${scope}|${dateKey ?? (scope === 'year' ? defaultYear : 'latest')}|${reloadTick}`)
   436→      })
   437→    return () => ctrl.abort()
   438→  }, [scope, dateKey, reloadTick, defaultYear])
   439→
   440→  /* ИИ-анализ периода: запускается, когда журнал загружен, и следует за
   441→     выбранным периодом и ульем. Анализ делает /api/history/ai — агент смотрит
   442→     на те же данные, что и графики, и объясняет, как исправить отклонения. */
   443→  const [aiData, setAiData] = useState<HistoryAiResponse | null>(null)
   444→  const [aiKey, setAiKey] = useState<string | null>(null)
   445→  const [aiErr, setAiErr] = useState<string | null>(null)
   446→  const expectedAiKey = data ? `${scope}|${data.date}|${pointFilter}|${reloadTick}` : null
   447→  const aiLoading = expectedAiKey !== null && aiKey !== expectedAiKey
   448→
   449→  useEffect(() => {
   450→    if (!data || loading) return
   451→    const key = `${scope}|${data.date}|${pointFilter}|${reloadTick}`
   452→    if (aiKey === key) return
   453→    const ctrl = new AbortController()
   454→    const dateParam =
   455→      scope === 'day' ? `date=${data.date}` : scope === 'month' ? `month=${data.date}` : `year=${data.date}`
   456→    const pointQ = pointFilter !== 'all' ? `&pointId=${encodeURIComponent(pointFilter)}` : ''
   457→    fetch(`/api/history/ai?scope=${scope}&${dateParam}${pointQ}`, { cache: 'no-store', signal: ctrl.signal })
   458→      .then(async (res) => {
   459→        const j = (await res.json().catch(() => ({}))) as HistoryAiResponse & { error?: string }
   460→        if (!res.ok) throw new Error(j.error ?? `HTTP ${res.status}`)
   461→        return j
   462→      })
   463→      .then((j) => {
   464→        if (ctrl.signal.aborted) return
   465→        setAiData(j)
   466→        setAiKey(key)
   467→        setAiErr(null)
   468→      })
   469→      .catch((e: unknown) => {
   470→        if (e instanceof DOMException && e.name === 'AbortError') return
   471→        setAiKey(key)
   472→        setAiErr(e instanceof Error ? e.message : 'Не удалось получить анализ')
   473→      })
   474→    return () => ctrl.abort()
   475→  }, [data, loading, scope, pointFilter, reloadTick, aiKey])
   476→
   477→  /* Всплывающие таблицы */
   478→  const openPop = useCallback(
   479→    (el: HTMLElement, title: string, subtitle: string | undefined, table: React.ReactNode) => {
   480→      if (leaveTimer.current) {
   481→        clearTimeout(leaveTimer.current)
   482→        leaveTimer.current = null
   483→      }
   484→      const r = el.getBoundingClientRect()
   485→      setHoverPop({ rect: { left: r.left, right: r.right, top: r.top }, title, subtitle, table })
   486→    },
   487→    [],
   488→  )
   489→  const closePopSoon = useCallback(() => {
   490→    if (leaveTimer.current) clearTimeout(leaveTimer.current)
   491→    leaveTimer.current = setTimeout(() => setHoverPop(null), 180)
   492→  }, [])
   493→  const keepPop = useCallback(() => {
   494→    if (leaveTimer.current) {
   495→      clearTimeout(leaveTimer.current)
   496→      leaveTimer.current = null
   497→    }
   498→  }, [])
   499→  const pinPop = useCallback(
   500→    (el: HTMLElement, title: string, subtitle: string | undefined, table: React.ReactNode) => {

   502→      const st = { rect: { left: r.left, right: r.right, top: r.top }, title, subtitle, table }

   504→        prev && prev.title === title && prev.rect.top === st.rect.top ? null : st,
   505→      )
   506→    },
   507→    [],
   508→  )
   509→  useEffect(() => () => void (leaveTimer.current && clearTimeout(leaveTimer.current)), [])
   510→
   511→  /* Данные журнала с каскадным фильтром «пасека → улей» */
   512→  const groups = data?.groups ?? []
   513→  const pointName = useMemo(() => {

   519→  /** Попадает ли улей в текущий выбор пасеки/улья. */

   578→    const avgT = pts.reduce((s, p) => s + p.avgTempC * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)
   579→    const avgH = pts.reduce((s, p) => s + p.avgHumidityPct * p.logs, 0) / pts.reduce((s, p) => s + p.logs, 0)

   581→    const alarms = pts.reduce((s, p) => s + p.alarms, 0)
   582→    const logs = pts.reduce((s, p) => s + p.logs, 0)

   595→          fmtTime(l.recordedAt),
   596→          `${l.weightKg.toFixed(1)} кг`,
   597→          <Delta key="dw" v={prev ? l.weightKg - prev.weightKg : 0} unit="кг" />,
   598→          `${l.tempC.toFixed(1)}°C`,
   599→          <Delta key="dt" v={prev ? l.tempC - prev.tempC : 0} unit="°C" />,
   600→          `${l.humidityPct}%`,
   601→          <Delta key="dh" v={prev ? l.humidityPct - prev.humidityPct : 0} unit="%" />,
   602→        ]

   604→      return (
   605→        <MiniTable
   606→          head={['Час', 'Вес', 'Δ', 'Темп.', 'Δ', 'Влажн.', 'Δ']}

   611→    },
   612→    [data],
   613→  )

   615→  /** Снимок всех ульев в конкретный час (в рамках выбранной пасеки/улья). */
   616→  const hourSnapshotTable = useCallback(
   617→    (hour: number) => {
   618→      const logs = filteredLogs.filter((l) => new Date(l.recordedAt).getUTCHours() === hour)
   619→      const rows = logs.map((l) => [
   620→        pointName.get(l.pointId) ?? 'Улей',
   621→        `${l.weightKg.toFixed(1)} кг`,
   622→        `${l.tempC.toFixed(1)}°C`,
   623→        `${l.humidityPct}%`,
   624→        l.alarm ? <span key="a" className="inline-flex items-center gap-1 font-bold text-red-600"><TriangleAlert className="h-3 w-3" /> тревога</span> : <span key="a" className="text-emerald-600">норма</span>,
   625→      ])
   626→      return <MiniTable head={['Улей', 'Вес', 'Темп.', 'Влажн.', 'Статус']} rows={rows} />
   627→    },
   628→    [filteredLogs, pointName],
   629→  )
   630→
   631→  /** Сводка по ульям за день (месячный вид). */
   632→  const monthDayTable = useCallback(
   633→    (date: string) => {
   634→      const day = data?.days?.find((d) => d.date === date)
   635→      const per = (day?.perPoint ?? []).filter((p) => matchPoint(p.pointId))
   636→      const rows = per.map((p) => [
   637→        pointName.get(p.pointId) ?? 'Улей',
   638→        `${p.startWeightKg.toFixed(1)} кг`,
   639→        `${p.endWeightKg.toFixed(1)} кг`,

   644→        `${p.minTempC.toFixed(1)}…${p.maxTempC.toFixed(1)}°C`,

   646→        `${p.logs}`,
   647→        p.alarms > 0 ? <span key="a" className="font-bold text-red-600">{p.alarms}</span> : <span key="a" className="text-gray-400">0</span>,
   648→      ])
   649→      return (
   650→        <MiniTable
   651→          head={['Улей', 'Вес утро', 'Вес вечер', 'Δ за день', 'Ср t°', 't макс/мин', 'Ср влажн.', 'Записей', 'Событий']}
   652→          rows={rows}
   653→        />
   654→      )
   655→    },
   656→    [data, matchPoint, pointName],

   660→  const yearMonthTable = useCallback(
   661→    (month: string) => {
   662→      const m = data?.months?.find((x) => x.month === month)
   663→      const per = (m?.perPoint ?? []).filter((p) => matchPoint(p.pointId))
   664→      const rows = per.map((p) => [
   665→        pointName.get(p.pointId) ?? 'Улей',
   666→        `${p.avgTempC.toFixed(1)}°C`,
   667→        `${p.avgHumidityPct}%`,
   668→        <span key="d" className={cn('font-bold', p.deltaWeightKg >= 0 ? 'text-emerald-600' : 'text-red-600')}>
   669→          {p.deltaWeightKg >= 0 ? '+' : ''}{p.deltaWeightKg.toFixed(1)} кг

   672→        p.alarms > 0 ? <span key="a" className="font-bold text-red-600">{p.alarms}</span> : <span key="a" className="text-gray-400">0</span>,
   673→      ])

   675→    },
   676→    [data, matchPoint, pointName],

   679→  /* Переключение периодов */

   681→  const clampDayKey = (key: string) =>

   685→  const asDayKey = (k: string) => {
   686→    if (k.length === 10) return k
   687→    if (k.length === 7) return `${k}-01`
   688→    return `${k}-${range.from.slice(5, 7)}-01`
   689→  }
   690→
   691→  const clampMonthKey = (key: string) => {
   692→    const from = range.from.slice(0, 7)
   693→    const to = range.to.slice(0, 7)

   695→  }
   696→
   697→  const switchScope = (next: HistoryScope) => {
   698→    if (next === scope) return

   701→    if (next === 'day') key = clampDayKey(asDayKey(cur))
   702→    else if (next === 'month') key = clampMonthKey(cur.slice(0, 7))
   703→    else {
   704→      const y = Number(cur.slice(0, 4))
   705→      const minY = Number(range.from.slice(0, 4))
   706→      const maxY = Number(range.to.slice(0, 4))
   707→      key = String(Math.max(minY, Math.min(maxY, y)))

   709→    setScope(next)
   710→    setDateKey(key)
   711→  }

   718→      setDateKey(mk < range.from.slice(0, 7) ? range.to.slice(0, 7) : mk > range.to.slice(0, 7) ? range.from.slice(0, 7) : mk)
   719→    } else {
   720→      const y = Number(cur) + dir
   721→      const minY = Number(range.from.slice(0, 4))
   722→      const maxY = Number(range.to.slice(0, 4))
   723→      setDateKey(String(Math.max(minY, Math.min(maxY, y))))
   724→    }
   725→  }

   728→    ? 'Последние записи'

   734→

   736→    ? false
   737→    : scope === 'day'
   738→      ? dateKey <= range.from

   740→        ? dateKey <= range.from.slice(0, 7)
   741→        : Number(dateKey) <= Number(range.from.slice(0, 4))

   743→    ? false
   744→    : scope === 'day'
   745→      ? dateKey >= range.to
   746→      : scope === 'month'
   747→        ? dateKey >= range.to.slice(0, 7)
   748→        : Number(dateKey) >= Number(range.to.slice(0, 4))

   753→    return parseDayKey(key)
   754→  }, [dateKey, scope, calOpen, range])
   755→
   756→  const hasSensorPoints = (data?.points.length ?? 0) > 0
   757→  const showPop = pinnedPop ?? hoverPop
   758→
   759→  /* ---------------------------------------------------------------- */
   760→
   761→  return (

   766→        .history-scroll::-webkit-scrollbar-thumb { background: #f0d5c8; border-radius: 999px }
   767→        .history-scroll::-webkit-scrollbar-track { background: transparent }

   772→      <div className="flex flex-wrap items-center justify-between gap-2">
   773→        <div>
   774→          <h1 className="flex items-center gap-2 text-2xl font-extrabold tracking-tight text-[#2A1A12]">
   775→            <History className="h-6 w-6 text-red-600" /> История событий
   776→          </h1>
   777→          <p className="mt-1 max-w-2xl text-sm text-gray-500">
   778→            Журнал датчиков ульев: температура, влажность и масса фиксируются каждый час.
   779→            Выберите пасеку и улей кнопкой справа, наведите на запись — всплывёт таблица изменений,
   780→            а ИИ-агент следит за показателями и подсказывает, что делать при отклонениях.
   781→          </p>
   782→        </div>
   783→        <span className="inline-flex items-center gap-1.5 rounded-full border border-red-100 bg-white px-3 py-1.5 text-[11px] font-semibold text-gray-400 shadow-sm dark:bg-[#221814]">
   784→          <MousePointerClick className="h-3.5 w-3.5 text-red-500" /> клик закрепляет таблицу
   785→        </span>
   786→      </div>
   787→
   788→      {/* Панель периодов */}
   789→      <Card className="border-red-100 dark:bg-[#221814]">
   790→        <CardContent className="flex flex-wrap items-center gap-2.5 p-3.5">
   791→          {/* День / Месяц / Год */}
   792→          <div className="flex rounded-full border border-red-100 bg-red-50/50 p-1 dark:border-[#2C211C] dark:bg-[#2C211C]">
   793→            {([
   794→              { id: 'day', label: 'День' },
   795→              { id: 'month', label: 'Месяц' },
   796→              { id: 'year', label: 'Год' },
   797→            ] as const).map((t) => (
   798→              <button
   799→                key={t.id}
   800→                onClick={() => switchScope(t.id)}
   801→                aria-pressed={scope === t.id}
   802→                className={cn(
   803→                  'rounded-full px-4 py-1.5 text-xs font-extrabold transition-all',
   804→                  scope === t.id
   805→                    ? 'bg-gradient-to-r from-red-600 to-orange-500 text-white shadow-md shadow-red-100'
   806→                    : 'text-gray-500 hover:text-red-600',
   807→                )}
   808→              >
   809→                {t.label}
   810→              </button>
   811→            ))}
   812→          </div>
   813→
   814→          {/* Шаг назад/вперёд */}
   815→          <div className="flex items-center gap-1">

   818→              variant="outline"
   819→              className="h-9 w-9 rounded-full border-red-100"

   821→              onClick={() => step(-1)}
   822→              aria-label="Предыдущий период"

   824→              <ChevronLeft className="h-4 w-4" />

   826→            <span className="min-w-[170px] text-center text-sm font-extrabold tabular-nums text-[#2A1A12] dark:text-[#E8D9CF]">
   827→              {periodLabel}
   828→            </span>
   829→            <Button
   830→              size="icon"
   831→              variant="outline"
   832→              className="h-9 w-9 rounded-full border-red-100"
   833→              disabled={atEnd}
   834→              onClick={() => step(1)}
   835→              aria-label="Следующий период"

   837→              <ChevronRight className="h-4 w-4" />
