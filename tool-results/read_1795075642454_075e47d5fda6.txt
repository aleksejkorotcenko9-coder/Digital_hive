   115→   115→function lastPoint(h: Hive) {
   116→   116→  return h.series[h.series.length - 1] ?? null
   117→   117→}
   118→   118→
   119→   119→function lastWeight(h: Hive): number | null {
   120→   120→  return lastPoint(h)?.weightKg ?? null
   121→   121→}
   122→   122→
   123→   123→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   124→   124→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   125→   125→function weightDelta(h: Hive, backDays: number): number | null {
   126→   126→  const s = h.series
   127→   127→  if (s.length === 0) return null
   128→   128→  const lastPoint = s[s.length - 1]
   129→   129→  if (lastPoint.weightKg === null) return null
   130→   130→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   131→   131→  let best: number | null = null
   132→   132→  let bestDiff = Infinity
   133→   133→  for (const p of s) {
   134→   134→    if (p.weightKg === null || p === lastPoint) continue
   135→   135→    const diff = Math.abs(new Date(p.date).getTime() - target)
   136→   136→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   137→   137→      bestDiff = diff
   138→   138→      best = p.weightKg
   139→   139→    }
   140→   140→  }
   141→   141→  if (best === null) return null
   142→   142→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   143→   143→}
   144→   144→
   145→   145→const delta2d = (h: Hive) => weightDelta(h, 2)
   146→   146→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   147→   147→
   148→   148→function maxTempRecent(h: Hive, n = 3): number | null {
   149→   149→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   150→   150→  return temps.length ? Math.max(...temps) : null
   151→   151→}

   153→   153→function lastHumidity(h: Hive): number | null {
   154→   154→  return lastPoint(h)?.humidityPct ?? null
   155→   155→}

   157→   157→/** Лидер пасеки по 30-дневному привесу. */
   158→   158→function leaderOf(list: Hive[]): Hive | null {
   159→   159→  const withW = list.filter((x) => delta30d(x) !== null)
   160→   160→  if (!withW.length) return null
   161→   161→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   162→   162→}
   163→   163→
   164→   164→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   165→   165→function spanDays(h: Hive): number {
   166→   166→  if (h.series.length < 2) return h.series.length
   167→   167→  const a = new Date(h.series[0].date).getTime()
   168→   168→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   169→   169→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   170→   170→  return Math.round((b - a) / 86400000) + 1
   171→   171→}
   172→   172→
   173→   173→function periodLabel(h: Hive): string {
   174→   174→  const d = spanDays(h)
   175→   175→  if (d >= 25) return 'за месяц'
   176→   176→  if (d === 1) return 'за день'
   177→   177→  if (d === 0) return '—'
   178→   178→  return `за ${d} дн.`
   179→   179→}

   187→   187→}

   330→   330→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   331→   331→  const w = ctx.weather
   332→   332→  if (!w || w.length === 0) return null
   333→   333→  const good = w.filter((d) => d.flight === 'good')
   334→   334→  const bad = w.filter((d) => d.flight === 'bad')
   335→   335→  const tMax = Math.max(...w.map((d) => d.tMax))
   336→   336→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   337→   337→  return {
   338→   338→    id: 'weather',
   339→   339→    scenario: 'weather',
   340→   340→    hiveId: null,
   341→   341→    title: good.length
   342→   342→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   343→   343→      : 'Ближайшие дни — не для лёта',

   399→   399→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]

  1189→  1189→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
  1190→  1190→export function askAgent(
  1191→  1191→  question: string,
  1192→  1192→  currentHiveId: number,
  1193→  1193→  ctx: AgentContext,
  1194→  1194→): AgentMessage {
  1195→  1195→  const q = norm(question)
  1196→  1196→  const hiveId = extractHiveId(q, ctx)
  1197→  1197→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
  1198→  1198→
  1199→  1199→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
  1200→  1200→
  1201→  1201→  // Приветствие и сервисные фразы
  1202→  1202→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
  1203→  1203→    return greetingAnswer(ctx)
  1204→  1204→  }
  1205→  1205→  // Вопрос о самом агенте: модель и устройство не раскрываем
  1206→  1206→  if (has(/кто ты|кто такой|как тебя зовут|ты бот|чат ?бот|какая (ты )?модель|какой (ты )?модель|на (чем|чём) ты (работаешь|сделан|написан)|что ты за|нейросет|искусственн|\bllm\b|\bgpt\b/)) {
  1207→  1207→    return identityAnswer()
  1208→  1208→  }
  1209→  1209→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
  1210→  1210→    return thanksAnswer()
  1211→  1211→  }
  1212→  1212→
  1213→  1213→  // Сценарий пестицидов
  1214→  1214→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
  1215→  1215→    return pesticideAnswer(ctx)
  1216→  1216→  }
  1217→  1217→
  1218→  1218→  // Кочёвка
  1219→  1219→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
  1220→  1220→    return migrationAnswer(ctx)
  1221→  1221→  }
  1222→  1222→
  1223→  1223→  // Погода и медосбор (включая температуру ВОЗДЕУХА — не путать с температурой улья)
  1224→  1224→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|снег|мороз|когда вывоз|когда летать|температура (воздуха|за окном|на улице|снаружи)/)) {
  1225→  1225→    return weatherAnswer(ctx)
  1226→  1226→  }
  1227→  1227→
  1228→  1228→  if (!hive) {
  1229→  1229→    return fallbackAnswer()
  1230→  1230→  }
  1231→  1231→
  1232→  1232→  // Темы по ульям: если номер улья не назван — отвечаем сразу по ВСЕМ ульям
  1233→  1233→  if (has(/влажн|сырост|сыро/)) {
  1234→  1234→    return hiveId ? humidityAnswer(hive) : humidityAllAnswer(ctx)
  1235→  1235→  }
  1236→  1236→  if (has(/шум|децибел|гул\b/)) {
  1237→  1237→    return noiseAnswer(ctx)
  1238→  1238→  }
  1239→  1239→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
  1240→  1240→    return hiveId ? swarmAnswer(hive, ctx) : swarmAllAnswer(ctx)
  1241→  1241→  }
  1242→  1242→  if (has(/температ|жар|тепло внутри|градус/)) {
  1243→  1243→    return hiveId ? climateAnswer(hive) : climateAllAnswer(ctx)
  1244→  1244→  }
  1245→  1245→  if (has(/вес|мед\b|меда|привес|набор|тяже|лидер/)) {
  1246→  1246→    return hiveId ? weightAnswer(hive, ctx) : weightAllAnswer(ctx)
  1247→  1247→  }
  1248→  1248→  if (has(/датчик|сенсор/)) {
  1249→  1249→    return sensorsAnswer(ctx)
  1250→  1250→  }
  1251→  1251→  if (has(/все.{0,8}уль|всех ульев|всей пасек|пасек|обзор|сводк|итог|общее состояни|как дела|как себя|что происходит|состояни|статус|проверь|проблем|тревог|расскажи|что с улеем|как улей/)) {

  1254→  1254→

  1256→  1256→  if (hiveId !== null) {
  1257→  1257→    return statusAnswer(hive, ctx)
  1258→  1258→  }
  1259→  1259→
  1260→  1260→  return fallbackAnswer()
