   144→   144→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; key: string; label: string; cls: string } {
   145→   145→  if (pct > 3) return { level: 'crit', key: 'dash.varroa.status.crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   146→   146→  if (pct >= 2) return { level: 'risk', key: 'dash.varroa.status.risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   147→   147→  return { level: 'ok', key: 'dash.varroa.status.ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   148→   148→}

   700→   700→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   701→   701→function seeded(seed: number): () => number {
   702→   702→  let a = seed >>> 0
   703→   703→  return () => {
   704→   704→    a |= 0
   705→   705→    a = (a + 0x6d2b79f5) | 0
   706→   706→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   707→   707→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   708→   708→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   709→   709→  }
   710→   710→}

   761→   761→export interface VarroaPoint {
   762→   762→  label: string
   763→   763→  /** Историческое значение заклещенности, %. */
   764→   764→  pct: number | null
   765→   765→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   766→   766→  forecast: number | null
   767→   767→  /** Отметка проведенной обработки. */
   768→   768→  treated: boolean
   769→   769→}
   770→   770→
   771→   771→/**
   772→   772→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   773→   773→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   774→   774→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   775→   775→ *
   776→   776→ * opts.overridePct — «текущее» значение после проведения обработки или нового
   777→   777→ * замера (кривая сходится к нему); opts.treatedToday — сегодня была обработка:
   778→   778→ * на последней точке рисуется дип, а прогноз считается от нового значения.
   779→   779→ */
   780→   780→export function demoVarroaSeries(
   781→   781→  h: DemoHive,
   782→   782→  days = 30,
   783→   783→  opts?: { overridePct?: number; treatedToday?: boolean },
   784→   784→): VarroaPoint[] {
   785→   785→  const rnd = seeded(3000 + h.id * 57)
   786→   786→  const out: VarroaPoint[] = []
   787→   787→  const nowMs = Date.now()
   788→   788→
   789→   789→  const curPct = opts?.overridePct ?? h.base.varroaPct
   790→   790→  const treatedToday = opts?.treatedToday ?? false
   791→   791→  // прогноз: после обработки семья восстанавливается (+0,1 за 14 дн),
   792→   792→  // без обработки критический улей продолжает расти (+1,6)
   793→   793→  const futurePct =
   794→   794→    opts?.overridePct !== undefined || treatedToday
   795→   795→      ? Math.round((curPct + (treatedToday ? 0.1 : 0.2)) * 10) / 10
   796→   796→      : h.base.varroaForecast14d
   797→   797→
   798→   798→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   799→   799→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   800→   800→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   801→   801→
   802→   802→  for (let k = days - 1; k >= 0; k--) {
   803→   803→    const t = new Date(nowMs - k * 86400000)
   804→   804→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   805→   805→    let pct = startPct + (curPct - startPct) * progress
   806→   806→    // дип после обработки: −1,2 % в течение 3 дней
   807→   807→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   808→   808→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   809→   809→    }
   810→   810→    pct += (rnd() - 0.5) * 0.12
   811→   811→    out.push({
   812→   812→      label: format(t, 'd.MM'),
   813→   813→      pct: Math.max(0.2, Math.round(pct * 10) / 10),
   814→   814→      forecast: null,
   815→   815→      treated: treatDay > 0 && k === treatDay,
   816→   816→    })
   817→   817→  }
   818→   818→  out[out.length - 1].pct = curPct
   819→   819→
   820→   820→  // свежая обработка: дип на последней точке (кривая резко падает)
   821→   821→  if (treatedToday && out.length >= 2) {
   822→   822→    out[out.length - 2].treated = true
   823→   823→  }
   824→   824→
   825→   825→  // прогноз: через 14 дней (пунктир от текущей точки)
   826→   826→  const future = new Date(nowMs + 14 * 86400000)
   827→   827→  out.push({
   828→   828→    label: format(future, 'd.MM'),
   829→   829→    pct: null,
   830→   830→    forecast: futurePct,
   831→   831→    treated: false,
   832→   832→  })
   833→   833→  // сегмент начинается с последней исторической точки
   834→   834→  out[out.length - 2].forecast = curPct
   835→   835→  return out
   836→   836→}
   837→   837→
   838→   838→/**
   839→   839→ * Привесы за 7 дней: столбцы суточного привеса + ночная температура
   840→   840→ * (ночные похолодания объясняют провалы приноса нектара).
   841→   841→ */
   842→   842→export function demoGainSeries(h: DemoHive, days = 7): Array<{ label: string; gainKg: number; nightTempC: number }> {
   843→   843→  const rnd = seeded(4000 + h.id * 83)
   844→   844→  const out: Array<{ label: string; gainKg: number; nightTempC: number }> = []
   845→   845→  const nowMs = Date.now()
   846→   846→  for (let k = days - 1; k >= 0; k--) {
   847→   847→    const t = new Date(nowMs - k * 86400000)
   848→   848→    // привес качается вокруг базового: погода и сила семьи
   849→   849→    const gainKg = k === 0 ? h.base.dailyGainKg : Math.max(-0.1, h.base.dailyGainKg + (rnd() - 0.45) * 0.5)
   850→   850→    // ночная температура: волна + случайные похолодания (2 прохладные ночи)
   851→   851→    const wave = Math.sin(((days - k) / days) * Math.PI * 1.4) * 3
   852→   852→    const dip = k === days - 2 || k === 1 ? -3.5 : 0
   853→   853→    const nightTempC = 14 + wave + dip + (rnd() - 0.5) * 2
   854→   854→    out.push({
   855→   855→      label: format(t, 'd.MM'),
   856→   856→      gainKg: Math.round(gainKg * 100) / 100,
   857→   857→      nightTempC: Math.round(nightTempC * 10) / 10,
   858→   858→    })
   859→   859→  }
   860→   860→  return out
   861→   861→}

   932→   932→/**
