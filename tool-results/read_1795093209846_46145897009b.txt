   161→function liveForSelection(live: LiveTelemetry, hive: DemoHive): LiveDemoValues {
   162→  if (hive.id !== 0) return liveForHive(live, hive)
   163→  const vals = DEMO_HIVES.map((h) => live.values[h.id] ?? h.base)
   164→  const avg = (f: (v: LiveDemoValues) => number) => vals.reduce((s, v) => s + f(v), 0) / vals.length
   165→  return {
   166→    tempC: avg((v) => v.tempC),
   167→    humidityPct: avg((v) => v.humidityPct),
   168→    weightKg: vals.reduce((s, v) => s + v.weightKg, 0),
   169→    soundHz: avg((v) => v.soundHz),
   170→    co2Ppm: avg((v) => v.co2Ppm),
   171→    vocIndex: avg((v) => v.vocIndex),
   172→    trafficPerMin: avg((v) => v.trafficPerMin),
   173→    varroaPct: avg((v) => v.varroaPct),
   174→  }
   175→}

   183→  label,
   184→  value,

   186→  digits = 1,
   187→  suffix,
   188→  hint,

   212→      </p>
