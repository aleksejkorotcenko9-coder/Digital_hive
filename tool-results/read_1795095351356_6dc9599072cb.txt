   178→function liveForSelection(live: LiveTelemetry, hive: DemoHive): LiveDemoValues {
   179→  if (hive.id !== 0) return liveForHive(live, hive)
   180→  const vals = DEMO_HIVES.map((h) => live.values[h.id] ?? h.base)
   181→  const avg = (f: (v: LiveDemoValues) => number) => vals.reduce((s, v) => s + f(v), 0) / vals.length
   182→  return {
   183→    tempC: avg((v) => v.tempC),
   184→    humidityPct: avg((v) => v.humidityPct),
   185→    weightKg: vals.reduce((s, v) => s + v.weightKg, 0),
   186→    soundHz: avg((v) => v.soundHz),
   187→    co2Ppm: avg((v) => v.co2Ppm),
   188→    vocIndex: avg((v) => v.vocIndex),
   189→    trafficPerMin: avg((v) => v.trafficPerMin),
   190→    varroaPct: avg((v) => v.varroaPct),
   191→  }
   192→}

   198→function LiveTile({
   199→  icon,
   200→  label,
   201→  value,
   202→  num,
   203→  digits = 1,
   204→  suffix,
   205→  hint,
   206→  badge,
   207→  tone = 'red',
   208→}: {
   209→  icon: React.ReactNode
   210→  label: string
   211→  value?: string
   212→  num?: number
   213→  digits?: number
   214→  suffix?: string
   215→  hint?: string
   216→  badge?: React.ReactNode
   217→  tone?: 'red' | 'amber' | 'emerald'
   218→}) {
   219→  const tones = {
   220→    red: 'border-red-50 bg-red-50/40 text-red-400',
   221→    amber: 'border-amber-100 bg-amber-50/50 text-amber-500',
   222→    emerald: 'border-emerald-100 bg-emerald-50/50 text-emerald-600',
   223→  } as const
   224→  return (
   225→    <div className={cn('rounded-xl border px-3 py-2.5', tones[tone])}>
   226→      <p className="flex items-center gap-1.5 text-[11px] font-semibold text-gray-500">
   227→        <span>{icon}</span>
   228→        {label}
   229→      </p>
   230→      <p className="mt-1 flex flex-wrap items-baseline gap-1.5 text-lg font-extrabold leading-none text-gray-900">
   231→        {num !== undefined ? (
   232→          <AnimatedNumber value={num} digits={digits} />
   233→        ) : (
   234→          <LiveValue value={value ?? ''} />
   235→        )}
   236→        {suffix && <span className="text-[11px] font-bold text-gray-400">{suffix}</span>}
   237→        {badge}
   238→      </p>
   239→      {hint && <p className="mt-1 text-[10px] leading-tight text-gray-400">{hint}</p>}
   240→    </div>
   241→  )
