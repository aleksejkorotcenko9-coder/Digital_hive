   103→   103→export function DashboardView() {

   108→   108→  const { status, data, reload } = useRealData()
   109→   109→
   110→   110→  // Синхронизация с картой в ОБОИХ режимах: ульи дашборда = точки на карте
   111→   111→  // («Демо» — участок у Чишм, «Мои данные» — собственная пасека в Стерлитамаке),
   112→   112→  // показания — с датчиков. Опрос каждые 5 секунд — числа «живут» вместе с картой.
   113→   113→  // Режим хранится вместе со списком: пока идёт загрузка той пасеки, чужие
   114→   114→  // точки не показываются (синхронизация без «вспышек» чужих данных).
   115→   115→  const [pointsByMode, setPointsByMode] = useState<{ mode: 'demo' | 'real'; points: MapPointDto[] } | null>(null)
   116→   116→  useEffect(() => {
   117→   117→    let alive = true
   118→   118→    const load = async () => {
   119→   119→      try {
   120→   120→        const res = await fetch(`/api/map-points?mode=${mode}`, { cache: 'no-store' })
   121→   121→        if (!res.ok) return
   122→   122→        const j = (await res.json()) as { points: MapPointDto[] }
   123→   123→        if (alive) setPointsByMode({ mode, points: j.points })
   124→   124→      } catch {
   125→   125→        // датчики подтянутся на следующем опросе
   126→   126→      }
   127→   127→    }
   128→   128→    void load()
   129→   129→    const timer = setInterval(() => void load(), 5000)
   130→   130→    return () => {
   131→   131→      alive = false
   132→   132→      clearInterval(timer)
   133→   133→    }
   134→   134→  }, [mode])
   135→   135→  const points = pointsByMode && pointsByMode.mode === mode ? pointsByMode.points : null
   136→   136→
   137→   137→  // Погода той пасеки, к которой относится режим (реальные данные из интернета):
   138→   138→  // «Демо» — Чишмы, «Мои данные» — Стерлитамак. Та же, что в панели погоды.

   141→   141→    let alive = true
   142→   142→    const [lat, lon] = APIARY_PLACES[mode].center

   145→   145→      try {
   146→   146→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   147→   147→        if (!res.ok) return
   148→   148→        const j = (await res.json()) as WeatherResponse
   149→   149→        if (alive) setWByMode({ mode, data: j })
   150→   150→      } catch {
   151→   151→        // агент честно скажет, если погоды нет
   152→   152→      }
   153→   153→    }
   154→   154→    void load()
   155→   155→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   156→   156→    return () => {
   157→   157→      alive = false
   158→   158→      clearInterval(timer)
   159→   159→    }
   160→   160→  }, [mode])
   161→   161→  const wResp = wByMode && wByMode.mode === mode ? wByMode.data : null
   162→   162→
   163→   163→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   164→   164→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   165→   165→
   166→   166→  // выбранный улей всегда существует в синхронизированном наборе
   167→   167→  useEffect(() => {

   169→   169→      setHive(syncedHives[0].id)
   170→   170→    }
   171→   171→  }, [syncedHives, hiveId, setHive])
   172→   172→
   173→   173→  if (mode === 'real') {
   174→   174→    if (status === 'idle' || status === 'loading') {
   175→   175→      return <DashboardSkeleton />
   176→   176→    }
   177→   177→    if (status === 'error') {
   178→   178→      return (
   179→   179→        <Card className="mx-auto max-w-lg border-red-100">
   180→   180→          <CardContent className="p-6 text-center">
   181→   181→            <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   182→   182→            <h2 className="mt-2 font-bold text-gray-900">Не удалось загрузить данные</h2>
   183→   183→            <p className="mt-1 text-sm text-gray-500">
   184→   184→              Проверьте, что база данных доступна, и попробуйте ещё раз.
   185→   185→            </p>
   186→   186→            <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   187→   187→              Повторить загрузку
   188→   188→            </Button>
   189→   189→          </CardContent>
   190→   190→        </Card>
   191→   191→      )
   192→   192→    }
   193→   193→    // Схема и диаграммы «Моих данных» — ТО ЖЕ точки, что на карте пасеки
   194→   194→    // (Стерлитамак): один набор ульев для карты, вкладки «Датчики» и дашборда.
   195→   195→    // Из базы берём журнал и площадки кочёвки.
   196→   196→    if (!points || !data || syncedHives === null) return <DashboardSkeleton />
   197→   197→    if (syncedHives.length === 0) {
   198→   198→      return <EmptyMapState onOpenMap={() => setView('map')} real />
   199→   199→    }
   200→   200→    return (
   201→   201→      <DashboardContent
   202→   202→        mode="real"
   203→   203→        hives={syncedHives}
   204→   204→        sites={data.sites.map(siteDtoToSite)}
   205→   205→        weather={agentWeather?.days ?? null}
   206→   206→        pesticide={null}
   207→   207→        events={data.events}
   208→   208→      />
   209→   209→    )
   210→   210→  }
   211→   211→
   212→   212→  if (!syncedHives) return <DashboardSkeleton />
   213→   213→  if (syncedHives.length === 0) return <EmptyMapState onOpenMap={() => setView('map')} />
   214→   214→
   215→   215→  return (
   216→   216→    <DashboardContent
   217→   217→      mode="demo"
   218→   218→      hives={syncedHives}
   219→   219→      sites={migrationSites}
   220→   220→      weather={agentWeather?.days ?? null}
   221→   221→      pesticide={pesticideAlert}
   222→   222→      events={[]}
   223→   223→    />
   224→   224→  )
   225→   225→}

   689→   689→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   690→   690→function weightTotal(h: Hive): number {
   691→   691→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   692→   692→  const first = h.series[0]?.weightKg ?? null
   693→   693→  if (w === null || first === null) return 0
   694→   694→  return Math.round((w - first) * 10) / 10
   695→   695→}
