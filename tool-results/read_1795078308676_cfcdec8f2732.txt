   356→   356→      method: 'PATCH',

   358→   358→      body: JSON.stringify(body),
   359→   359→    })

   361→   361→      toast({ title: 'Не получилось', variant: 'destructive' })
   362→   362→      return

   366→   366→    setPoints((prev) =>
   367→   367→      prev
   368→   368→        ? prev.map((x) => {
   369→   369→            if (x.id !== p.id) return x
   370→   370→            const hasSensor = body.hasSensor ?? x.hasSensor
   371→   371→            const sensorOn = body.sensorOn ?? (body.hasSensor !== undefined ? body.hasSensor : x.sensorOn)
   372→   372→            return { ...x, hasSensor, sensorOn }
   373→   373→          })
   374→   374→        : prev,
   375→   375→    )
