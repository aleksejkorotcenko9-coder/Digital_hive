   144→function lastPoint(h: Hive) {

   148→function lastWeight(h: Hive): number | null {
   149→  return lastPoint(h)?.weightKg ?? null

   152→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   153→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   154→function weightDelta(h: Hive, backDays: number): number | null {
   155→  const s = h.series
   156→  if (s.length === 0) return null
   157→  const lastPoint = s[s.length - 1]
   158→  if (lastPoint.weightKg === null) return null
   159→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   160→  let best: number | null = null
   161→  let bestDiff = Infinity
   162→  for (const p of s) {
   163→    if (p.weightKg === null || p === lastPoint) continue
   164→    const diff = Math.abs(new Date(p.date).getTime() - target)
   165→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   166→      bestDiff = diff
   167→      best = p.weightKg
   168→    }
   169→  }
   170→  if (best === null) return null
   171→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   172→}
   173→
   174→const delta2d = (h: Hive) => weightDelta(h, 2)
   175→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   176→
   177→/**
   178→ * Произвольный период из вопроса: «за 8 дней», «за 12 дней», «за 2 недели»,
   179→ * «за месяц», «за 45 дней». Возвращает число дней или null.
   180→ * (п. 7 ТЗ: любые периоды вместо трёх фиксированных)
   181→ */
   182→export function extractPeriodDays(q: string): number | null {
   183→  const days = q.match(/за (\d+)\s*(дн|день|дня|дней|сутк)/)
   184→  if (days) return Math.max(1, Math.min(365, Number(days[1])))
   185→  const weeks = q.match(/за (\d+)\s*(недел)/)
   186→  if (weeks) return Math.max(1, Math.min(104, Number(weeks[1]) * 7))
   187→  const months = q.match(/за (\d+)\s*(мес)/)
   188→  if (months) return Math.max(1, Math.min(24, Number(months[1]) * 30))
   189→  if (/за неделю/.test(q)) return 7
   190→  if (/за месяц/.test(q)) return 30
   191→  return null
   192→}

   203→/** Лидер пасеки по 30-дневному привесу. */
   204→function leaderOf(list: Hive[]): Hive | null {
   205→  const withW = list.filter((x) => delta30d(x) !== null)
   206→  if (!withW.length) return null
   207→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   208→}

   210→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   211→function spanDays(h: Hive): number {
   212→  if (h.series.length < 2) return h.series.length
   213→  const a = new Date(h.series[0].date).getTime()
   214→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   215→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   216→  return Math.round((b - a) / 86400000) + 1
   217→}

   219→function periodLabel(h: Hive, lang: Lang): string {
   220→  const d = spanDays(h)
   221→  if (d >= 25) return tr(lang, 'agent.period.month')
   222→  if (d === 1) return tr(lang, 'agent.period.day')
   223→  if (d === 0) return tr(lang, 'agent.period.dash')
   224→  return tr(lang, 'agent.period.days', { n: d })
   225→}

   253→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   254→  const l = ctx.lang ?? 'ru'
   255→  const status = hiveStatus(h, ctx.hives)
   256→  const d2 = delta2d(h)
   257→  const d30 = delta30d(h)
   258→  const t = lastPoint(h)?.tempInC ?? null
   259→  const hum = lastHumidity(h)
   260→
   261→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   262→    return {
   263→      id: `nodata-${h.id}`,
   264→      scenario: 'status',
   265→      hiveId: h.id,
   266→      title: tr(l, 'agent.find.nodata.title', { n: h.id }),
   267→      text: tr(l, 'agent.find.nodata.text', { n: h.id }),
   268→      basis: [tr(l, 'agent.basis.db'), tr(l, 'agent.basis.measurements', { n: h.id })],
   269→      urgency: 'this-week',
   270→      urgencyNote: tr(l, 'agent.note.loadData'),
   271→      probability: tr(l, 'agent.prob.nodata'),
   272→      metrics: [{ label: tr(l, 'agent.metric.measurements'), value: String(h.series.length) }],
   273→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   274→    }
   275→  }
   276→
   277→  if (status.level === 'risk' && d2 !== null) {
   278→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   279→    const now = lastWeight(h) ?? peak
   280→    return {
   281→      id: `swarm-${h.id}`,
   282→      scenario: 'swarm',
   283→      hiveId: h.id,
   284→      title: tr(l, 'agent.find.swarm.title'),
   285→      text: tr(l, 'agent.find.swarm.text', {
   286→        n: h.id,
   287→        w: fmtNum(Math.abs(d2)),
   288→        peak: fmtNum(peak),
   289→        now: fmtNum(now),
   290→        temp: t !== null ? tr(l, 'agent.find.swarm.temp', { t: fmtNum(t) }) : '',
   291→        hum: hum !== null ? tr(l, 'agent.find.swarm.hum', { h: fmtNum(hum, 0) }) : '',
   292→      }),
   293→      basis: [
   294→        tr(l, 'agent.basis.weightChart', { n: h.id }),
   295→        tr(l, 'agent.basis.weight2d', { w: fmtSigned(d2) }),
   296→        ...(t !== null ? [tr(l, 'agent.basis.tempNormCheck', { t: fmtNum(t) })] : []),
   297→        tr(l, 'agent.basis.seasonJuly'),
   298→      ],
   299→      urgency: 'today',
   300→      urgencyNote: tr(l, 'agent.note.inspectTodayEvening'),
   301→      probability: swarmProbability(d2, l),
   302→      metrics: [
   303→        { label: tr(l, 'agent.metric.weight2d'), value: tr(l, 'agent.value.kg', { w: fmtSigned(d2) }) },
   304→        ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   305→        ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),
   306→      ],
   307→      prefill: tr(l, 'agent.prefill.what', { n: h.id }),
   308→    }
   309→  }
   310→
   311→  if (status.level === 'watch' && h.label !== undefined) {
   312→    const leader = leaderOf(ctx.hives)
   313→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   314→      ? tr(l, 'agent.find.slow.reasonSensor')
   315→      : tr(l, 'agent.find.slow.reasonLeader')
   316→    return {
   317→      id: `slow-${h.id}`,
   318→      scenario: 'status',
   319→      hiveId: h.id,
   320→      title: tr(l, 'agent.find.slow.title'),
   321→      text: tr(l, 'agent.find.slow.text', {
   322→        n: h.id,
   323→        w: fmtSigned(d30 ?? 0),
   324→        period: periodLabel(h, l),
   325→        leader:
   326→          leader && leader.id !== h.id
   327→            ? tr(l, 'agent.find.slow.leader', { n: leader.id, w: fmtSigned(delta30d(leader) ?? 0) })
   328→            : '',
   329→        temp: t !== null ? tr(l, 'agent.inline.temp', { t: fmtNum(t) }) : '',
   330→        reason: cap(reason),
   331→        extra: h.humidityOfflineFrom
   332→          ? tr(l, 'agent.find.slow.offline', { date: h.humidityOfflineFrom })
   333→          : tr(l, 'agent.find.slow.compare'),
   334→      }),
   335→      basis: [
   336→        tr(l, 'agent.basis.gainCompare'),
   337→        tr(l, 'agent.basis.slowGain', { n: h.id, w: fmtSigned(d30 ?? 0) }),
   338→        ...(h.humidityOfflineFrom ? [tr(l, 'agent.basis.humOffline', { date: h.humidityOfflineFrom })] : []),
   339→      ],
   340→      urgency: 'this-week',
   341→      urgencyNote: tr(l, 'agent.note.checkSensorDays'),
   342→      probability: tr(l, 'agent.prob.indirect'),
   343→      metrics: [
   344→        { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   345→        ...(leader && leader.id !== h.id
   346→          ? [{ label: tr(l, 'agent.metric.leaderN', { n: leader.id }), value: tr(l, 'agent.value.kg', { w: fmtSigned(delta30d(leader) ?? 0) }) }]
   347→          : []),
   348→        ...(hum !== null
   349→          ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }]
   350→          : [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.metric.nodata') }]),

   352→      prefill: tr(l, 'agent.prefill.slowGain', { n: h.id }),

   355→
   356→  return {
   357→    id: `ok-${h.id}`,
   358→    scenario: 'status',
   359→    hiveId: h.id,
   360→    title: tr(l, 'agent.find.ok.title'),
   361→    text: tr(l, 'agent.find.ok.text', {
   362→      n: h.id,
   363→      w: fmtNum(lastWeight(h) ?? 0),
   364→      g: fmtSigned(d30 ?? 0),
   365→      period: periodLabel(h, l),
   366→      d2: fmtSigned(d2 ?? 0),
   367→      temp: t !== null ? tr(l, 'agent.find.ok.temp', { t: fmtNum(t) }) : '',
   368→    }),
   369→    basis: [
   370→      tr(l, 'agent.basis.weightChart', { n: h.id }),
   371→      tr(l, 'agent.basis.gain', { w: fmtSigned(d30 ?? 0) }),
   372→      tr(l, 'agent.basis.tempOkFamily'),
   373→    ],
   374→    urgency: 'ok',
   375→    urgencyNote: tr(l, 'agent.note.observe'),
   376→    metrics: [
   377→      { label: tr(l, 'agent.metric.gainPeriod', { period: periodLabel(h, l) }), value: tr(l, 'agent.value.kg', { w: fmtSigned(d30 ?? 0) }) },
   378→      ...(t !== null ? [{ label: tr(l, 'agent.metric.tempIn'), value: tr(l, 'agent.value.tempC', { t: fmtNum(t) }) }] : []),
   379→      ...(hum !== null ? [{ label: tr(l, 'agent.metric.humidity'), value: tr(l, 'agent.value.pct', { h: fmtNum(hum, 0) }) }] : []),

   419→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   420→  const w = ctx.weather
   421→  if (!w || w.length === 0) return null
   422→  const good = w.filter((d) => d.flight === 'good')
   423→  const bad = w.filter((d) => d.flight === 'bad')
   424→  const tMax = Math.max(...w.map((d) => d.tMax))
   425→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   426→  const l = ctx.lang ?? 'ru'
   427→  return {
   428→    id: 'weather',
   429→    scenario: 'weather',
   430→    hiveId: null,
   431→    title: good.length
   432→      ? tr(l, 'agent.find.weather.title', { days: good.map((d) => dayMonth(d.date, l)).join(', ').slice(0, 80) })
   433→      : tr(l, 'agent.find.weather.titleBad'),
   434→    text: good.length
   435→      ? tr(l, 'agent.find.weather.textGood', {
   436→          list: good.map((d) => d.full).join(', '),
   437→          tmax: fmtSigned(tMax, 0),
   438→          wind: Math.max(...good.map((d) => d.windMs)),

   471→function migrationFindingOf(ctx: AgentContext): AgentFinding | null {
   472→  const sites = [...ctx.sites].sort((a, b) => b.score - a.score)

   478→    l,

   514→/** Список находок: по выбранному улью + глобальные сценарии, отсортированы по срочности. */
   515→export function findingsFor(hiveId: number, ctx: AgentContext): AgentFinding[] {
   516→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   517→  const list: AgentFinding[] = []
   518→  if (hive) list.push(hiveFinding(hive, ctx))
   519→  const pesticide = pesticideFindingOf(ctx)
   520→  if (pesticide) list.push(pesticide)
   521→  const weatherF = weatherFindingOf(ctx)
   522→  if (weatherF) list.push(weatherF)
   523→  const migrationF = migrationFindingOf(ctx)
   524→  if (migrationF) list.push(migrationF)
   525→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   526→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   527→}

  1099→/**
  1100→ * Календарь цветения медоносов (п. 3 ТЗ): эталонные окна культур средней
  1101→ * полосы + окна полей Минсельхоза, пересчитанные от фактических дат посева.
  1102→ * Зимой календарь — план на следующий сезон; летом сверяется с GDD-прогнозом.
  1103→ * Отвечает и на вопрос про конкретный медонос («когда цветёт липа»).
  1104→ */
  1105→function bloomCalendarAnswer(question: string, ctx: AgentContext): AgentMessage {
  1106→  const l = ctx.lang ?? 'ru'
  1107→  const q = norm(question)
  1108→  const year = new Date().getFullYear()
  1109→
  1110→  // Эталонные окна всех культур справочника, отсортированные по дате старта.
  1111→  const cultures = Object.values(AGRO_CULTURES)
  1112→    .map((c) => ({ c, w: estimateBloomWindow(c.key, null, year) as BloomWindow }))
  1113→    .filter((x) => x.w !== null)
  1114→    .sort((a, b) => a.w.start.getTime() - b.w.start.getTime())
  1115→
  1116→  // Названный в вопросе медонос — отдельной строкой сверху с деталями.
  1117→  // Согласование: «гречихи», «липу», «донника» — ищем корень слова (имя без окончания).
  1118→  const stemOf = (name: string) => norm(name).slice(0, Math.max(4, norm(name).length - 2))
  1119→  const named =
  1120→    cultures.find(({ c }) => {
  1121→      const stem = stemOf(c.name)
  1122→      return q.includes(stem)
  1123→    }) ?? null
  1124→  const ordered = named ? [named, ...cultures.filter((x) => x !== named)] : cultures
  1125→
  1126→  const yieldOf = (c: AgroCulture) => `${c.honeyYieldMin}–${c.honeyYieldMax}`
  1127→  const rows = ordered
  1128→    .map(({ c, w }) => tr(l, 'agent.bloom.row', { name: c.name, window: formatBloomWindow(w), yield: yieldOf(c) }))
  1129→    .join('\n')
  1130→
  1131→  // Окна полей Минсельхоза: расчёт от фактической даты посева + отклонение от эталона.
  1132→  const fields = (ctx.cropFields ?? []).filter((f) => f.sowingDate && f.bloomText)
  1133→  const corrRows = fields
  1134→    .map((f) => {
  1135→      const bloomYear = f.bloomStart ? new Date(f.bloomStart + 'T00:00:00').getFullYear() : year
  1136→      const etalon = estimateBloomWindow(f.culture, null, bloomYear)
  1137→      let dev = ''
  1138→      if (etalon && f.bloomStart) {
  1139→        const d = Math.round((new Date(f.bloomStart + 'T00:00:00').getTime() - etalon.start.getTime()) / 86400000)
  1140→        if (d > 0) dev = tr(l, 'agent.bloom.devLate', { d })
  1141→        else if (d < 0) dev = tr(l, 'agent.bloom.devEarly', { d: -d })
  1142→      }
  1143→      return tr(l, 'agent.bloom.corrRow', { name: f.name, culture: f.cultureName, bloom: f.bloomText, dev })
  1144→    })
  1145→    .join('\n')
  1146→
  1147→  // Совет про два сорта мёда: пара медоносов с максимальным перекрытием окон.
  1148→  let tip = ''
  1149→  let bestOverlap = 0
  1150→  let tipPair: [AgroCulture, AgroCulture] | null = null
  1151→  for (let i = 0; i < cultures.length; i++) {
  1152→    for (let j = i + 1; j < cultures.length; j++) {
  1153→      const a = cultures[i]
  1154→      const b = cultures[j]
  1155→      const overlap = Math.min(a.w.end.getTime(), b.w.end.getTime()) - Math.max(a.w.start.getTime(), b.w.start.getTime())
  1156→      if (overlap > bestOverlap) {
  1157→        bestOverlap = overlap
  1158→        tipPair = [a.c, b.c]
  1159→      }
  1160→    }
  1161→  }
  1162→  if (tipPair && bestOverlap >= 10 * 86400000) {
  1163→    tip = tr(l, 'agent.bloom.tip', { a: tipPair[0].name, b: tipPair[1].name })
  1164→  }
  1165→
  1166→  const winter = seasonPhase(new Date()) === 'winter_rest'
  1167→  const season = winter ? tr(l, 'agent.bloom.seasonWinter') : tr(l, 'agent.bloom.seasonActive')
  1168→
  1169→  const parts: string[] = []
  1170→  if (named) {
  1171→    const note = named.c.note.charAt(0).toUpperCase() + named.c.note.slice(1)
  1172→    parts.push(
  1173→      tr(l, 'agent.bloom.single', {
  1174→        name: named.c.name,
  1175→        window: formatBloomWindow(named.w),
  1176→        yield: yieldOf(named.c),
  1177→        note,
  1178→      }),
  1179→    )
  1180→    parts.push(tr(l, 'agent.bloom.fullList', { rows }))
  1181→  } else {
  1182→    parts.push(tr(l, 'agent.bloom.etalonTitle', { rows }))
  1183→  }
  1184→  if (corrRows) parts.push(tr(l, 'agent.bloom.corrected', { rows: corrRows }))
  1185→  if (tip) parts.push(tip)
  1186→  parts.push(season)
  1187→
  1188→  const basis = [tr(l, 'agent.basis.agroCalendar')]
  1189→  if (corrRows) basis.push(tr(l, 'agent.basis.minselhoz'), tr(l, 'agent.basis.bloomFromSowing'))
  1190→
  1191→  const earliest = cultures[0]
  1192→  return {
  1193→    scenario: 'bloom',
  1194→    hiveId: null,
  1195→    text: parts.join('\n'),
  1196→    basis,
  1197→    urgency: winter ? 'ok' : 'this-week',
  1198→    urgencyNote: winter ? tr(l, 'agent.crop.winterUrgency') : tr(l, 'agent.note.planMigration'),
  1199→    metrics: [
  1200→      { label: tr(l, 'agent.metric.bloomCultures'), value: String(cultures.length) },
  1201→      ...(earliest
  1202→        ? [{ label: tr(l, 'agent.metric.bloomEarliest'), value: `${earliest.c.name} — ${formatBloomWindow(earliest.w)}` }]
  1203→        : []),
  1204→      ...(fields.length ? [{ label: tr(l, 'agent.metric.bloomFields'), value: String(fields.length) }] : []),
  1205→    ],
  1206→  }
  1207→}
