    13→ * — Список точек справа: переключение датчика, удаление, перелёт к точке.

    83→  const ymRef = useRef<YmapsApi | null>(null)
    84→  const pointsColRef = useRef<YmapsGeoCollection | null>(null)

    94→
    95→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
    96→  const [selected, setSelected] = useState<{ lat: number; lon: number } | null>(null)
    97→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
    98→  const [weatherLoading, setWeatherLoading] = useState(false)
    99→
   100→  // диалог привязки улья по коду датчика
   101→  const [dialogOpen, setDialogOpen] = useState(false)
   102→  const [code, setCode] = useState('')
   103→  const [name, setName] = useState('')
   104→  const [busy, setBusy] = useState(false)
   105→
   106→  // Место улья: по GPS датчика (по умолчанию) или точка, выбранная кликом по карте.
   107→  // pick — режим «ждём клик по карте»: место для нового улья или перенос существующего.
   108→  const [placeMode, setPlaceMode] = useState<'gps' | 'manual'>('gps')
   109→  const [manualPos, setManualPos] = useState<{ lat: number; lon: number } | null>(null)
   110→  const [pick, setPick] = useState<PickMode | null>(null)
   111→  useEffect(() => {
   112→    pickRef.current = pick
   113→  }, [pick])
   114→
   115→  // Как только введён код из каталога — сразу знаем GPS-позицию датчика
   116→  const found = useMemo(() => findSensor(code), [code])
   117→  const showNotFound = normalizeCode(code).length >= 4 && !found
   118→
   119→  /* ---------- карта (Яндекс Карты) ---------- */
   120→  useEffect(() => {
   121→    let disposed = false
   122→    void (async () => {
   123→      let ym: YmapsApi
   124→      try {
   125→        ym = await loadYmaps()
   126→      } catch (err) {
   127→        if (!disposed) {
   128→          setMapError(err instanceof Error ? err.message : 'Не удалось загрузить карту — проверьте соединение')
   129→        }
   130→        return
   131→      }
   132→      if (disposed || !containerRef.current) return
   133→
   134→      // controls: переключатель «Схема / Спутник / Гибрид» + масштаб и геолокация;
   135→      // suppressMapOpenBlock — без блока «Открыть в Яндекс Картах»
   136→      const map = new ym.Map(containerRef.current, {
   137→        // стартовый центр — запасной; как только точки загрузятся, перелетаем к пасеке
   138→        center: [...DEFAULT_PLACE.center] as YmapsCoords,
   139→        zoom: 12,
   140→        controls: ['zoomControl', 'typeSelector', 'fullscreenControl', 'geolocationControl', 'rulerControl'],
   141→        suppressMapOpenBlock: true,
   142→      })
   143→      map.behaviors.enable('scrollZoom')
   144→
   145→      // События меток всплывают до карты: выбираем точку только по клику «в пустое место»
   146→      map.events.add('click', (e) => {
   147→        if (e.get('target') !== map) return
   148→        const coords = e.get('coords') as YmapsCoords | undefined
   149→        if (!coords) return
   150→        clickRef.current(coords[0], coords[1])
   151→      })
   152→
   153→      const pointsCol = new ym.GeoObjectCollection()
   154→      const selCol = new ym.GeoObjectCollection()
   155→      const prevCol = new ym.GeoObjectCollection()
   156→      map.geoObjects.add(pointsCol).add(selCol).add(prevCol)
   157→
   158→      // HTML-макет метки улья (круг 🐝 + подпись с числом ульев)
   159→      layoutRef.current = ym.templateLayoutFactory.createClass(HIVE_ICON_TPL)
   160→
   161→      ymRef.current = ym
   162→      mapRef.current = map
   163→      pointsColRef.current = pointsCol
   164→      selColRef.current = selCol
   165→      prevColRef.current = prevCol
   166→      setMapReady(true)
   167→    })()
   168→    return () => {
   169→      disposed = true
   170→      mapRef.current?.destroy()
   171→      mapRef.current = null
   172→      pointsColRef.current = null
   173→      selColRef.current = null
   174→      prevColRef.current = null
   175→      markerRefs.current.clear()
   176→      setMapReady(false)
   177→    }
   178→  }, [])
   179→
   180→  /* ---------- данные точек ---------- */
   181→  const load = useCallback(async () => {
   182→    try {
   183→      const res = await fetch('/api/map-points', { cache: 'no-store' })
   184→      if (!res.ok) return
   185→      const j = (await res.json()) as { points: MapPointDto[] }
   186→      setPoints(j.points)
   187→    } catch {
   188→      // карта и погода продолжают работать, точки подтянутся при следующем обновлении
   189→    }
   190→  }, [])
   191→
   192→  useEffect(() => {
   193→    void load()
   194→  }, [load])
   195→
   196→  // Первый успешный приход точек: подгоняем камеру так, чтобы КАЖДЫЙ улей был
   197→  // в кадре. Раньше камера летела к среднему центру с зумом 12 — ульи,
   198→  // разбросанные на десятки километров, оставались за кадром и выглядели
   199→  // «непривязанными». Один улей — крупный зум на его место. fittedRef —
   200→  // чтобы не дёргать камеру на каждом опросе.
   201→  const fittedRef = useRef(false)
   202→  useEffect(() => {
   203→    if (!mapReady || fittedRef.current) return
   204→    if (!points || points.length === 0) return
   205→    fittedRef.current = true
   206→    const map = mapRef.current
   207→    if (!map) return
   208→    if (points.length === 1) {
   209→      map.setCenter([points[0].lat, points[0].lon], 13)
   210→      return
   211→    }
   212→    const lats = points.map((p) => p.lat)
   213→    const lons = points.map((p) => p.lon)
   214→    map.setBounds(
   215→      [
   216→        [Math.min(...lats), Math.min(...lons)],
   217→        [Math.max(...lats), Math.max(...lons)],
   218→      ],
   219→      { checkZoomRange: true, zoomMargin: 50, duration: 700 },
   220→    )
   221→  }, [points, mapReady])
   222→
   223→  /* ---------- полёт камеры, перенос улья и выбор места ---------- */
   224→  /** Плавный перелёт к точке: зум до 12, полёт Яндекса, затем открываем балун метки. */
   225→  const flyMapTo = useCallback((lat: number, lon: number, balloonId?: string) => {
   226→    const map = mapRef.current
   227→    if (!map) return
   228→    if (map.getZoom() < 12) map.setZoom(12)
   229→    map.panTo([lat, lon], { duration: 700, flying: true })
   230→    if (balloonId) {
   231→      setTimeout(() => markerRefs.current.get(balloonId)?.balloon.open(), 750)
   232→    }
   233→  }, [])
   234→
   235→  /** Перенос существующего улья на новое место (клик по карте в режиме переноса). */
   236→  const movePoint = useCallback(
   237→    async (id: string, lat: number, lon: number) => {
   238→      try {
   239→        const res = await fetch(`/api/map-points/${id}`, {
   240→          method: 'PATCH',
   241→          headers: { 'Content-Type': 'application/json' },
   242→          body: JSON.stringify({ lat, lon }),
   243→        })
   244→        const j = (await res.json().catch(() => ({}))) as { error?: string }
   245→        if (!res.ok) {
   246→          toast({ title: 'Не получилось', description: j.error ?? 'Ошибка сервера', variant: 'destructive' })
   247→          return
   248→        }
   249→        setPoints((prev) => (prev ? prev.map((x) => (x.id === id ? { ...x, lat, lon } : x)) : prev))
   250→        toast({ title: 'Улей перенесён', description: `Новое место: ${lat.toFixed(4)}, ${lon.toFixed(4)}` })
   251→        flyMapTo(lat, lon, id)
   252→        void load()
   253→      } catch {
   254→        toast({ title: 'Не получилось', description: 'Ошибка сети', variant: 'destructive' })
   255→      }
   256→    },
   257→    [toast, flyMapTo, load],
   258→  )
   259→
   260→  /** Кнопка «Выбрать место на карте»: прячем диалог, ждём клик по карте. */
   261→  const startPickPlace = () => {
   262→    setPick({ kind: 'new' })
   263→    setDialogOpen(false)
   264→  }
   265→
   266→  /** Кнопка «Перенести» в карточке улья: ждём клик по карте. */
   267→  const startMove = (p: MapPointDto) => {
   268→    setPick({ kind: 'move', id: p.id, name: p.name })
   269→  }
   270→
   271→  const onMapClick = useCallback(
   272→    (lat: number, lon: number) => {
   273→      const cur = pickRef.current
   274→      if (cur?.kind === 'new') {
   275→        // выбрали место для нового улья — возвращаемся в диалог привязки
   276→        setManualPos({ lat, lon })
   277→        setPick(null)
   278→        setDialogOpen(true)
   279→        return
   280→      }
   281→      if (cur?.kind === 'move') {
   282→        setPick(null)
   283→        void movePoint(cur.id, lat, lon)
   284→        return
   285→      }
   286→      // Клик по карте — выбор точки для погоды; место улья задаётся в диалоге
   287→      // (GPS датчика или клик по карте в режиме выбора)
   288→      setSelected({ lat, lon })
   289→    },
   290→    [movePoint],
   291→  )
   292→
   293→  useEffect(() => {
   294→    clickRef.current = onMapClick
   295→  }, [onMapClick])
   296→
   297→  // Курсор: ладошка в обычном режиме, перекрестие при выборе места на карте
   298→  useEffect(() => {
   299→    if (!mapReady) return
   300→    const map = mapRef.current
   301→    if (!map) return
   302→    cursorRef.current?.remove()
   303→    cursorRef.current = map.cursors.push(pick ? 'crosshair' : 'grab')
   304→  }, [mapReady, pick])
   305→
   306→  /* ---------- маркеры точек ---------- */
   307→  useEffect(() => {
   308→    const ym = ymRef.current
   309→    const col = pointsColRef.current
   310→    const layout = layoutRef.current
   311→    if (!mapReady || !ym || !col || layout === null || !points) return
   312→    col.removeAll()
   313→    markerRefs.current.clear()

   317→      const readings = active
   318→        ? `<br>🌡 ${p.sensor.tempC} °C · 💧 ${p.sensor.humidityPct} %<br>⚖ ${p.sensor.weightKg} кг · 🔊 ${p.sensor.noiseDb} дБ`
   319→        : `<br><i>Данных об улье нет — датчик ${SENSOR_LABEL[p.sensor.reason]}.</i>`
   320→      const pm = new ym.Placemark(
   321→        [p.lat, p.lon],
   322→        {
   323→          color,
   324→          label: `${p.hiveCount} ул.`,
   325→          balloonContentHeader: p.name,
   326→          balloonContentBody: `Датчик: <b>${p.sensorCode ?? '—'}</b><br>GPS: ${p.lat.toFixed(4)}, ${p.lon.toFixed(4)}<br>Состояние: ${SENSOR_LABEL[p.sensor.reason]}${readings}`,
   327→        },
   328→        {
   329→          iconLayout: layout,
   330→          iconShape: HIVE_ICON_SHAPE,
   331→          iconOffset: [-17, -17],
   332→          hideIconOnBalloonOpen: false,
   333→        },
   334→      )
   335→      col.add(pm)
   336→      markerRefs.current.set(p.id, pm)
   337→    }
   338→  }, [points, mapReady])
   339→
   340→  /* ---------- маркер выбранной точки ---------- */
   341→  useEffect(() => {
   342→    const ym = ymRef.current
   343→    const col = selColRef.current
   344→    if (!mapReady || !ym || !col) return
   345→    col.removeAll()
   346→    if (!selected) return
   347→    col.add(
   348→      new ym.Placemark(
   349→        [selected.lat, selected.lon],
   350→        { balloonContentBody: 'Выбранная точка — погода справа' },
   351→        { preset: 'islands#redDotIcon', iconColor: '#ef4444', hideIconOnBalloonOpen: false },
   352→      ),
   353→    )
   354→  }, [selected, mapReady])
   355→
   356→  /* ---------- превью позиции: GPS датчика или выбранное на карте место ---------- */
   357→  useEffect(() => {
   358→    const ym = ymRef.current
   359→    const col = prevColRef.current
   360→    const map = mapRef.current
   361→    if (!mapReady || !ym || !col) return
   362→    col.removeAll()
   363→    // Пока ждём клик по карте — превью не показываем
   364→    if (pick) return
   365→    let target: { lat: number; lon: number } | null = null
   366→    let header = ''
   367→    let body = ''
   368→    if (dialogOpen && placeMode === 'manual') {
   369→      if (!manualPos) return
   370→      target = manualPos
   371→      header = 'Место, выбранное на карте'
   372→      body = `Улей с датчиком <b>${normalizeCode(code) || '—'}</b> встанет сюда<br>Место: ${manualPos.lat.toFixed(4)}, ${manualPos.lon.toFixed(4)}`
   373→    } else if (dialogOpen && found) {
   374→      // Синяя метка «здесь будет улей»: позиция, которую передаёт датчик
   375→      target = { lat: found.lat, lon: found.lon }
   376→      header = 'Позиция по GPS датчика'
   377→      body = `Датчик: <b>${found.code}</b> · ${found.model}<br>GPS: ${found.lat.toFixed(4)}, ${found.lon.toFixed(4)}<br>${found.note}`
   378→    }
   379→    if (!target) return
   380→    col.add(
   381→      new ym.Placemark(
   382→        [target.lat, target.lon],
   383→        { balloonContentHeader: header, balloonContentBody: body },
   384→        { preset: 'islands#blueCircleDotIcon', iconColor: '#2563eb', hideIconOnBalloonOpen: false },
   385→      ),
   386→    )
   387→    if (map) {
   388→      if (map.getZoom() < 12) map.setZoom(12)
   389→      map.panTo([target.lat, target.lon], { duration: 700, flying: true })
   390→    }
   391→  }, [dialogOpen, found, manualPos, placeMode, mapReady, pick, code])
   392→
   393→  /* ---------- погода выбранной точки ---------- */
   394→  useEffect(() => {
   395→    if (!selected) {
   396→      setWeather(null)
   397→      return
   398→    }
   399→    const ctl = new AbortController()
   400→    setWeatherLoading(true)
   401→    setWeather(null)
   402→    fetch(`/api/weather?lat=${selected.lat}&lon=${selected.lon}`, {
   403→      signal: ctl.signal,
   404→      cache: 'no-store',
   405→    })
   406→      .then((r) => r.json() as Promise<WeatherResponse>)
   407→      .then((j) => setWeather(j))
   408→      .catch(() => undefined)
   409→      .finally(() => {
   410→        if (!ctl.signal.aborted) setWeatherLoading(false)
   411→      })
   412→    return () => ctl.abort()
   413→  }, [selected])
   414→
   415→  /** Плавный перелёт к точке: зум до 12, полёт Яндекса, затем открываем балун метки. */
   416→  // flyMapTo объявлен выше — рядом с переносом улья и выбором места
   417→
   418→  /* ---------- привязать улей по коду датчика ---------- */
   419→  const place = async () => {
   420→    if (!found) return
   421→    if (placeMode === 'manual' && !manualPos) return
   422→    setBusy(true)
   423→    try {
   424→      const manual = placeMode === 'manual' && manualPos !== null
   425→      const res = await fetch('/api/map-points', {
   426→        method: 'POST',
   427→        headers: { 'Content-Type': 'application/json' },
   428→        body: JSON.stringify(
   429→          manual && manualPos
   430→            ? { sensorCode: found.code, name: name.trim(), lat: manualPos.lat, lon: manualPos.lon }
   431→            : { sensorCode: found.code, name: name.trim() },
   432→        ),
   433→      })
