   543→   543→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
   544→   544→export function askAgent(question: string, currentHiveId: number): AgentMessage {
   545→   545→  const q = norm(question)
   546→   546→  const hiveId = extractHiveId(q)
   547→   547→  const h = getHive(hiveId ?? currentHiveId)
   548→   548→
   549→   549→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
   550→   550→
   551→   551→  // Приветствие и сервисные фразы
   552→   552→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
   553→   553→    return greetingAnswer()
   554→   554→  }
   555→   555→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
   556→   556→    return thanksAnswer()
   557→   557→  }
   558→   558→
   559→   559→  // Сценарий пестицидов
   560→   560→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
   561→   561→    return pesticideAnswer()
   562→   562→  }
   563→   563→
   564→   564→  // Кочёвка
   565→   565→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
   566→   566→    return migrationAnswer()
   567→   567→  }
   568→   568→
   569→   569→  // Погода и медосбор
   570→   570→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|когда вывоз|когда летать/)) {
   571→   571→    return weatherAnswer()
   572→   572→  }
   573→   573→
   574→   574→  // Вопрос про конкретный или выбранный улей
   575→   575→  if (has(/влажн|сырост|сыро/)) {
   576→   576→    return humidityAnswer(h)
   577→   577→  }
   578→   578→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
   579→   579→    return swarmAnswer(h)
   580→   580→  }
   581→   581→  if (has(/температ|жар|тепло внутри/)) {
   582→   582→    return climateAnswer(h)
   583→   583→  }
   584→   584→  if (has(/вес|мед\b|меда|привес|набор|тяже/)) {
   585→   585→    return weightAnswer(h)
   586→   586→  }
   587→   587→  if (has(/что происходит|состояни|как (себя|дела)|статус|проверь|проблем|тревог|что с улеем|как улей|расскажи про улей/)) {
   588→   588→    return statusAnswer(h)
   589→   589→  }
   590→   590→
   591→   591→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
   592→   592→  if (hiveId !== null) {
   593→   593→    return statusAnswer(h)
