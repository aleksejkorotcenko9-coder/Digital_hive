    86→  const mode = useAppState((s) => s.mode)

    93→function RealDashboard() {
    94→  const setView = useAppState((s) => s.setView)
    95→  const hiveId = useAppState((s) => s.hiveId)
    96→  const setHive = useAppState((s) => s.setHive)
    97→  const lang = useAppState((s) => s.lang)
    98→  const t = useT()
    99→  const { status, data, reload } = useRealData()
   100→
   101→  // Синхронизация с картой: ульи дашборда = точки на карте, показания — с
   102→  // датчиков. Опрос каждые 5 секунд — числа «живут» вместе с картой.
   103→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
   104→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
   105→  useEffect(() => {
   106→    let alive = true
   107→    const load = async () => {
   108→      try {
   109→        const res = await fetch('/api/map-points', { cache: 'no-store' })
   110→        if (!res.ok) return
   111→        const j = (await res.json()) as { points: MapPointDto[] }
   112→        if (alive) setPoints(j.points)
   113→      } catch {
   114→        // датчики подтянутся на следующем опросе
   115→      }
   116→    }
   117→    void load()
   118→    const timer = setInterval(() => void load(), 5000)
   119→    return () => {
   120→      alive = false
   121→      clearInterval(timer)
   122→    }
   123→  }, [])
   124→
   125→  // Погода по фактическому центру пасеки (среднее координат точек).
   126→  // Эффект зависит от ЧИСЕЛ координат: пересчёт центра при каждом опросе
   127→  // точек не вызывает перезапрос погоды, пока координаты не изменились.
   128→  const center = useMemo<[number, number]>(
   129→    () =>
   130→      (points && points.length > 0 ? pointsCenter(points) : null) ?? DEFAULT_PLACE.center,
   131→    [points],
   132→  )
   133→  const [wLat, wLon] = center
   134→  useEffect(() => {
   135→    let alive = true
   136→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
   137→    const load = async () => {
   138→      try {
   139→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   140→        if (!res.ok) return
   141→        const j = (await res.json()) as WeatherResponse
   142→        if (alive) setWeather(j)
   143→      } catch {
   144→        // агент честно скажет, если погоды нет
   145→      }
   146→    }
   147→    void load()
   148→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   149→    return () => {
   150→      alive = false
   151→      clearInterval(timer)
   152→    }
   153→  }, [wLat, wLon])
   154→  const agentWeather = useMemo(() => (weather ? weatherResponseToDays(weather) : null), [weather])
   155→
   156→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   157→
   158→  // выбранный улей всегда существует в синхронизированном наборе
   159→  useEffect(() => {
   160→    if (syncedHives && syncedHives.length > 0 && !syncedHives.some((h) => h.id === hiveId)) {
   161→      setHive(syncedHives[0].id)
   162→    }
   163→  }, [syncedHives, hiveId, setHive])
   164→
   165→  const headerNote =
   166→    status === 'error' ? t('real.note.error') : syncedHives?.length === 0 ? t('real.note.unbound') : null
   167→
   168→  if (status === 'idle' || status === 'loading') {
   169→    return <DashboardSkeleton />
   170→  }
   171→  if (status === 'error') {
   172→    return (
   173→      <div className="space-y-5">
   174→        <RealHeaderWithSwitcher note={t('real.note.error')} />
   175→        <Card className="mx-auto max-w-lg border-red-100">
   176→          <CardContent className="p-6 text-center">
   177→            <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   178→            <h2 className="mt-2 font-bold text-gray-900">{t('real.error.title')}</h2>
   179→            <p className="mt-1 text-sm text-gray-500">{t('real.error.text')}</p>
   180→            <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   181→              {t('real.error.retry')}
   182→            </Button>
   183→          </CardContent>
   184→        </Card>
   185→      </div>
   186→    )
   187→  }
   188→  // Схема и диаграммы — ТО ЖЕ точки, что на карте пасеки: один набор ульев
   189→  // для карты, вкладки «Датчики» и дашборда. Из базы — журнал и площадки кочёвки.
   190→  if (!points || !data || syncedHives === null) return <DashboardSkeleton />
   191→  if (syncedHives.length === 0) {
   192→    return (
   193→      <div className="space-y-5">
   194→        <RealHeaderWithSwitcher note={t('real.note.unbound')} />
   195→        <EmptyMapState onOpenMap={() => setView('map')} />
   196→      </div>
   197→    )
   198→  }
   199→  return (
   200→    <DashboardContent
   201→      hives={syncedHives}
   202→      sites={data.sites.map(siteDtoToSite)}
   203→      weather={agentWeather?.days ?? null}
   204→      events={data.events}
   205→      center={center}
   206→      lang={lang}
   207→    />
   208→  )
   209→}

   274→function DashboardContent({ hives, sites, weather, events, center, lang }: ContentProps) {
   275→  const t = useT()

   277→  const setHive = useAppState((s) => s.setHive)
   278→  const setView = useAppState((s) => s.setView)
   279→  const ask = useAppState((s) => s.ask)
   280→  const reminders = useAppState((s) => s.reminders)
   281→  const [openFinding, setOpenFinding] = useState<string | null>(null)
   282→
   283→  const hive = hives.find((x) => x.id === hiveId) ?? hives[0]
   284→  const status = hiveStatus(hive, hives)
   285→  const ctx = makeContext(hives, { weather, sites, apiaryName: 'Моя пасека' })
   286→  const findings = findingsFor(hive.id, ctx)
   287→
   288→  const riskyHive = hives.find((x) => hiveStatus(x, hives).level === 'risk')
   289→  const watchHive = hives.find((x) => hiveStatus(x, hives).level === 'watch' && x.id !== riskyHive?.id)
   290→
   291→  const lastPoint = hive.series[hive.series.length - 1]
   292→  const d2 = weightDeltaSafe(hive, 2)
   293→  const lastWeight = lastPoint?.weightKg ?? null
   294→  const lastTemp = lastPoint?.tempInC ?? null
   295→  const lastHum = lastPoint?.humidityPct ?? null
   296→
   297→  // длительность наблюдений — честно из дат датчика
   298→  const periodDays = useMemo(() => {
   299→    if (hive.series.length < 2) return null
   300→    const a = new Date(hive.series[0].date).getTime()
   301→    const b = new Date(hive.series[hive.series.length - 1].date).getTime()
   302→    return Math.round((b - a) / 86400000) + 1
   303→  }, [hive.series])
   304→  const periodLabel =
   305→    periodDays === null
   306→      ? t('real.passport.fewDays')
   307→      : periodDays >= 25
   308→        ? t('real.passport.month')
   309→        : t('real.passport.days', { n: periodDays })
   310→
   311→  // показатели пасеки целиком
   312→  const withTelemetry = hives.filter((h) => h.series.length > 0 && h.series[h.series.length - 1].weightKg !== null)
   313→  const gauge = apiaryGauge(hives.map((h) => hiveStatus(h, hives).level), hives.length)
   314→  const leader = bestHiveOf(hives)
   315→  const leaderGain = weightDeltaSafe(leader, 30) ?? weightTotal(leader)
   316→
   317→  const upcomingReminders = reminders
   318→    .filter((r) => !r.done)
   319→    .sort((a, b) => a.remindAt.localeCompare(b.remindAt))
   320→    .slice(0, 4)
   321→
   322→  const todayLabel = format(new Date(), 'EEEE, d MMMM yyyy', { locale: LANG_DATE_LOCALE[lang] })
   323→
   324→  return (
   325→    <div className="space-y-4">
   326→      {/* ── Баннер срочности по данным датчиков ── */}
   327→      <Rise delay={0}>

   329→          <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-[#D93E00] to-[#FF5A1F] p-4 text-white shadow-lg shadow-red-100 md:p-4.5">
   330→            <TrendingUp className="absolute -right-3 -top-3 h-28 w-28 rotate-12 text-white/15" />
   331→            <div className="relative flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   332→              <div className="flex items-start gap-3">
   333→                <AlertTriangle className="animate-heartbeat mt-0.5 h-6 w-6 shrink-0" />
   334→                <div>
   335→                  <p className="text-[13px] font-extrabold uppercase tracking-wide text-white/95">
   336→                    {t('real.banner.risk.badge')}
   337→                  </p>
   338→                  <p className="mt-1 max-w-3xl text-sm leading-snug">
   339→                    {t('real.banner.risk.text', { n: riskyHive.id, w: fmtSigned(weightDeltaSafe(riskyHive, 2) ?? 0) })}
   340→                  </p>
   341→                </div>
   342→              </div>
   343→              <Button
   344→                className="w-fit shrink-0 bg-white font-bold text-red-700 hover:bg-red-50"
   345→                onClick={() => ask(`Что происходит с ульем №${riskyHive.id}?`)}
   346→              >
   347→                {t('real.banner.cta')} <ArrowRight className="h-4 w-4" />
   348→              </Button>
   349→            </div>
   350→          </div>

   353→            <AlertTriangle className="h-5 w-5 shrink-0 text-amber-500" />
   354→            <p className="text-sm">
   355→              {t('real.banner.watch', { n: watchHive.id, label: hiveStatus(watchHive, hives).label.toLowerCase() })}

   851→      {note && <p className="mt-1 text-[11px] text-gray-400">{note}</p>}
   852→    </div>

   856→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   857→function weightTotal(h: Hive): number {
   858→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   859→  const first = h.series[0]?.weightKg ?? null
   860→  if (w === null || first === null) return 0
   861→  return Math.round((w - first) * 10) / 10
   862→}
