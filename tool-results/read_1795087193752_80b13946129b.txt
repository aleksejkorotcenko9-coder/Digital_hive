   110→   110→/** Статус заклещенности: Норма <2 % / Риск 2–3 % / Критично >3 %. */
   111→   111→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; label: string; cls: string } {
   112→   112→  if (pct > 3) return { level: 'crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   113→   113→  if (pct >= 2) return { level: 'risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   114→   114→  return { level: 'ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   115→   115→}
   116→   116→
   117→   117→/** Состояние воздуха по индексу ЛОС (0…100). */
   118→   118→export function vocState(v: number): { label: string; cls: string } {
   119→   119→  if (v > 60) return { label: 'Духота — семье не хватает объема', cls: 'text-red-600' }
   120→   120→  if (v > 30) return { label: 'Умеренная загрузка воздухом', cls: 'text-amber-600' }
   121→   121→  return { label: 'Воздух чистый', cls: 'text-emerald-600' }
   122→   122→}

   124→   124→/** Состояние трафика на летке относительно нормы пасеки (45 пч/мин). */
   125→   125→export function trafficState(v: number): { label: string; cls: string } {
   126→   126→  if (v >= 55) return { label: 'Интенсивный лёт', cls: 'text-emerald-600' }
   127→   127→  if (v >= 30) return { label: 'Активный лёт', cls: 'text-emerald-600' }
   128→   128→  if (v >= 18) return { label: 'Лёт ослаблен', cls: 'text-amber-600' }
   129→   129→  return { label: 'Лёт резко снижен', cls: 'text-red-600' }
   130→   130→}

   133→   133→export function demoGaugeInfo(value: number): { label: string; tone: 'ok' | 'normal' | 'watch' | 'alert' } {

   137→   137→  return { label: 'Отличное состояние', tone: 'ok' }
   138→   138→}

   556→   556→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   557→   557→function seeded(seed: number): () => number {
   558→   558→  let a = seed >>> 0
   559→   559→  return () => {
   560→   560→    a |= 0
   561→   561→    a = (a + 0x6d2b79f5) | 0
   562→   562→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   563→   563→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   564→   564→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   565→   565→  }
   566→   566→}

   568→   568→/** Температура и влажность за 14 дней: колебания вокруг базовых значений. */
   569→   569→export function demoClimateSeries(h: DemoHive, days = 14): DayPoint[] {
   570→   570→  const rnd = seeded(1000 + h.id * 97)
   571→   571→  const out: DayPoint[] = []
   572→   572→  const nowMs = Date.now()
   573→   573→  for (let k = days - 1; k >= 0; k--) {
   574→   574→    const t = new Date(nowMs - k * 86400000)
   575→   575→    const recent = k <= 1
   576→   576→    const tempC = recent
   577→   577→      ? h.base.tempC + (rnd() - 0.5) * 0.2
   578→   578→      : h.base.tempC - 0.4 + rnd() * 0.8
   579→   579→    const humidityPct = recent
   580→   580→      ? h.base.humidityPct + (rnd() - 0.5) * 2
   581→   581→      : h.base.humidityPct - 4 + rnd() * 8
   582→   582→    out.push({
   583→   583→      date: format(t, 'yyyy-MM-dd'),
   584→   584→      label: format(t, 'd.MM'),
   585→   585→      full: format(t, 'd MMMM', { locale: ru }),
   586→   586→      weightKg: null,
   587→   587→      tempInC: Math.round(tempC * 10) / 10,
   588→   588→      humidityPct: Math.round(humidityPct),
   589→   589→    })
   590→   590→  }
   591→   591→  // последние две точки точно соответствуют «сейчас»
   592→   592→  out[out.length - 1].tempInC = h.base.tempC
   593→   593→  out[out.length - 1].humidityPct = h.base.humidityPct
   594→   594→  return out
   595→   595→}

   617→   617→export interface VarroaPoint {
   618→   618→  label: string
   619→   619→  /** Историческое значение заклещенности, %. */
   620→   620→  pct: number | null
   621→   621→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   622→   622→  forecast: number | null
   623→   623→  /** Отметка проведенной обработки. */
   624→   624→  treated: boolean
   625→   625→}
   626→   626→
   627→   627→/**
   628→   628→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   629→   629→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   630→   630→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   631→   631→ */
   632→   632→export function demoVarroaSeries(h: DemoHive, days = 30): VarroaPoint[] {
   633→   633→  const rnd = seeded(3000 + h.id * 57)
   634→   634→  const out: VarroaPoint[] = []
   635→   635→  const nowMs = Date.now()
   636→   636→
   637→   637→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   638→   638→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   639→   639→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   640→   640→
   641→   641→  for (let k = days - 1; k >= 0; k--) {
   642→   642→    const t = new Date(nowMs - k * 86400000)
   643→   643→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   644→   644→    let pct = startPct + (h.base.varroaPct - startPct) * progress
   645→   645→    // дип после обработки: −1,2 % в течение 3 дней
   646→   646→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   647→   647→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   648→   648→    }
   649→   649→    pct += (rnd() - 0.5) * 0.12
   650→   650→    out.push({
   651→   651→      label: format(t, 'd.MM'),
   652→   652→      pct: Math.round(pct * 10) / 10,
   653→   653→      forecast: null,
   654→   654→      treated: treatDay > 0 && k === treatDay,
   655→   655→    })
   656→   656→  }
   657→   657→  out[out.length - 1].pct = h.base.varroaPct
   658→   658→
   659→   659→  // прогноз: через 14 дней (пунктир от текущей точки)
   660→   660→  const future = new Date(nowMs + 14 * 86400000)
   661→   661→  out.push({
   662→   662→    label: format(future, 'd.MM'),
   663→   663→    pct: null,
   664→   664→    forecast: h.base.varroaForecast14d,
   665→   665→    treated: false,
   666→   666→  })
   667→   667→  // сегмент начинается с последней исторической точки
   668→   668→  out[out.length - 2].forecast = h.base.varroaPct
   669→   669→  return out
   670→   670→}

   694→   694→  return GDD_CROPS.map((c) => {
   695→   695→    const daysLeft = Math.max(1, Math.round((c.need - GDD_CURRENT) / GDD_PER_DAY))
   696→   696→    return {
   697→   697→      ...c,
   698→   698→      pct: Math.min(100, Math.round((GDD_CURRENT / c.need) * 100)),
   699→   699→      daysLeft,
   700→   700→      soon: daysLeft <= 7,
   701→   701→    }
   702→   702→  })
   703→   703→}
