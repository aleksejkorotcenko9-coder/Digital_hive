    98→export function DashboardView() {
    99→  const mode = useAppState((s) => s.mode)
   100→  // «Демо» — сценарные ульи с живой телеметрией; «Мои данные» — прежний режим,

   106→function RealDashboard() {
   107→  const setView = useAppState((s) => s.setView)
   108→  const hiveId = useAppState((s) => s.hiveId)
   109→  const setHive = useAppState((s) => s.setHive)
   110→  const { status, data, reload } = useRealData()
   111→
   112→  // Синхронизация с картой: ульи дашборда = точки на карте, показания — с
   113→  // датчиков. Опрос каждые 5 секунд — числа «живут» вместе с картой.
   114→  const [points, setPoints] = useState<MapPointDto[] | null>(null)
   115→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
   116→  useEffect(() => {
   117→    let alive = true
   118→    const load = async () => {
   119→      try {
   120→        const res = await fetch('/api/map-points', { cache: 'no-store' })
   121→        if (!res.ok) return
   122→        const j = (await res.json()) as { points: MapPointDto[] }
   123→        if (alive) setPoints(j.points)
   124→      } catch {
   125→        // датчики подтянутся на следующем опросе
   126→      }
   127→    }
   128→    void load()
   129→    const timer = setInterval(() => void load(), 5000)
   130→    return () => {
   131→      alive = false
   132→      clearInterval(timer)
   133→    }
   134→  }, [])
   135→
   136→  // Погода по фактическому центру пасеки (среднее координат точек).
   137→  // Эффект зависит от ЧИСЕЛ координат: пересчёт центра при каждом опросе
   138→  // точек не вызывает перезапрос погоды, пока координаты не изменились.
   139→  const center = useMemo<[number, number]>(
   140→    () =>
   141→      (points && points.length > 0 ? pointsCenter(points) : null) ?? DEFAULT_PLACE.center,
   142→    [points],
   143→  )
   144→  const [wLat, wLon] = center
   145→  useEffect(() => {
   146→    let alive = true
   147→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
   148→    const load = async () => {
   149→      try {
   150→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   151→        if (!res.ok) return
   152→        const j = (await res.json()) as WeatherResponse
   153→        if (alive) setWeather(j)
   154→      } catch {
   155→        // агент честно скажет, если погоды нет
   156→      }
   157→    }
   158→    void load()
   159→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   160→    return () => {
   161→      alive = false
   162→      clearInterval(timer)
   163→    }
   164→  }, [wLat, wLon])
   165→  const agentWeather = useMemo(() => (weather ? weatherResponseToDays(weather) : null), [weather])
   166→
   167→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   168→
   169→  // выбранный улей всегда существует в синхронизированном наборе
   170→  useEffect(() => {
   171→    if (syncedHives && syncedHives.length > 0 && !syncedHives.some((h) => h.id === hiveId)) {
   172→      setHive(syncedHives[0].id)
   173→    }
   174→  }, [syncedHives, hiveId, setHive])
   175→
   176→  if (status === 'idle' || status === 'loading') {
   177→    return <DashboardSkeleton />
   178→  }
   179→  if (status === 'error') {
   180→    return (
   181→      <div className="space-y-5">
   182→        <RealHeaderWithSwitcher note="данные недоступны" />
   183→        <Card className="mx-auto max-w-lg border-red-100">
   184→          <CardContent className="p-6 text-center">
   185→            <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   186→            <h2 className="mt-2 font-bold text-gray-900">Не удалось загрузить данные</h2>
   187→            <p className="mt-1 text-sm text-gray-500">
   188→              Проверьте, что база данных доступна, и попробуйте ещё раз.
   189→            </p>
   190→            <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   191→              Повторить загрузку
   192→            </Button>
   193→          </CardContent>
   194→        </Card>
   195→      </div>
   196→    )
   197→  }
   198→  // Схема и диаграммы — ТО ЖЕ точки, что на карте пасеки: один набор ульев
   199→  // для карты, вкладки «Датчики» и дашборда. Из базы — журнал и площадки кочёвки.
   200→  if (!points || !data || syncedHives === null) return <DashboardSkeleton />
   201→  if (syncedHives.length === 0) {
   202→    return (
   203→      <div className="space-y-5">
   204→        <RealHeaderWithSwitcher note="ульи не привязаны" />
   205→        <EmptyMapState onOpenMap={() => setView('map')} />
   206→      </div>
   207→    )
   208→  }
   209→  return (
   210→    <DashboardContent
   211→      hives={syncedHives}
   212→      sites={data.sites.map(siteDtoToSite)}
   213→      weather={agentWeather?.days ?? null}
   214→      events={data.events}
   215→      center={center}
   216→    />
   217→  )
   218→}

   283→function DashboardContent({ hives, sites, weather, events, center }: ContentProps) {
   284→  const hiveId = useAppState((s) => s.hiveId)
   285→  const setHive = useAppState((s) => s.setHive)
   286→  const setView = useAppState((s) => s.setView)
   287→  const ask = useAppState((s) => s.ask)
   288→  const reminders = useAppState((s) => s.reminders)
   289→  const [openFinding, setOpenFinding] = useState<string | null>(null)
   290→
   291→  const hive = hives.find((x) => x.id === hiveId) ?? hives[0]
   292→  const status = hiveStatus(hive, hives)
   293→  const ctx = makeContext(hives, { weather, sites, apiaryName: 'Моя пасека' })
   294→  const findings = findingsFor(hive.id, ctx)
   295→
   296→  const riskyHive = hives.find((x) => hiveStatus(x, hives).level === 'risk')
   297→  const watchHive = hives.find((x) => hiveStatus(x, hives).level === 'watch' && x.id !== riskyHive?.id)
   298→
   299→  const lastPoint = hive.series[hive.series.length - 1]
   300→  const d2 = weightDeltaSafe(hive, 2)
   301→  const lastWeight = lastPoint?.weightKg ?? null
   302→  const lastTemp = lastPoint?.tempInC ?? null
   303→  const lastHum = lastPoint?.humidityPct ?? null
   304→  const period = periodCaption(hive)
   305→
   306→  // показатели пасеки целиком
   307→  const withTelemetry = hives.filter((h) => h.series.length > 0 && h.series[h.series.length - 1].weightKg !== null)
   308→  const gauge = apiaryGauge(hives.map((h) => hiveStatus(h, hives).level), hives.length)
   309→  const leader = bestHiveOf(hives)
   310→
   311→  const upcomingReminders = reminders
   312→    .filter((r) => !r.done)
   313→    .sort((a, b) => a.remindAt.localeCompare(b.remindAt))
   314→    .slice(0, 4)
   315→
   316→  const todayLabel = format(new Date(), 'EEEE, d MMMM yyyy', { locale: ru })
   317→
   318→  return (
   319→    <div className="space-y-5">
   320→      {/* Верхний баннер срочности */}
   321→      {riskyHive ? (
   322→        <div className="relative overflow-hidden rounded-2xl bg-gradient-to-r from-[#D93E00] to-[#FF5A1F] p-4 text-white shadow-lg shadow-red-100 md:p-5">
   323→          <Split className="absolute -right-3 -top-3 h-28 w-28 rotate-12 text-white/15" />
   324→          <div className="relative flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   325→            <div className="flex items-start gap-3">
   326→              <AlertTriangle className="mt-0.5 h-6 w-6 shrink-0" />
   327→              <div>
   328→                <p className="text-sm font-bold uppercase tracking-wide text-white/90">
   329→                  Сигнал данных · возможное роение
   330→                </p>
   331→                <p className="mt-1 text-sm leading-snug md:text-base">
   332→                  Улей №{riskyHive.id}: вес снизился на {fmtSigned(weightDeltaSafe(riskyHive, 2) ?? 0)} кг за 2 дня при
   333→                  тёплой погоде — похоже на подготовку семьи к делению.
   334→                </p>
   335→              </div>
   336→            </div>
   337→            <Button
   338→              className="w-fit shrink-0 bg-white font-semibold text-orange-700 hover:bg-orange-50"
   339→              onClick={() => ask(`Что происходит с ульем №${riskyHive.id}?`)}
   340→            >
   341→              Что делать? <ArrowRight className="h-4 w-4" />
   342→            </Button>
   343→          </div>
   344→        </div>
   345→      ) : watchHive ? (
   346→        <div className="flex items-center gap-3 rounded-2xl border border-amber-200 bg-amber-50 px-4 py-3 text-amber-800">
   347→          <AlertTriangle className="h-5 w-5 shrink-0 text-amber-500" />
   348→          <p className="text-sm">
   349→            Улей №{watchHive.id}: {hiveStatus(watchHive, hives).label.toLowerCase()} — проверьте семью в ближайшие дни.
   350→          </p>
   351→        </div>
   352→      ) : (
   353→        <div className="flex items-center gap-3 rounded-2xl border border-emerald-200 bg-emerald-50 px-4 py-3 text-emerald-800">
   354→          <Sun className="h-5 w-5 shrink-0 text-emerald-500" />
   355→          <p className="text-sm">
   356→            По данным датчиков тревожных сигналов нет — все семьи с телеметрией работают штатно.
   357→          </p>
   358→        </div>
   359→      )}
   360→
   361→      {/* Заголовок пасеки */}
   362→      <div className="flex flex-wrap items-end justify-between gap-3">
   363→        <div>
   364→          <h1 className="text-2xl font-extrabold tracking-tight text-[#2A1A12] md:text-3xl">

   434→            {/* Выбор улья */}

   436→              {hives.map((h) => {

   669→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   670→function weightTotal(h: Hive): number {
   671→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   672→  const first = h.series[0]?.weightKg ?? null
   673→  if (w === null || first === null) return 0
   674→  return Math.round((w - first) * 10) / 10
   675→}
