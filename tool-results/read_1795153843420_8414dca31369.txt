     3→/**
     4→ * Экран «ИИ-агент»: чат пчеловода с цифровым помощником.
     5→ * Агент отвечает сразу по всем ульям пасеки: телеметрия из базы, датчики точек
     6→ * на карте (температура, вес, влажность, шум) и реальная погода из интернета.
     7→ * Темы строго ограничены: погода и ульи — отвечает, всё остальное — вежливо
     8→ * отказывается. Модель, на которой работает агент, пользователю не показывается.
     9→ * Голосовой ввод работает как рация (push-to-talk): кнопку микрофона нужно
    10→ * зажать и говорить — пока она удерживается, идёт запись; при отпускании запись
    11→ * завершается, речь распознаётся (Web Speech или сервер /api/asr) и текст
    12→ * подставляется в поле ввода.
    13→ * Улии и показания синхронизированы с картой: агент говорит по тем же точкам
    14→ * и датчикам, что видит пчеловод на карте и дашборде.
    15→ * История переписки сохраняется в SQLite (общая лента пасеки).
    16→ */

   189→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   190→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   191→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   192→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   193→  const srClass = useMemo(() => getSpeechRecognition(), [])
   194→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   195→  const [voice, setVoice] = useState<VoiceState>('idle')

   202→  // push-to-talk: кнопка микрофона зажата и момент начала удержания

   207→  const clearAutoStop = () => {

   218→        speechRef.current.stop()

   221→      }
   222→      return
   223→    }
   224→    const rec = recorderRef.current
   225→    if (rec && rec.state !== 'inactive') rec.stop()

   256→    }

   258→    try {
   259→      const wav = await blobToWavBase64(blob)
   260→      const res = await fetch('/api/asr', {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },

   274→      toast({ title: 'Голос распознан', description: text })
   275→    } catch {
   276→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   277→    } finally {
   278→      setVoice('idle')
   279→    }
   280→  }, [toast])
   281→
   282→  const startVoice = useCallback(async () => {
   283→    // запись уже идёт — второй старт (второй палец/указатель) не нужен
   284→    if (speechRef.current || recorderRef.current) return
   285→    clearAutoStop()
   286→    baseInputRef.current = input.trim()
   287→
   288→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   289→    if (srClass) {
   290→      const rec = new srClass()
   291→      rec.lang = VOICE_LANG
   292→      rec.continuous = true
   293→      rec.interimResults = true
   294→      rec.maxAlternatives = 1
   295→      let finalText = ''
   296→      rec.onresult = (e) => {
   297→        let interim = ''
   298→        for (let i = e.resultIndex; i < e.results.length; i++) {
   299→          const r = e.results[i]
   300→          const t = r[0]?.transcript ?? ''
   301→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   302→          else interim += t
   303→        }
   304→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   305→        if (text) {
   306→          const base = baseInputRef.current
   307→          setInput(base ? `${base} ${text}` : text)
   308→        }
   309→      }
   310→      rec.onerror = (ev) => {
   311→        const code = ev.error ?? ''
   312→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   313→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   314→        } else if (code === 'no-speech') {
   315→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   316→        } else if (code !== 'aborted') {
   317→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   318→        }
   319→      }
   320→      rec.onend = () => {
   321→        speechRef.current = null
   322→        clearAutoStop()
   323→        setVoice('idle')
   324→      }
   325→      speechRef.current = rec
   326→      try {
   327→        rec.start()
   328→      } catch {
   329→        toast({ title: 'Не получилось начать', description: 'Зажмите кнопку микрофона ещё раз.', variant: 'destructive' })
   330→        return
   331→      }
   332→      setVoice('recording')
   333→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   334→      // кнопку отпустили раньше, чем распознавание успело стартовать
   335→      if (!holdActiveRef.current) stopVoice()
   336→      return
   337→    }
   338→
   339→    // 2) запасной путь: запись с микрофона → серверное распознавание
   340→    try {
   341→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   342→      streamRef.current = stream
   343→      chunksRef.current = []
   344→      const rec = new MediaRecorder(stream)
   345→      rec.ondataavailable = (e) => {
   346→        if (e.data.size > 0) chunksRef.current.push(e.data)
   347→      }
   348→      rec.onstop = () => void finishVoice()
   349→      rec.start()
   350→      recorderRef.current = rec
   351→      setVoice('recording')
   352→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   353→      // кнопку отпустили, пока запрашивался доступ к микрофону
   354→      if (!holdActiveRef.current) stopVoice()
   355→    } catch {
   356→      toast({
   357→        title: 'Микрофон недоступен',
   358→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   359→        variant: 'destructive',
   360→      })
   361→    }
   362→  }, [srClass, input, finishVoice, stopVoice, toast])
   363→
   364→  // при размонтировании прерываем распознавание и закрываем поток
   365→  useEffect(
   366→    () => () => {
   367→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   368→      try {
   369→        speechRef.current?.abort()
   370→      } catch {
   371→        // нет активного распознавания
   372→      }
   373→      streamRef.current?.getTracks().forEach((t) => t.stop())
   374→    },
   375→    [],
   376→  )
   377→
   378→  /* История из базы — общая лента пасеки, без привязки к улью */
   379→  useEffect(() => {
   380→    if (!user || realStatus !== 'ready' || !ctx) return
   381→    let cancelled = false
   382→    void (async () => {
   383→      try {
   384→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   385→        if (!res.ok) throw new Error('history')
   386→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   387→        if (cancelled) return
   388→        const restored: ChatItem[] = j.items.map((m) => ({
   389→          role: m.role,
   390→          text: m.text,
   391→          ...(m.role === 'agent' && m.agentJson
   392→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   393→            : {}),
   394→        }))
   395→        if (restored.length === 0) {
   396→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   397→        } else {
   398→          setItems(restored.slice(-30))
   399→        }
   400→      } catch {
   401→        if (!cancelled) {
   402→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   403→        }
   404→      }
   405→    })()
   406→    return () => {
   407→      cancelled = true
   408→    }
   409→  }, [user, realStatus, ctx])
   410→
   411→  const sendUser = useCallback(
   412→    (rawText: string) => {
   413→      const trimmed = rawText.trim()
   414→      if (!trimmed || thinkingRef.current) return
   415→      // отправка текста прерывает голосовой ввод (кнопка могла быть ещё зажата)
   416→      if (holdActiveRef.current) {
   417→        holdActiveRef.current = false
   418→        stopVoice()
   419→      }
   420→      const context = ctx
   421→      if (!context) return
   422→      thinkingRef.current = true
   423→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   424→      setInput('')
   425→      setThinking(true)
   426→      const started = Date.now()
   427→
   428→      const run = async () => {
   429→        let answer: AgentMessage
   430→        try {
   431→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   432→          const res = await fetch('/api/agent', {
   433→            method: 'POST',
   434→            headers: { 'Content-Type': 'application/json' },
   435→            body: JSON.stringify({ question: trimmed }),
   436→          })
   437→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   438→          const j = (await res.json()) as { answer: AgentMessage }
   439→          answer = j.answer
   440→        } catch {
   441→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   442→        }
   443→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   444→        const elapsed = Date.now() - started
   445→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   446→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   447→        thinkingRef.current = false
   448→        setThinking(false)
   449→        // сохраняем переписку в базу — общая лента пасеки
   450→        if (useAppState.getState().user) {
   451→          void fetch('/api/chat', {
   452→            method: 'POST',
   453→            headers: { 'Content-Type': 'application/json' },
   454→            body: JSON.stringify({
   455→              userText: trimmed,
   456→              agentText: answer.text,
   457→              agent: answer,
   458→            }),
   459→          }).catch(() => undefined)
   460→        }
