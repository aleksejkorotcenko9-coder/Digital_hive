   154→   154→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */

   164→   164→  const chunksRef = useRef<Blob[]>([])
   165→   165→  const autoStopRef = useRef<number | null>(null)

   167→   167→
   168→   168→  const clearAutoStop = () => {

   220→   220→    } catch {
   221→   221→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   222→   222→    } finally {
   223→   223→      setVoice('idle')
   224→   224→    }

   227→   227→  const startVoice = useCallback(async () => {

   242→   242→          const r = e.results[i]

   244→   244→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   245→   245→          else interim += t
   246→   246→        }
   247→   247→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   248→   248→        if (text) {
   249→   249→          const base = baseInputRef.current
   250→   250→          setInput(base ? `${base} ${text}` : text)
   251→   251→        }
   252→   252→      }
   253→   253→      rec.onerror = (ev) => {
   254→   254→        const code = ev.error ?? ''
   255→   255→        if (code === 'not-allowed' || code === 'service-not-allowed') {

   257→   257→        } else if (code === 'no-speech') {
   258→   258→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   259→   259→        } else if (code !== 'aborted') {
   260→   260→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   261→   261→        }
   262→   262→      }
   263→   263→      rec.onend = () => {
   264→   264→        speechRef.current = null
   265→   265→        clearAutoStop()
   266→   266→        setVoice('idle')
   267→   267→      }
   268→   268→      speechRef.current = rec

   270→   270→        rec.start()
   271→   271→      } catch {
   272→   272→        toast({ title: 'Не получилось начать', description: 'Нажмите микрофон ещё раз.', variant: 'destructive' })
   273→   273→        return
   274→   274→      }
   275→   275→      setVoice('recording')
   276→   276→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   277→   277→      return
   278→   278→    }
   279→   279→

   293→   293→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)

   296→   296→        title: 'Микрофон недоступен',
   297→   297→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   298→   298→        variant: 'destructive',
   299→   299→      })
   300→   300→    }
   301→   301→  }, [srClass, input, finishVoice, stopVoice, toast])
   302→   302→

   328→   328→          role: m.role,
   329→   329→          text: m.text,

   333→   333→        }))
   334→   334→        if (restored.length === 0) {
   335→   335→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   336→   336→        } else {
   337→   337→          setItems(restored.slice(-30))
   338→   338→        }
   339→   339→      } catch {
   340→   340→        if (!cancelled) {
   341→   341→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   342→   342→        }
   343→   343→      }
   344→   344→    })()
   345→   345→    return () => {
   346→   346→      cancelled = true
   347→   347→    }
   348→   348→  }, [mode, user, realStatus, ctx])
   349→   349→
   350→   350→  /* Демо: приветствие один раз (история в демо не сохраняется) */
   351→   351→  useEffect(() => {
   352→   352→    if (mode !== 'demo' || !ctx) return

   354→   354→      setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   355→   355→    }
   356→   356→  }, [mode, ctx, items.length])
   357→   357→
   358→   358→  const sendUser = useCallback(
   359→   359→    (rawText: string) => {
   360→   360→      const trimmed = rawText.trim()
   361→   361→      if (!trimmed || thinkingRef.current) return
   362→   362→      const context =
   363→   363→        mode === 'demo'
   364→   364→          ? demoCtx
   365→   365→          : data
   366→   366→            ? buildRealContext(data, agentWeather?.days ?? null, agentWeather?.now ?? null)
   367→   367→            : null
   368→   368→      if (!context) return
   369→   369→      thinkingRef.current = true
   370→   370→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   371→   371→      setInput('')
   372→   372→      setThinking(true)
   373→   373→      const started = Date.now()
   374→   374→
   375→   375→      const run = async () => {
   376→   376→        let answer: AgentMessage
   377→   377→        try {
   378→   378→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   379→   379→          const res = await fetch('/api/agent', {
   380→   380→            method: 'POST',
   381→   381→            headers: { 'Content-Type': 'application/json' },
   382→   382→            body: JSON.stringify({ question: trimmed, mode }),
   383→   383→          })
   384→   384→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   385→   385→          const j = (await res.json()) as { answer: AgentMessage }
   386→   386→          answer = j.answer
   387→   387→        } catch {
   388→   388→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   389→   389→        }
   390→   390→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   391→   391→        const elapsed = Date.now() - started
   392→   392→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   393→   393→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   394→   394→        thinkingRef.current = false
   395→   395→        setThinking(false)
   396→   396→        // сохраняем переписку в базу (реальный режим) — общая лента пасеки
   397→   397→        if (mode === 'real' && useAppState.getState().user) {
   398→   398→          void fetch('/api/chat', {
   399→   399→            method: 'POST',
   400→   400→            headers: { 'Content-Type': 'application/json' },
   401→   401→            body: JSON.stringify({
