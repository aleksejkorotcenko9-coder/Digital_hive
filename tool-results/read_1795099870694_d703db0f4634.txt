    32→2. Логика кодов — src/lib/verification.ts (полностью)
    33→ts
    34→/**
    35→ * Подтверждение email: коды и токены.
    36→ *
    37→ * КОД (основной способ): 6 цифр, crypto.randomInt. В БД хранится ТОЛЬКО
    38→ * SHA-256 хеш кода (утечка базы не раскрывает код), сырой код уходит
    39→ * пользователю в письме. Срок жизни — 30 минут, код одноразовый;
    40→ * не более 5 неверных попыток ввода.
    41→ *
    42→ * ТОКЕН-ССЫЛКА (legacy, остался для /api/auth/verify): randomBytes(32) → hex,
    43→ * в БД SHA-256 хеш, срок 24 часа, одноразовый.
    44→ *
    45→ * Плюс rate limit повторной отправки: не более 3 запросов на один email в час.
    46→ * Хранение в памяти процесса (Map) — достаточно для этого приложения.
    47→ */
    48→import { createHash, randomBytes, randomInt } from 'node:crypto'
    49→import { db } from './db'
    50→
    51→export const VERIFICATION_TTL_MS = 24 * 3600 * 1000 // 24 часа (для токена-ссылки)
    52→export const CODE_TTL_MS = 30 * 60 * 1000 // 30 минут (для кода из письма)
    53→const MAX_CODE_ATTEMPTS = 5
    54→
    55→function hashToken(raw: string): string {
    56→  return createHash('sha256').update(raw).digest('hex')
    57→}
    58→
    59→/**
    60→ * Создаёт новый токен подтверждения для пользователя. Предыдущие токены
    61→ * пользователя удаляются — действительна всегда только последняя ссылка.
    62→ * Возвращает СЫРОЙ токен (только для ссылки в письме).
    63→ */
    64→export async function issueVerificationToken(userId: string): Promise<string> {
    65→  const raw = randomBytes(32).toString('hex')
    66→  await db.verificationToken.deleteMany({ where: { userId } })
    67→  await db.verificationToken.create({
    68→    data: {
    69→      userId,
    70→      tokenHash: hashToken(raw),
    71→      expiresAt: new Date(Date.now() + VERIFICATION_TTL_MS),
    72→    },
    73→  })
    74→  return raw
    75→}
    76→
    77→export type ConsumeResult = { ok: true; userId: string } | { ok: false }
    78→
    79→/* ------------------------------------------------------------------ */
    80→/* КОД ПОДТВЕРЖДЕНИЯ (основной способ)                                  */
    81→/* ------------------------------------------------------------------ */
    82→
    83→/** userId → счётчик неверных попыток ввода кода. */
    84→const codeAttempts = new Map<string, { count: number; firstAt: number }>()
    85→
    86→function resetAttempts(userId: string): void {
    87→  codeAttempts.delete(userId)
    88→}
    89→
    90→function failAttempt(userId: string): number {
    91→  const now = Date.now()
    92→  const rec = codeAttempts.get(userId)
    93→  if (!rec || now - rec.firstAt > CODE_TTL_MS) {
    94→    codeAttempts.set(userId, { count: 1, firstAt: now })
    95→    return 1
    96→  }
    97→  rec.count += 1
    98→  return rec.count
    99→}
   100→
   101→function attemptsLeft(userId: string): number {
   102→  const rec = codeAttempts.get(userId)
   103→  if (!rec || Date.now() - rec.firstAt > CODE_TTL_MS) return MAX_CODE_ATTEMPTS
   104→  return Math.max(0, MAX_CODE_ATTEMPTS - rec.count)
   105→}
   106→
   107→/**
   108→ * Создаёт новый 6-значный код подтверждения для пользователя. Предыдущие
   109→ * записи пользователя удаляются — действителен всегда только последний код.
   110→ * Возвращает СЫРОЙ код (только для письма).
   111→ */
   112→export async function issueVerificationCode(userId: string): Promise<string> {
   113→  const code = String(randomInt(100000, 1000000))
   114→  await db.verificationToken.deleteMany({ where: { userId } })
   115→  await db.verificationToken.create({
   116→    data: {
   117→      userId,
   118→      tokenHash: hashToken(code),
   119→      expiresAt: new Date(Date.now() + CODE_TTL_MS),
   120→    },
   121→  })
   122→  resetAttempts(userId)
   123→  return code
   124→}
   125→
   126→export type CodeConsumeResult = 'ok' | 'invalid' | 'expired' | 'locked'
   127→
   128→/**
   129→ * Проверяет код из письма по email. Валидный и не истёкший — ставит
   130→ * User.emailVerified и удаляет запись (одноразовость).
   131→ * 'invalid' — неверный код (или аккаунта нет); 'expired' — срок вышел;
   132→ * 'locked' — слишком много неверных попыток.
   133→ */
   134→export async function consumeVerificationCode(email: string, code: string): Promise<CodeConsumeResult> {
   135→  const user = await db.user.findUnique({ where: { email } })
   136→  if (!user) return 'invalid'
   137→
   138→  const record = await db.verificationToken.findFirst({ where: { userId: user.id } })
   139→  if (!record) return 'expired'
   140→
   141→  if (record.expiresAt.getTime() < Date.now()) {
   142→    await db.verificationToken.delete({ where: { id: record.id } }).catch(() => undefined)
   143→    return 'expired'
   144→  }
   145→
   146→  if (attemptsLeft(user.id) <= 0) return 'locked'
   147→
   148→  if (hashToken(code) !== record.tokenHash) {
   149→    const fails = failAttempt(user.id)
   150→    return fails >= MAX_CODE_ATTEMPTS ? 'locked' : 'invalid'
   151→  }
   152→
   153→  await db.$transaction([
   154→    db.user.update({ where: { id: user.id }, data: { emailVerified: new Date() } }),
   155→    db.verificationToken.delete({ where: { id: record.id } }),
   156→  ])
   157→  resetAttempts(user.id)
   158→  return 'ok'
   159→}
   160→
   161→/**
   162→ * Проверяет сырой токен из ссылки. Если он валиден и не истёк — ставит
   163→ * User.emailVerified = текущая дата и удаляет запись токена (одноразовость).
   164→ * Невалидный / истёкший / отсутствующий → { ok: false }.
   165→ */
   166→export async function consumeVerificationToken(raw: string | null): Promise<ConsumeResult> {
   167→  if (!raw) return { ok: false }
   168→  const record = await db.verificationToken.findUnique({ where: { tokenHash: hashToken(raw) } })
   169→  if (!record) return { ok: false }
   170→
   171→  if (record.expiresAt.getTime() < Date.now()) {
   172→    // Истёкший токен сразу чистим, чтобы не копился мусор.
   173→    await db.verificationToken.delete({ where: { id: record.id } }).catch(() => undefined)
   174→    return { ok: false }
   175→  }
   176→
   177→  await db.$transaction([
   178→    db.user.update({ where: { id: record.userId }, data: { emailVerified: new Date() } }),
   179→    db.verificationToken.delete({ where: { id: record.id } }),
   180→  ])
   181→  return { ok: true, userId: record.userId }
   182→}

   184→/* ------------------------------------------------------------------ */
   185→/* Rate limit повторной отправки: 3 запроса на email в час              */
   186→/* ------------------------------------------------------------------ */
   187→
   188→const RESEND_WINDOW_MS = 3600 * 1000 // 1 час
   189→const RESEND_MAX_PER_WINDOW = 3
   190→
   191→/** email (в нижнем регистре) → времена запросов внутри текущего часа. */
   192→const resendHits = new Map<string, number[]>()
   193→
   194→export function checkResendRateLimit(email: string): { allowed: boolean; retryAfterSec: number } {
   195→  const now = Date.now()
   196→  const list = (resendHits.get(email) ?? []).filter((t) => now - t < RESEND_WINDOW_MS)
   197→
   198→  if (list.length >= RESEND_MAX_PER_WINDOW) {
   199→    const retryAfterSec = Math.max(1, Math.ceil((RESEND_WINDOW_MS - (now - list[0])) / 1000))
   200→    resendHits.set(email, list)
   201→    return { allowed: false, retryAfterSec }
   202→  }
   203→
   204→  list.push(now)
   205→  resendHits.set(email, list)
   206→  return { allowed: true, retryAfterSec: 0 }
   207→}

   293→export type SendResult = 'sent' | 'dev-logged'
   294→
   295→/**
   296→ * Отправляет письмо с кодом подтверждения. Бросает исключение только при
   297→ * настроенном, но неудавшемся SMTP-вызове — вызывающий код решает, как реагировать.
   298→ */
   299→export async function sendVerificationEmail(to: string, name: string, code: string): Promise<SendResult> {
   300→  // SMTP не настроен — dev-fallback: печатаем код в консоль сервера.
   301→  if (!isSmtpConfigured()) {
   302→    console.log(`[email] SMTP не настроен, письмо НЕ отправлено. Код подтверждения для ${to}: ${code}`)
   303→    return 'dev-logged'
   304→  }
   305→
   306→  const port = Number(process.env.EMAIL_SERVER_PORT ?? 587)
   307→  const transporter = nodemailer.createTransport({
   308→    host: process.env.EMAIL_SERVER_HOST,
   309→    port,
   310→    secure: port === 465, // 465 — implicit TLS, 587 — STARTTLS
   311→    auth: process.env.EMAIL_SERVER_USER
   312→      ? {
   313→          user: process.env.EMAIL_SERVER_USER,
   314→          pass: process.env.EMAIL_SERVER_PASSWORD,
   315→        }
   316→      : undefined,
   317→    // CHANGED: Mail.ru обрывает TLS-рукопожатие без явного минимального TLS 1.2
   318→    tls: { minVersion: 'TLSv1.2' },
   319→  })
   320→
   321→  const info = await transporter.sendMail({
   322→    from: process.env.EMAIL_FROM,
   323→    to,
   324→    subject: `Код подтверждения — ${APP_NAME}`,
   325→    html: buildVerificationHtml(name, code),
   326→    text: buildVerificationText(name, code),
   327→    // CHANGED: просим Gmail прислать отчёт при отказе/задержке — придёт в ящик отправителя
   328→    dsn: { notify: ['FAILURE', 'DELAY'], ret: 'HDRS' },
   329→  })
   330→  console.log(`[email] письмо с кодом отправлено на ${to} (${info.messageId}). Код: ${code}`)
   331→  return 'sent'
   332→}

   344→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/
   345→
   346→/**
   347→ * Человекочитаемое предупреждение при неудачной отправке кода.
   348→ * CHANGED: Mail.ru проверяет существование ящика получателя и отклоняет
   349→ * несуществующие адреса (550 non-local recipient verification failed) —
   350→ * показываем это пользователю сразу, чтобы он поправил опечатку в email.
   351→ */
   352→function smtpWarning(e: unknown): string {
   353→  const err = e as { responseCode?: number; code?: string }
   354→  if (err.responseCode === 550 || err.code === 'EENVELOPE') {
   355→    return 'Код не удалось отправить: почтовый сервер отклонил этот адрес. Проверьте правильность email и отправьте код повторно.'
   356→  }
   357→  return 'Не удалось отправить код на почту. Воспользуйтесь кнопкой «Отправить код повторно».'
   358→}

   360→/**
   361→ * Регистрация нового пользователя: имя + email + пароль.
   362→ * CHANGED: сессия/cookie НЕ выдаются — доступ откроется только после
   363→ * подтверждения email кодом из письма.
   364→ */
   365→export async function POST(req: Request) {
   366→  try {
   367→    await ensureSeeded()
   368→    const body = (await req.json()) as { name?: string; email?: string; password?: string; role?: string }
   369→    const name = (body.name ?? '').trim()
   370→    const email = (body.email ?? '').trim().toLowerCase()
   371→    const password = body.password ?? ''
   372→    const role = body.role === 'BEEKEEPER' ? 'BEEKEEPER' : 'HELPER'
   373→
   374→    if (name.length < 2 || name.length > 40) {
   375→      return NextResponse.json({ error: 'Имя: от 2 до 40 символов' }, { status: 400 })
   376→    }
   377→    if (!EMAIL_RE.test(email)) {
   378→      return NextResponse.json({ error: 'Введите корректный email' }, { status: 400 })
   379→    }
   380→    if (password.length < 4) {
   381→      return NextResponse.json({ error: 'Пароль — минимум 4 символа' }, { status: 400 })
   382→    }
   383→
   384→    const res = await registerUser(name, email, password, role as Role)
   385→    if (!res.ok) {
   386→      return NextResponse.json({ error: res.error }, { status: 400 })
   387→    }
   388→
   389→    // Код подтверждения: в БД только SHA-256 хеш, сырой код уходит в письме.
   390→    const code = await issueVerificationCode(res.user.id)
   391→    let warning: string | null = null
   392→    try {
   393→      const send = await sendVerificationEmail(email, name, code)
   394→      if (send === 'dev-logged') {
   395→        console.log(`[register] письмо для ${email} не отправлено (SMTP не настроен) — код в консоли сервера`)
   396→      }
   397→    } catch (e) {
   398→      // Аккаунт уже создан: не откатываем регистрацию, пусть пользователь
   399→      // воспользуется «Отправить код повторно».
   400→      console.error('register: не удалось отправить письмо', e)
   401→      warning = smtpWarning(e)
   402→    }
   403→
   404→    // Никакой сессии и никакого редиректа в приложение — только сообщение.
   405→    return NextResponse.json({
   406→      message: 'Регистрация успешна. Проверьте почту для подтверждения аккаунта',
   407→      ...(warning ? { warning } : {}),
   408→    })
   409→  } catch (e) {
   410→    console.error('register error', e)
   411→    return NextResponse.json({ error: 'Ошибка сервера при регистрации' }, { status: 500 })
   412→  }

   416→ts
   417→import { NextResponse } from 'next/server'
   418→import { consumeVerificationCode } from '@/lib/verification'
   419→
   420→const CODE_RE = /^\d{6}$/
   421→
   422→/**
   423→ * POST /api/auth/verify-code  { email, code }
   424→ * Подтверждение email кодом из письма (основной способ).
   425→ *   • ok      → 200 «Email подтверждён. Теперь вы можете войти»
   426→ *   • invalid → 400 «Неверный код»
   427→ *   • expired → 400 «Код истёк. Запросите новый код»
   428→ *   • locked  → 429 после 5 неверных попыток
   429→ */
   430→export async function POST(req: Request) {
   431→  try {
   432→    const body = (await req.json().catch(() => ({}))) as { email?: string; code?: string }
   433→    const email = (body.email ?? '').trim().toLowerCase()
   434→    const code = (body.code ?? '').replace(/\s+/g, '')
   435→
   436→    if (!CODE_RE.test(code)) {
   437→      return NextResponse.json({ error: 'Введите 6-значный код из письма' }, { status: 400 })
   438→    }
   439→
   440→    const result = await consumeVerificationCode(email, code)
   441→
   442→    if (result === 'ok') {
   443→      return NextResponse.json({ message: 'Email подтверждён. Теперь вы можете войти' })
   444→    }
   445→    if (result === 'expired') {
   446→      return NextResponse.json({ error: 'Код истёк. Запросите новый код' }, { status: 400 })
   447→    }

   468→/**
   469→ * POST /api/auth/resend-verification  { email }
   470→ * Повторная отправка письма подтверждения.
   471→ *   • Rate limit: не более 3 запросов на один email в час (Map в памяти процесса).
   472→ *   • Пользователь не найден → всё равно 200 (не раскрываем существование аккаунта).
   473→ *   • Email уже подтверждён → 200 «Email уже подтверждён».
   474→ *   • Иначе — новый код (старые записи пользователя удаляются) и письмо.
   475→ */
   476→export async function POST(req: Request) {
   477→  try {
   478→    const body = (await req.json().catch(() => ({}))) as { email?: string }
   479→    const email = (body.email ?? '').trim().toLowerCase()
   480→    if (!email) {
   481→      return NextResponse.json({ error: 'Укажите email' }, { status: 400 })
   482→    }
   483→
   484→    // Rate limit считаем по каждому запросу, до проверок существования аккаунта.
   485→    const rl = checkResendRateLimit(email)
   486→    if (!rl.allowed) {
   487→      const minutes = Math.max(1, Math.ceil(rl.retryAfterSec / 60))
   488→      return NextResponse.json(
   489→        { error: `Слишком много запросов. Попробуйте ещё раз примерно через ${minutes} мин.` },
   490→        { status: 429 },
   491→      )
   492→    }
   493→
   494→    const user = await db.user.findUnique({ where: { email } })
   495→    if (!user || !user.email) {
   496→      // Не раскрываем, существует ли аккаунт с таким email.
   497→      return NextResponse.json({ message: 'Если такой email зарегистрирован и не подтверждён, мы отправили письмо' })
   498→    }
   499→    if (user.emailVerified) {
   500→      return NextResponse.json({ message: 'Email уже подтверждён' })
   501→    }
   502→
   503→    const code = await issueVerificationCode(user.id)
   504→    try {
   505→      const send = await sendVerificationEmail(user.email, user.name, code)
   506→      if (send === 'dev-logged') {
   507→        console.log(`[resend] письмо для ${user.email} не отправлено (SMTP не настроен) — код в консоли сервера`)
   508→      }
   509→    } catch (e) {
   510→      console.error('resend-verification: не удалось отправить письмо', e)
   511→      // CHANGED: Mail.ru отклоняет несуществующие адреса — сообщаем пользователю
   512→      const err = e as { responseCode?: number; code?: string }
   513→      if (err.responseCode === 550 || err.code === 'EENVELOPE') {
   514→        return NextResponse.json({
   515→          message: 'Если такой email зарегистрирован и не подтверждён, мы отправили код',
   516→          warning: 'Код не удалось отправить: почтовый сервер отклонил этот адрес. Проверьте правильность email.',
   517→        })
   518→      }
   519→    }
   520→
   521→    return NextResponse.json({ message: 'Если такой email зарегистрирован и не подтверждён, мы отправили код' })
   522→  } catch (e) {
   523→    console.error('resend-verification error', e)
   524→    return NextResponse.json({ error: 'Ошибка сервера. Попробуйте позже.' }, { status: 500 })

   576→import { NextResponse } from 'next/server'
   577→import { consumeVerificationToken } from '@/lib/verification'
   578→
   579→/**
   580→ * GET /api/auth/verify?token={rawToken} — переход по ссылке из письма.
   581→ * Ответ — ТОЛЬКО редирект (не JSON):
   582→ *   • токен валиден и не истёк → User.emailVerified = текущая дата, запись
   583→ *     токена удалена, редирект на /login?verified=true
   584→ *   • токен невалиден / истёк / отсутствует → редирект на /login?error=invalid_token
   585→ */
   586→export async function GET(req: Request) {
   587→  // Базу редиректа берём из самого запроса (учитывая x-forwarded-proto за прокси),
   588→  // чтобы браузер остался на том же домене, с которого открыл ссылку.
   589→  const url = new URL(req.url)
   590→  const proto = req.headers.get('x-forwarded-proto') ?? url.protocol.replace(':', '')
   591→  const host = req.headers.get('host') ?? url.host
   592→  const base = `${proto}://${host}`
   593→
   594→  const token = url.searchParams.get('token')
   595→  const ok = await consumeVerificationToken(token).catch(() => ({ ok: false as const }))
   596→
   597→  if (!ok.ok) {
   598→    return NextResponse.redirect(`${base}/login?error=invalid_token`)
   599→  }
   600→  return NextResponse.redirect(`${base}/login?verified=true`)
   601→}
