     1→/**

   115→function lastPoint(h: Hive) {
   116→  return h.series[h.series.length - 1] ?? null
   117→}
   118→
   119→function lastWeight(h: Hive): number | null {
   120→  return lastPoint(h)?.weightKg ?? null
   121→}
   122→
   123→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   124→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   125→function weightDelta(h: Hive, backDays: number): number | null {
   126→  const s = h.series
   127→  if (s.length === 0) return null
   128→  const lastPoint = s[s.length - 1]
   129→  if (lastPoint.weightKg === null) return null
   130→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   131→  let best: number | null = null
   132→  let bestDiff = Infinity
   133→  for (const p of s) {
   134→    if (p.weightKg === null || p === lastPoint) continue
   135→    const diff = Math.abs(new Date(p.date).getTime() - target)
   136→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   137→      bestDiff = diff
   138→      best = p.weightKg
   139→    }
   140→  }
   141→  if (best === null) return null
   142→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   143→}
   144→
   145→const delta2d = (h: Hive) => weightDelta(h, 2)
   146→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   147→
   148→function maxTempRecent(h: Hive, n = 3): number | null {
   149→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   150→  return temps.length ? Math.max(...temps) : null
   151→}
   152→
   153→function lastHumidity(h: Hive): number | null {
   154→  return lastPoint(h)?.humidityPct ?? null
   155→}
   156→
   157→/** Лидер пасеки по 30-дневному привесу. */
   158→function leaderOf(list: Hive[]): Hive | null {
   159→  const withW = list.filter((x) => delta30d(x) !== null)
   160→  if (!withW.length) return null
   161→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   162→}
   163→
   164→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   165→function spanDays(h: Hive): number {
   166→  if (h.series.length < 2) return h.series.length
   167→  const a = new Date(h.series[0].date).getTime()
   168→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   169→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   170→  return Math.round((b - a) / 86400000) + 1
   171→}
   172→
   173→function periodLabel(h: Hive): string {
   174→  const d = spanDays(h)
   175→  if (d >= 25) return 'за месяц'
   176→  if (d === 1) return 'за день'
   177→  if (d === 0) return '—'
   178→  return `за ${d} дн.`
   179→}

   182→const SWARM_DROP = -1.0

   184→function swarmProbability(drop: number): string {
   185→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   186→  return `Вероятность подготовки к роению — около ${p} %`
   187→}

   193→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   194→  const status = hiveStatus(h, ctx.hives)
   195→  const d2 = delta2d(h)
   196→  const d30 = delta30d(h)
   197→  const t = lastPoint(h)?.tempInC ?? null
   198→  const hum = lastHumidity(h)
   199→
   200→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   201→    return {
   202→      id: `nodata-${h.id}`,
   203→      scenario: 'status',
   204→      hiveId: h.id,
   205→      title: `Улей №${h.id}: пока нет данных`,
   206→      text: `По улью №${h.id} в базе нет измерений. Я не буду гадать: как только появятся показания — загрузите их в разделе «Данные» (CSV, вручную или через API), и я сразу начну следить за весом, температурой и влажностью.`,
   207→      basis: ['База данных платформы', 'Измерения улья №' + h.id],
   208→      urgency: 'this-week',
   209→      urgencyNote: 'Загрузить данные',
   210→      probability: 'Оценка невозможна — данных нет',
   211→      metrics: [{ label: 'Измерений', value: String(h.series.length) }],
   212→      prefill: `Что происходит с ульем №${h.id}?`,
   213→    }
   214→  }
   215→
   216→  if (status.level === 'risk' && d2 !== null) {
   217→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   218→    const now = lastWeight(h) ?? peak
   219→    return {
   220→      id: `swarm-${h.id}`,
   221→      scenario: 'swarm',
   222→      hiveId: h.id,
   223→      title: 'Похоже, семья готовится к роению',
   224→      text: `За последние 2 дня вес улья №${h.id} снизился на ${fmtNum(Math.abs(d2))} кг — с ${fmtNum(peak)} до ${fmtNum(now)} кг, хотя до этого семья стабильно набирала мед.${
   225→        t !== null ? ` Температура внутри поднималась до ${fmtNum(t)} °C (норма около 34,5 °C).` : ''
   226→      }${hum !== null ? ` Влажность держится на отметке ${fmtNum(hum, 0)} % — выше обычной для этого улья.` : ''} Июль — пик сезона роения: часть пчёл с маткой может улететь, и медосбор остановится на 2–3 недели.`,
   227→      basis: [
   228→        `График веса улья №${h.id}`,
   229→        `Вес: ${fmtSigned(d2)} кг за 2 дня`,
   230→        ...(t !== null ? [`Температура: ${fmtNum(t)} °C при норме 34,5 °C`] : []),
   231→        'Сезон: июль — период роения',
   232→      ],
   233→      urgency: 'today',
   234→      urgencyNote: 'Осмотреть улей сегодня до вечера',
   235→      probability: swarmProbability(d2),
   236→      metrics: [
   237→        { label: 'Вес за 2 дня', value: `${fmtSigned(d2)} кг` },
   238→        ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   239→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   240→      ],
   241→      prefill: `Что происходит с ульем №${h.id}?`,
   242→    }
   243→  }
   244→
   245→  if (status.level === 'watch' && h.label !== undefined) {
   246→    const leader = leaderOf(ctx.hives)
   247→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   248→      ? 'отдельно: датчик влажности не отвечает'
   249→      : 'привес хуже, чем у лидера пасеки'
   250→    return {
   251→      id: `slow-${h.id}`,
   252→      scenario: 'status',
   253→      hiveId: h.id,
   254→      title: 'Слабый набор веса',
   255→      text: `Улей №${h.id} прибавил только ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}${
   256→        leader && leader.id !== h.id ? ` — у лидера (улей №${leader.id}) ${fmtSigned(delta30d(leader) ?? 0)} кг` : ''
   257→      }. Температура${t !== null ? ` ${fmtNum(t)} °C` : ''} в норме. Причин может быть две: слабая семья или бедная кормовая база вокруг. ${reason.charAt(0).toUpperCase() + reason.slice(1)}${
   258→        h.humidityOfflineFrom ? ` — молчит с ${h.humidityOfflineFrom}, возможно, сел аккумулятор` : ' — стоит сверить привес с соседними ульями'
   259→      }.`,
   260→      basis: [
   261→        'Сравнение привеса по ульям пасеки',
   262→        `Привес улья №${h.id}: ${fmtSigned(d30 ?? 0)} кг`,
   263→        ...(h.humidityOfflineFrom ? [`Датчик влажности: нет данных с ${h.humidityOfflineFrom}`] : []),
   264→      ],
   265→      urgency: 'this-week',
   266→      urgencyNote: 'Проверить датчик и семью в ближайшие дни',
   267→      probability: 'Диагноз по косвенным признакам — нужна проверка',
   268→      metrics: [
   269→        { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   270→        ...(leader && leader.id !== h.id
   271→          ? [{ label: `У лидера (№${leader.id})`, value: `${fmtSigned(delta30d(leader) ?? 0)} кг` }]
   272→          : []),
   273→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : [{ label: 'Влажность', value: 'нет данных' }]),
   274→      ],
   275→      prefill: `Почему в улье №${h.id} слабый привес?`,
   276→    }
   277→  }
   278→
   279→  return {
   280→    id: `ok-${h.id}`,
   281→    scenario: 'status',
   282→    hiveId: h.id,
   283→    title: 'Семья работает штатно',
   284→    text: `Улей №${h.id} развивается штатно: вес ${fmtNum(lastWeight(h) ?? 0)} кг, привес ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}, за последние 2 дня ${fmtSigned(d2 ?? 0)} кг.${
   285→      t !== null ? ` Температура ${fmtNum(t)} °C — в норме.` : ''
   286→    } Признаков роения нет: вес растёт ровно, без провалов.`,
   287→    basis: [
   288→      `График веса улья №${h.id}`,
   289→      `Привес: ${fmtSigned(d30 ?? 0)} кг`,
   290→      'Температура в норме семьи',
   291→    ],
   292→    urgency: 'ok',
   293→    urgencyNote: 'Продолжать наблюдение',
   294→    metrics: [
   295→      { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   296→      ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   297→      ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   298→    ],
   299→    prefill: `Что происходит с улье №${h.id}?`,
   300→  }
   301→}

   330→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   331→  const w = ctx.weather
   332→  if (!w || w.length === 0) return null
   333→  const good = w.filter((d) => d.flight === 'good')
   334→  const bad = w.filter((d) => d.flight === 'bad')
   335→  const tMax = Math.max(...w.map((d) => d.tMax))
   336→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   337→  return {
   338→    id: 'weather',
   339→    scenario: 'weather',
   340→    hiveId: null,
   341→    title: good.length
   342→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   343→      : 'Ближайшие дни — не для лёта',
   344→    text: good.length
   345→      ? `Ближайшие дни с хорошими условиями: ${good.map((d) => d.full).join(', ')} — до ${fmtSigned(tMax, 0)} °C, ветер до ${Math.max(...good.map((d) => d.windMs))} м/с, осадки маловероятны.${
   346→          bad.length ? ` Затем, ${bad.map((d) => d.full).join(' и ')}, — дожди (до ${rainProb} %) и ветер до ${Math.max(...bad.map((d) => d.windMs))} м/с: пчёлы останутся в ульях, привес остановится.` : ''
   347→        }`
   348→      : `По прогнозу на неделю погода не летная: дожди и ветер. Пчёлы останутся в ульях, привес остановится — следите за запасами корма по графику веса.`,
   349→    basis: ['Прогноз погоды платформы', 'Календарь цветения медоносов — ГИС'],
   350→    urgency: 'ok',
   351→    urgencyNote: good.length ? 'Использовать окно лёта' : 'Наблюдать за погодой',
   352→    metrics: [
   353→      { label: 'Максимум', value: `${fmtSigned(tMax, 0)} °C` },
   354→      ...(bad.length ? [{ label: 'Дождь', value: `до ${rainProb} %` }] : []),
   355→      ...(ctx.sites.length
   356→        ? [{ label: 'Кандидатов кочёвки', value: String(ctx.sites.length) }]
   357→        : []),
   358→    ],
   359→    prefill: 'Когда пчёлам лучше летать за взятком?',
   360→  }
   361→}

   363→function monthGen(date: string): string {
   364→  // «2026-07-20» → «июля» (родительный падеж)
   365→  const months = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря']
   366→  const m = Number(date.slice(5, 7))
   367→  return months[m - 1] ?? ''
   368→}

   370→function migrationFindingOf(ctx: AgentContext): AgentFinding | null {
   371→  const sites = [...ctx.sites].sort((a, b) => b.score - a.score)
   372→  if (!sites.length) return null
   373→  const best = sites[0]
   374→  const worst = sites[sites.length - 1]
   375→  return {
   376→    id: 'migration',
   377→    scenario: 'migration',
   378→    hiveId: null,
   379→    title: `Лучшее место для кочёвки — «${best.name}»`,
   380→    text: `Сравнил ${sites.length} ${
   381→      sites.length === 1 ? 'участок' : sites.length < 5 ? 'участка' : 'участков'

   398→export function findingsFor(hiveId: number, ctx: AgentContext): AgentFinding[] {
   399→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   400→  const list: AgentFinding[] = []
   401→  if (hive) list.push(hiveFinding(hive, ctx))
   402→  const pesticide = pesticideFindingOf(ctx)
   403→  if (pesticide) list.push(pesticide)
   404→  const weatherF = weatherFindingOf(ctx)
   405→  if (weatherF) list.push(weatherF)
   406→  const migrationF = migrationFindingOf(ctx)
   407→  if (migrationF) list.push(migrationF)
   408→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   409→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   410→}

   983→/** Температура сразу по всем ульям + погода снаружи. */
   984→function climateAllAnswer(ctx: AgentContext): AgentMessage {
   985→  const lines = ctx.hives.map((h) => {
   986→    const t = lastPoint(h)?.tempInC ?? null
   987→    return `№${h.id} — ${t !== null ? `${fmtNum(t)} °C` : 'данных нет'}`
   988→  })
   989→  const temps = ctx.hives
   990→    .map((h) => lastPoint(h)?.tempInC ?? null)
   991→    .filter((t): t is number => t !== null)
   992→  const maxT = temps.length ? Math.max(...temps) : null
   993→  const overheated = maxT !== null && maxT > 35.7
   994→  const now = ctx.weatherNow
   995→  const w0 = ctx.weather?.[0]
   996→  return {
   997→    scenario: overheated ? 'swarm' : 'status',
   998→    hiveId: null,

  1189→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
  1190→export function askAgent(
  1191→  question: string,
  1192→  currentHiveId: number,
  1193→  ctx: AgentContext,
  1194→): AgentMessage {
  1195→  const q = norm(question)
  1196→  const hiveId = extractHiveId(q, ctx)
  1197→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
  1198→
  1199→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
  1200→
  1201→  // Приветствие и сервисные фразы
  1202→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
  1203→    return greetingAnswer(ctx)
  1204→  }
  1205→  // Вопрос о самом агенте: модель и устройство не раскрываем
  1206→  if (has(/кто ты|кто такой|как тебя зовут|ты бот|чат ?бот|какая (ты )?модель|какой (ты )?модель|на (чем|чём) ты (работаешь|сделан|написан)|что ты за|нейросет|искусственн|\bllm\b|\bgpt\b/)) {
  1207→    return identityAnswer()
  1208→  }
  1209→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
  1210→    return thanksAnswer()
  1211→  }
  1212→
  1213→  // Сценарий пестицидов
  1214→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
  1215→    return pesticideAnswer(ctx)
  1216→  }
  1217→
  1218→  // Кочёвка
  1219→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
  1220→    return migrationAnswer(ctx)
  1221→  }
  1222→
  1223→  // Погода и медосбор (включая температуру ВОЗДЕУХА — не путать с температурой улья)
  1224→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|снег|мороз|когда вывоз|когда летать|температура (воздуха|за окном|на улице|снаружи)/)) {
  1225→    return weatherAnswer(ctx)
  1226→  }
  1227→
  1228→  if (!hive) {
  1229→    return fallbackAnswer()
  1230→  }
  1231→
  1232→  // Темы по ульям: если номер улья не назван — отвечаем сразу по ВСЕМ ульям
  1233→  if (has(/влажн|сырост|сыро/)) {
  1234→    return hiveId ? humidityAnswer(hive) : humidityAllAnswer(ctx)
  1235→  }
  1236→  if (has(/шум|децибел|гул\b/)) {
  1237→    return noiseAnswer(ctx)
  1238→  }
  1239→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
  1240→    return hiveId ? swarmAnswer(hive, ctx) : swarmAllAnswer(ctx)
  1241→  }
  1242→  if (has(/температ|жар|тепло внутри|градус/)) {
  1243→    return hiveId ? climateAnswer(hive) : climateAllAnswer(ctx)
  1244→  }
  1245→  if (has(/вес|мед\b|меда|привес|набор|тяже|лидер/)) {
  1246→    return hiveId ? weightAnswer(hive, ctx) : weightAllAnswer(ctx)
  1247→  }
  1248→  if (has(/датчик|сенсор/)) {
  1249→    return sensorsAnswer(ctx)
  1250→  }
  1251→  if (has(/все.{0,8}уль|всех ульев|всей пасек|пасек|обзор|сводк|итог|общее состояни|как дела|как себя|что происходит|состояни|статус|проверь|проблем|тревог|расскажи|что с улеем|как улей/)) {
  1252→    return hiveId ? statusAnswer(hive, ctx) : allHivesAnswer(ctx)
  1253→  }
  1254→
  1255→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
  1256→  if (hiveId !== null) {
  1257→    return statusAnswer(hive, ctx)
  1258→  }
  1259→
  1260→  return fallbackAnswer()
  1261→}
