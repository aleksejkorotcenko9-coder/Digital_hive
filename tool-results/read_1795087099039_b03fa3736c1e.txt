   121→function LiveTile({
   122→  icon,
   123→  label,
   124→  value,
   125→  num,
   126→  digits = 1,
   127→  suffix,
   128→  hint,
   129→}: {
   130→  icon: React.ReactNode
   131→  label: string
   132→  value?: string
   133→  num?: number
   134→  digits?: number
   135→  suffix?: string
   136→  hint?: string
   137→}) {
   138→  return (
   139→    <div className="rounded-xl border border-red-50 bg-red-50/40 px-3 py-2.5 dark:bg-[#2C211C]">
   140→      <p className="flex items-center gap-1.5 text-[11px] font-semibold text-gray-500">
   141→        <span className="text-red-400">{icon}</span>
   142→        {label}
   143→      </p>
   144→      <p className="mt-1 flex items-baseline gap-1 text-lg font-extrabold leading-none text-gray-900">
   145→        {num !== undefined ? (
   146→          <AnimatedNumber value={num} digits={digits} />
   147→        ) : (
   148→          <LiveValue value={value ?? ''} />
   149→        )}
   150→        {suffix && <span className="text-[11px] font-bold text-gray-400">{suffix}</span>}
   151→      </p>
