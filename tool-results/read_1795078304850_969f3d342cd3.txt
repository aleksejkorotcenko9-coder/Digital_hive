    89→  const [points, setPoints] = useState<MapPointDto[] | null>(null)

    91→  const [weather, setWeather] = useState<WeatherResponse | null>(null)
    92→  const [weatherLoading, setWeatherLoading] = useState(false)
    93→
    94→  // диалог привязки улья по коду датчика
    95→  const [dialogOpen, setDialogOpen] = useState(false)
    96→  const [code, setCode] = useState('')

    98→  const [busy, setBusy] = useState(false)

   100→  // Как только введён код из каталога — сразу знаем GPS-позицию датчика

   104→  /* ---------- карта (Яндекс Карты) ---------- */

   118→
   119→      // controls: переключатель «Схема / Спутник / Гибрид» + масштаб и геолокация;

   149→      selColRef.current = selCol
   150→      prevColRef.current = prevCol

   155→      mapRef.current?.destroy()

   157→      pointsColRef.current = null
   158→      selColRef.current = null
   159→      prevColRef.current = null
   160→      markerRefs.current.clear()

   163→  }, [])

   173→      // карта и погода продолжают работать, точки подтянутся при следующем обновлении

   176→
   177→  useEffect(() => {

   188→    const map = mapRef.current
   189→    if (!map) return

   191→    const lon = points.reduce((s, p) => s + p.lon, 0) / points.length

   203→    clickRef.current = onMapClick

   205→
   206→  useEffect(() => {

   209→    // улей привязывается по коду датчика в диалоге
   210→    mapRef.current?.cursors.push('grab')
   211→  }, [mapReady])
   212→
   213→  /* ---------- маркеры точек ---------- */

   216→    const col = pointsColRef.current

   222→      const active = p.sensor.reason === 'active'

   224→      const readings = active

   226→        : `<br><i>Данных об улье нет — датчик ${SENSOR_LABEL[p.sensor.reason]}.</i>`
   227→      const pm = new ym.Placemark(
   228→        [p.lat, p.lon],

   231→          label: `${p.hiveCount} ул.`,
   232→          balloonContentHeader: p.name,
   233→          balloonContentBody: `Датчик: <b>${p.sensorCode ?? '—'}</b><br>GPS: ${p.lat.toFixed(4)}, ${p.lon.toFixed(4)}<br>Состояние: ${SENSOR_LABEL[p.sensor.reason]}${readings}`,
   234→        },
   235→        {
   236→          iconLayout: layout,
   237→          iconShape: HIVE_ICON_SHAPE,
   238→          iconOffset: [-17, -17],
   239→          hideIconOnBalloonOpen: false,
   240→        },
   241→      )
   242→      col.add(pm)
   243→      markerRefs.current.set(p.id, pm)
   244→    }
   245→  }, [points, mapReady])
   246→
   247→  /* ---------- маркер выбранной точки ---------- */
   248→  useEffect(() => {
   249→    const ym = ymRef.current
   250→    const col = selColRef.current
   251→    if (!mapReady || !ym || !col) return
   252→    col.removeAll()

   254→    col.add(
   255→      new ym.Placemark(
   256→        [selected.lat, selected.lon],
   257→        { balloonContentBody: 'Выбранная точка — погода справа' },
   258→        { preset: 'islands#redDotIcon', iconColor: '#ef4444', hideIconOnBalloonOpen: false },
   259→      ),
   260→    )
   261→  }, [selected, mapReady])
   262→
   263→  /* ---------- превью GPS-позиции датчика при вводе кода ---------- */
   264→  useEffect(() => {
   265→    const ym = ymRef.current
   266→    const col = prevColRef.current
   267→    const map = mapRef.current
   268→    if (!mapReady || !ym || !col) return
   269→    col.removeAll()

   273→      new ym.Placemark(
   274→        [found.lat, found.lon],
   275→        {

   284→      map.panTo([found.lat, found.lon], { duration: 700, flying: true })
   285→    }
   286→  }, [dialogOpen, found, mapReady])

   288→  /* ---------- погода выбранной точки ---------- */
   289→  useEffect(() => {

   350→  const patchSensor = async (
   351→    p: MapPointDto,
   352→    body: { sensorOn?: boolean; hasSensor?: boolean },
   353→    okMsg: string,
   354→  ) => {
   355→    const res = await fetch(`/api/map-points/${p.id}`, {
   356→      method: 'PATCH',
   357→      headers: { 'Content-Type': 'application/json' },
   358→      body: JSON.stringify(body),
   359→    })
   360→    if (!res.ok) {
   361→      toast({ title: 'Не получилось', variant: 'destructive' })
   362→      return
   363→    }
   364→    toast({ title: okMsg })
   365→    // оптимистично правим карточку, серверный ответ уточнит детали
   366→    setPoints((prev) =>
   367→      prev
   368→        ? prev.map((x) => {
   369→            if (x.id !== p.id) return x
   370→            const hasSensor = body.hasSensor ?? x.hasSensor
   371→            const sensorOn = body.sensorOn ?? (body.hasSensor !== undefined ? body.hasSensor : x.sensorOn)
   372→            return { ...x, hasSensor, sensorOn }
   373→          })
   374→        : prev,
   375→    )
   376→    void load()
   377→  }

   386→    toast({ title: 'Точка удалена с карты' })
   387→  }
   388→
   389→  const flyTo = (p: MapPointDto) => flyMapTo(p.lat, p.lon, p.id)
   390→
   391→  const list = points ?? []

   394→    const sensors = list.filter((p) => p.hasSensor && p.sensorOn).length

   396→  }, [list])
   397→
   398→  return (
   399→    <div className="mx-auto max-w-6xl space-y-5">
   400→      {/* Шапка */}
   401→      <div className="flex flex-wrap items-center justify-between gap-2">

   404→            Карта пасеки
   405→          </h1>
   406→          <p className="mt-1 max-w-2xl text-sm text-gray-500">
   407→            Кликните по карте — покажу погоду в этой точке. Улей привязывается по коду датчика:
   408→            позицию задаёт GPS самого датчика.
   409→          </p>
   410→        </div>
   411→        {isBeekeeper && (
   412→          <Button
   413→            onClick={() => setDialogOpen(true)}
   414→            className="gap-1.5 bg-gradient-to-r from-red-600 to-orange-500 font-bold active:scale-95"
   415→          >
   416→            <MapPinPlus className="h-4 w-4" />
   417→            Добавить улей
   418→          </Button>
   419→        )}
   420→      </div>
   421→
   422→      {/* Карта + правая колонка */}
   423→      <div className="grid gap-4 xl:grid-cols-[1fr_340px]">
   424→        <motion.div
   425→          initial={{ opacity: 0, scale: 0.98 }}
   426→          animate={{ opacity: 1, scale: 1 }}
   427→          transition={ease}
   428→          className="relative overflow-hidden rounded-2xl border border-red-100 shadow-sm"

   430→          <div ref={containerRef} className="h-[440px] w-full md:h-[520px]" />

   432→            <div className="absolute inset-0 flex items-center justify-center gap-2 bg-red-50/60 text-xs font-semibold text-gray-500">
   433→              <RefreshCcw className="h-4 w-4 animate-spin" /> Загружаем карту…

   435→          )}
   436→          {mapError && (
   437→            <div className="absolute inset-0 flex items-center justify-center bg-red-50/90 px-6 text-center text-xs font-semibold text-red-700">
   438→              {mapError}
   439→            </div>
   440→          )}
   441→        </motion.div>
   442→
   443→        <div className="space-y-4">
   444→          {/* Погода выбранной точки */}
   445→          <Card className="border-red-100 shadow-sm">
   446→            <CardContent className="p-4">
   447→              <h2 className="flex items-center gap-2 text-sm font-bold text-[#2A1A12]">
   448→                <MapPin className="h-4 w-4 text-red-600" />
   449→                Погода в точке
