  1779→export type Dict = Record<string, string>
  1780→
  1781→const BASE_DICTS: Record<Lang, Dict> = { ru: RU, en: EN, fr: FR, kk: KK, be: BE, zh: ZH }
  1782→
  1783→/**
  1784→ * Дополнительные словари (по группам экранов) — подключаются из модулей
  1785→ * i18n-extra-*.ts, чтобы переводить ВСЁ: оставшиеся экраны, ответы агента
  1786→ * и содержательные тексты демо-данных. Отсутствующие ключи падают на русский.
  1787→ */
  1788→type ExtraModule = Record<Lang, Dict>
  1789→
  1790→const EXTRA_MODULES: ExtraModule[] = [
  1791→  REAL_DICTS,
  1792→  SHELL_DICTS,
  1793→  HISTORY_DICTS,
  1794→  MAP_DICTS,
  1795→  DATA_DICTS,
  1796→  AGENT_DICTS,
  1797→  DEMO_DICTS,
  1798→]
  1799→
  1800→const DICTS: Record<Lang, Dict> = Object.fromEntries(
  1801→  (Object.keys(BASE_DICTS) as Lang[]).map((l) => [
  1802→    l,
  1803→    Object.assign({}, BASE_DICTS[l], ...EXTRA_MODULES.map((m) => m[l] ?? {})),
  1804→  ]),
  1805→) as Record<Lang, Dict>
  1806→
  1807→/** Перевод строки с подстановкой переменных {name}. */
  1808→export function tr(lang: Lang, key: string, vars?: Record<string, string | number>): string {
  1809→  const dict = DICTS[lang] ?? RU
  1810→  let s = dict[key] ?? RU[key] ?? key
  1811→  if (vars) {
  1812→    for (const [k, v] of Object.entries(vars)) s = s.replaceAll(`{${k}}`, String(v))
  1813→  }
  1814→  return s
  1815→}
  1816→
  1817→/** Хук перевода для клиентских компонентов: t('key', {n: 1}). */
  1818→export function useT() {
  1819→  const lang = useAppState((s) => s.lang)
  1820→  return useCallback((key: string, vars?: Record<string, string | number>) => tr(lang, key, vars), [lang])
  1821→}
