   116→   116→function lastPoint(h: Hive) {
   117→   117→  return h.series[h.series.length - 1] ?? null
   118→   118→}
   119→   119→
   120→   120→function lastWeight(h: Hive): number | null {
   121→   121→  return lastPoint(h)?.weightKg ?? null
   122→   122→}
   123→   123→
   124→   124→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   125→   125→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   126→   126→function weightDelta(h: Hive, backDays: number): number | null {
   127→   127→  const s = h.series
   128→   128→  if (s.length === 0) return null
   129→   129→  const lastPoint = s[s.length - 1]
   130→   130→  if (lastPoint.weightKg === null) return null
   131→   131→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   132→   132→  let best: number | null = null
   133→   133→  let bestDiff = Infinity
   134→   134→  for (const p of s) {
   135→   135→    if (p.weightKg === null || p === lastPoint) continue
   136→   136→    const diff = Math.abs(new Date(p.date).getTime() - target)
   137→   137→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   138→   138→      bestDiff = diff
   139→   139→      best = p.weightKg
   140→   140→    }
   141→   141→  }
   142→   142→  if (best === null) return null
   143→   143→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   144→   144→}
   145→   145→
   146→   146→const delta2d = (h: Hive) => weightDelta(h, 2)
   147→   147→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   148→   148→
   149→   149→function maxTempRecent(h: Hive, n = 3): number | null {
   150→   150→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   151→   151→  return temps.length ? Math.max(...temps) : null
   152→   152→}

   154→   154→function lastHumidity(h: Hive): number | null {

   158→   158→/** Лидер пасеки по 30-дневному привесу. */
   159→   159→function leaderOf(list: Hive[]): Hive | null {
   160→   160→  const withW = list.filter((x) => delta30d(x) !== null)
   161→   161→  if (!withW.length) return null
   162→   162→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   163→   163→}
   164→   164→
   165→   165→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   166→   166→function spanDays(h: Hive): number {
   167→   167→  if (h.series.length < 2) return h.series.length
   168→   168→  const a = new Date(h.series[0].date).getTime()
   169→   169→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   170→   170→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   171→   171→  return Math.round((b - a) / 86400000) + 1
   172→   172→}

   174→   174→function periodLabel(h: Hive): string {
   175→   175→  const d = spanDays(h)
   176→   176→  if (d >= 25) return 'за месяц'
   177→   177→  if (d === 1) return 'за день'
   178→   178→  if (d === 0) return '—'
   179→   179→  return `за ${d} дн.`
   180→   180→}

   188→   188→}

   331→   331→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   332→   332→  const w = ctx.weather
   333→   333→  if (!w || w.length === 0) return null
   334→   334→  const good = w.filter((d) => d.flight === 'good')
   335→   335→  const bad = w.filter((d) => d.flight === 'bad')
   336→   336→  const tMax = Math.max(...w.map((d) => d.tMax))
   337→   337→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   338→   338→  return {
   339→   339→    id: 'weather',
   340→   340→    scenario: 'weather',
   341→   341→    hiveId: null,
   342→   342→    title: good.length
   343→   343→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   344→   344→      : 'Ближайшие дни — не для лёта',
