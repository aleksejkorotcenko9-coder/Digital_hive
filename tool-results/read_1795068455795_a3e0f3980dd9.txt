   103→export function DashboardView() {
   104→  const mode = useAppState((s) => s.mode)
   105→  const setView = useAppState((s) => s.setView)
   106→  const hiveId = useAppState((s) => s.hiveId)
   107→  const setHive = useAppState((s) => s.setHive)
   108→  const { status, data, reload } = useRealData()
   109→
   110→  // Синхронизация с картой в ОБОИХ режимах: ульи дашборда = точки на карте
   111→  // («Демо» — участок у Чишм, «Мои данные» — собственная пасека в Стерлитамаке),
   112→  // показания — с датчиков. Опрос каждые 5 секунд — числа «живут» вместе с картой.
   113→  // Режим хранится вместе со списком: пока идёт загрузка той пасеки, чужие
   114→  // точки не показываются (синхронизация без «вспышек» чужих данных).
   115→  const [pointsByMode, setPointsByMode] = useState<{ mode: 'demo' | 'real'; points: MapPointDto[] } | null>(null)
   116→  useEffect(() => {
   117→    let alive = true
   118→    const load = async () => {
   119→      try {
   120→        const res = await fetch(`/api/map-points?mode=${mode}`, { cache: 'no-store' })
   121→        if (!res.ok) return
   122→        const j = (await res.json()) as { points: MapPointDto[] }
   123→        if (alive) setPointsByMode({ mode, points: j.points })
   124→      } catch {
   125→        // датчики подтянутся на следующем опросе
   126→      }
   127→    }
   128→    void load()
   129→    const timer = setInterval(() => void load(), 5000)
   130→    return () => {
   131→      alive = false
   132→      clearInterval(timer)
   133→    }
   134→  }, [mode])
   135→  const points = pointsByMode && pointsByMode.mode === mode ? pointsByMode.points : null
   136→
   137→  // Погода той пасеки, к которой относится режим (реальные данные из интернета):
   138→  // «Демо» — Чишмы, «Мои данные» — Стерлитамак. Та же, что в панели погоды.
   139→  const [wByMode, setWByMode] = useState<{ mode: 'demo' | 'real'; data: WeatherResponse } | null>(null)
   140→  useEffect(() => {
   141→    let alive = true
   142→    const [lat, lon] = APIARY_PLACES[mode].center
   143→    const sp = new URLSearchParams({ lat: String(lat), lon: String(lon), place: APIARY_PLACES[mode].name })
   144→    const load = async () => {
   145→      try {
   146→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   147→        if (!res.ok) return
   148→        const j = (await res.json()) as WeatherResponse
   149→        if (alive) setWByMode({ mode, data: j })
   150→      } catch {
   151→        // агент честно скажет, если погоды нет
   152→      }
   153→    }
   154→    void load()
   155→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   156→    return () => {
   157→      alive = false
   158→      clearInterval(timer)
   159→    }
   160→  }, [mode])
   161→  const wResp = wByMode && wByMode.mode === mode ? wByMode.data : null
   162→
   163→  const syncedHives = useMemo(() => (points ? pointsToHives(points) : null), [points])
   164→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   165→
   166→  // выбранный улей всегда существует в синхронизированном наборе
   167→  useEffect(() => {
   168→    if (syncedHives && syncedHives.length > 0 && !syncedHives.some((h) => h.id === hiveId)) {
   169→      setHive(syncedHives[0].id)
   170→    }
   171→  }, [syncedHives, hiveId, setHive])
   172→
   173→  if (mode === 'real') {
   174→    if (status === 'idle' || status === 'loading') {
   175→      return <DashboardSkeleton />
   176→    }
   177→    if (status === 'error') {
   178→      return (
   179→        <Card className="mx-auto max-w-lg border-red-100">
   180→          <CardContent className="p-6 text-center">
   181→            <AlertTriangle className="mx-auto h-8 w-8 text-red-400" />
   182→            <h2 className="mt-2 font-bold text-gray-900">Не удалось загрузить данные</h2>
   183→            <p className="mt-1 text-sm text-gray-500">
   184→              Проверьте, что база данных доступна, и попробуйте ещё раз.
   185→            </p>
   186→            <Button onClick={reload} className="mt-4 bg-red-600 hover:bg-red-700">
   187→              Повторить загрузку
   188→            </Button>
   189→          </CardContent>
   190→        </Card>
   191→      )
   192→    }
   193→    // Схема и диаграммы «Моих данных» — ТО ЖЕ точки, что на карте пасеки
   194→    // (Стерлитамак): один набор ульев для карты, вкладки «Датчики» и дашборда.
   195→    // Из базы берём журнал и площадки кочёвки.
   196→    if (!points || !data || syncedHives === null) return <DashboardSkeleton />
   197→    if (syncedHives.length === 0) {
   198→      return <EmptyMapState onOpenMap={() => setView('map')} real />
   199→    }
   200→    return (
   201→      <DashboardContent
   202→        mode="real"
   203→        hives={syncedHives}
   204→        sites={data.sites.map(siteDtoToSite)}
   205→        weather={agentWeather?.days ?? null}
   206→        pesticide={null}
   207→        events={data.events}
   208→      />
   209→    )
   210→  }
   211→
   212→  if (!syncedHives) return <DashboardSkeleton />
   213→  if (syncedHives.length === 0) return <EmptyMapState onOpenMap={() => setView('map')} />
   214→
   215→  return (
   216→    <DashboardContent
   217→      mode="demo"
   218→      hives={syncedHives}
   219→      sites={migrationSites}
   220→      weather={agentWeather?.days ?? null}
   221→      pesticide={pesticideAlert}
   222→      events={[]}
   223→    />
   224→  )
   225→}

   231→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/50" />
   232→      <div className="h-96 animate-pulse rounded-2xl bg-red-50/40" />

   689→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   690→function weightTotal(h: Hive): number {
   691→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   692→  const first = h.series[0]?.weightKg ?? null
   693→  if (w === null || first === null) return 0
   694→  return Math.round((w - first) * 10) / 10
   695→}
