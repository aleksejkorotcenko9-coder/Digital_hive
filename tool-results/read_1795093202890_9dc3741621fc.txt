   132→/** Статус заклещенности: Норма <2 % / Риск 2–3 % / Критично >3 %. */
   133→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; label: string; cls: string } {
   134→  if (pct > 3) return { level: 'crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   135→  if (pct >= 2) return { level: 'risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   136→  return { level: 'ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   137→}
   138→
   139→/** Состояние воздуха по индексу ЛОС (0…100). */
   140→export function vocState(v: number): { label: string; cls: string } {
   141→  if (v > 60) return { label: 'Духота — семье не хватает объема', cls: 'text-red-600' }
   142→  if (v > 30) return { label: 'Умеренная загрузка воздухом', cls: 'text-amber-600' }
   143→  return { label: 'Воздух чистый', cls: 'text-emerald-600' }
   144→}
   145→
   146→/** Состояние трафика на летке относительно нормы пасеки (45 пч/мин). */
   147→export function trafficState(v: number): { label: string; cls: string } {
   148→  if (v >= 55) return { label: 'Интенсивный лёт', cls: 'text-emerald-600' }
   149→  if (v >= 30) return { label: 'Активный лёт', cls: 'text-emerald-600' }
   150→  if (v >= 18) return { label: 'Лёт ослаблен', cls: 'text-amber-600' }
   151→  return { label: 'Лёт резко снижен', cls: 'text-red-600' }
   152→}

   154→/** Индекс 0…10 → подпись и тон шкалы. */
   155→export function demoGaugeInfo(value: number): { label: string; tone: 'ok' | 'normal' | 'watch' | 'alert' } {
   156→  if (value < 4) return { label: 'Тревожное состояние', tone: 'alert' }
   157→  if (value < 6) return { label: 'Требует внимания', tone: 'watch' }
   158→  if (value < 8.5) return { label: 'Нормальное состояние', tone: 'normal' }
   159→  return { label: 'Отличное состояние', tone: 'ok' }
   160→}

   607→export function demoHiveById(id: number): DemoHive {
   608→  if (id === 0) return demoAggregateHive()

   610→}

   613→/* Сводный вид «Вся пасека» (id = 0): средние показатели по 4 ульям     */
   614→/* ------------------------------------------------------------------ */

   616→const avg = (xs: number[]) => xs.reduce((s, x) => s + x, 0) / xs.length
   617→
   618→/**
   619→ * Сводный улей: показатели пасеки целиком — средние климат и акустика,
   620→ * суммарные вес и привес, средняя заклещенность. Предписания — самые важные

   672→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   673→function seeded(seed: number): () => number {
   674→  let a = seed >>> 0
   675→  return () => {
   676→    a |= 0
   677→    a = (a + 0x6d2b79f5) | 0
   678→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   679→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   680→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   681→  }
   682→}

   684→/** Температура и влажность за 14 дней: колебания вокруг базовых значений. */
   685→export function demoClimateSeries(h: DemoHive, days = 14): DayPoint[] {
   686→  const rnd = seeded(1000 + h.id * 97)
   687→  const out: DayPoint[] = []
   688→  const nowMs = Date.now()
   689→  for (let k = days - 1; k >= 0; k--) {
   690→    const t = new Date(nowMs - k * 86400000)
   691→    const recent = k <= 1
   692→    const tempC = recent
   693→      ? h.base.tempC + (rnd() - 0.5) * 0.2
   694→      : h.base.tempC - 0.4 + rnd() * 0.8
   695→    const humidityPct = recent
   696→      ? h.base.humidityPct + (rnd() - 0.5) * 2
   697→      : h.base.humidityPct - 4 + rnd() * 8
   698→    out.push({
   699→      date: format(t, 'yyyy-MM-dd'),
   700→      label: format(t, 'd.MM'),
   701→      full: format(t, 'd MMMM', { locale: ru }),
   702→      weightKg: null,
   703→      tempInC: Math.round(tempC * 10) / 10,
   704→      humidityPct: Math.round(humidityPct),
   705→    })
   706→  }
   707→  // последние две точки точно соответствуют «сейчас»
   708→  out[out.length - 1].tempInC = h.base.tempC
   709→  out[out.length - 1].humidityPct = h.base.humidityPct
   710→  return out
   711→}

   733→export interface VarroaPoint {
   734→  label: string
   735→  /** Историческое значение заклещенности, %. */
   736→  pct: number | null
   737→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   738→  forecast: number | null
   739→  /** Отметка проведенной обработки. */
   740→  treated: boolean
   741→}
   742→
   743→/**
   744→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   745→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   746→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   747→ *
   748→ * opts.overridePct — «текущее» значение после проведения обработки или нового
   749→ * замера (кривая сходится к нему); opts.treatedToday — сегодня была обработка:
   750→ * на последней точке рисуется дип, а прогноз считается от нового значения.
   751→ */
   752→export function demoVarroaSeries(
   753→  h: DemoHive,
   754→  days = 30,
   755→  opts?: { overridePct?: number; treatedToday?: boolean },
   756→): VarroaPoint[] {
   757→  const rnd = seeded(3000 + h.id * 57)
   758→  const out: VarroaPoint[] = []
   759→  const nowMs = Date.now()
   760→
   761→  const curPct = opts?.overridePct ?? h.base.varroaPct
   762→  const treatedToday = opts?.treatedToday ?? false
   763→  // прогноз: после обработки семья восстанавливается (+0,1 за 14 дн),
   764→  // без обработки критический улей продолжает расти (+1,6)
   765→  const futurePct =
   766→    opts?.overridePct !== undefined || treatedToday
   767→      ? Math.round((curPct + (treatedToday ? 0.1 : 0.2)) * 10) / 10
   768→      : h.base.varroaForecast14d
   769→
   770→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   771→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   772→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   773→
   774→  for (let k = days - 1; k >= 0; k--) {
   775→    const t = new Date(nowMs - k * 86400000)
   776→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   777→    let pct = startPct + (curPct - startPct) * progress
   778→    // дип после обработки: −1,2 % в течение 3 дней
   779→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   780→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   781→    }
   782→    pct += (rnd() - 0.5) * 0.12
   783→    out.push({
   784→      label: format(t, 'd.MM'),
   785→      pct: Math.max(0.2, Math.round(pct * 10) / 10),
   786→      forecast: null,
   787→      treated: treatDay > 0 && k === treatDay,
   788→    })
   789→  }
   790→  out[out.length - 1].pct = curPct
   791→
   792→  // свежая обработка: дип на последней точке (кривая резко падает)
   793→  if (treatedToday && out.length >= 2) {
   794→    out[out.length - 2].treated = true
   795→  }
   796→
   797→  // прогноз: через 14 дней (пунктир от текущей точки)
   798→  const future = new Date(nowMs + 14 * 86400000)
   799→  out.push({
   800→    label: format(future, 'd.MM'),
   801→    pct: null,
   802→    forecast: futurePct,
   803→    treated: false,
   804→  })
   805→  // сегмент начинается с последней исторической точки
   806→  out[out.length - 2].forecast = curPct
   807→  return out
   808→}
   809→
   810→/**
   811→ * Привесы за 7 дней: столбцы суточного привеса + ночная температура
   812→ * (ночные похолодания объясняют провалы приноса нектара).
   813→ */
   814→export function demoGainSeries(h: DemoHive, days = 7): Array<{ label: string; gainKg: number; nightTempC: number }> {
   815→  const rnd = seeded(4000 + h.id * 83)
   816→  const out: Array<{ label: string; gainKg: number; nightTempC: number }> = []
   817→  const nowMs = Date.now()
   818→  for (let k = days - 1; k >= 0; k--) {
   819→    const t = new Date(nowMs - k * 86400000)
   820→    // привес качается вокруг базового: погода и сила семьи
   821→    const gainKg = k === 0 ? h.base.dailyGainKg : Math.max(-0.1, h.base.dailyGainKg + (rnd() - 0.45) * 0.5)
   822→    // ночная температура: волна + случайные похолодания (2 прохладные ночи)
   823→    const wave = Math.sin(((days - k) / days) * Math.PI * 1.4) * 3
   824→    const dip = k === days - 2 || k === 1 ? -3.5 : 0
   825→    const nightTempC = 14 + wave + dip + (rnd() - 0.5) * 2
   826→    out.push({
   827→      label: format(t, 'd.MM'),
   828→      gainKg: Math.round(gainKg * 100) / 100,
   829→      nightTempC: Math.round(nightTempC * 10) / 10,
   830→    })
   831→  }
   832→  return out
   833→}

   854→  { key: 'buckwheat', name: 'Гречиха', need: 760, note: '60–360 кг/га, взяток 3–7 кг в день' },
   855→]
   856→
   857→export function gddForecast(): Array<GddCrop & { pct: number; daysLeft: number; soon: boolean }> {
   858→  return GDD_CROPS.map((c) => {
   859→    const daysLeft = Math.max(1, Math.round((c.need - GDD_CURRENT) / GDD_PER_DAY))
   860→    return {
   861→      ...c,
   862→      pct: Math.min(100, Math.round((GDD_CURRENT / c.need) * 100)),
   863→      daysLeft,
   864→      soon: daysLeft <= 7,
   865→    }
   866→  })
   867→}

   870→export function nearestBloom(): { name: string; daysLeft: number; pct: number; note: string } {
   871→  const f = gddForecast().sort((a, b) => a.daysLeft - b.daysLeft)[0]
   872→  return { name: f.name, daysLeft: f.daysLeft, pct: f.pct, note: f.note }
   873→}
