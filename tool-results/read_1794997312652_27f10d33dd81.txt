    64→function scenarioIcon(scenario: Scenario, className: string) {
    65→  const cls = cn('h-4 w-4', className)
    66→  switch (scenario) {
    67→    case 'swarm':
    68→      return <Split className={cls} />
    69→    case 'pesticide':
    70→      return <SprayCan className={cls} />
    71→    case 'weather':
    72→      return <CloudSun className={cls} />
    73→    case 'migration':
    74→      return <Truck className={cls} />
    75→    default:
    76→      return <Activity className={cls} />
    77→  }
    78→}

    80→function WeatherIcon({ kind, className }: { kind: 'sun' | 'sun-cloud' | 'cloud' | 'rain'; className?: string }) {
    81→  const cls = cn('h-5 w-5', className)
    82→  if (kind === 'sun') return <Sun className={cn(cls, 'text-amber-500')} />
    83→  if (kind === 'sun-cloud') return <CloudSun className={cn(cls, 'text-amber-500')} />
    84→  if (kind === 'cloud') return <Cloud className={cn(cls, 'text-gray-400')} />
    85→  return <CloudRain className={cn(cls, 'text-sky-600')} />

   610→/** Итоговое изменение веса за период наблюдений (для KPI «привес»). */
   611→function weightTotal(h: Hive): number {
   612→  const w = h.series[h.series.length - 1]?.weightKg ?? null
   613→  const first = h.series[0]?.weightKg ?? null
   614→  if (w === null || first === null) return 0
   615→  return Math.round((w - first) * 10) / 10
   616→}
