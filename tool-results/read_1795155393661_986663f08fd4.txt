     3→/**
     4→ * Экран «ИИ-агент»: чат пчеловода с цифровым помощником.
     5→ * Агент отвечает сразу по всем ульям пасеки: телеметрия из базы, датчики точек
     6→ * на карте (температура, вес, влажность, шум) и реальная погода из интернета.
     7→ * Темы строго ограничены: погода и ульи — отвечает, всё остальное — вежливо
     8→ * отказывается. Модель, на которой работает агент, пользователю не показывается.
     9→ * Голосовой ввод работает как рация (push-to-talk): кнопку микрофона нужно
    10→ * зажать и говорить — пока она удерживается, идёт запись; при отпускании запись
    11→ * завершается, речь распознаётся (Web Speech или сервер /api/asr) и текст
    12→ * подставляется в поле ввода.
    13→ * Улии и показания синхронизированы с картой: агент говорит по тем же точкам
    14→ * и датчикам, что видит пчеловод на карте и дашборде.
    15→ * Чаты изолированы по пользователям: у каждого свой список чатов (SQLite),
    16→ * первый вопрос даёт чату заголовок.
    17→ */

   150→        const j = (await res.json()) as WeatherResponse
   151→        if (!cancelled) setWResp(j)

   156→    void load()
   157→    const timer = setInterval(() => void load(), 10 * 60 * 1000)

   160→      clearInterval(timer)
   161→    }
   162→  }, [wLat, wLon])
   163→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
   164→
   165→  // Ульи и показания — те же точки, что на карте (синхронизация с /api/map-points)
   166→  const pointsCtx = useMemo<AgentContext | null>(() => {

   178→    return makeContext(pointsToHives(apiaryPoints), {
   179→      weather: agentWeather?.days ?? null,
   180→      weatherNow: agentWeather?.now ?? null,
   181→      pesticide: null, // реестр обработок не подключён — агент честно об этом скажет
   182→      sites: data ? data.sites.map(siteDtoToSite) : [],
   183→      points,
   184→      cropFields,
   185→      apiaryName: 'Моя пасека',
   186→      lang: useAppState.getState().lang,
   187→    })
   188→  }, [apiaryPoints, agentWeather, data, cropFields])
   189→
   190→  // контекст мемоизирован, чтобы не пересоздаваться на каждом рендере
   191→  // (иначе эффекты с ctx в зависимостях зацикливаются)
   192→  const ctx: AgentContext | null = pointsCtx
   193→
   194→  const [items, setItems] = useState<ChatItem[]>([])
   195→  const [input, setInput] = useState('')
   196→  const [thinking, setThinking] = useState(false)
   197→  const thinkingRef = useRef(false)
   198→  const scrollRef = useRef<HTMLDivElement>(null)
   199→
   200→  /* ---------- чаты пользователя: у каждого свой список ---------- */
   201→  const [sessions, setSessions] = useState<ChatSessionInfo[]>([])
   202→  const [activeSessionId, setActiveSessionId] = useState<string | null>(null)
   203→  // актуальные значения для колбэков: sendUser пишет в текущий чат
   204→  const activeSessionRef = useRef<string | null>(null)
   205→  activeSessionRef.current = activeSessionId
   206→  const ctxRef = useRef<AgentContext | null>(null)
   207→  useEffect(() => {
   208→    ctxRef.current = ctx
   209→  }, [ctx])
   210→
   211→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   212→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   213→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   214→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   215→  const srClass = useMemo(() => getSpeechRecognition(), [])
   216→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   217→  const [voice, setVoice] = useState<VoiceState>('idle')
   218→  const speechRef = useRef<SpeechRecognitionLike | null>(null)
   219→  const recorderRef = useRef<MediaRecorder | null>(null)
   220→  const streamRef = useRef<MediaStream | null>(null)
   221→  const chunksRef = useRef<Blob[]>([])
   222→  const autoStopRef = useRef<number | null>(null)
   223→  const baseInputRef = useRef('')
   224→  // push-to-talk: кнопка микрофона зажата и момент начала удержания
   225→  // (по длительности отличаем намеренное удержание от случайного тапа)
   226→  const holdActiveRef = useRef(false)
   227→  const holdStartedAtRef = useRef(0)
   228→
   229→  const clearAutoStop = () => {
   230→    if (autoStopRef.current !== null) {
   231→      window.clearTimeout(autoStopRef.current)
   232→      autoStopRef.current = null
   233→    }
   234→  }
   235→
   236→  const stopVoice = useCallback(() => {
   237→    clearAutoStop()
   238→    if (speechRef.current) {
   239→      try {
   240→        speechRef.current.stop()
   241→      } catch {
   242→        // уже остановлен
   243→      }
   244→      return
   245→    }
   246→    const rec = recorderRef.current
   247→    if (rec && rec.state !== 'inactive') rec.stop()
   248→  }, [])
   249→
   250→  /** Отпускание кнопки (push-to-talk): завершаем запись. */
   251→  const releaseHold = useCallback(() => {
   252→    if (!holdActiveRef.current) return
   253→    holdActiveRef.current = false
   254→    const wasRecording = voice === 'recording'
   255→    const heldMs = Date.now() - holdStartedAtRef.current
   256→    stopVoice()
   257→    // мгновенный тап вместо удержания — вежливо подсказываем, как пользоваться
   258→    if (wasRecording && heldMs < 600) {
   259→      toast({
   260→        title: 'Короткое нажатие',
   261→        description: 'Зажмите кнопку микрофона и говорите, отпустите — вопрос появится в поле ввода.',
   262→      })
   263→    }
   264→  }, [stopVoice, voice, toast])
   265→
   266→  /** Запасной путь: собранный blob → WAV → /api/asr → текст. */
   267→  const finishVoice = useCallback(async () => {
   268→    // поток больше не нужен
   269→    streamRef.current?.getTracks().forEach((t) => t.stop())
   270→    streamRef.current = null
   271→    recorderRef.current = null
   272→    const blob = new Blob(chunksRef.current, { type: chunksRef.current[0]?.type || 'audio/webm' })
   273→    chunksRef.current = []
   274→    if (blob.size < 2048) {
   275→      setVoice('idle')
   276→      toast({ title: 'Запись пустая', description: 'Зажмите кнопку микрофона и скажите вопрос голосом.' })
   277→      return
   278→    }
   279→    setVoice('transcribing')
   280→    try {
   281→      const wav = await blobToWavBase64(blob)
   282→      const res = await fetch('/api/asr', {
   283→        method: 'POST',
   284→        headers: { 'Content-Type': 'application/json' },
   285→        body: JSON.stringify({ audio: wav }),
   286→      })
   287→      const j = (await res.json().catch(() => ({}))) as { text?: string; error?: string }
   288→      if (!res.ok || !j.text?.trim()) {
   289→        toast({ title: 'Речь не распознана', description: j.error ?? 'Попробуйте сказать чуть чётче.', variant: 'destructive' })
   290→        return
   291→      }
   292→      // текст подставляем в поле ввода — пчеловод может поправить и отправить
   293→      const base = baseInputRef.current
   294→      const text = j.text.trim()
   295→      setInput(base ? `${base} ${text}` : text)
   296→      toast({ title: 'Голос распознан', description: text })
   297→    } catch {
   298→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   299→    } finally {
   300→      setVoice('idle')
   301→    }
   302→  }, [toast])
   303→
   304→  const startVoice = useCallback(async () => {
   305→    // запись уже идёт — второй старт (второй палец/указатель) не нужен
   306→    if (speechRef.current || recorderRef.current) return
   307→    clearAutoStop()
   308→    baseInputRef.current = input.trim()
   309→
   310→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   311→    if (srClass) {
   312→      const rec = new srClass()
   313→      rec.lang = VOICE_LANG
   314→      rec.continuous = true
   315→      rec.interimResults = true
   316→      rec.maxAlternatives = 1
   317→      let finalText = ''
   318→      rec.onresult = (e) => {
   319→        let interim = ''
   320→        for (let i = e.resultIndex; i < e.results.length; i++) {
   321→          const r = e.results[i]
   322→          const t = r[0]?.transcript ?? ''
   323→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   324→          else interim += t
   325→        }
   326→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   327→        if (text) {
   328→          const base = baseInputRef.current
   329→          setInput(base ? `${base} ${text}` : text)
   330→        }
   331→      }
   332→      rec.onerror = (ev) => {
   333→        const code = ev.error ?? ''
   334→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   335→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   336→        } else if (code === 'no-speech') {
   337→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   338→        } else if (code !== 'aborted') {
   339→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   340→        }
   341→      }
   342→      rec.onend = () => {
   343→        speechRef.current = null
   344→        clearAutoStop()
   345→        setVoice('idle')
   346→      }
   347→      speechRef.current = rec
   348→      try {
   349→        rec.start()
   350→      } catch {
   351→        toast({ title: 'Не получилось начать', description: 'Зажмите кнопку микрофона ещё раз.', variant: 'destructive' })
   352→        return
   353→      }
   354→      setVoice('recording')
   355→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   356→      // кнопку отпустили раньше, чем распознавание успело стартовать
   357→      if (!holdActiveRef.current) stopVoice()
   358→      return
   359→    }
   360→
   361→    // 2) запасной путь: запись с микрофона → серверное распознавание
   362→    try {
   363→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   364→      streamRef.current = stream
   365→      chunksRef.current = []
   366→      const rec = new MediaRecorder(stream)
   367→      rec.ondataavailable = (e) => {
   368→        if (e.data.size > 0) chunksRef.current.push(e.data)
   369→      }
   370→      rec.onstop = () => void finishVoice()
   371→      rec.start()
   372→      recorderRef.current = rec
   373→      setVoice('recording')
   374→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   375→      // кнопку отпустили, пока запрашивался доступ к микрофону
   376→      if (!holdActiveRef.current) stopVoice()
   377→    } catch {
   378→      toast({
   379→        title: 'Микрофон недоступен',
   380→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   381→        variant: 'destructive',
   382→      })
   383→    }
   384→  }, [srClass, input, finishVoice, stopVoice, toast])
   385→
   386→  // при размонтировании прерываем распознавание и закрываем поток
   387→  useEffect(
   388→    () => () => {
   389→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   390→      try {
   391→        speechRef.current?.abort()
   392→      } catch {
   393→        // нет активного распознавания
   394→      }
   395→      streamRef.current?.getTracks().forEach((t) => t.stop())
   396→    },
   397→    [],
   398→  )
   399→
   400→  /* ---------- чаты: список пользователя + сообщения выбранного ---------- */
   401→  const ctxReady = !!ctx && ctx.hives.length > 0
   402→
   403→  // Список чатов текущего пользователя; если пусто — создаём первый.
   404→  useEffect(() => {
   405→    if (!user || realStatus !== 'ready') return
   406→    let cancelled = false
   407→    void (async () => {
   408→      try {
   409→        const res = await fetch('/api/chat/sessions', { cache: 'no-store' })
   410→        if (!res.ok) throw new Error('sessions')
   411→        let list = ((await res.json()) as { sessions: ChatSessionInfo[] }).sessions
   412→        if (list.length === 0) {
   413→          const created = await fetch('/api/chat/sessions', { method: 'POST' })
   414→          if (created.ok) {
   415→            const j = (await created.json()) as { session: ChatSessionInfo }
   416→            list = [j.session]
   417→          }
   418→        }
   419→        if (!cancelled) {
   420→          setSessions(list)
   421→          setActiveSessionId((prev) => (prev && list.some((s) => s.id === prev) ? prev : (list[0]?.id ?? null)))
   422→        }
   423→      } catch {
   424→        // без списка чатов чат всё равно работает — история подтянется позже
   425→      }
   426→    })()
   427→    return () => {
   428→      cancelled = true
   429→    }
   430→  }, [user, realStatus])
   431→
   432→  // Сообщения выбранного чата: история из базы или приветствие в новом.
   433→  useEffect(() => {
   434→    if (!user || realStatus !== 'ready' || !activeSessionId || !ctxReady) return
   435→    let cancelled = false
   436→    void (async () => {
   437→      try {
   438→        const res = await fetch(`/api/chat?sessionId=${encodeURIComponent(activeSessionId)}`, { cache: 'no-store' })
   439→        if (!res.ok) throw new Error('history')
   440→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   441→        if (cancelled) return
   442→        const restored: ChatItem[] = j.items.map((m) => ({
   443→          role: m.role,
   444→          text: m.text,
   445→          ...(m.role === 'agent' && m.agentJson
   446→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   447→            : {}),
   448→        }))
   449→        if (restored.length === 0) {
   450→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctxRef.current?.hives[0]?.id ?? 1, ctxRef.current!) }])
   451→        } else {
   452→          setItems(restored.slice(-30))
   453→        }
   454→      } catch {
   455→        if (!cancelled) {
