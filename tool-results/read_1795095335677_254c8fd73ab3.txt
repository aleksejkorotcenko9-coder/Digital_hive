   134→   134→ * Статус заклещенности: Норма <2 % / Риск 2–3 % / Критично >3 %.

   138→   138→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; key: string; label: string; cls: string } {
   139→   139→  if (pct > 3) return { level: 'crit', key: 'dash.varroa.status.crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   140→   140→  if (pct >= 2) return { level: 'risk', key: 'dash.varroa.status.risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   141→   141→  return { level: 'ok', key: 'dash.varroa.status.ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   142→   142→}

   694→   694→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   695→   695→function seeded(seed: number): () => number {
   696→   696→  let a = seed >>> 0
   697→   697→  return () => {
   698→   698→    a |= 0
   699→   699→    a = (a + 0x6d2b79f5) | 0
   700→   700→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   701→   701→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   702→   702→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   703→   703→  }
   704→   704→}

   755→   755→export interface VarroaPoint {
   756→   756→  label: string
   757→   757→  /** Историческое значение заклещенности, %. */
   758→   758→  pct: number | null
   759→   759→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   760→   760→  forecast: number | null
   761→   761→  /** Отметка проведенной обработки. */
   762→   762→  treated: boolean
   763→   763→}
   764→   764→
   765→   765→/**
   766→   766→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   767→   767→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   768→   768→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   769→   769→ *
   770→   770→ * opts.overridePct — «текущее» значение после проведения обработки или нового
   771→   771→ * замера (кривая сходится к нему); opts.treatedToday — сегодня была обработка:
   772→   772→ * на последней точке рисуется дип, а прогноз считается от нового значения.
   773→   773→ */
   774→   774→export function demoVarroaSeries(
   775→   775→  h: DemoHive,
   776→   776→  days = 30,
   777→   777→  opts?: { overridePct?: number; treatedToday?: boolean },
   778→   778→): VarroaPoint[] {
   779→   779→  const rnd = seeded(3000 + h.id * 57)
   780→   780→  const out: VarroaPoint[] = []
   781→   781→  const nowMs = Date.now()
   782→   782→
   783→   783→  const curPct = opts?.overridePct ?? h.base.varroaPct
   784→   784→  const treatedToday = opts?.treatedToday ?? false
   785→   785→  // прогноз: после обработки семья восстанавливается (+0,1 за 14 дн),
   786→   786→  // без обработки критический улей продолжает расти (+1,6)
   787→   787→  const futurePct =
   788→   788→    opts?.overridePct !== undefined || treatedToday
   789→   789→      ? Math.round((curPct + (treatedToday ? 0.1 : 0.2)) * 10) / 10
   790→   790→      : h.base.varroaForecast14d
   791→   791→
   792→   792→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   793→   793→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   794→   794→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   795→   795→
   796→   796→  for (let k = days - 1; k >= 0; k--) {
   797→   797→    const t = new Date(nowMs - k * 86400000)
   798→   798→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   799→   799→    let pct = startPct + (curPct - startPct) * progress
   800→   800→    // дип после обработки: −1,2 % в течение 3 дней
   801→   801→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   802→   802→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   803→   803→    }
   804→   804→    pct += (rnd() - 0.5) * 0.12
   805→   805→    out.push({
   806→   806→      label: format(t, 'd.MM'),
   807→   807→      pct: Math.max(0.2, Math.round(pct * 10) / 10),
   808→   808→      forecast: null,
   809→   809→      treated: treatDay > 0 && k === treatDay,
   810→   810→    })
   811→   811→  }
   812→   812→  out[out.length - 1].pct = curPct
   813→   813→
   814→   814→  // свежая обработка: дип на последней точке (кривая резко падает)
   815→   815→  if (treatedToday && out.length >= 2) {
   816→   816→    out[out.length - 2].treated = true
   817→   817→  }
   818→   818→
   819→   819→  // прогноз: через 14 дней (пунктир от текущей точки)
   820→   820→  const future = new Date(nowMs + 14 * 86400000)
   821→   821→  out.push({
   822→   822→    label: format(future, 'd.MM'),
   823→   823→    pct: null,
   824→   824→    forecast: futurePct,
   825→   825→    treated: false,
   826→   826→  })
   827→   827→  // сегмент начинается с последней исторической точки
   828→   828→  out[out.length - 2].forecast = curPct
   829→   829→  return out
   830→   830→}
   831→   831→
   832→   832→/**
   833→   833→ * Привесы за 7 дней: столбцы суточного привеса + ночная температура
   834→   834→ * (ночные похолодания объясняют провалы приноса нектара).
   835→   835→ */
   836→   836→export function demoGainSeries(h: DemoHive, days = 7): Array<{ label: string; gainKg: number; nightTempC: number }> {
   837→   837→  const rnd = seeded(4000 + h.id * 83)
   838→   838→  const out: Array<{ label: string; gainKg: number; nightTempC: number }> = []
   839→   839→  const nowMs = Date.now()
   840→   840→  for (let k = days - 1; k >= 0; k--) {
   841→   841→    const t = new Date(nowMs - k * 86400000)
   842→   842→    // привес качается вокруг базового: погода и сила семьи
   843→   843→    const gainKg = k === 0 ? h.base.dailyGainKg : Math.max(-0.1, h.base.dailyGainKg + (rnd() - 0.45) * 0.5)
   844→   844→    // ночная температура: волна + случайные похолодания (2 прохладные ночи)
   845→   845→    const wave = Math.sin(((days - k) / days) * Math.PI * 1.4) * 3
   846→   846→    const dip = k === days - 2 || k === 1 ? -3.5 : 0
   847→   847→    const nightTempC = 14 + wave + dip + (rnd() - 0.5) * 2
   848→   848→    out.push({
   849→   849→      label: format(t, 'd.MM'),
   850→   850→      gainKg: Math.round(gainKg * 100) / 100,
   851→   851→      nightTempC: Math.round(nightTempC * 10) / 10,
   852→   852→    })
   853→   853→  }
   854→   854→  return out
   855→   855→}
