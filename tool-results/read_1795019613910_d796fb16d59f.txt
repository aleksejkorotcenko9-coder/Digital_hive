   163→/* Метрики по рядам (безопасны к пропускам значений)                   */

   165→
   166→function lastPoint(h: Hive) {
   167→  return h.series[h.series.length - 1] ?? null
   168→}
   169→
   170→function lastWeight(h: Hive): number | null {
   171→  return lastPoint(h)?.weightKg ?? null
   172→}
   173→
   174→/** Изменение веса за `backDays` календарных дней; null, если точек не хватает.
   175→ *  Календарный поиск с допуском ±1,5 дня — корректен при пропусках дней. */
   176→function weightDelta(h: Hive, backDays: number): number | null {
   177→  const s = h.series
   178→  if (s.length === 0) return null
   179→  const lastPoint = s[s.length - 1]
   180→  if (lastPoint.weightKg === null) return null
   181→  const target = new Date(lastPoint.date).getTime() - backDays * 86400000
   182→  let best: number | null = null
   183→  let bestDiff = Infinity
   184→  for (const p of s) {
   185→    if (p.weightKg === null || p === lastPoint) continue
   186→    const diff = Math.abs(new Date(p.date).getTime() - target)
   187→    if (diff <= 1.5 * 86400000 && diff < bestDiff) {
   188→      bestDiff = diff
   189→      best = p.weightKg
   190→    }
   191→  }
   192→  if (best === null) return null
   193→  return Math.round((lastPoint.weightKg - best) * 10) / 10
   194→}
   195→
   196→const delta2d = (h: Hive) => weightDelta(h, 2)
   197→const delta30d = (h: Hive) => weightDelta(h, Math.min(29, Math.max(1, spanDays(h) - 1)))
   198→
   199→function maxTempRecent(h: Hive, n = 3): number | null {
   200→  const temps = h.series.slice(-n).map((p) => p.tempInC).filter((t): t is number => t !== null)
   201→  return temps.length ? Math.max(...temps) : null
   202→}
   203→
   204→function lastHumidity(h: Hive): number | null {
   205→  return lastPoint(h)?.humidityPct ?? null
   206→}
   207→
   208→/** Лидер пасеки по 30-дневному привесу. */
   209→function leaderOf(list: Hive[]): Hive | null {
   210→  const withW = list.filter((x) => delta30d(x) !== null)
   211→  if (!withW.length) return null
   212→  return [...withW].sort((a, b) => (delta30d(b) ?? 0) - (delta30d(a) ?? 0))[0]
   213→}
   214→
   215→/** Сколько дней охватывает ряд (для честных подписей «за N дн.»). */
   216→function spanDays(h: Hive): number {
   217→  if (h.series.length < 2) return h.series.length
   218→  const a = new Date(h.series[0].date).getTime()
   219→  const b = new Date(h.series[h.series.length - 1].date).getTime()
   220→  if (Number.isNaN(a) || Number.isNaN(b)) return h.series.length
   221→  return Math.round((b - a) / 86400000) + 1
   222→}
   223→
   224→function periodLabel(h: Hive): string {
   225→  const d = spanDays(h)
   226→  if (d >= 25) return 'за месяц'
   227→  if (d === 1) return 'за день'
   228→  if (d === 0) return '—'
   229→  return `за ${d} дн.`
   230→}

   233→const SWARM_DROP = -1.0

   235→function swarmProbability(drop: number): string {
   236→  const p = drop <= -3 ? 80 : drop <= -2 ? 70 : drop <= -1.5 ? 62 : 55
   237→  return `Вероятность подготовки к роению — около ${p} %`
   238→}

   244→function hiveFinding(h: Hive, ctx: AgentContext): AgentFinding {
   245→  const status = hiveStatus(h, ctx.hives)
   246→  const d2 = delta2d(h)
   247→  const d30 = delta30d(h)
   248→  const t = lastPoint(h)?.tempInC ?? null
   249→  const hum = lastHumidity(h)
   250→
   251→  if (h.series.length === 0 || (lastWeight(h) === null && t === null)) {
   252→    return {
   253→      id: `nodata-${h.id}`,
   254→      scenario: 'status',
   255→      hiveId: h.id,
   256→      title: `Улей №${h.id}: пока нет данных`,
   257→      text: `По улью №${h.id} в базе нет измерений. Я не буду гадать: как только появятся показания — загрузите их в разделе «Данные» (CSV, вручную или через API), и я сразу начну следить за весом, температурой и влажностью.`,
   258→      basis: ['База данных платформы', 'Измерения улья №' + h.id],
   259→      urgency: 'this-week',
   260→      urgencyNote: 'Загрузить данные',
   261→      probability: 'Оценка невозможна — данных нет',
   262→      metrics: [{ label: 'Измерений', value: String(h.series.length) }],
   263→      prefill: `Что происходит с ульем №${h.id}?`,
   264→    }
   265→  }
   266→
   267→  if (status.level === 'risk' && d2 !== null) {
   268→    const peak = Math.max(...h.series.slice(-3).map((p) => p.weightKg ?? -Infinity))
   269→    const now = lastWeight(h) ?? peak
   270→    return {
   271→      id: `swarm-${h.id}`,
   272→      scenario: 'swarm',
   273→      hiveId: h.id,
   274→      title: 'Похоже, семья готовится к роению',
   275→      text: `За последние 2 дня вес улья №${h.id} снизился на ${fmtNum(Math.abs(d2))} кг — с ${fmtNum(peak)} до ${fmtNum(now)} кг, хотя до этого семья стабильно набирала мед.${
   276→        t !== null ? ` Температура внутри поднималась до ${fmtNum(t)} °C (норма около 34,5 °C).` : ''
   277→      }${hum !== null ? ` Влажность держится на отметке ${fmtNum(hum, 0)} % — выше обычной для этого улья.` : ''} Июль — пик сезона роения: часть пчёл с маткой может улететь, и медосбор остановится на 2–3 недели.`,
   278→      basis: [
   279→        `График веса улья №${h.id}`,
   280→        `Вес: ${fmtSigned(d2)} кг за 2 дня`,
   281→        ...(t !== null ? [`Температура: ${fmtNum(t)} °C при норме 34,5 °C`] : []),
   282→        'Сезон: июль — период роения',
   283→      ],
   284→      urgency: 'today',
   285→      urgencyNote: 'Осмотреть улей сегодня до вечера',
   286→      probability: swarmProbability(d2),
   287→      metrics: [
   288→        { label: 'Вес за 2 дня', value: `${fmtSigned(d2)} кг` },
   289→        ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   290→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   291→      ],
   292→      prefill: `Что происходит с ульем №${h.id}?`,
   293→    }
   294→  }
   295→
   296→  if (status.level === 'watch' && h.label !== undefined) {
   297→    const leader = leaderOf(ctx.hives)
   298→    const reason = !h.humiditySensorOk && h.humidityOfflineFrom
   299→      ? 'отдельно: датчик влажности не отвечает'
   300→      : 'привес хуже, чем у лидера пасеки'
   301→    return {
   302→      id: `slow-${h.id}`,
   303→      scenario: 'status',
   304→      hiveId: h.id,
   305→      title: 'Слабый набор веса',
   306→      text: `Улей №${h.id} прибавил только ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}${
   307→        leader && leader.id !== h.id ? ` — у лидера (улей №${leader.id}) ${fmtSigned(delta30d(leader) ?? 0)} кг` : ''
   308→      }. Температура${t !== null ? ` ${fmtNum(t)} °C` : ''} в норме. Причин может быть две: слабая семья или бедная кормовая база вокруг. ${reason.charAt(0).toUpperCase() + reason.slice(1)}${
   309→        h.humidityOfflineFrom ? ` — молчит с ${h.humidityOfflineFrom}, возможно, сел аккумулятор` : ' — стоит сверить привес с соседними ульями'
   310→      }.`,
   311→      basis: [
   312→        'Сравнение привеса по ульям пасеки',
   313→        `Привес улья №${h.id}: ${fmtSigned(d30 ?? 0)} кг`,
   314→        ...(h.humidityOfflineFrom ? [`Датчик влажности: нет данных с ${h.humidityOfflineFrom}`] : []),
   315→      ],
   316→      urgency: 'this-week',
   317→      urgencyNote: 'Проверить датчик и семью в ближайшие дни',
   318→      probability: 'Диагноз по косвенным признакам — нужна проверка',
   319→      metrics: [
   320→        { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   321→        ...(leader && leader.id !== h.id
   322→          ? [{ label: `У лидера (№${leader.id})`, value: `${fmtSigned(delta30d(leader) ?? 0)} кг` }]
   323→          : []),
   324→        ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : [{ label: 'Влажность', value: 'нет данных' }]),
   325→      ],
   326→      prefill: `Почему в улье №${h.id} слабый привес?`,
   327→    }
   328→  }
   329→
   330→  return {
   331→    id: `ok-${h.id}`,
   332→    scenario: 'status',
   333→    hiveId: h.id,
   334→    title: 'Семья работает штатно',
   335→    text: `Улей №${h.id} развивается штатно: вес ${fmtNum(lastWeight(h) ?? 0)} кг, привес ${fmtSigned(d30 ?? 0)} кг ${periodLabel(h)}, за последние 2 дня ${fmtSigned(d2 ?? 0)} кг.${
   336→      t !== null ? ` Температура ${fmtNum(t)} °C — в норме.` : ''
   337→    } Признаков роения нет: вес растёт ровно, без провалов.`,
   338→    basis: [
   339→      `График веса улья №${h.id}`,
   340→      `Привес: ${fmtSigned(d30 ?? 0)} кг`,
   341→      'Температура в норме семьи',
   342→    ],
   343→    urgency: 'ok',
   344→    urgencyNote: 'Продолжать наблюдение',
   345→    metrics: [
   346→      { label: `Привес ${periodLabel(h)}`, value: `${fmtSigned(d30 ?? 0)} кг` },
   347→      ...(t !== null ? [{ label: 'Темп. внутри', value: `${fmtNum(t)} °C` }] : []),
   348→      ...(hum !== null ? [{ label: 'Влажность', value: `${fmtNum(hum, 0)} %` }] : []),
   349→    ],
   350→    prefill: `Что происходит с улье №${h.id}?`,
   351→  }

   381→function weatherFindingOf(ctx: AgentContext): AgentFinding | null {
   382→  const w = ctx.weather
   383→  if (!w || w.length === 0) return null
   384→  const good = w.filter((d) => d.flight === 'good')
   385→  const bad = w.filter((d) => d.flight === 'bad')
   386→  const tMax = Math.max(...w.map((d) => d.tMax))
   387→  const rainProb = bad.length ? Math.max(...bad.map((d) => d.precipProb)) : 0
   388→  return {
   389→    id: 'weather',
   390→    scenario: 'weather',
   391→    hiveId: null,
   392→    title: good.length
   393→      ? `Окно лёта: ${good.map((d) => `${Number(d.date.slice(8))} ${monthGen(d.date)}`).join(', ').slice(0, 80)}`
   394→      : 'Ближайшие дни — не для лёта',
   395→    text: good.length
   396→      ? `Ближайшие дни с хорошими условиями: ${good.map((d) => d.full).join(', ')} — до ${fmtSigned(tMax, 0)} °C, ветер до ${Math.max(...good.map((d) => d.windMs))} м/с, осадки маловероятны.${
   397→          bad.length ? ` Затем, ${bad.map((d) => d.full).join(' и ')}, — дожди (до ${rainProb} %) и ветер до ${Math.max(...bad.map((d) => d.windMs))} м/с: пчёлы останутся в ульях, привес остановится.` : ''
   398→        }`
   399→      : `По прогнозу на неделю погода не летная: дожди и ветер. Пчёлы останутся в ульях, привес остановится — следите за запасами корма по графику веса.`,

   449→export function findingsFor(hiveId: number, ctx: AgentContext = DEMO_CONTEXT): AgentFinding[] {
   450→  const hive = ctx.hives.find((x) => x.id === hiveId) ?? ctx.hives[0]
   451→  const list: AgentFinding[] = []
   452→  if (hive) list.push(hiveFinding(hive, ctx))
   453→  const pesticide = pesticideFindingOf(ctx)
   454→  if (pesticide) list.push(pesticide)
   455→  const weatherF = weatherFindingOf(ctx)
   456→  if (weatherF) list.push(weatherF)
   457→  const migrationF = migrationFindingOf(ctx)
   458→  if (migrationF) list.push(migrationF)
   459→  const rank: Record<Urgency, number> = { 'act-now': 0, today: 1, 'this-week': 2, ok: 3 }
   460→  return list.sort((a, b) => rank[a.urgency] - rank[b.urgency])
   461→}

  1014→      probability: swarmProbability(delta2d(risky[0]) ?? SWARM_DROP),

  1020→  }

  1240→/** Главная функция: вопрос пчеловода → структурированный ответ агента. */
  1241→export function askAgent(
  1242→  question: string,
  1243→  currentHiveId: number,
  1244→  ctx: AgentContext = DEMO_CONTEXT,
  1245→): AgentMessage {
  1246→  const q = norm(question)
  1247→  const hiveId = extractHiveId(q, ctx)
  1248→  const hive = ctx.hives.find((x) => x.id === (hiveId ?? currentHiveId)) ?? ctx.hives[0]
  1249→
  1250→  const has = (...patterns: RegExp[]) => patterns.some((p) => p.test(q))
  1251→
  1252→  // Приветствие и сервисные фразы
  1253→  if (has(/привет|здравств|добрый день|добрый вечер|доброе утро|что ты умеешь|чем ты можешь|помощь\b|справк/)) {
  1254→    return greetingAnswer(ctx)
  1255→  }
  1256→  // Вопрос о самом агенте: модель и устройство не раскрываем
  1257→  if (has(/кто ты|кто такой|как тебя зовут|ты бот|чат ?бот|какая (ты )?модель|какой (ты )?модель|на (чем|чём) ты (работаешь|сделан|написан)|что ты за|нейросет|искусственн|\bllm\b|\bgpt\b/)) {
  1258→    return identityAnswer()
  1259→  }
  1260→  if (has(/спасибо|благодар|класс|супер|отлично/)) {
  1261→    return thanksAnswer()
  1262→  }
  1263→
  1264→  // Сценарий пестицидов
  1265→  if (has(/пестицид|обработк|хими|опрыск|травят|фунгицид|инсектицид/)) {
  1266→    return pesticideAnswer(ctx)
  1267→  }
  1268→
  1269→  // Кочёвка
  1270→  if (has(/кочев|перевез|перевоз|переех|перемест|новое место|куда (вывезти|ехать|ставить)|где лучше|выбор места|участк/)) {
  1271→    return migrationAnswer(ctx)
  1272→  }
  1273→
  1274→  // Погода и медосбор (включая температуру ВОЗДЕУХА — не путать с температурой улья)
  1275→  if (has(/погод|медосбор|взятк|летат|лёт|лету|дожд|ветер|осадк|цветен|медонос|снег|мороз|когда вывоз|когда летать|температура (воздуха|за окном|на улице|снаружи)/)) {
  1276→    return weatherAnswer(ctx)
  1277→  }
  1278→
  1279→  if (!hive) {
  1280→    return fallbackAnswer()
  1281→  }
  1282→
  1283→  // Темы по ульям: если номер улья не назван — отвечаем сразу по ВСЕМ ульям
  1284→  if (has(/влажн|сырост|сыро/)) {
  1285→    return hiveId ? humidityAnswer(hive) : humidityAllAnswer(ctx)
  1286→  }
  1287→  if (has(/шум|децибел|гул\b/)) {
  1288→    return noiseAnswer(ctx)
  1289→  }
  1290→  if (has(/ро(ение|иться|й|и\b)|риск роения|делит|делени/)) {
  1291→    return hiveId ? swarmAnswer(hive, ctx) : swarmAllAnswer(ctx)
  1292→  }
  1293→  if (has(/температ|жар|тепло внутри|градус/)) {
  1294→    return hiveId ? climateAnswer(hive) : climateAllAnswer(ctx)
  1295→  }
  1296→  if (has(/вес|мед\b|меда|привес|набор|тяже|лидер/)) {
  1297→    return hiveId ? weightAnswer(hive, ctx) : weightAllAnswer(ctx)
  1298→  }
  1299→  if (has(/датчик|сенсор/)) {
  1300→    return sensorsAnswer(ctx)
  1301→  }
  1302→  if (has(/все.{0,8}уль|всех ульев|всей пасек|пасек|обзор|сводк|итог|общее состояни|как дела|как себя|что происходит|состояни|статус|проверь|проблем|тревог|расскажи|что с улеем|как улей/)) {
  1303→    return hiveId ? statusAnswer(hive, ctx) : allHivesAnswer(ctx)
  1304→  }
  1305→
  1306→  // Номер улья без распознанной темы — считаем вопросом «что происходит»
  1307→  if (hiveId !== null) {
  1308→    return statusAnswer(hive, ctx)
  1309→  }
  1310→
  1311→  return fallbackAnswer()
  1312→}
