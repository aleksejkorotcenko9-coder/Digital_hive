   139→/**
   140→ * Статус заклещенности: Норма <2 % / Риск 2–3 % / Критично >3 %.
   141→ * label — русский фолбэк из данных; key — ссылка на базовый словарь i18n
   142→ * (dash.varroa.status.* уже переведены на все языки в i18n.ts).
   143→ */
   144→export function varroaStatus(pct: number): { level: 'ok' | 'risk' | 'crit'; key: string; label: string; cls: string } {
   145→  if (pct > 3) return { level: 'crit', key: 'dash.varroa.status.crit', label: 'Критично', cls: 'bg-red-50 text-red-700 border-red-200' }
   146→  if (pct >= 2) return { level: 'risk', key: 'dash.varroa.status.risk', label: 'Риск', cls: 'bg-amber-50 text-amber-700 border-amber-200' }
   147→  return { level: 'ok', key: 'dash.varroa.status.ok', label: 'Норма', cls: 'bg-emerald-50 text-emerald-700 border-emerald-200' }
   148→}

   150→/** Состояние воздуха по индексу ЛОС (0…100). key → demo.voc.* в словаре DEMO_DICTS. */
   151→export function vocState(v: number): { key: string; label: string; cls: string } {
   152→  if (v > 60) return { key: 'demo.voc.hot', label: 'Духота — семье не хватает объема', cls: 'text-red-600' }
   153→  if (v > 30) return { key: 'demo.voc.mid', label: 'Умеренная загрузка воздухом', cls: 'text-amber-600' }
   154→  return { key: 'demo.voc.ok', label: 'Воздух чистый', cls: 'text-emerald-600' }
   155→}

   157→/** Состояние трафика на летке относительно нормы пасеки (45 пч/мин). key → demo.traffic.* в словаре DEMO_DICTS. */
   158→export function trafficState(v: number): { key: string; label: string; cls: string } {
   159→  if (v >= 55) return { key: 'demo.traffic.high', label: 'Интенсивный лёт', cls: 'text-emerald-600' }
   160→  if (v >= 30) return { key: 'demo.traffic.active', label: 'Активный лёт', cls: 'text-emerald-600' }
   161→  if (v >= 18) return { key: 'demo.traffic.weak', label: 'Лёт ослаблен', cls: 'text-amber-600' }
   162→  return { key: 'demo.traffic.low', label: 'Лёт резко снижен', cls: 'text-red-600' }
   163→}

   165→/** Индекс 0…10 → подпись и тон шкалы. */
   166→export function demoGaugeInfo(value: number): { label: string; tone: 'ok' | 'normal' | 'watch' | 'alert' } {
   167→  if (value < 4) return { label: 'Тревожное состояние', tone: 'alert' }
   168→  if (value < 6) return { label: 'Требует внимания', tone: 'watch' }
   169→  if (value < 8.5) return { label: 'Нормальное состояние', tone: 'normal' }
   170→  return { label: 'Отличное состояние', tone: 'ok' }
   171→}

   700→/** Детерминированный генератор псевдослучайных чисел (mulberry32). */
   701→function seeded(seed: number): () => number {
   702→  let a = seed >>> 0
   703→  return () => {
   704→    a |= 0
   705→    a = (a + 0x6d2b79f5) | 0
   706→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
   707→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   708→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   709→  }
   710→}

   712→/** Температура и влажность за 14 дней: колебания вокруг базовых значений. */
   713→export function demoClimateSeries(h: DemoHive, days = 14): DayPoint[] {
   714→  const rnd = seeded(1000 + h.id * 97)
   715→  const out: DayPoint[] = []

   719→    const recent = k <= 1
   720→    const tempC = recent
   721→      ? h.base.tempC + (rnd() - 0.5) * 0.2
   722→      : h.base.tempC - 0.4 + rnd() * 0.8
   723→    const humidityPct = recent
   724→      ? h.base.humidityPct + (rnd() - 0.5) * 2
   725→      : h.base.humidityPct - 4 + rnd() * 8
   726→    out.push({
   727→      date: format(t, 'yyyy-MM-dd'),
   728→      label: format(t, 'd.MM'),
   729→      full: format(t, 'd MMMM', { locale: ru }),
   730→      weightKg: null,
   731→      tempInC: Math.round(tempC * 10) / 10,
   732→      humidityPct: Math.round(humidityPct),

   735→  // последние две точки точно соответствуют «сейчас»
   736→  out[out.length - 1].tempInC = h.base.tempC
   737→  out[out.length - 1].humidityPct = h.base.humidityPct
   738→  return out
   739→}

   761→export interface VarroaPoint {
   762→  label: string
   763→  /** Историческое значение заклещенности, %. */
   764→  pct: number | null
   765→  /** Пунктирный сегмент прогноза: последнее значение → прогноз +14 дней. */
   766→  forecast: number | null
   767→  /** Отметка проведенной обработки. */
   768→  treated: boolean
   769→}
   770→
   771→/**
   772→ * Динамика заклещенности за 30 дней + прогноз на 14 дней вперед.
   773→ * Ульи №1 и №2 обработаны щавелевой кислотой 12 дней назад (виден дип),
   774→ * №3 — лечения не было, показатель растет, №4 — обработка 20 дней назад.
   775→ *
   776→ * opts.overridePct — «текущее» значение после проведения обработки или нового
   777→ * замера (кривая сходится к нему); opts.treatedToday — сегодня была обработка:
   778→ * на последней точке рисуется дип, а прогноз считается от нового значения.
   779→ */
   780→export function demoVarroaSeries(
   781→  h: DemoHive,
   782→  days = 30,
   783→  opts?: { overridePct?: number; treatedToday?: boolean },
   784→): VarroaPoint[] {
   785→  const rnd = seeded(3000 + h.id * 57)
   786→  const out: VarroaPoint[] = []
   787→  const nowMs = Date.now()
   788→
   789→  const curPct = opts?.overridePct ?? h.base.varroaPct
   790→  const treatedToday = opts?.treatedToday ?? false
   791→  // прогноз: после обработки семья восстанавливается (+0,1 за 14 дн),
   792→  // без обработки критический улей продолжает расти (+1,6)
   793→  const futurePct =
   794→    opts?.overridePct !== undefined || treatedToday
   795→      ? Math.round((curPct + (treatedToday ? 0.1 : 0.2)) * 10) / 10
   796→      : h.base.varroaForecast14d
   797→
   798→  // траектория «назад»: стартовая точка → текущая, с дипом после обработки
   799→  const treatDay = h.scenario === 'norm' || h.scenario === 'flow' ? 12 : h.scenario === 'swarm' ? 20 : -1
   800→  const startPct = h.scenario === 'varroa' ? 1.2 : h.scenario === 'swarm' ? 2.6 : 1.4
   801→
   802→  for (let k = days - 1; k >= 0; k--) {
   803→    const t = new Date(nowMs - k * 86400000)
   804→    const progress = 1 - k / (days - 1) // 0 → 1 к «сегодня»
   805→    let pct = startPct + (curPct - startPct) * progress
   806→    // дип после обработки: −1,2 % в течение 3 дней
   807→    if (treatDay > 0 && k <= treatDay && k > treatDay - 4) {
   808→      pct -= 1.2 * ((treatDay - k + 1) / 4)
   809→    }
   810→    pct += (rnd() - 0.5) * 0.12
   811→    out.push({
   812→      label: format(t, 'd.MM'),
   813→      pct: Math.max(0.2, Math.round(pct * 10) / 10),
   814→      forecast: null,
   815→      treated: treatDay > 0 && k === treatDay,
   816→    })
   817→  }
   818→  out[out.length - 1].pct = curPct
   819→
   820→  // свежая обработка: дип на последней точке (кривая резко падает)
   821→  if (treatedToday && out.length >= 2) {
   822→    out[out.length - 2].treated = true
   823→  }
   824→
   825→  // прогноз: через 14 дней (пунктир от текущей точки)
   826→  const future = new Date(nowMs + 14 * 86400000)
   827→  out.push({
   828→    label: format(future, 'd.MM'),
   829→    pct: null,
   830→    forecast: futurePct,
   831→    treated: false,
   832→  })
   833→  // сегмент начинается с последней исторической точки
   834→  out[out.length - 2].forecast = curPct
   835→  return out
   836→}
   837→
   838→/**
   839→ * Привесы за 7 дней: столбцы суточного привеса + ночная температура
   840→ * (ночные похолодания объясняют провалы приноса нектара).
   841→ */
   842→export function demoGainSeries(h: DemoHive, days = 7): Array<{ label: string; gainKg: number; nightTempC: number }> {
   843→  const rnd = seeded(4000 + h.id * 83)
   844→  const out: Array<{ label: string; gainKg: number; nightTempC: number }> = []
   845→  const nowMs = Date.now()
   846→  for (let k = days - 1; k >= 0; k--) {
   847→    const t = new Date(nowMs - k * 86400000)
   848→    // привес качается вокруг базового: погода и сила семьи
   849→    const gainKg = k === 0 ? h.base.dailyGainKg : Math.max(-0.1, h.base.dailyGainKg + (rnd() - 0.45) * 0.5)
   850→    // ночная температура: волна + случайные похолодания (2 прохладные ночи)
   851→    const wave = Math.sin(((days - k) / days) * Math.PI * 1.4) * 3
   852→    const dip = k === days - 2 || k === 1 ? -3.5 : 0
   853→    const nightTempC = 14 + wave + dip + (rnd() - 0.5) * 2
   854→    out.push({
   855→      label: format(t, 'd.MM'),
   856→      gainKg: Math.round(gainKg * 100) / 100,
   857→      nightTempC: Math.round(nightTempC * 10) / 10,
   858→    })
   859→  }
   860→  return out
   861→}
