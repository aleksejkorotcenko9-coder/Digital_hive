    14→ * и датчикам, что видит пчеловод на карте и дашборде.

    80→    const timer = setInterval(() => void load(), 5000)

    82→      alive = false

    86→
    87→  // Погода по фактическому центру пасеки (среднее координат точек), те же
    88→  // координаты, что на дашборде, чтобы агент называл те же числа, что видит
    89→  // пчеловод. Эффект зависит от ЧИСЕЛ координат: опрос точек каждые 5 с
    90→  // не вызывает перезапрос погоды, пока координаты не изменились.
    91→  const center = useMemo<[number, number]>(
    92→    () =>
    93→      (apiaryPoints && apiaryPoints.length > 0 ? pointsCenter(apiaryPoints) : null) ??
    94→      DEFAULT_PLACE.center,
    95→    [apiaryPoints],
    96→  )
    97→  const [wLat, wLon] = center
    98→  useEffect(() => {
    99→    let cancelled = false
   100→    const sp = new URLSearchParams({ lat: String(wLat), lon: String(wLon), place: 'Моя пасека' })
   101→    const load = async () => {
   102→      try {
   103→        const res = await fetch(`/api/weather?${sp.toString()}`, { cache: 'no-store' })
   104→        if (!res.ok) throw new Error('weather')
   105→        const j = (await res.json()) as WeatherResponse
   106→        if (!cancelled) setWResp(j)
   107→      } catch {
   108→        // погода не критична: агент честно скажет, если данных нет

   110→    }
   111→    void load()
   112→    const timer = setInterval(() => void load(), 10 * 60 * 1000)
   113→    return () => {
   114→      cancelled = true
   115→      clearInterval(timer)
   116→    }

   123→    const points: AgentPoint[] = apiaryPoints.map((p) => ({
   124→      id: p.id,

   132→    }))
   133→    return makeContext(pointsToHives(apiaryPoints), {
   134→      weather: agentWeather?.days ?? null,
   135→      weatherNow: agentWeather?.now ?? null,
   136→      pesticide: null, // реестр обработок не подключён — агент честно об этом скажет
   137→      sites: data ? data.sites.map(siteDtoToSite) : [],
   138→      points,
   139→      apiaryName: 'Моя пасека',
   140→      lang: useAppState.getState().lang,
   141→    })
   142→  }, [apiaryPoints, agentWeather, data])
   143→
   144→  // контекст мемоизирован, чтобы не пересоздаваться на каждом рендере
   145→  // (иначе эффекты с ctx в зависимостях зацикливаются)
   146→  const ctx: AgentContext | null = pointsCtx
   147→
   148→  const [items, setItems] = useState<ChatItem[]>([])
   149→  const [input, setInput] = useState('')
   150→  const [thinking, setThinking] = useState(false)
   151→  const thinkingRef = useRef(false)
   152→  const scrollRef = useRef<HTMLDivElement>(null)
   153→
   154→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   155→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   156→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   157→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   158→  const srClass = useMemo(() => getSpeechRecognition(), [])
   159→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   160→  const [voice, setVoice] = useState<VoiceState>('idle')
   161→  const speechRef = useRef<SpeechRecognitionLike | null>(null)
   162→  const recorderRef = useRef<MediaRecorder | null>(null)
   163→  const streamRef = useRef<MediaStream | null>(null)
   164→  const chunksRef = useRef<Blob[]>([])
   165→  const autoStopRef = useRef<number | null>(null)
   166→  const baseInputRef = useRef('')
   167→  // push-to-talk: кнопка микрофона зажата и момент начала удержания
   168→  // (по длительности отличаем намеренное удержание от случайного тапа)
   169→  const holdActiveRef = useRef(false)
   170→  const holdStartedAtRef = useRef(0)
   171→
   172→  const clearAutoStop = () => {
   173→    if (autoStopRef.current !== null) {
   174→      window.clearTimeout(autoStopRef.current)
   175→      autoStopRef.current = null
   176→    }
   177→  }
   178→
   179→  const stopVoice = useCallback(() => {
   180→    clearAutoStop()
   181→    if (speechRef.current) {
   182→      try {
   183→        speechRef.current.stop()
   184→      } catch {
   185→        // уже остановлен
   186→      }
   187→      return
   188→    }
   189→    const rec = recorderRef.current
   190→    if (rec && rec.state !== 'inactive') rec.stop()
   191→  }, [])
   192→
   193→  /** Отпускание кнопки (push-to-talk): завершаем запись. */
   194→  const releaseHold = useCallback(() => {
   195→    if (!holdActiveRef.current) return
   196→    holdActiveRef.current = false
   197→    const wasRecording = voice === 'recording'
   198→    const heldMs = Date.now() - holdStartedAtRef.current
   199→    stopVoice()
   200→    // мгновенный тап вместо удержания — вежливо подсказываем, как пользоваться
   201→    if (wasRecording && heldMs < 600) {
   202→      toast({
   203→        title: 'Короткое нажатие',
   204→        description: 'Зажмите кнопку микрофона и говорите, отпустите — вопрос появится в поле ввода.',
   205→      })
   206→    }
   207→  }, [stopVoice, voice, toast])
   208→
   209→  /** Запасной путь: собранный blob → WAV → /api/asr → текст. */
   210→  const finishVoice = useCallback(async () => {
   211→    // поток больше не нужен
   212→    streamRef.current?.getTracks().forEach((t) => t.stop())
   213→    streamRef.current = null
   214→    recorderRef.current = null
   215→    const blob = new Blob(chunksRef.current, { type: chunksRef.current[0]?.type || 'audio/webm' })
   216→    chunksRef.current = []
   217→    if (blob.size < 2048) {
   218→      setVoice('idle')
   219→      toast({ title: 'Запись пустая', description: 'Зажмите кнопку микрофона и скажите вопрос голосом.' })
   220→      return
   221→    }
   222→    setVoice('transcribing')
   223→    try {
   224→      const wav = await blobToWavBase64(blob)
   225→      const res = await fetch('/api/asr', {
   226→        method: 'POST',
   227→        headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify({ audio: wav }),
   229→      })
   230→      const j = (await res.json().catch(() => ({}))) as { text?: string; error?: string }
   231→      if (!res.ok || !j.text?.trim()) {
   232→        toast({ title: 'Речь не распознана', description: j.error ?? 'Попробуйте сказать чуть чётче.', variant: 'destructive' })
   233→        return
   234→      }
   235→      // текст подставляем в поле ввода — пчеловод может поправить и отправить
   236→      const base = baseInputRef.current
   237→      const text = j.text.trim()
   238→      setInput(base ? `${base} ${text}` : text)
   239→      toast({ title: 'Голос распознан', description: text })
   240→    } catch {
   241→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   242→    } finally {
   243→      setVoice('idle')
   244→    }
   245→  }, [toast])
   246→
   247→  const startVoice = useCallback(async () => {
   248→    // запись уже идёт — второй старт (второй палец/указатель) не нужен
   249→    if (speechRef.current || recorderRef.current) return
   250→    clearAutoStop()
   251→    baseInputRef.current = input.trim()
   252→
   253→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   254→    if (srClass) {
   255→      const rec = new srClass()
   256→      rec.lang = VOICE_LANG
   257→      rec.continuous = true
   258→      rec.interimResults = true
   259→      rec.maxAlternatives = 1
   260→      let finalText = ''
   261→      rec.onresult = (e) => {
   262→        let interim = ''
   263→        for (let i = e.resultIndex; i < e.results.length; i++) {
   264→          const r = e.results[i]
   265→          const t = r[0]?.transcript ?? ''
   266→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   267→          else interim += t
   268→        }
   269→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   270→        if (text) {
   271→          const base = baseInputRef.current
   272→          setInput(base ? `${base} ${text}` : text)
   273→        }
   274→      }
   275→      rec.onerror = (ev) => {
   276→        const code = ev.error ?? ''
   277→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   278→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   279→        } else if (code === 'no-speech') {
   280→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   281→        } else if (code !== 'aborted') {
   282→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   283→        }
   284→      }
   285→      rec.onend = () => {
   286→        speechRef.current = null
   287→        clearAutoStop()
   288→        setVoice('idle')
   289→      }
   290→      speechRef.current = rec
   291→      try {
   292→        rec.start()
   293→      } catch {
   294→        toast({ title: 'Не получилось начать', description: 'Зажмите кнопку микрофона ещё раз.', variant: 'destructive' })
   295→        return
   296→      }
   297→      setVoice('recording')
   298→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   299→      // кнопку отпустили раньше, чем распознавание успело стартовать
   300→      if (!holdActiveRef.current) stopVoice()
   301→      return
   302→    }
   303→
   304→    // 2) запасной путь: запись с микрофона → серверное распознавание
   305→    try {
   306→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   307→      streamRef.current = stream
   308→      chunksRef.current = []
   309→      const rec = new MediaRecorder(stream)
   310→      rec.ondataavailable = (e) => {
   311→        if (e.data.size > 0) chunksRef.current.push(e.data)
   312→      }
   313→      rec.onstop = () => void finishVoice()
   314→      rec.start()
   315→      recorderRef.current = rec
   316→      setVoice('recording')
   317→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   318→      // кнопку отпустили, пока запрашивался доступ к микрофону
   319→      if (!holdActiveRef.current) stopVoice()
   320→    } catch {
   321→      toast({
   322→        title: 'Микрофон недоступен',
   323→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   324→        variant: 'destructive',
   325→      })
   326→    }
   327→  }, [srClass, input, finishVoice, stopVoice, toast])
   328→
   329→  // при размонтировании прерываем распознавание и закрываем поток
   330→  useEffect(
   331→    () => () => {
   332→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   333→      try {
   334→        speechRef.current?.abort()
   335→      } catch {
   336→        // нет активного распознавания
   337→      }
   338→      streamRef.current?.getTracks().forEach((t) => t.stop())
   339→    },
   340→    [],
   341→  )
   342→
   343→  /* История из базы — общая лента пасеки, без привязки к улью */
   344→  useEffect(() => {
   345→    if (!user || realStatus !== 'ready' || !ctx) return
   346→    let cancelled = false
   347→    void (async () => {
   348→      try {
   349→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   350→        if (!res.ok) throw new Error('history')
   351→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   352→        if (cancelled) return
   353→        const restored: ChatItem[] = j.items.map((m) => ({
   354→          role: m.role,
   355→          text: m.text,
   356→          ...(m.role === 'agent' && m.agentJson
   357→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   358→            : {}),
   359→        }))
   360→        if (restored.length === 0) {
   361→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   362→        } else {
   363→          setItems(restored.slice(-30))
   364→        }
   365→      } catch {
   366→        if (!cancelled) {
   367→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   368→        }
   369→      }
   370→    })()
   371→    return () => {
   372→      cancelled = true
   373→    }
   374→  }, [user, realStatus, ctx])
   375→
   376→  const sendUser = useCallback(
   377→    (rawText: string) => {
   378→      const trimmed = rawText.trim()
   379→      if (!trimmed || thinkingRef.current) return
   380→      // отправка текста прерывает голосовой ввод (кнопка могла быть ещё зажата)
   381→      if (holdActiveRef.current) {
   382→        holdActiveRef.current = false
   383→        stopVoice()
   384→      }
   385→      const context = ctx
   386→      if (!context) return
   387→      thinkingRef.current = true
   388→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   389→      setInput('')
   390→      setThinking(true)
   391→      const started = Date.now()
   392→
   393→      const run = async () => {
   394→        let answer: AgentMessage
   395→        try {
   396→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   397→          const res = await fetch('/api/agent', {
   398→            method: 'POST',
   399→            headers: { 'Content-Type': 'application/json' },
   400→            body: JSON.stringify({ question: trimmed }),
   401→          })
   402→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   403→          const j = (await res.json()) as { answer: AgentMessage }
   404→          answer = j.answer
   405→        } catch {
   406→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   407→        }
   408→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   409→        const elapsed = Date.now() - started
   410→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   411→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   412→        thinkingRef.current = false
   413→        setThinking(false)
   414→        // сохраняем переписку в базу — общая лента пасеки
   415→        if (useAppState.getState().user) {
   416→          void fetch('/api/chat', {
   417→            method: 'POST',
   418→            headers: { 'Content-Type': 'application/json' },
   419→            body: JSON.stringify({
   420→              userText: trimmed,
   421→              agentText: answer.text,
   422→              agent: answer,
   423→            }),
   424→          }).catch(() => undefined)
   425→        }
   426→      }
   427→      void run()
   428→    },
   429→    [ctx, stopVoice],
   430→  )
   431→
   432→  // consume prefill, заданного до монтирования + подписка на новые вопросы из дашборда
   433→  useEffect(() => {
   434→    useAppState.setState({ chatPrefill: null })
   435→    const unsub = useAppState.subscribe((state, prev) => {
   436→      if (state.chatPrefill && state.chatPrefill !== prev.chatPrefill) {
   437→        useAppState.setState({ chatPrefill: null })
   438→        sendUser(state.chatPrefill)
   439→      }
   440→    })
   441→    return unsub
   442→  }, [sendUser])
   443→
   444→  useEffect(() => {
   445→    const el = scrollRef.current
   446→    if (el) el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' })
   447→  }, [items, thinking])
   448→
   449→  /* Контекст ещё не готов — экран загрузки; пустая карта — честная подсказка */
   450→  if (!ctx || ctx.hives.length === 0) {
   451→    const noPoints = apiaryPoints !== null && apiaryPoints.length === 0
   452→    return (
   453→      <Card className="border-red-100">
   454→        <CardContent className="p-10 text-center">
   455→          <HexChip className="mx-auto h-12 w-12 animate-pulse bg-red-100">
