     3→/**
     4→ * Экран «ИИ-агент»: чат пчеловода с цифровым помощником.
     5→ * Агент отвечает сразу по всем ульям пасеки: телеметрия из базы, датчики точек
     6→ * на карте (температура, вес, влажность, шум) и реальная погода Open-Meteo.
     7→ * Темы строго ограничены: погода и ульи — отвечает, всё остальное — вежливо
     8→ * отказывается. Модель, на которой работает агент, пользователю не показывается.
     9→ * Есть голосовой ввод: кнопка микрофона записывает вопрос, речь распознаётся
    10→ * на сервере (/api/asr) и подставляется в поле ввода.
    11→ * Улии и показания синхронизированы с картой: и в демо, и в «Моих данных» агент
    12→ * говорит по тем же точкам и датчикам, что видит пчеловод на карте и дашборде.
    13→ * История переписки в реальном режиме сохраняется в SQLite (общая лента пасеки).
    14→ */

    51→export function AgentChatView() {
    52→  const mode = useAppState((s) => s.mode)
    53→  const user = useAppState((s) => s.user)
    54→  const { toast } = useToast()
    55→  const { status: realStatus, data } = useRealData()
    56→
    57→  // Погода для контекста агента: Open-Meteo через /api/weather (в обоих режимах —
    58→  // та же панель погоды, что на дашборде, чтобы агент называл те же числа)
    59→  const [wResp, setWResp] = useState<WeatherResponse | null>(null)
    60→  useEffect(() => {
    61→    let cancelled = false
    62→    void (async () => {
    63→      try {
    64→        const res = await fetch('/api/weather', { cache: 'no-store' })
    65→        if (!res.ok) throw new Error('weather')
    66→        const j = (await res.json()) as WeatherResponse
    67→        if (!cancelled) setWResp(j)
    68→      } catch {
    69→        // погода не критична: агент честно скажет, если данных нет
    70→      }
    71→    })()
    72→    const timer = setInterval(() => {
    73→      void (async () => {
    74→        try {
    75→          const res = await fetch('/api/weather', { cache: 'no-store' })
    76→          if (!res.ok) return
    77→          const j = (await res.json()) as WeatherResponse
    78→          if (!cancelled) setWResp(j)
    79→        } catch {
    80→          // оставляем предыдущий прогноз
    81→        }
    82→      })()
    83→    }, 10 * 60 * 1000)
    84→    return () => {
    85→      cancelled = true
    86→      clearInterval(timer)
    87→    }
    88→  }, [])
    89→  const agentWeather = useMemo(() => (wResp ? weatherResponseToDays(wResp) : null), [wResp])
    90→
    91→  // Демо: ульи и показания — те же точки, что на карте (синхронизация с /api/map-points)
    92→  const [demoPoints, setDemoPoints] = useState<MapPointDto[] | null>(null)
    93→  useEffect(() => {
    94→    if (mode !== 'demo') return
    95→    let alive = true
    96→    const load = async () => {
    97→      try {
    98→        const res = await fetch('/api/map-points', { cache: 'no-store' })
    99→        if (!res.ok) return
   100→        const j = (await res.json()) as { points: MapPointDto[] }
   101→        if (alive) setDemoPoints(j.points)
   102→      } catch {
   103→        // датчики подтянутся на следующем опросе
   104→      }
   105→    }
   106→    void load()
   107→    const timer = setInterval(() => void load(), 5000)
   108→    return () => {
   109→      alive = false
   110→      clearInterval(timer)
   111→    }
   112→  }, [mode])
   113→
   114→  const demoCtx = useMemo<AgentContext | null>(() => {
   115→    if (!demoPoints) return null
   116→    const points: AgentPoint[] = demoPoints.map((p) => ({
   117→      id: p.id,
   118→      name: p.name,
   119→      lat: p.lat,
   120→      lon: p.lon,
   121→      hiveCount: p.hiveCount,
   122→      hasSensor: p.hasSensor,
   123→      sensorOn: p.sensorOn,
   124→      sensor: p.sensor,
   125→    }))
   126→    return makeContext(pointsToHives(demoPoints), {
   127→      weather: agentWeather?.days ?? null,
   128→      weatherNow: agentWeather?.now ?? null,
   129→      pesticide: pesticideAlert, // сценарий демо: реестр обработок к пасеке не подключён
   130→      sites: migrationSites,
   131→      points,
   132→      apiaryName: apiary.name,
   133→    })
   134→  }, [demoPoints, agentWeather])
   135→
   136→  // контекст мемоизирован, чтобы не пересоздаваться на каждом рендере
   137→  // (иначе эффекты с ctx в зависимостях зацикливаются)
   138→  const ctx: AgentContext | null = useMemo(
   139→    () =>
   140→      mode === 'demo'
   141→        ? demoCtx
   142→        : data
   143→          ? buildRealContext(data, agentWeather?.days ?? null, agentWeather?.now ?? null)
   144→          : null,
   145→    [mode, demoCtx, data, agentWeather],
   146→  )
   147→
   148→  const [items, setItems] = useState<ChatItem[]>([])
   149→  const [input, setInput] = useState('')
   150→  const [thinking, setThinking] = useState(false)
   151→  const thinkingRef = useRef(false)
   152→  const scrollRef = useRef<HTMLDivElement>(null)
   153→
   154→  /* ---------- голосовой ввод: микрофон → текст в поле ввода ---------- */
   155→  // Основной путь — Web Speech API (Chrome/Edge/Яндекс.Браузер, ru-RU): текст
   156→  // появляется в поле ввода прямо во время разговора. Запасной путь (браузеры
   157→  // без нативного распознавания) — запись MediaRecorder → WAV → /api/asr.
   158→  const srClass = useMemo(() => getSpeechRecognition(), [])
   159→  const voiceOk = useMemo(() => !!srClass || voiceInputSupported(), [srClass])
   160→  const [voice, setVoice] = useState<VoiceState>('idle')
   161→  const speechRef = useRef<SpeechRecognitionLike | null>(null)
   162→  const recorderRef = useRef<MediaRecorder | null>(null)
   163→  const streamRef = useRef<MediaStream | null>(null)
   164→  const chunksRef = useRef<Blob[]>([])
   165→  const autoStopRef = useRef<number | null>(null)
   166→  const baseInputRef = useRef('')
   167→
   168→  const clearAutoStop = () => {
   169→    if (autoStopRef.current !== null) {
   170→      window.clearTimeout(autoStopRef.current)
   171→      autoStopRef.current = null
   172→    }
   173→  }
   174→
   175→  const stopVoice = useCallback(() => {
   176→    clearAutoStop()
   177→    if (speechRef.current) {
   178→      try {
   179→        speechRef.current.stop()
   180→      } catch {
   181→        // уже остановлен
   182→      }
   183→      return
   184→    }
   185→    const rec = recorderRef.current
   186→    if (rec && rec.state !== 'inactive') rec.stop()
   187→  }, [])
   188→
   189→  /** Запасной путь: собранный blob → WAV → /api/asr → текст. */
   190→  const finishVoice = useCallback(async () => {
   191→    // поток больше не нужен
   192→    streamRef.current?.getTracks().forEach((t) => t.stop())
   193→    streamRef.current = null
   194→    recorderRef.current = null
   195→    const blob = new Blob(chunksRef.current, { type: chunksRef.current[0]?.type || 'audio/webm' })
   196→    chunksRef.current = []
   197→    if (blob.size < 2048) {
   198→      setVoice('idle')
   199→      toast({ title: 'Запись пустая', description: 'Нажмите микрофон и скажите вопрос голосом.' })
   200→      return
   201→    }
   202→    setVoice('transcribing')
   203→    try {
   204→      const wav = await blobToWavBase64(blob)
   205→      const res = await fetch('/api/asr', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ audio: wav }),
   209→      })
   210→      const j = (await res.json().catch(() => ({}))) as { text?: string; error?: string }
   211→      if (!res.ok || !j.text?.trim()) {
   212→        toast({ title: 'Речь не распознана', description: j.error ?? 'Попробуйте сказать чуть чётче.', variant: 'destructive' })
   213→        return
   214→      }
   215→      // текст подставляем в поле ввода — пчеловод может поправить и отправить
   216→      const base = baseInputRef.current
   217→      const text = j.text.trim()
   218→      setInput(base ? `${base} ${text}` : text)
   219→      toast({ title: 'Голос распознан', description: text })
   220→    } catch {
   221→      toast({ title: 'Не получилось', description: 'Не удалось распознать запись — попробуйте ещё раз.', variant: 'destructive' })
   222→    } finally {
   223→      setVoice('idle')
   224→    }
   225→  }, [toast])
   226→
   227→  const startVoice = useCallback(async () => {
   228→    clearAutoStop()
   229→    baseInputRef.current = input.trim()
   230→
   231→    // 1) нативное распознавание в браузере — русский отлично работает в Chrome/Яндексе
   232→    if (srClass) {
   233→      const rec = new srClass()
   234→      rec.lang = VOICE_LANG
   235→      rec.continuous = true
   236→      rec.interimResults = true
   237→      rec.maxAlternatives = 1
   238→      let finalText = ''
   239→      rec.onresult = (e) => {
   240→        let interim = ''
   241→        for (let i = e.resultIndex; i < e.results.length; i++) {
   242→          const r = e.results[i]
   243→          const t = r[0]?.transcript ?? ''
   244→          if (r.isFinal) finalText = finalText ? `${finalText} ${t.trim()}` : t.trim()
   245→          else interim += t
   246→        }
   247→        const text = (finalText + (interim ? ` ${interim}` : '')).trim()
   248→        if (text) {
   249→          const base = baseInputRef.current
   250→          setInput(base ? `${base} ${text}` : text)
   251→        }
   252→      }
   253→      rec.onerror = (ev) => {
   254→        const code = ev.error ?? ''
   255→        if (code === 'not-allowed' || code === 'service-not-allowed') {
   256→          toast({ title: 'Микрофон недоступен', description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.', variant: 'destructive' })
   257→        } else if (code === 'no-speech') {
   258→          toast({ title: 'Речи не слышно', description: 'Скажите вопрос чуть ближе к микрофону.' })
   259→        } else if (code !== 'aborted') {
   260→          toast({ title: 'Распознавание прервано', description: 'Скажите вопрос ещё раз.', variant: 'destructive' })
   261→        }
   262→      }
   263→      rec.onend = () => {
   264→        speechRef.current = null
   265→        clearAutoStop()
   266→        setVoice('idle')
   267→      }
   268→      speechRef.current = rec
   269→      try {
   270→        rec.start()
   271→      } catch {
   272→        toast({ title: 'Не получилось начать', description: 'Нажмите микрофон ещё раз.', variant: 'destructive' })
   273→        return
   274→      }
   275→      setVoice('recording')
   276→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   277→      return
   278→    }
   279→
   280→    // 2) запасной путь: запись с микрофона → серверное распознавание
   281→    try {
   282→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   283→      streamRef.current = stream
   284→      chunksRef.current = []
   285→      const rec = new MediaRecorder(stream)
   286→      rec.ondataavailable = (e) => {
   287→        if (e.data.size > 0) chunksRef.current.push(e.data)
   288→      }
   289→      rec.onstop = () => void finishVoice()
   290→      rec.start()
   291→      recorderRef.current = rec
   292→      setVoice('recording')
   293→      autoStopRef.current = window.setTimeout(() => stopVoice(), VOICE_MAX_SECONDS * 1000)
   294→    } catch {
   295→      toast({
   296→        title: 'Микрофон недоступен',
   297→        description: 'Разрешите доступ к микрофону в браузере и попробуйте ещё раз.',
   298→        variant: 'destructive',
   299→      })
   300→    }
   301→  }, [srClass, input, finishVoice, stopVoice, toast])
   302→
   303→  // при размонтировании прерываем распознавание и закрываем поток
   304→  useEffect(
   305→    () => () => {
   306→      if (autoStopRef.current !== null) window.clearTimeout(autoStopRef.current)
   307→      try {
   308→        speechRef.current?.abort()
   309→      } catch {
   310→        // нет активного распознавания
   311→      }
   312→      streamRef.current?.getTracks().forEach((t) => t.stop())
   313→    },
   314→    [],
   315→  )
   316→
   317→  /* История из базы (реальный режим) — общая лента пасеки, без привязки к улью */
   318→  useEffect(() => {
   319→    if (mode !== 'real' || !user || realStatus !== 'ready' || !ctx) return
   320→    let cancelled = false
   321→    void (async () => {
   322→      try {
   323→        const res = await fetch('/api/chat?scope=general', { cache: 'no-store' })
   324→        if (!res.ok) throw new Error('history')
   325→        const j = (await res.json()) as { items: ChatHistoryItem[] }
   326→        if (cancelled) return
   327→        const restored: ChatItem[] = j.items.map((m) => ({
   328→          role: m.role,
   329→          text: m.text,
   330→          ...(m.role === 'agent' && m.agentJson
   331→            ? { agent: JSON.parse(m.agentJson) as AgentMessage }
   332→            : {}),
   333→        }))
   334→        if (restored.length === 0) {
   335→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   336→        } else {
   337→          setItems(restored.slice(-30))
   338→        }
   339→      } catch {
   340→        if (!cancelled) {
   341→          setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   342→        }
   343→      }
   344→    })()
   345→    return () => {
   346→      cancelled = true
   347→    }
   348→  }, [mode, user, realStatus, ctx])
   349→
   350→  /* Демо: приветствие один раз (история в демо не сохраняется) */
   351→  useEffect(() => {
   352→    if (mode !== 'demo' || !ctx) return
   353→    if (items.length === 0) {
   354→      setItems([{ role: 'agent', text: '', agent: askAgent('привет', ctx.hives[0]?.id ?? 1, ctx) }])
   355→    }
   356→  }, [mode, ctx, items.length])
   357→
   358→  const sendUser = useCallback(
   359→    (rawText: string) => {
   360→      const trimmed = rawText.trim()
   361→      if (!trimmed || thinkingRef.current) return
   362→      const context =
   363→        mode === 'demo'
   364→          ? demoCtx
   365→          : data
   366→            ? buildRealContext(data, agentWeather?.days ?? null, agentWeather?.now ?? null)
   367→            : null
   368→      if (!context) return
   369→      thinkingRef.current = true
   370→      setItems((prev) => [...prev, { role: 'user', text: trimmed }])
   371→      setInput('')
   372→      setThinking(true)
   373→      const started = Date.now()
   374→
   375→      const run = async () => {
   376→        let answer: AgentMessage
   377→        try {
   378→          // настоящий LLM-агент на сервере; при сбое — встроенные правила
   379→          const res = await fetch('/api/agent', {
   380→            method: 'POST',
   381→            headers: { 'Content-Type': 'application/json' },
   382→            body: JSON.stringify({ question: trimmed, mode }),
   383→          })
   384→          if (!res.ok) throw new Error(`HTTP ${res.status}`)
   385→          const j = (await res.json()) as { answer: AgentMessage }
   386→          answer = j.answer
   387→        } catch {
   388→          answer = askAgent(trimmed, context.hives[0]?.id ?? 1, context)
   389→        }
   390→        // лёгкая пауза, чтобы индикатор «думает» не мигал при быстрых ответах
   391→        const elapsed = Date.now() - started
   392→        if (elapsed < 600) await new Promise((r) => setTimeout(r, 600 - elapsed))
   393→        setItems((prev) => [...prev, { role: 'agent', text: '', agent: answer }])
   394→        thinkingRef.current = false
   395→        setThinking(false)
   396→        // сохраняем переписку в базу (реальный режим) — общая лента пасеки
   397→        if (mode === 'real' && useAppState.getState().user) {
   398→          void fetch('/api/chat', {
   399→            method: 'POST',
   400→            headers: { 'Content-Type': 'application/json' },
   401→            body: JSON.stringify({
   402→              userText: trimmed,
   403→              agentText: answer.text,
   404→              agent: answer,
   405→            }),
   406→          }).catch(() => undefined)
   407→        }
   408→      }
   409→      void run()
   410→    },
   411→    [mode, demoCtx, data, agentWeather],
   412→  )
   413→
   414→  // consume prefill, заданного до монтирования + подписка на новые вопросы из дашборда
   415→  useEffect(() => {
   416→    useAppState.setState({ chatPrefill: null })
   417→    const unsub = useAppState.subscribe((state, prev) => {
   418→      if (state.chatPrefill && state.chatPrefill !== prev.chatPrefill) {
   419→        useAppState.setState({ chatPrefill: null })
   420→        sendUser(state.chatPrefill)
   421→      }
   422→    })
   423→    return unsub
   424→  }, [sendUser])
   425→
   426→  useEffect(() => {
   427→    const el = scrollRef.current
   428→    if (el) el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' })
   429→  }, [items, thinking])
   430→
   431→  /* Контекст ещё не готов — экран загрузки; пустая карта — честная подсказка */
   432→  if (!ctx || ctx.hives.length === 0) {
   433→    const demoEmpty = mode === 'demo' && demoPoints !== null && demoPoints.length === 0
   434→    return (
   435→      <Card className="border-red-100">
   436→        <CardContent className="p-10 text-center">
   437→          <HexChip className="mx-auto h-12 w-12 animate-pulse bg-red-100">
   438→            <Bot className="h-6 w-6 text-red-600" />
   439→          </HexChip>
   440→          <p className="mt-3 text-sm text-gray-500">
   441→            {demoEmpty
   442→              ? 'На карте пасеки ещё нет ульев с датчиками — привяжите улей по коду с корпуса датчика, и агент начнёт отвечать по его показаниям.'
   443→              : mode === 'real'
   444→                ? 'Ждём данные из базы: агент отвечает по телеметрии ваших ульев.'
   445→                : 'Агент загружается…'}
   446→          </p>
   447→        </CardContent>
   448→      </Card>
   449→    )
   450→  }
   451→
   452→  return (
   453→    <div className="grid gap-4 lg:grid-cols-[1fr_280px]">
   454→      {/* Чат */}
   455→      <Card className="flex flex-col border-red-100 shadow-sm lg:h-[calc(100vh-190px)] lg:min-h-[520px]">
   456→        <CardContent className="flex min-h-0 flex-1 flex-col p-0">
   457→          {/* шапка чата */}
   458→          <div className="flex flex-wrap items-center gap-2 border-b border-red-100 px-4 py-3">
   459→            <HexChip className="h-9 w-9 bg-gradient-to-br from-[#E93A2D] to-[#9B4DFF]">
   460→              <Sparkles className="h-4.5 w-4.5" />
   461→            </HexChip>
   462→            <div>
   463→              <p className="text-sm font-bold text-[#2A1A12]">ИИ-агент пасеки</p>
   464→              <p className="text-[11px] text-gray-400">
   465→                {mode === 'demo' ? 'данные — ульи и датчики на карте пасеки' : 'отвечает по телеметрии из базы данных'}
   466→              </p>
   467→            </div>
   468→            <div className="ml-auto flex items-center gap-1.5">
